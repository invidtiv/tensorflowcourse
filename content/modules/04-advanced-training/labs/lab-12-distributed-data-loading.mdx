# Distributed Data Loading

```python
"""
Lab 3.3: Distributed Data Loading
==================================
Implement data pipelines for distributed training.
"""

import tensorflow as tf
import numpy as np

def create_distributed_dataset(strategy, x, y, batch_size=64, is_training=True):
    """Create a dataset optimized for distributed training."""
    global_batch_size = batch_size
    
    def dataset_fn(input_context):
        local_batch_size = input_context.get_per_replica_batch_size(global_batch_size)
        dataset = tf.data.Dataset.from_tensor_slices((x, y))
        dataset = dataset.shard(input_context.num_input_pipelines, input_context.input_pipeline_id)
        
        if is_training:
            dataset = dataset.shuffle(buffer_size=10000)
            dataset = dataset.repeat()
        
        dataset = dataset.batch(local_batch_size)
        dataset = dataset.prefetch(tf.data.AUTOTUNE)
        return dataset
    
    return strategy.distribute_datasets_from_function(dataset_fn)

def multi_gpu_training_example():
    gpus = tf.config.list_physical_devices('GPU')
    print(f"Available GPUs: {len(gpus)}")
    
    if len(gpus) > 1:
        strategy = tf.distribute.MirroredStrategy()
    else:
        strategy = tf.distribute.OneDeviceStrategy("/gpu:0" if gpus else "/cpu:0")
    
    print(f"Number of devices: {strategy.num_replicas_in_sync}")
    
    (x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
    x_train, x_test = x_train / 255.0, x_test / 255.0
    y_train = tf.keras.utils.to_categorical(y_train, 10)
    y_test = tf.keras.utils.to_categorical(y_test, 10)
    
    global_batch_size = 64 * strategy.num_replicas_in_sync
    
    train_dist_dataset = create_distributed_dataset(
        strategy, x_train, y_train, global_batch_size, is_training=True)
    test_dist_dataset = create_distributed_dataset(
        strategy, x_test, y_test, global_batch_size, is_training=False)
    
    with strategy.scope():
        model = tf.keras.Sequential([
            tf.keras.layers.Conv2D(32, 3, activation='relu', input_shape=(32, 32, 3)),
            tf.keras.layers.MaxPooling2D(),
            tf.keras.layers.Conv2D(64, 3, activation='relu'),
            tf.keras.layers.MaxPooling2D(),
            tf.keras.layers.Flatten(),
            tf.keras.layers.Dense(128, activation='relu'),
            tf.keras.layers.Dense(10, activation='softmax')
        ])
        
        model.compile(optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
                      loss='categorical_crossentropy', metrics=['accuracy'])
    
    steps_per_epoch = len(x_train) // global_batch_size
    
    model.fit(train_dist_dataset, epochs=10, steps_per_epoch=steps_per_epoch,
              validation_data=test_dist_dataset,
              validation_steps=len(x_test) // global_batch_size)

if tf.config.list_physical_devices('GPU'):
    multi_gpu_training_example()
else:
    print("No GPUs available for distributed training demo")
```
