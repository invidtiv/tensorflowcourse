# Custom Optimizer Implementation

```python
"""
Lab 1.4: Custom Optimizer Implementation
========================================
Implement a custom optimizer by extending tf.keras.optimizers.Optimizer.
"""

import tensorflow as tf
import numpy as np

class CustomSGDWithMomentum(tf.keras.optimizers.Optimizer):
    """
    Custom SGD optimizer with momentum and gradient accumulation.
    Demonstrates how to implement a custom optimizer.
    """
    
    def __init__(self, learning_rate=0.01, momentum=0.9, accumulation_steps=4,
                 name="CustomSGDWithMomentum", **kwargs):
        super().__init__(name, **kwargs)
        self._learning_rate = self._build_learning_rate(learning_rate)
        self._momentum = momentum
        self._accumulation_steps = accumulation_steps
        
    def build(self, var_list):
        super().build(var_list)
        if hasattr(self, '_built') and self._built:
            return
        
        self._momentums = []
        self._grad_accumulators = []
        self._accum_step_counter = self.add_variable(
            shape=(), dtype=tf.int32, name="accum_step_counter",
            initializer=tf.keras.initializers.Zeros())
        
        for var in var_list:
            self._momentums.append(self.add_variable_from_reference(
                model_variable=var, variable_name="m", initial_value=tf.zeros_like(var)))
            self._grad_accumulators.append(self.add_variable_from_reference(
                model_variable=var, variable_name="grad_accum", initial_value=tf.zeros_like(var)))
        
        self._built = True
    
    def _resource_apply_dense(self, grad, variable):
        var_key = self._var_key(variable)
        momentum = self._momentums[self._index_dict[var_key]]
        grad_accum = self._grad_accumulators[self._index_dict[var_key]]
        lr = tf.cast(self._learning_rate, variable.dtype)
        momentum_coeff = tf.cast(self._momentum, variable.dtype)
        
        grad_accum.assign_add(grad)
        self._accum_step_counter.assign_add(1)
        
        def apply_update():
            avg_grad = grad_accum / tf.cast(self._accumulation_steps, grad_accum.dtype)
            momentum.assign(momentum_coeff * momentum - lr * avg_grad)
            variable.assign_add(momentum)
            grad_accum.assign(tf.zeros_like(grad_accum))
            self._accum_step_counter.assign(0)
            return tf.no_op()
        
        tf.cond(tf.greater_equal(self._accum_step_counter, self._accumulation_steps),
                apply_update, lambda: tf.no_op())
    
    def get_config(self):
        config = super().get_config()
        config.update({"learning_rate": self._serialize_hyperparameter("learning_rate"),
                       "momentum": self._momentum,
                       "accumulation_steps": self._accumulation_steps})
        return config

def test_custom_optimizer():
    print("Testing Custom SGD with Gradient Accumulation")
    print("="*50)
    
    (x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
    x_train = x_train[:1000].reshape(-1, 784).astype('float32') / 255.0
    y_train = tf.keras.utils.to_categorical(y_train[:1000], 10)
    
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(128, activation='relu', input_shape=(784,)),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    
    model.compile(optimizer=CustomSGDWithMomentum(learning_rate=0.01, momentum=0.9, accumulation_steps=4),
                  loss='categorical_crossentropy', metrics=['accuracy'])
    
    print("\nTraining with custom optimizer...")
    history = model.fit(x_train, y_train, epochs=10, batch_size=32, verbose=1)
    
    print(f"\nFinal Loss: {history.history['loss'][-1]:.4f}")
    print(f"Final Accuracy: {history.history['accuracy'][-1]:.4f}")
    return history

test_custom_optimizer()
```

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| `_resource_apply_dense` never triggers the update branch | `self._accum_step_counter` is a scalar `tf.Variable` but the `tf.cond` predicate may be evaluated in graph mode before the variable is incremented in eager mode | Move gradient accumulation logic inside `update_step` (TF 2.x API) or verify the counter with `tf.print(self._accum_step_counter)` inside `apply_update()` |
| `build()` called multiple times, creating duplicate slot variables | `hasattr(self, '_built')` check may fail if the optimizer is serialized and restored, re-entering `build()` | Replace the guard with `if self._built: return` using the parent class `_built` flag, which is reliably set by `super().build(var_list)` |
| Gradient accumulator retains stale values across epochs | `grad_accum` is reset to zero only inside `apply_update()`; if total steps is not divisible by `accumulation_steps`, the last partial batch's gradients bleed into the next epoch | Add an `on_epoch_end` callback that manually zeros all `_grad_accumulators` and resets `_accum_step_counter` |
| `accumulation_steps=4` effectively multiplies the learning rate | Averaging 4 gradients before applying the momentum step changes the effective scale of updates compared to step-by-step SGD | Divide the learning rate by `accumulation_steps` when comparing against standard SGD, or document the effective LR = `lr × accumulation_steps` |
| `get_config` missing `accumulation_steps` causes load failure | Model checkpointing calls `optimizer.get_config()` to reconstruct the optimizer; if a hyperparameter is absent, `from_config` raises `TypeError` | Verify the saved config contains all three keys: `learning_rate`, `momentum`, `accumulation_steps`, then call `CustomSGDWithMomentum.from_config(optimizer.get_config())` in a round-trip test |

## Stretch Goals

1. **Extend to AdaGrad accumulation**: Modify `CustomSGDWithMomentum` to accumulate squared gradients in a second slot variable and divide the momentum update by `sqrt(accumulated_sq_grad + epsilon)`, effectively implementing a custom AdaGrad with gradient accumulation.
2. **Comparative experiment vs standard SGD with large batch**: Run `CustomSGDWithMomentum(accumulation_steps=4)` on CIFAR-10 with batch_size=32 and compare it against standard `SGD` with batch_size=128 (same effective batch size); verify that gradient accumulation produces nearly identical weight updates.
3. **Wrap with LookAhead**: Apply a LookAhead outer loop (`k=5` steps, `alpha=0.5`) around `CustomSGDWithMomentum` by overriding `apply_gradients`; measure whether slow-weight interpolation reduces oscillation in the loss curve on MNIST.
4. **Implement Sharpness-Aware Minimization (SAM) step**: Override `apply_gradients` to perform a two-step update — first perturb weights along the gradient direction to find the adversarial perturbation, then compute gradients at the perturbed point, and apply the momentum update with those gradients to seek flatter minima.
