# Adaptive Optimizer Deep Dive - Adam Variants

```python
"""
Lab 1.2: Adam Optimizer Variants Comparison
============================================
Deep dive into Adam variants and their characteristics.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

tf.random.set_seed(42)
np.random.seed(42)

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
x_train, x_test = x_train / 255.0, x_test / 255.0
y_train = tf.keras.utils.to_categorical(y_train, 10)
y_test = tf.keras.utils.to_categorical(y_test, 10)

def create_simple_model():
    return tf.keras.Sequential([
        tf.keras.layers.Conv2D(32, 3, activation='relu', input_shape=(32, 32, 3)),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, 3, activation='relu'),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, 3, activation='relu'),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ])

def get_adam_variants():
    return {
        'Adam': {
            'optimizer': tf.keras.optimizers.Adam(learning_rate=0.001),
            'description': 'Standard Adam with adaptive learning rates'
        },
        'AdamW': {
            'optimizer': tf.keras.optimizers.AdamW(learning_rate=0.001, weight_decay=0.004),
            'description': 'Adam with decoupled weight decay'
        },
        'Adamax': {
            'optimizer': tf.keras.optimizers.Adamax(learning_rate=0.002),
            'description': 'Adam with infinity norm (more stable)'
        },
        'Nadam': {
            'optimizer': tf.keras.optimizers.Nadam(learning_rate=0.001),
            'description': 'Adam with Nesterov momentum'
        },
        'AMSGrad': {
            'optimizer': tf.keras.optimizers.AMSGrad(learning_rate=0.001),
            'description': 'Adam with monotonic improvement guarantee'
        }
    }

def train_and_compare_adam_variants(epochs=20):
    variants = get_adam_variants()
    results = {}
    
    for name, config in variants.items():
        print(f"\nTraining with {name}: {config['description']}")
        
        model = create_simple_model()
        model.compile(optimizer=config['optimizer'],
                      loss='categorical_crossentropy', metrics=['accuracy'])
        
        history = model.fit(x_train, y_train, batch_size=128, epochs=epochs,
                           validation_data=(x_test, y_test), verbose=0)
        
        test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
        results[name] = {'history': history.history, 'test_accuracy': test_acc}
        print(f"  Test Accuracy: {test_acc:.4f}")
    
    return results

def visualize_adam_comparison(results):
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    for name, data in results.items():
        axes[0].plot(data['history']['val_accuracy'], label=name, linewidth=2)
    axes[0].set_title('Adam Variants - Validation Accuracy', fontweight='bold')
    axes[0].set_xlabel('Epoch'); axes[0].set_ylabel('Accuracy')
    axes[0].legend(); axes[0].grid(True, alpha=0.3)
    
    for name, data in results.items():
        axes[1].plot(data['history']['val_loss'], label=name, linewidth=2)
    axes[1].set_title('Adam Variants - Validation Loss', fontweight='bold')
    axes[1].set_xlabel('Epoch'); axes[1].set_ylabel('Loss')
    axes[1].legend(); axes[1].grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.savefig('adam_variants_comparison.png', dpi=150)
    plt.show()

results = train_and_compare_adam_variants(epochs=20)
visualize_adam_comparison(results)
```

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| AMSGrad converges slower than Adam on this task | AMSGrad's monotonically non-decreasing second moment bound is conservative; it undershoots the effective LR that Adam uses | Increase AMSGrad's LR to 0.003 and compare; AMSGrad's advantage appears in non-convex settings with noisy gradients, not short CIFAR-10 runs |
| AdamW weight decay hurts accuracy when set too high | `weight_decay=0.004` applies L2 penalty to all parameters including bias and BN gamma/beta, over-regularising the small model | Exclude BN and bias from weight decay by passing `exclude_from_weight_decay=['bias', 'gamma', 'beta']`, or sweep wd ∈ {0.0001, 0.001, 0.01} |
| Nadam and Adam produce identical curves | Both use the same β₁/β₂ defaults; the Nesterov look-ahead in Nadam only differs in the gradient used for the first moment update | Inspect per-step loss curves (not epoch averages); Nadam's advantage is most visible at high LR (try lr=0.003) |
| All variants plateau at ~75% after epoch 15 | The simple 3-conv architecture is the bottleneck, not the optimizer | Add BatchNormalization after each Conv2D layer and re-run; the ceiling rises to ~80%, and optimizer differences become clearer |
| β₁/β₂ hyperparameters ignored in comparison | Using every optimizer's default β₁=0.9, β₂=0.999 makes the comparison partially unfair; Adamax benefits from lower β₂ | Run a secondary sweep with β₂ ∈ {0.99, 0.999} to isolate the impact of the second-moment decay rate from the variant design |
