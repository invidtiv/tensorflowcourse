# SGD Momentum Analysis

```python
"""
Lab 1.3: SGD Momentum Analysis
==============================
Analyze the effect of different momentum values on training.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

tf.random.set_seed(42)

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
x_train, x_test = x_train / 255.0, x_test / 255.0
y_train = tf.keras.utils.to_categorical(y_train, 10)
y_test = tf.keras.utils.to_categorical(y_test, 10)

def create_model():
    return tf.keras.Sequential([
        tf.keras.layers.Conv2D(32, 3, activation='relu', input_shape=(32, 32, 3)),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, 3, activation='relu'),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ])

def compare_momentum_values():
    momentum_configs = {
        'SGD (no momentum)': tf.keras.optimizers.SGD(learning_rate=0.01),
        'SGD momentum=0.5': tf.keras.optimizers.SGD(learning_rate=0.01, momentum=0.5),
        'SGD momentum=0.9': tf.keras.optimizers.SGD(learning_rate=0.01, momentum=0.9),
        'SGD momentum=0.99': tf.keras.optimizers.SGD(learning_rate=0.01, momentum=0.99),
        'Nesterov momentum=0.9': tf.keras.optimizers.SGD(
            learning_rate=0.01, momentum=0.9, nesterov=True),
    }
    
    results = {}
    
    for name, optimizer in momentum_configs.items():
        print(f"\nTraining: {name}")
        
        model = create_model()
        model.compile(optimizer=optimizer, loss='categorical_crossentropy', metrics=['accuracy'])
        
        history = model.fit(x_train, y_train, batch_size=128, epochs=25,
                           validation_data=(x_test, y_test), verbose=0)
        
        test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
        results[name] = {'history': history.history, 'test_accuracy': test_acc}
        print(f"  Test Accuracy: {test_acc:.4f}")
    
    return results

def visualize_momentum_effects(results):
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    for name, data in results.items():
        axes[0].plot(data['history']['accuracy'], label=name, linewidth=2)
    axes[0].set_title('Effect of Momentum on Training', fontweight='bold')
    axes[0].set_xlabel('Epoch'); axes[0].set_ylabel('Training Accuracy')
    axes[0].legend(); axes[0].grid(True, alpha=0.3)
    
    for name, data in results.items():
        axes[1].plot(data['history']['val_accuracy'], label=name, linewidth=2)
    axes[1].set_title('Effect of Momentum on Validation', fontweight='bold')
    axes[1].set_xlabel('Epoch'); axes[1].set_ylabel('Validation Accuracy')
    axes[1].legend(); axes[1].grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.savefig('momentum_analysis.png', dpi=150)
    plt.show()

results = compare_momentum_values()
visualize_momentum_effects(results)
```

## Expected Output

Training 5 SGD configurations (no momentum, momentum 0.5/0.9/0.99, Nesterov 0.9) on CIFAR-10 with a 2-conv CNN for 25 epochs at lr=0.01.

| SGD momentum=0.5 | 0.56–0.62 | 0.65–0.70 | Moderate | Smooth |

Nesterov with momentum=0.9 should achieve the highest val accuracy. momentum=0.99 will show visible oscillations in the loss curve.

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| momentum=0.99 accuracy is worse than momentum=0.9 at epoch 25 | High momentum accumulates a very large velocity vector; with lr=0.01 the effective step size overshoots minima causing persistent oscillation | Reduce lr to 0.001 when using momentum≥0.95, or add gradient clipping (`clipnorm=0.5`) to cap the velocity norm |
| No visible difference between momentum=0.9 and Nesterov=0.9 on training curves | 25 epochs with batch size 128 gives only ~9,765 steps; Nesterov's look-ahead advantage is most pronounced during rapid loss descent in early training | Plot per-step loss (not per-epoch) for the first 500 steps; Nesterov typically shows faster initial descent that blurs when averaged over an epoch |
| SGD (no momentum) loss curve is jagged despite low LR | Without momentum, each mini-batch gradient update is independent; noisy label imbalance in CIFAR-10 batches causes high variance | Increase batch size to 256 or apply gradient averaging over 2 mini-batches to smooth the no-momentum baseline |
| Momentum hyperparameter interaction with BatchNormalization statistics | BN running mean/variance are updated every step; high momentum SGD interacts with BN's `momentum` parameter (default 0.99), causing conflicting update rates | Explicitly set `tf.keras.layers.BatchNormalization(momentum=0.9)` to align BN decay with optimizer momentum, or remove BN to isolate the effect |
| Comparing momentum values without tuning LR per configuration | lr=0.01 is near-optimal for momentum=0.9 but suboptimal for momentum=0.5 and too large for momentum=0.99 | Run a grid search over (momentum, lr) pairs using `{0.5: 0.05, 0.9: 0.01, 0.99: 0.001}` to ensure a fair comparison |
