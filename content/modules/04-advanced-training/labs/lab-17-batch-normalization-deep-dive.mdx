# Batch Normalization Deep Dive

```python
"""
Lab 5.1: Batch Normalization Deep Dive
=======================================
Comprehensive analysis of batch normalization.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

tf.random.set_seed(42)

class CustomBatchNorm(tf.keras.layers.Layer):
    """
    Custom batch normalization implementation.
    Formula:
    - Training: y = gamma * (x - batch_mean) / sqrt(batch_var + epsilon) + beta
    - Inference: y = gamma * (x - moving_mean) / sqrt(moving_var + epsilon) + beta
    """
    
    def __init__(self, epsilon=1e-5, momentum=0.99, **kwargs):
        super().__init__(**kwargs)
        self.epsilon = epsilon
        self.momentum = momentum
        
    def build(self, input_shape):
        self.gamma = self.add_weight(name='gamma', shape=input_shape[-1:],
                                     initializer='ones', trainable=True)
        self.beta = self.add_weight(name='beta', shape=input_shape[-1:],
                                    initializer='zeros', trainable=True)
        self.moving_mean = self.add_weight(name='moving_mean', shape=input_shape[-1:],
                                           initializer='zeros', trainable=False)
        self.moving_var = self.add_weight(name='moving_var', shape=input_shape[-1:],
                                          initializer='ones', trainable=False)
        super().build(input_shape)
    
    def call(self, inputs, training=None):
        if training:
            batch_mean = tf.reduce_mean(inputs, axis=0, keepdims=True)
            batch_var = tf.reduce_variance(inputs, axis=0, keepdims=True)
            
            self.moving_mean.assign(self.momentum * self.moving_mean + (1 - self.momentum) * batch_mean)
            self.moving_var.assign(self.momentum * self.moving_var + (1 - self.momentum) * batch_var)
            
            normalized = (inputs - batch_mean) / tf.sqrt(batch_var + self.epsilon)
        else:
            normalized = (inputs - self.moving_mean) / tf.sqrt(self.moving_var + self.epsilon)
        
        return self.gamma * normalized + self.beta
    
    def get_config(self):
        config = super().get_config()
        config.update({'epsilon': self.epsilon, 'momentum': self.momentum})
        return config

def analyze_batch_size_effect():
    (x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
    x_train, x_test = x_train / 255.0, x_test / 255.0
    y_train = tf.keras.utils.to_categorical(y_train, 10)
    y_test = tf.keras.utils.to_categorical(y_test, 10)
    
    def create_model():
        return tf.keras.Sequential([
            tf.keras.layers.Conv2D(32, 3, padding='same', input_shape=(32, 32, 3)),
            tf.keras.layers.BatchNormalization(),
            tf.keras.layers.Activation('relu'),
            tf.keras.layers.MaxPooling2D(),
            tf.keras.layers.Conv2D(64, 3, padding='same'),
            tf.keras.layers.BatchNormalization(),
            tf.keras.layers.Activation('relu'),
            tf.keras.layers.MaxPooling2D(),
            tf.keras.layers.Flatten(),
            tf.keras.layers.Dense(128),
            tf.keras.layers.BatchNormalization(),
            tf.keras.layers.Activation('relu'),
            tf.keras.layers.Dense(10, activation='softmax')
        ])
    
    batch_sizes = [8, 16, 32, 64, 128, 256]
    results = {}
    
    for bs in batch_sizes:
        print(f"\nTraining with batch size: {bs}")
        
        model = create_model()
        model.compile(optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
                      loss='categorical_crossentropy', metrics=['accuracy'])
        
        history = model.fit(x_train, y_train, batch_size=bs, epochs=20,
                           validation_data=(x_test, y_test), verbose=0)
        
        test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
        results[bs] = {'history': history.history, 'test_accuracy': test_acc}
        print(f"  Test Accuracy: {test_acc:.4f}")
    
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    for bs, data in results.items():
        axes[0].plot(data['history']['val_accuracy'], label=f'BS={bs}', linewidth=2)
    axes[0].set_title('Validation Accuracy vs Batch Size', fontweight='bold')
    axes[0].set_xlabel('Epoch'); axes[0].set_ylabel('Accuracy')
    axes[0].legend(); axes[0].grid(True, alpha=0.3)
    
    batch_sizes_list = list(results.keys())
    test_accs = [results[bs]['test_accuracy'] for bs in batch_sizes_list]
    axes[1].plot(batch_sizes_list, test_accs, 'o-', linewidth=2, markersize=10)
    axes[1].set_title('Final Test Accuracy vs Batch Size', fontweight='bold')
    axes[1].set_xlabel('Batch Size'); axes[1].set_ylabel('Test Accuracy')
    axes[1].set_xscale('log', base=2); axes[1].grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.savefig('batch_size_effect.png', dpi=150)
    plt.show()
    
    return results

results = analyze_batch_size_effect()
```

## Expected Output

### 1. CustomBatchNorm — Shape Invariants

| Input shape | Output shape | Notes |
|---|---|---|
| `(batch, features)` | `(batch, features)` | Normalizes across axis=0 (batch axis) per feature |
| `(32, 128)` | `(32, 128)` | gamma/beta shapes: `(128,)` each |
| `(1, 128)` — inference | `(1, 128)` | Uses moving_mean / moving_var — no batch stats needed |

The `gamma` weight is initialized to **1.0** (identity scale) and `beta` to **0.0** (zero shift), so the very first forward pass outputs approximately zero-mean, unit-variance activations before learning adjusts them.

### 2. Moving Statistics Convergence

After training for several epochs on a large dataset, the `moving_mean` and `moving_var` buffers should converge close to the true per-feature statistics of the training set. With `momentum=0.99` (slow EMA), convergence takes more steps than PyTorch's default of `0.1`. Expect:

```
Epoch 1, batch 0:   moving_mean ≈ 0.0     (initial — dominated by zeros)
Epoch 1, batch 50:  moving_mean ≈ 0.003   (slowly updating)
Epoch 5, batch 390: moving_mean ≈ true μ  (≈ converged for most features)
```

> **Note:** Keras `BatchNormalization` uses `momentum=0.99` by default (slow update). PyTorch's `BatchNorm2d` uses `momentum=0.1` (fast update — effectively 10× more aggressive). If you port a PyTorch model to Keras and see stale moving stats, this is why.

### 3. Batch Size Comparison — Expected Accuracy Ranges (CIFAR-10, 20 epochs)

| Batch size | Expected val_accuracy (20 ep) | Typical train-val gap | Notes |
|---|---|---|---|
| 8 | 0.55 – 0.65 | wide (0.10–0.20) | Noisy BN stats degrade training |
| 16 | 0.60 – 0.68 | moderate | Still noisy; EMA updates are volatile |
| 32 | 0.65 – 0.72 | moderate | Sweet spot lower bound |
| 64 | 0.68 – 0.74 | small | Common default; good BN stability |
| 128 | 0.69 – 0.75 | small | Near-optimal for this architecture |
| 256 | 0.68 – 0.74 | very small | May slightly underfit; LR often needs scaling |

> **If your bs=8 accuracy is below 0.50:** the small-batch BN statistics are too noisy. Try replacing `BatchNormalization` with `LayerNormalization` for small-batch regimes. If bs=256 underperforms bs=128 significantly (>3%), consider the linear LR scaling rule: multiply LR by `bs_new / bs_reference`.

## Common Pitfalls

### 1. `training=None` Silently Runs in Inference Mode

The `call(inputs, training=None)` signature uses Python truthiness: `if training:` is `False` when `training` is `None`. This means if you call `layer(x)` without specifying the training flag (as happens in pure eager Python scripts outside Keras model training), the layer silently uses moving statistics — even during what you intend to be a training forward pass. This is particularly dangerous when you manually call `model(x)` in a custom training loop without `training=True`.

**Fix:** Always pass `training=True` in custom loops:
```python
# ❌ Uses inference stats silently
logits = model(x_batch)

# ✅ Uses batch stats correctly
logits = model(x_batch, training=True)
```

### 2. Forgetting to Freeze BN During Fine-Tuning

When you load a pretrained model and freeze its layers for fine-tuning, Keras's `model.trainable = False` does NOT stop `BatchNormalization` layers from updating their moving stats — it only stops gradient updates to gamma/beta. The moving stats continue updating via `assign` inside `call`. On a small fine-tuning dataset this corrupts the pretrained statistics and degrades accuracy.

**Fix:** Explicitly set BN layers to inference mode:
```python
for layer in base_model.layers:
    if isinstance(layer, tf.keras.layers.BatchNormalization):
        layer.trainable = False
# Then call with training=False during fine-tuning
```

### 3. Small Batch Sizes Degrade BN Quality

With `bs=8` on CIFAR-10 (3072 features per image), each mini-batch mean and variance is computed from only 8 samples — extremely noisy estimates. The resulting normalized activations have high variance, making optimization erratic. The `moving_mean` EMA also takes much longer to stabilize because each update incorporates noisy batch statistics.

**Diagnostic:** If val_accuracy at bs=8 is more than 10 percentage points below bs=64, it's a BN issue — not a learning rate issue.

### 4. Keras BN Momentum ≠ PyTorch BN Momentum

Keras `BatchNormalization(momentum=0.99)` means: `moving_mean = 0.99 * moving_mean + 0.01 * batch_mean`. PyTorch `BatchNorm2d(momentum=0.1)` means: `running_mean = 0.1 * batch_mean + 0.9 * running_mean`. Despite the reversed naming convention, PyTorch's default is effectively a **10× faster** EMA update. If you reproduce a PyTorch paper result in Keras and the moving stats converge slowly, set `momentum=0.9` (Keras) to match PyTorch's default behavior.

### 5. BN Placement — Before or After Activation?

The original BN paper (Ioffe & Szegedy, 2015) places BN **before** the activation: `Conv → BN → ReLU`. However, several follow-up studies find that **post-activation BN** (`Conv → ReLU → BN`) can work equally well or better for residual networks, because ReLU's non-negative output violates the Gaussian assumption that BN was designed for. The code in this lab uses the original pre-activation order (correct for the original paper). Switching to post-activation is a valid experiment but inconsistently supported across architectures.
