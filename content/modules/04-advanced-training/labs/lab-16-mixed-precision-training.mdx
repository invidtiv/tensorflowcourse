# Mixed Precision Training

```python
"""
Lab 4.4: Mixed Precision Training
==================================
Implement and benchmark mixed precision training.
"""

import tensorflow as tf
import numpy as np
import time
import matplotlib.pyplot as plt

print(f"TensorFlow version: {tf.__version__}")
print(f"Mixed precision available: {tf.keras.mixed_precision.global_policy().name}")

def setup_mixed_precision():
    policy = tf.keras.mixed_precision.Policy('mixed_float16')
    tf.keras.mixed_precision.set_global_policy(policy)
    print(f"Compute dtype: {policy.compute_dtype}")
    print(f"Variable dtype: {policy.variable_dtype}")

def reset_to_float32():
    policy = tf.keras.mixed_precision.Policy('float32')
    tf.keras.mixed_precision.set_global_policy(policy)

def create_model_with_dtype():
    inputs = tf.keras.layers.Input(shape=(32, 32, 3))
    
    x = tf.keras.layers.Conv2D(32, 3, activation='relu')(inputs)
    x = tf.keras.layers.MaxPooling2D()(x)
    x = tf.keras.layers.Conv2D(64, 3, activation='relu')(x)
    x = tf.keras.layers.MaxPooling2D()(x)
    x = tf.keras.layers.Conv2D(128, 3, activation='relu')(x)
    x = tf.keras.layers.MaxPooling2D()(x)
    x = tf.keras.layers.Flatten()(x)
    x = tf.keras.layers.Dense(256, activation='relu')(x)
    x = tf.keras.layers.Dropout(0.5)(x)
    
    outputs = tf.keras.layers.Dense(10, activation='softmax', dtype='float32')(x)
    return tf.keras.Model(inputs, outputs)

def benchmark_precision_mode(precision_mode, epochs=10):
    print(f"\n{'='*50}")
    print(f"Benchmarking: {precision_mode}")
    print(f"{'='*50}")
    
    if precision_mode == 'mixed_float16':
        setup_mixed_precision()
    else:
        reset_to_float32()
    
    (x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
    x_train, x_test = x_train / 255.0, x_test / 255.0
    y_train = tf.keras.utils.to_categorical(y_train, 10)
    y_test = tf.keras.utils.to_categorical(y_test, 10)
    
    model = create_model_with_dtype()
    
    if precision_mode == 'mixed_float16':
        optimizer = tf.keras.optimizers.Adam(learning_rate=0.001)
        optimizer = tf.keras.mixed_precision.LossScaleOptimizer(optimizer)
    else:
        optimizer = tf.keras.optimizers.Adam(learning_rate=0.001)
    
    model.compile(optimizer=optimizer, loss='categorical_crossentropy', metrics=['accuracy'])
    
    model.fit(x_train[:1000], y_train[:1000], epochs=1, verbose=0)  # Warmup
    
    start_time = time.time()
    history = model.fit(x_train, y_train, batch_size=128, epochs=epochs,
                       validation_data=(x_test, y_test), verbose=1)
    training_time = time.time() - start_time
    
    test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
    model_size = sum([tf.size(w).numpy() * w.dtype.size for w in model.weights])
    
    print(f"\nResults for {precision_mode}:")
    print(f"  Training Time: {training_time:.2f}s")
    print(f"  Test Accuracy: {test_acc:.4f}")
    print(f"  Model Size: {model_size / (1024**2):.2f} MB")
    
    return {
        'training_time': training_time,
        'test_accuracy': test_acc,
        'model_size_mb': model_size / (1024**2),
        'history': history.history
    }

def compare_mixed_precision():
    results = {}
    
    for mode in ['float32', 'mixed_float16']:
        try:
            results[mode] = benchmark_precision_mode(mode, epochs=10)
        except Exception as e:
            print(f"Error with {mode}: {e}")
            results[mode] = None
    
    if all(v is not None for v in results.values()):
        fig, axes = plt.subplots(1, 3, figsize=(15, 5))
        
        modes = list(results.keys())
        
        times = [results[m]['training_time'] for m in modes]
        axes[0].bar(modes, times, color=['skyblue', 'lightgreen'], edgecolor='black')
        axes[0].set_ylabel('Time (seconds)')
        axes[0].set_title('Training Time Comparison', fontweight='bold')
        for i, t in enumerate(times):
            axes[0].text(i, t + 5, f'{t:.1f}s', ha='center', fontweight='bold')
        
        accs = [results[m]['test_accuracy'] for m in modes]
        axes[1].bar(modes, accs, color=['skyblue', 'lightgreen'], edgecolor='black')
        axes[1].set_ylabel('Accuracy')
        axes[1].set_title('Test Accuracy Comparison', fontweight='bold')
        for i, a in enumerate(accs):
            axes[1].text(i, a + 0.01, f'{a:.3f}', ha='center', fontweight='bold')
        
        sizes = [results[m]['model_size_mb'] for m in modes]
        axes[2].bar(modes, sizes, color=['skyblue', 'lightgreen'], edgecolor='black')
        axes[2].set_ylabel('Size (MB)')
        axes[2].set_title('Model Size Comparison', fontweight='bold')
        for i, s in enumerate(sizes):
            axes[2].text(i, s + 0.5, f'{s:.1f}MB', ha='center', fontweight='bold')
        
        plt.tight_layout()
        plt.savefig('mixed_precision_comparison.png', dpi=150)
        plt.show()
        
        speedup = results['float32']['training_time'] / results['mixed_float16']['training_time']
        print(f"\nMixed Precision Speedup: {speedup:.2f}x")
    
    return results

results = compare_mixed_precision()
```

## Expected Output

The lab compares float32 vs mixed_float16 training on a CNN.

Expected results (example run on a T4/Turing GPU):
```
Policy          Train Acc   Val Acc   Epoch Time   Speedup          Peak Memory
float32         0.88        0.85      14.2 s       1.0×             1.8 GB
mixed_float16   0.88        0.85      9.1 s        1.56× (T4 obs.)  1.2 GB
```

The 1.56× figure above is a single observed T4 result. Actual speedup is hardware-dependent:
- T4 / Turing Tensor Cores: **1.3–1.6×**
- V100 / Volta: **1.5–1.9×**
- A100 / Ampere: **1.8–2.3×**
- Non-Tensor-Core GPUs (K80, P100, CPU): **~1.0–1.2×** (negligible or no speedup)

Signals:
- **Accuracy** should be nearly identical (< 0.5% difference)
- **Speedup** of 1.3–1.8× is expected on Volta/Turing Tensor Cores; negligible on older hardware without Tensor Cores
- On CPU or pre-Volta, speedup ≈ 1.0× or even slower
- **Loss scale** should start at 2^15 and stabilize
- If loss goes to NaN with mixed precision, the dynamic loss scaler is failing — usually means the initial scale is too high or gradients are genuinely huge

A healthy mixed-precision run on a **Volta / Ampere / Hopper** GPU (or a TPU for BF16) produces these canonical signals:

- **Throughput speedup (vs FP32 baseline, same batch size, same model):**
  - `mixed_float16` on V100 / T4 / RTX 20-series: **1.5×–2.0×** steps/sec
  - `mixed_float16` on A100 / H100: **2.0×–3.0×** steps/sec
  - `mixed_bfloat16` on A100 / H100 / TPU v3+: **2.0×–3.0×** steps/sec
  - **CPU or pre-Volta GPU: ~0.95×–1.05×** (no Tensor Cores → no real speedup; if you see a *slowdown* ≥ 10%, disable MP — the cast overhead is dominating).
- **Peak GPU memory:** **~25–40% lower** than FP32 for the same batch size, because activations and optimizer momentum are stored in FP16/BF16. Measured via `tf.config.experimental.get_memory_info('GPU:0')['peak']`. Use the freed memory to ~1.5–2× the batch size for additional throughput.
- **Accuracy delta:** final test accuracy for `mixed_float16` and `mixed_bfloat16` should be **within ±1–2%** of the FP32 baseline on CIFAR-10 / Fashion-MNIST. If you see a >2% drop, the cause is almost always a missing `dtype='float32'` on the output layer or a numerically unstable loss (see Pitfall 1).
- **Loss scale trajectory (FP16 only):** `LossScaleOptimizer`'s dynamic scale typically settles in the range **`2^14` to `2^16`** within the first 200 steps. If it keeps halving (bouncing off `1.0`) you have chronic overflow — reduce learning rate or check for exploding activations. If it stays pinned at the initial `2^15` without ever adjusting, gradients are too small and silent underflow is likely — log unscaled gradient histograms.
- **Dtype audit:** after `set_global_policy('mixed_float16')`, intermediate layer compute dtype is `float16` and variable dtype is `float32` (master weights). The **output Dense layer must report `dtype == float32`** or the softmax/cross-entropy path is unsafe.
- **Wall-clock (CIFAR-10, 3-conv CNN, batch 128, 10 epochs):** FP32 baseline ~**90–120 s** on T4; `mixed_float16` ~**50–70 s** on T4 / ~**30–45 s** on A100; `mixed_bfloat16` ~**30–45 s** on A100.

**Diagnostic tree if speedup is missing:**

1. **No speedup at all (≤ 1.05×):** verify `tf.config.list_physical_devices('GPU')` shows a Volta-or-newer card; check `nvidia-smi` for compute capability ≥ 7.0; confirm the batch size is large enough to saturate Tensor Cores (typically ≥ 64 per GPU).
2. **Speedup on training but not inference:** inference often runs at batch 1, where cast overhead dominates. Batch ≥ 32 at inference to see MP gains.
3. **NaN loss after 1–5 steps with `mixed_float16`:** missing `dtype='float32'` on the output layer, or a custom loss that computes in FP16. Wrap loss computation explicitly in `tf.cast(..., tf.float32)`.
4. **Accuracy drop > 3% vs FP32:** try `mixed_bfloat16` if available (no loss scaling needed); otherwise increase initial loss scale and verify no custom layers use unsafe FP16 ops (e.g. large reductions, log/exp on tiny values).

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| NaN loss within first few steps | Output Dense layer computes softmax in FP16; overflow/underflow corrupts gradients | Add `dtype='float32'` to the final Dense layer so Keras auto-casts before loss computation |
| No speedup on CPU or pre-Volta GPU | Tensor Cores that accelerate FP16 matrix multiply are only present from NVIDIA Volta (V100) onward | Check `tf.config.list_physical_devices('GPU')` and `nvidia-smi` compute capability; disable MP if < 7.0 |
| `LossScaleOptimizer` error in Keras 3 | Manual wrapping is no longer required or supported when a `mixed_float16` policy is active | Remove the manual `LossScaleOptimizer` wrap; the policy handles scaling automatically in TF ≥ 2.14 |
| Loss scale keeps halving to 1.0 | Chronic gradient overflow means the dynamic scaler cannot find a stable scale | Lower the learning rate, add gradient clipping (`clipnorm=1.0`), or switch to `mixed_bfloat16` which needs no scaling |
| Intermediate variables stored in FP32 unexpectedly | A custom layer explicitly passes `dtype='float32'` to `add_weight`, defeating FP16 activation savings | Omit explicit dtype in `add_weight` for activation tensors; only force FP32 for the final output and running statistics |

<Callout type="warning" title="Common Pitfalls">

**Pitfall 1: Forgetting `dtype='float32'` on the output layer.**
When mixed precision is active, all layers default to float16. The final Dense layer + softmax computed in FP16 is numerically unstable — softmax can overflow/underflow, producing NaN loss. Always explicitly set `dtype='float32'` on the last Dense (or any loss-sensitive) layer. Keras will handle the automatic up-cast.

**Pitfall 2: `LossScaleOptimizer` is deprecated in Keras 3.**
In TF 2.x / Keras 3, wrapping the optimizer manually with `tf.keras.mixed_precision.LossScaleOptimizer(optimizer)` is no longer needed when using `Policy('mixed_float16')` — the policy handles loss scaling automatically. Manual wrapping in Keras 3 raises a warning or error. Check your TF version: if `tf.__version__ >= '2.14'`, rely on the policy alone.

**Pitfall 3: Enabling mixed precision on CPU or non-Tensor-Core GPUs.**
Mixed precision provides no speedup (and adds overhead) on CPU or on older NVIDIA GPUs (pre-Volta, i.e., before GTX 1080 Ti). Tensor Core acceleration requires at least Volta (V100) for FP16, Ampere (A100) for BF16. Always check `tf.config.list_physical_devices('GPU')` and confirm your hardware before enabling MP in production.

**Pitfall 4: Using `mixed_float16` when BF16 hardware is available.**
If you have an A100, H100, or a TPU, prefer `Policy('mixed_bfloat16')` — BF16 has the same exponent range as FP32 so it avoids underflow without any loss scaling. `mixed_float16` on these devices is not wrong but is unnecessarily complex.
