"""
Lab 10.7: TensorFlow Serving Setup
==================================
Complete guide to setting up TensorFlow Serving for model deployment.
"""

import tensorflow as tf
import numpy as np
import os
import subprocess
import json

print("=" * 60)
print("Lab 10.7: TensorFlow Serving Setup")
print("=" * 60)

# ============================================================
# PART 1: Create and Export a Model
# ============================================================

print("\n" + "=" * 60)
print("PART 1: Creating and Exporting a Model")
print("=" * 60)

# Create a simple model
def create_servable_model():
    """Create a model for serving."""
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(128, activation='relu', input_shape=(10,)),
        tf.keras.layers.Dropout(0.2),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(5, activation='softmax', name='predictions')
    ])
    return model

model = create_servable_model()
model.compile(optimizer='adam',
              loss='categorical_crossentropy',
              metrics=['accuracy'])

# Train briefly
x_dummy = np.random.random((1000, 10))
y_dummy = tf.keras.utils.to_categorical(np.random.randint(0, 5, 1000), 5)

print("Training model...")
model.fit(x_dummy, y_dummy, epochs=5, verbose=0)

# ============================================================
# PART 2: Export Model for Serving
# ============================================================

print("\n" + "=" * 60)
print("PART 2: Exporting Model for Serving")
print("=" * 60)

"""
TensorFlow Serving requires models in SavedModel format with specific structure:

models/
└── <model_name>/
    └── <version>/
        ├── saved_model.pb
        └── variables/
            ├── variables.data-00000-of-00001
            └── variables.index
"""

# Create model directory structure
MODEL_NAME = "my_model"
VERSION = "1"

model_base_path = f"models/{MODEL_NAME}"
version_path = f"{model_base_path}/{VERSION}"

# Create directories
os.makedirs(version_path, exist_ok=True)

# Save model in SavedModel format
tf.saved_model.save(model, version_path)

print(f"Model exported to: {version_path}")
print(f"Directory structure:")
for root, dirs, files in os.walk(model_base_path):
    level = root.replace(model_base_path, '').count(os.sep)
    indent = ' ' * 2 * level
    print(f'{indent}{os.path.basename(root)}/')
    subindent = ' ' * 2 * (level + 1)
    for file in files:
        print(f'{subindent}{file}')

# ============================================================
# PART 3: Model Configuration
# ============================================================

print("\n" + "=" * 60)
print("PART 3: Model Configuration")
print("=" * 60)

# Create model configuration file
model_config = {
    "model_config_list": [
        {
            "name": "my_model",
            "base_path": "/models/my_model",
            "model_platform": "tensorflow",
            "model_version_policy": {
                "all": {}
            }
        }
    ]
}

# Save configuration
config_path = "models/models.config"
with open(config_path, 'w') as f:
    json.dump(model_config, f, indent=2)

print(f"Model config saved to: {config_path}")
print("\nConfiguration content:")
print(json.dumps(model_config, indent=2))

# ============================================================
# PART 4: TensorFlow Serving Startup Commands
# ============================================================

print("\n" + "=" * 60)
print("PART 4: TensorFlow Serving Startup")
print("=" * 60)

print("""
# Option 1: Using Docker (Recommended)
# ====================================

docker run -p 8501:8501 \
  --mount type=bind,source=$(pwd)/models,target=/models \
  --mount type=bind,source=$(pwd)/models/models.config,target=/models/models.config \
  -e MODEL_NAME=my_model \
  -t tensorflow/serving

# With model configuration file:
docker run -p 8501:8501 \
  --mount type=bind,source=$(pwd)/models,target=/models \
  -t tensorflow/serving \
  --model_config_file=/models/models.config

# Option 2: Local Installation
# =============================

# Install TensorFlow Serving
# (Ubuntu/Debian)
echo "deb [arch=amd64] http://storage.googleapis.com/tensorflow-serving-apt stable tensorflow-model-server tensorflow-model-server-universal" | sudo tee /etc/apt/sources.list.d/tensorflow-serving.list
curl https://storage.googleapis.com/tensorflow-serving-apt/tensorflow-serving.release.pub.gpg | sudo apt-key add -
sudo apt-get update
sudo apt-get install tensorflow-model-server

# Start TensorFlow Serving
tensorflow_model_server \
  --model_name=my_model \
  --model_base_path=$(pwd)/models/my_model \
  --rest_api_port=8501

# Option 3: Using Docker Compose
# ===============================
""")

# Create docker-compose.yml
docker_compose = """version: '3'
services:
  tensorflow-serving:
    image: tensorflow/serving
    ports:
      - "8501:8501"
    volumes:
      - ./models:/models
    environment:
      - MODEL_NAME=my_model
    command: tensorflow_model_server --model_name=my_model --model_base_path=/models/my_model --rest_api_port=8501
"""

with open('docker-compose.yml', 'w') as f:
    f.write(docker_compose)

print("Docker Compose file created: docker-compose.yml")

# ============================================================
# PART 5: Model Versioning
# ============================================================

print("\n" + "=" * 60)
print("PART 5: Model Versioning")
print("=" * 60)

# Export version 2 (simulated improvement)
print("\nCreating version 2 of the model...")

# Create slightly different model

model_v2.compile(optimizer='adam',
                 loss='categorical_crossentropy',
                 metrics=['accuracy'])

# Export version 2
version_2_path = f"{model_base_path}/2"

print(f"Version 2 exported to: {version_2_path}")

# Show all versions
print("\nAvailable model versions:")
versions = sorted([d for d in os.listdir(model_base_path) 
                   if os.path.isdir(os.path.join(model_base_path, d))])
for v in versions:
    print(f"  - Version {v}")

# ============================================================

# Load and inspect SavedModel
loaded = tf.saved_model.load(version_path)

# Get serving signature
infer = loaded.signatures['serving_default']

print(infer)

print("\nInput specifications:")
for input_name, input_spec in infer.structured_input_signature[1].items():
    print(f"  {input_name}: {input_spec}")

print("\nOutput specifications:")
for output_name, output_spec in infer.structured_outputs.items():
    print(f"  {output_name}: {output_spec}")

## Expected Output

```
Saving model v1 to models/my_model/1/ ... ok
Saving model v2 to models/my_model/2/ ... ok
Starting TF Serving (docker run -p 8501:8501 -v $(pwd)/models:/models ...) ... ok
GET http://localhost:8501/v1/models/my_model → {"model_version_status":[{"version":"2","state":"AVAILABLE"}]}
POST http://localhost:8501/v1/models/my_model:predict → 200 OK (latency ≈ 8–14 ms)
```

Key observation: after both versions are written, TF Serving's default
`Latest` version policy auto-switches to version 2 within a few seconds
without a server restart — this is the whole point of the versioned
directory layout, and it is visible in the server logs as a
`Successfully loaded servable version` line.

<Callout type="warning" title="Common Pitfalls">

1. **Wrong directory layout.** TF Serving expects
   `models/<model_name>/<integer_version>/saved_model.pb`. A nested
   `models/my_model/v1/` or `models/my_model/1.0/` is silently ignored and
   the model shows as "not found" even though the file is on disk.
2. **Forgetting `MODEL_NAME` / `MODEL_BASE_PATH` env vars** when running the
   Docker image — the server starts but serves nothing, and the health
   endpoint returns an empty model list.
3. **Port collision on 8501.** Other tools frequently grab 8501; the
   container starts but connections silently route elsewhere. Always verify
   with `docker ps` and `curl`.
4. **Model config version-policy confusion.** `Latest { num_versions: 2 }`
   keeps the two newest versions loaded simultaneously; `Specific` pins an
   exact version. Using `Latest` when you meant `Specific` can cause
   surprising rollouts after any retraining job.
5. **REST vs gRPC port mismatch.** REST defaults to 8501, gRPC to 8500.
   Forgetting to publish 8500 while developing a gRPC client looks like a
   handshake failure rather than a port-not-published error.
6. **Saving to a `path` that already contains files from a previous run.**
   TF Serving will load a mix of old and new assets silently; always remove
   the target directory before re-exporting.

</Callout>

</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|---------------|-----|
| Wrong versioned directory layout | TF Serving expects `models/<model_name>/<integer_version>/saved_model.pb`; paths like `v1/` or `1.0/` are silently ignored and the model shows as "not found" | Use only integer version numbers as directory names: `models/my_model/1/`, `models/my_model/2/` |
| Missing `MODEL_NAME` / `MODEL_BASE_PATH` env vars in Docker | The container starts without error but serves nothing; the `/v1/models` health endpoint returns an empty list | Always pass `-e MODEL_NAME=my_model -e MODEL_BASE_PATH=/models` when running the TF Serving Docker image |
| Port 8501 already in use | Other local services frequently bind 8501; the Docker container starts but `curl` silently routes elsewhere with no obvious error | Verify with `docker ps` and `curl http://localhost:8501/v1/models/my_model` before any client code runs |
| `Latest` version policy causing unintended rollouts | `Latest { num_versions: 2 }` auto-loads every newly written version directory; a failed training run that writes a version directory triggers a silent rollout | Use `Specific` version policy in `model_config_list` for production; reserve `Latest` only for staging environments |
| Saving to a directory that still contains a previous export | TF Serving loads a mixture of old and new assets without warning, causing random prediction behavior that is extremely hard to debug | Always delete or version-stamp the target SavedModel directory before re-exporting; use `shutil.rmtree` in the export script |

## Stretch Goals

1. **Canary rollout via `ModelConfig`** — load v1 and v2 simultaneously with a `Specific` policy, then route 10% of traffic to v2 from a thin nginx or Envoy reverse proxy; compare per-version latency and error rates side-by-side.
2. **Warmup requests** — ship a `warmup_requests` SavedModel asset so the first production prediction does not pay the lazy-initialization latency cost; verify by measuring first-request latency before and after adding the warmup file.
3. **Prometheus + Grafana monitoring** — enable `--monitoring_config_file` with `prometheus_config { enable: true }` and scrape metrics into a local Grafana dashboard; create an alert on `request_latency_p99 > 50ms`.
4. **REST vs gRPC latency benchmark** — send 1000 identical predict requests over both transports and measure p50/p95/p99 latency; gRPC should outperform REST by 2–5x on small input payloads due to protobuf serialization efficiency.
