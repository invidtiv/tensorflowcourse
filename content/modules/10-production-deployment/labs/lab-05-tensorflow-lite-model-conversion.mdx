```python
"""
Lab 10.4: TensorFlow Lite Model Conversion
==========================================
Comprehensive guide to converting various model types
to TensorFlow Lite format with best practices.
"""

import tensorflow as tf
import numpy as np
import os

print("=" * 60)
print("Lab 10.4: TensorFlow Lite Model Conversion")
print("=" * 60)

# ============================================================
# PART 1: Basic Model Conversion
# ============================================================

print("\n" + "=" * 60)
print("PART 1: Basic Model Conversion")
print("=" * 60)

# Create a simple model
model = tf.keras.Sequential([
    tf.keras.layers.Dense(128, activation='relu', input_shape=(784,)),
    tf.keras.layers.Dropout(0.2),
    tf.keras.layers.Dense(64, activation='relu'),
    tf.keras.layers.Dense(10, activation='softmax')
])

model.compile(optimizer='adam',
              loss='sparse_categorical_crossentropy',
              metrics=['accuracy'])

# Train briefly
(x_train, y_train), _ = tf.keras.datasets.mnist.load_data()
x_train = x_train[:1000].reshape(-1, 784).astype('float32') / 255.0
y_train = y_train[:1000]

model.fit(x_train, y_train, epochs=2, verbose=0)

# Method 1: Convert from Keras model directly
print("\nMethod 1: From Keras Model")
converter = tf.lite.TFLiteConverter.from_keras_model(model)
tflite_model = converter.convert()

with open('model_from_keras.tflite', 'wb') as f:
    f.write(tflite_model)
print(f"Saved: model_from_keras.tflite ({len(tflite_model)/1024:.2f} KB)")

# Method 2: Convert from SavedModel format
print("\nMethod 2: From SavedModel")
model.save('saved_model_dir')
converter = tf.lite.TFLiteConverter.from_saved_model('saved_model_dir')
tflite_model = converter.convert()

with open('model_from_savedmodel.tflite', 'wb') as f:
    f.write(tflite_model)
print(f"Saved: model_from_savedmodel.tflite ({len(tflite_model)/1024:.2f} KB)")

# ============================================================
# PART 2: Converter Configuration Options
# ============================================================

print("\n" + "=" * 60)
print("PART 2: Converter Configuration Options")
print("=" * 60)

"""
Key Converter Options:
- optimizations: Enable optimizations (quantization)
- representative_dataset: For full integer quantization
- target_spec.supported_ops: Limit supported operations
- target_spec.supported_types: Enable specific types (float16)
- inference_input_type/output_type: Set I/O types
"""

# Option 1: Default optimization
print("\n1. Default Optimization")
converter = tf.lite.TFLiteConverter.from_keras_model(model)
converter.optimizations = [tf.lite.Optimize.DEFAULT]
tflite_optimized = converter.convert()

with open('model_optimized.tflite', 'wb') as f:
    f.write(tflite_optimized)
print(f"Optimized model size: {len(tflite_optimized)/1024:.2f} KB")

# Option 2: Allow custom ops
print("\n2. Allow Custom Operations")
converter = tf.lite.TFLiteConverter.from_keras_model(model)
converter.allow_custom_ops = True  # For unsupported operations
tflite_custom = converter.convert()

# Option 3: Select TF ops
print("\n3. Select TensorFlow Operations")
converter = tf.lite.TFLiteConverter.from_keras_model(model)
converter.target_spec.supported_ops = [
    tf.lite.OpsSet.TFLITE_BUILTINS,  # Default TFLite ops
    tf.lite.OpsSet.SELECT_TF_OPS     # Select TF ops for compatibility
]
tflite_select = converter.convert()

with open('model_select_ops.tflite', 'wb') as f:
    f.write(tflite_select)
print(f"Select ops model size: {len(tflite_select)/1024:.2f} KB")

# ============================================================
# PART 3: Handling Complex Models
# ============================================================

print("\n" + "=" * 60)
print("PART 3: Handling Complex Models")
print("=" * 60)

# Model with multiple inputs
print("\n1. Multi-Input Model")

input_a = tf.keras.Input(shape=(32,), name='input_a')
input_b = tf.keras.Input(shape=(16,), name='input_b')

x = tf.keras.layers.Dense(64, activation='relu')(input_a)
y = tf.keras.layers.Dense(32, activation='relu')(input_b)

combined = tf.keras.layers.concatenate([x, y])
z = tf.keras.layers.Dense(10, activation='softmax')(combined)

multi_input_model = tf.keras.Model(inputs=[input_a, input_b], outputs=z)

# Convert multi-input model
converter = tf.lite.TFLiteConverter.from_keras_model(multi_input_model)
tflite_multi = converter.convert()

with open('model_multi_input.tflite', 'wb') as f:
    f.write(tflite_multi)
print(f"Multi-input model converted: {len(tflite_multi)/1024:.2f} KB")

# Model with multiple outputs
print("\n2. Multi-Output Model")

inputs = tf.keras.Input(shape=(64,))
x = tf.keras.layers.Dense(128, activation='relu')(inputs)

output_1 = tf.keras.layers.Dense(10, activation='softmax', name='class')(x)
output_2 = tf.keras.layers.Dense(1, activation='sigmoid', name='confidence')(x)

multi_output_model = tf.keras.Model(inputs=inputs, outputs=[output_1, output_2])

converter = tf.lite.TFLiteConverter.from_keras_model(multi_output_model)
tflite_multi_out = converter.convert()

with open('model_multi_output.tflite', 'wb') as f:
    f.write(tflite_multi_out)
print(f"Multi-output model converted: {len(tflite_multi_out)/1024:.2f} KB")

# ============================================================
# PART 4: Validating Converted Models
# ============================================================

print("\n" + "=" * 60)
print("PART 4: Validating Converted Models")
print("=" * 60)

def validate_tflite_model(tflite_path, test_input):
    """Validate TFLite model by running inference."""
    
    # Load interpreter
    interpreter = tf.lite.Interpreter(model_path=tflite_path)
    interpreter.allocate_tensors()
    
    # Get I/O details
    input_details = interpreter.get_input_details()
    output_details = interpreter.get_output_details()
    
    print(f"\nModel: {tflite_path}")
    print(f"Input shape: {input_details[0]['shape']}")
    print(f"Input dtype: {input_details[0]['dtype']}")
    print(f"Output shape: {output_details[0]['shape']}")
    print(f"Output dtype: {output_details[0]['dtype']}")
    
    # Run inference
    interpreter.set_tensor(input_details[0]['index'], test_input)
    interpreter.invoke()
    output = interpreter.get_tensor(output_details[0]['index'])
    
    print(f"Output shape: {output.shape}")
    print(f"Output sum (should be ~1.0 for softmax): {np.sum(output):.4f}")
    
    return output

# Validate simple model
test_input = np.random.random((1, 784)).astype(np.float32)
output = validate_tflite_model('model_from_keras.tflite', test_input)

# Validate multi-input model
print("\nValidating multi-input model...")
interpreter = tf.lite.Interpreter(model_path='model_multi_input.tflite')
interpreter.allocate_tensors()

input_details = interpreter.get_input_details()
print(f"Number of inputs: {len(input_details)}")
for i, detail in enumerate(input_details):
    print(f"  Input {i}: {detail['name']} - {detail['shape']}")

interpreter.set_tensor(input_details[1]['index'], test_input_b)
interpreter.invoke()

output_details = interpreter.get_output_details()
output = interpreter.get_tensor(output_details[0]['index'])
print(f"Output shape: {output.shape}")

def analyze_tflite_model(tflite_path):
    """Analyze TFLite model structure."""
    interpreter = tf.lite.Interpreter(model_path=tflite_path)
    interpreter.allocate_tensors()
    
    # Get tensor details
    tensor_details = interpreter.get_tensor_details()
    
    print(f"\nModel Analysis: {tflite_path}")
    print(f"Total tensors: {len(tensor_details)}")
    
    # Count by type
    type_counts = {}
    for detail in tensor_details:
        dtype = detail['dtype']
        type_counts[dtype] = type_counts.get(dtype, 0) + 1
    
    print("Tensor types:")
    for dtype, count in type_counts.items():
        print(f"  {dtype}: {count}")
    
    # Get op details
    op_details = interpreter._get_ops_details()
    print(f"Total operations: {len(op_details)}")
    
    # Count op types
    op_types = {}
    for op in op_details:
        op_type = op['op_name']
        op_types[op_type] = op_types.get(op_type, 0) + 1
    
    print("Operation types:")
    for op_type, count in list(op_types.items())[:10]:  # Top 10
        print(f"  {op_type}: {count}")

analyze_tflite_model('model_from_keras.tflite')
```

<Callout type="info" title="Expected Output">

After running all the conversion pipelines in this lab you should observe:

- **Baseline Keras `.tflite` size:** roughly **4×** smaller than the parent `.h5` / `SavedModel` directory. Small MLPs land in the tens of KB; a MobileNetV2-class model lands around **~9–14 MB** pre-optimization.
- **Dynamic-range quantization:** another **~3.8–4×** size drop versus the float32 `.tflite`, with typically **<1 %** top-1 accuracy loss on classification tasks.
- **Full-integer (INT8) quantization:** same size drop as dynamic-range but **2–4× faster** on ARM CPUs and **required** for Edge TPU / Hexagon DSP deployment.
- **Float16 quantization:** exactly **2×** size drop, essentially **zero** accuracy loss, and measurably faster on mobile GPU delegates.
- **Inference parity check:** the mean absolute difference between Keras and TFLite float outputs on the same input batch should be **< 1e-5**. Anything larger indicates either a representative-dataset bug or an unsupported op silently falling back to CPU.

If any of the above is off, work through this diagnostic tree:

- **TFLite size is identical to the Keras file →** optimization flags were never attached. Check that `converter.optimizations = [tf.lite.Optimize.DEFAULT]` is set *before* `converter.convert()`.
- **Accuracy drop > 5 % after INT8 quantization →** representative dataset too small (fewer than ~100 batches) or not representative of the real input distribution. Rebuild it from held-out training data.
- **Conversion fails with `Some ops are not supported`  →** either the op needs `converter.target_spec.supported_ops = [TFLITE_BUILTINS, SELECT_TF_OPS]` (then accept the runtime-size cost), or you need to refactor the layer (e.g., replace `tf.strided_slice` with a `Lambda` wrapper).
- **`Invalid argument: tensorflow/lite/... NOT_EQUAL`  at interpreter load →** the `.tflite` was serialized against a newer TFLite runtime than the device has. Pin the converter and the target runtime to the same minor version.

</Callout>

<Callout type="warning" title="Common Pitfalls">

1. **Forgetting `tf.function` input signatures.** When converting from a Keras model that uses `@tf.function(input_signature=...)`, omitting the signature lets TFLite pick `None` batch dims and bake a non-reusable shape into the flatbuffer. Always supply a concrete `TensorSpec` for each input.
2. **Using the training-mode model for conversion.** Call `model.trainable = False` (or rebuild from the saved artifact) before handing it to the converter; BatchNorm moving statistics otherwise get captured in training mode and cause divergence at inference time.
3. **Representative dataset from a generator that ends after one pass.** Once exhausted the converter silently proceeds with fewer samples than you think, yielding unstable calibration. Wrap the generator with `itertools.cycle` or materialize a list of sample arrays.
4. **INT8 quantization of a model with a softmax *inside* the graph.** The post-softmax range [0, 1] confuses min/max calibration; either move the softmax to the consumer or use per-channel quantization on the final dense layer.
5. **`converter.inference_input_type = tf.uint8` without matching dequant on device.** When the device pipeline still feeds float32 RGB in [0, 1], the model receives raw bytes and predictions degrade silently. Keep input/output types as float32 unless the mobile/Edge TPU pipeline explicitly handles the quantized I/O.
6. **Using `SELECT_TF_OPS` without sizing the binary.** Each TF op pulled in inflates the TFLite runtime by several MB on device. Audit the final op list with the TFLite Model Analyzer before shipping — you may be able to rewrite one offending op rather than link the full Flex delegate.
7. **Not cleaning up intermediate `.tflite` artifacts on repeat runs.** Subsequent benchmark cells will happily read a stale file (especially after a crashed quantization) and report misleading numbers. Always `os.remove()` before re-emitting, or write to a run-stamped directory.
8. **Evaluating accuracy on training data.** The conversion pipeline is tempting to smoke-test against a handful of training examples, which hides quantization-induced regressions. Always reserve a held-out eval batch and compare Keras ↔ TFLite top-k on it.
9. **Ignoring delegate availability.** `Interpreter(model_path)` falls back to the CPU delegate by default. To benchmark the device path, construct with `tf.lite.Interpreter(model_path, experimental_delegates=[...])` and assert the delegate actually loaded — otherwise you are measuring CPU, not NNAPI / GPU / Hexagon.

</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|---------------|-----|
| `tf.function` input signature missing batch dimension | Without a concrete `TensorSpec`, the converter infers a dynamic batch but bakes `None` dims that some edge runtimes reject | Always specify `input_signature=[tf.TensorSpec(shape=[1, H, W, C], dtype=tf.float32)]` for single-image inference |
| Model converted from training mode (BatchNorm uses batch stats) | `model.trainable` defaults to `True`; training-mode BN captures batch statistics instead of moving averages, causing divergent outputs after conversion | Set `model.trainable = False` or rebuild from a SavedModel artifact before passing to `TFLiteConverter` |
| Representative dataset generator exhausted after one pass | `TFLiteConverter` calls the generator once; a generator that exhausts silently produces fewer calibration batches than expected | Wrap the generator in `itertools.cycle()` or yield from a pre-materialized list |
| `SELECT_TF_OPS` included without accounting for binary size | Each additional TF op pulled in inflates the on-device runtime by several MB | Audit the final op list with `tf.lite.experimental.Analyzer.analyze()` and refactor the offending op before shipping |
| Evaluating conversion quality on training data | Quantization regressions hide on training data due to overfitting; the held-out set is the only honest signal | Always reserve a separate eval batch and compare Keras vs. TFLite top-k accuracy on it |
