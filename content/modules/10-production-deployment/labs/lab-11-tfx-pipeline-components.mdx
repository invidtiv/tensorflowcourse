"""
Transform:
- Performs feature engineering
- Applies transformations consistently to train/eval data
- Saves transformation graph for serving
"""

from tfx.components import Transform

# Define preprocessing function
_transform_module_file = os.path.join(PIPELINE_ROOT, 'transform_module.py')

_transform_module_content = '''
import tensorflow as tf
import tensorflow_transform as tft

# Feature constants
NUMERIC_FEATURES = ['feature_1', 'feature_2']
CATEGORICAL_FEATURES = ['feature_3', 'category']
LABEL_KEY = 'target'

def preprocessing_fn(inputs):
    """Preprocess input features into transformed features."""
    outputs = {}
    
    # Scale numeric features
    for key in NUMERIC_FEATURES:
        # Fill missing values with mean
        filled = tft.sparse_tensor_to_dense_with_missing(inputs[key], default_value=0.0)
        # Z-score normalization
        outputs[key + '_scaled'] = tft.scale_to_z_score(filled)
    
    # Convert categorical features to integers
    for key in CATEGORICAL_FEATURES:
        outputs[key + '_integerized'] = tft.compute_and_apply_vocabulary(inputs[key])
    
    # Keep label as is
    outputs[LABEL_KEY] = inputs[LABEL_KEY]
    
    return outputs
'''

with open(_transform_module_file, 'w') as f:
    f.write(_transform_module_content)

"""
Trainer:
- Trains the TensorFlow model
- Uses transformed data from Transform component
- Outputs trained model for evaluation

from tfx.components import Trainer
from tfx.proto import trainer_pb2

# Define training module
_trainer_module_file = os.path.join(PIPELINE_ROOT, 'trainer_module.py')

_trainer_module_content = '''
import tensorflow as tf
from tfx import v1 as tfx
from tfx_bsl.public import tfxio
from tensorflow_transform import TFTransformOutput

# Feature constants
NUMERIC_FEATURES = ['feature_1_scaled', 'feature_2_scaled']
CATEGORICAL_FEATURES = ['feature_3_integerized', 'category_integerized']
LABEL_KEY = 'target'

def _input_fn(file_pattern, data_accessor, tf_transform_output, batch_size=128):
    """Generate features and labels for training."""
    dataset = data_accessor.tf_dataset_factory(
        file_pattern,
        tfxio.TensorFlowDatasetOptions(batch_size=batch_size),
        tf_transform_output.transformed_metadata.schema
    )
    
    def parse_example(example):
        features = {}
        for key in NUMERIC_FEATURES + CATEGORICAL_FEATURES:
            features[key] = example[key]
        label = example[LABEL_KEY]
        return features, label
    
    return dataset.map(parse_example)

def _build_keras_model():
    """Build a simple Keras model."""
    inputs = {}
    for key in NUMERIC_FEATURES:
        inputs[key] = tf.keras.Input(shape=(1,), name=key)
    for key in CATEGORICAL_FEATURES:
        inputs[key] = tf.keras.Input(shape=(1,), name=key, dtype=tf.int64)
    
    # Concatenate all features
    concatenated = tf.keras.layers.concatenate([
        tf.keras.layers.Dense(8, activation='relu')(inputs[key])
        for key in list(inputs.keys())
    ])
    
    x = tf.keras.layers.Dense(32, activation='relu')(concatenated)
    x = tf.keras.layers.Dropout(0.3)(x)
    outputs = tf.keras.layers.Dense(1, activation='sigmoid')(x)
    
    model = tf.keras.Model(inputs=inputs, outputs=outputs)
    
    model.compile(
        optimizer='adam',
        loss='binary_crossentropy',
        metrics=['accuracy']
    )
    
    return model

def run_fn(fn_args):
    """Train the model."""
    tf_transform_output = TFTransformOutput(fn_args.transform_output)
    
    train_dataset = _input_fn(
        fn_args.train_files,
        fn_args.data_accessor,
        tf_transform_output,
        batch_size=128
    )
    
    eval_dataset = _input_fn(
        fn_args.eval_files,
        fn_args.data_accessor,
        tf_transform_output,
        batch_size=128
    )
    
    model = _build_keras_model()
    
    model.fit(
        train_dataset,
        validation_data=eval_dataset,
        epochs=5,
        steps_per_epoch=100,
        validation_steps=20
    )
    
    # Save model
    model.save(fn_args.serving_model_dir, save_format='tf')
'''

with open(_trainer_module_file, 'w') as f:
