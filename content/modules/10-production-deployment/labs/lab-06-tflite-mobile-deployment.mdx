# Create a lightweight CNN for mobile
def create_mobile_classifier():
    """Create a mobile-optimized image classifier."""
    model = tf.keras.Sequential([
        # Mobile-friendly: depthwise separable convolutions
        tf.keras.layers.SeparableConv2D(32, 3, activation='relu', 
                                        input_shape=(224, 224, 3)),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.MaxPooling2D(2),
        
        tf.keras.layers.SeparableConv2D(64, 3, activation='relu'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.MaxPooling2D(2),
        
        tf.keras.layers.SeparableConv2D(128, 3, activation='relu'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.GlobalAveragePooling2D(),
        
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.3),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    return model

class TFLiteMobileClassifier:
    """
    Mobile-ready TFLite classifier wrapper.
    
    This class handles:
    - Model loading and initialization
    - Image preprocessing
    - Inference execution
    - Result postprocessing
    """
    
    def __init__(self, model_path, num_threads=4):
        """
        Initialize the classifier.
        
        Args:
            model_path: Path to TFLite model
            num_threads: Number of threads for inference (mobile optimization)
        """
        self.interpreter = tf.lite.Interpreter(
            model_path=model_path,
            num_threads=num_threads  # Optimize for mobile CPU
        )
        self.interpreter.allocate_tensors()
        
        self.input_details = self.interpreter.get_input_details()
        self.output_details = self.interpreter.get_output_details()
        
        # Get input specifications
        self.input_shape = self.input_details[0]['shape']
        self.input_dtype = self.input_details[0]['dtype']
        self.input_quantization = self.input_details[0].get('quantization_parameters')
        
        print(f"Model loaded successfully")
        print(f"Input shape: {self.input_shape}")
        print(f"Input dtype: {self.input_dtype}")
        
    def preprocess_image(self, image, target_size=(224, 224)):
        """
        Preprocess image for inference.
        
        Args:
            image: PIL Image, numpy array, or file path
            target_size: Target size (height, width)
        
        Returns:
            Preprocessed image array
        """
        # Load image if path provided
        if isinstance(image, str):
            image = Image.open(image)
        
        # Convert to numpy if PIL Image
        if isinstance(image, Image.Image):
            image = np.array(image)
        
        # Resize
        image = cv2.resize(image, target_size)
        
        # Convert to RGB if needed
        if len(image.shape) == 2:
            image = cv2.cvtColor(image, cv2.COLOR_GRAY2RGB)
        elif image.shape[2] == 4:
            image = cv2.cvtColor(image, cv2.COLOR_RGBA2RGB)
        
        # Normalize
        image = image.astype(np.float32) / 255.0
        
        # Add batch dimension
        image = np.expand_dims(image, axis=0)
        
        # Quantize if needed (for uint8 models)
        if self.input_dtype == np.uint8:
            if self.input_quantization:
                scale, zero_point = self.input_quantization['scales'][0], \
                                     self.input_quantization['zero_points'][0]
                image = (image / scale + zero_point).astype(np.uint8)
            else:
                image = (image * 255).astype(np.uint8)
        
        return image
    
    def predict(self, image):
        """
        Run inference on an image.
        
        Args:
            image: Input image
        
        Returns:
            Dictionary with predictions
        """
        # Preprocess
        processed = self.preprocess_image(image)
        
        # Set input tensor
        self.interpreter.set_tensor(
            self.input_details[0]['index'], 
            processed
        )
        
        # Run inference
        self.interpreter.invoke()
        
        # Get output
        output = self.interpreter.get_tensor(
            self.output_details[0]['index']
        )
        
        # Dequantize if needed
        if self.output_details[0]['dtype'] == np.uint8:
            quant_params = self.output_details[0].get('quantization_parameters')
            if quant_params:
                scale = quant_params['scales'][0]
                zero_point = quant_params['zero_points'][0]
                output = (output.astype(np.float32) - zero_point) * scale
        
        # Get top predictions
        top_indices = np.argsort(output[0])[-5:][::-1]
        
        return {
            'probabilities': output[0],
            'top_predictions': [
                {'class': int(idx), 'confidence': float(output[0][idx])}
                for idx in top_indices
            ]
        }
    
    def benchmark(self, num_runs=100):
        """Benchmark inference speed."""
        import time
        
        dummy_input = np.random.rand(*self.input_shape).astype(np.float32)
        
        # Warm-up
        for _ in range(10):
            self.interpreter.set_tensor(self.input_details[0]['index'], dummy_input)
            self.interpreter.invoke()
        
        # Benchmark
        times = []
        for _ in range(num_runs):
            start = time.time()
            self.interpreter.set_tensor(self.input_details[0]['index'], dummy_input)
            self.interpreter.invoke()
            times.append(time.time() - start)
        
        avg_time = np.mean(times) * 1000  # ms
        std_time = np.std(times) * 1000
        
        return {
            'average_ms': avg_time,
            'std_ms': std_time,
            'fps': 1000 / avg_time
        }

class MobileInputProcessor:
    """Process various input types for mobile deployment."""
    
    @staticmethod
    def process_camera_frame(frame, target_size=(224, 224)):
        """
        Process camera frame from mobile camera.
        
        Mobile cameras typically return:
        - RGB or YUV format
        - Various resolutions
        - May need rotation
        """
        # Handle rotation (mobile cameras often need 90-degree rotation)
        # frame = cv2.rotate(frame, cv2.ROTATE_90_CLOCKWISE)
        
        # Resize maintaining aspect ratio with padding
        h, w = frame.shape[:2]
        target_h, target_w = target_size
        
        scale = min(target_w / w, target_h / h)
        new_w = int(w * scale)
        new_h = int(h * scale)
        
        resized = cv2.resize(frame, (new_w, new_h))
        
        # Pad to target size
        pad_h = (target_h - new_h) // 2
        pad_w = (target_w - new_w) // 2
        
        padded = cv2.copyMakeBorder(
            resized, pad_h, target_h - new_h - pad_h,
            pad_w, target_w - new_w - pad_w,
            cv2.BORDER_CONSTANT, value=[0, 0, 0]
        )
        
        return padded
    
    @staticmethod
    def process_sensor_data(sensor_readings, window_size=100):
        """
        Process sensor data (accelerometer, gyroscope, etc.)
        
        Args:
            sensor_readings: List of (x, y, z) readings
            window_size: Number of readings to include
        """
        # Normalize sensor data
        data = np.array(sensor_readings[-window_size:])
        
        # Pad if needed
        if len(data) < window_size:
            pad_length = window_size - len(data)
            data = np.pad(data, ((pad_length, 0), (0, 0)), mode='edge')
        
        # Normalize
        mean = np.mean(data, axis=0)
        std = np.std(data, axis=0)
        normalized = (data - mean) / (std + 1e-8)
        
        return normalized.flatten()
