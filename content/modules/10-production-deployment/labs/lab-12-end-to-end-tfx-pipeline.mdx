# Transform module

transform_code = '''
import tensorflow as tf
import tensorflow_transform as tft

NUMERIC_FEATURES = [
    'customer_tenure', 'monthly_charges', 'total_charges'
]

CATEGORICAL_FEATURES = [
    'contract_type', 'payment_method', 'internet_service',
    'tech_support', 'online_security'
]

LABEL_KEY = 'churn'

def preprocessing_fn(inputs):
    """Transform raw inputs into transformed features."""
    outputs = {}
    
    # Process numeric features
    for key in NUMERIC_FEATURES:
        # Fill missing values
        filled = tft.sparse_tensor_to_dense_with_missing(
            inputs[key], default_value=0.0
        )
        # Scale to [0, 1]
        outputs[key + '_scaled'] = tft.scale_to_0_1(filled)
    
    # Process categorical features
    for key in CATEGORICAL_FEATURES:
        outputs[key + '_indexed'] = tft.compute_and_apply_vocabulary(
            inputs[key],
            num_oov_buckets=1
        )
    
    # Label
    outputs[LABEL_KEY] = inputs[LABEL_KEY]
    
    return outputs
'''

with open(TRANSFORM_MODULE, 'w') as f:

# Trainer module
TRAINER_MODULE = os.path.join(MODULES_PATH, 'trainer.py')
trainer_code = '''
import tensorflow as tf
from tfx import v1 as tfx
from tfx_bsl.public import tfxio
from tensorflow_transform import TFTransformOutput

NUMERIC_FEATURES = ['customer_tenure_scaled', 'monthly_charges_scaled', 'total_charges_scaled']
CATEGORICAL_FEATURES = ['contract_type_indexed', 'payment_method_indexed', 
                        'internet_service_indexed', 'tech_support_indexed', 'online_security_indexed']
LABEL_KEY = 'churn'

def _input_fn(file_pattern, data_accessor, tf_transform_output, batch_size=64):
    """Create input dataset."""
    dataset = data_accessor.tf_dataset_factory(
        file_pattern,
        tfxio.TensorFlowDatasetOptions(batch_size=batch_size, label_key=LABEL_KEY),
        tf_transform_output.transformed_metadata.schema
    )
    return dataset

def _build_model(tf_transform_output):
    """Build Keras model."""
    # Numeric inputs
    numeric_inputs = []
    for key in NUMERIC_FEATURES:
        numeric_inputs.append(tf.keras.Input(shape=(1,), name=key))
    
    # Categorical inputs with embeddings
    categorical_inputs = []
    vocab_sizes = {}
    for key in CATEGORICAL_FEATURES:
        vocab_size = tf_transform_output.vocabulary_size_by_name(key.replace('_indexed', ''))
        vocab_sizes[key] = vocab_size
        categorical_inputs.append(tf.keras.Input(shape=(1,), name=key, dtype=tf.int64))
    
    # Embedding layers for categorical features
    embedded = []
    for i, key in enumerate(CATEGORICAL_FEATURES):
        vocab_size = vocab_sizes[key]
        embed_dim = min(50, (vocab_size + 1) // 2)
        embed = tf.keras.layers.Embedding(vocab_size + 1, embed_dim)(categorical_inputs[i])
        embed = tf.keras.layers.Flatten()(embed)
        embedded.append(embed)
    
    # Concatenate all features
    all_features = numeric_inputs + embedded
    concatenated = tf.keras.layers.concatenate(all_features)
    
    # Hidden layers
    x = tf.keras.layers.Dense(128, activation='relu')(concatenated)
    x = tf.keras.layers.BatchNormalization()(x)
    x = tf.keras.layers.Dropout(0.3)(x)
    x = tf.keras.layers.Dense(64, activation='relu')(x)
    x = tf.keras.layers.Dropout(0.3)(x)
    x = tf.keras.layers.Dense(32, activation='relu')(x)
    
    # Output
    output = tf.keras.layers.Dense(1, activation='sigmoid', name='output')(x)
    
    # Create model
    model = tf.keras.Model(
        inputs=numeric_inputs + categorical_inputs,
        outputs=output
    )
    
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss='binary_crossentropy',
        metrics=[
            'accuracy',
            tf.keras.metrics.AUC(name='auc'),
            tf.keras.metrics.Precision(name='precision'),
            tf.keras.metrics.Recall(name='recall')
        ]
    )
    
    return model

def run_fn(fn_args):
    """Training function."""
    tf_transform_output = TFTransformOutput(fn_args.transform_output)
    
    train_dataset = _input_fn(
        fn_args.train_files,
        fn_args.data_accessor,
        tf_transform_output,
        batch_size=64
    )
    
    eval_dataset = _input_fn(
        fn_args.eval_files,
        fn_args.data_accessor,
        tf_transform_output,
        batch_size=64
    )
    
    model = _build_model(tf_transform_output)
    
    # Callbacks
    callbacks = [
        tf.keras.callbacks.EarlyStopping(
            monitor='val_auc',
            patience=5,
            restore_best_weights=True
        ),
        tf.keras.callbacks.ReduceLROnPlateau(
            monitor='val_loss',
            factor=0.5,
            patience=3
        )
    ]
    
    # Train
    history = model.fit(
        train_dataset,
        validation_data=eval_dataset,
        epochs=20,
        callbacks=callbacks,
        steps_per_epoch=100,
        validation_steps=20
    )
    
    # Save model
    model.save(fn_args.serving_model_dir, save_format='tf')
    
    # Print final metrics
    print("\\nFinal training metrics:")
    for metric, values in history.history.items():
        print(f"  {metric}: {values[-1]:.4f}")
'''

with open(TRAINER_MODULE, 'w') as f:
    f.write(trainer_code)

def create_pipeline(pipeline_name, pipeline_root, data_path, serving_path, 
                    transform_module, trainer_module):
    """Create TFX pipeline."""
    
    from tfx.proto import example_gen_pb2, trainer_pb2, pusher_pb2
    import tensorflow_model_analysis as tfma
    
    # 1. ExampleGen - Data ingestion
    example_gen = tfx.components.CsvExampleGen(
        input_base=data_path,
        output_config=example_gen_pb2.Output(
            split_config=example_gen_pb2.SplitConfig(splits=[
                example_gen_pb2.SplitConfig.Split(name='train', hash_buckets=8),
                example_gen_pb2.SplitConfig.Split(name='eval', hash_buckets=2)
            ])
        )
    )
    
    # 2. StatisticsGen - Compute statistics
    statistics_gen = tfx.components.StatisticsGen(
        examples=example_gen.outputs['examples']
    )
    
    # 3. SchemaGen - Generate schema
    schema_gen = tfx.components.SchemaGen(
        statistics=statistics_gen.outputs['statistics'],
        infer_feature_shape=True
    )
    
    # 4. ExampleValidator - Validate data
    example_validator = tfx.components.ExampleValidator(
        statistics=statistics_gen.outputs['statistics'],
        schema=schema_gen.outputs['schema']
    )
    
    # 5. Transform - Feature engineering
    transform = tfx.components.Transform(
        examples=example_gen.outputs['examples'],
        schema=schema_gen.outputs['schema'],
        module_file=transform_module
    )
    
    # 6. Trainer - Train model
    trainer = tfx.components.Trainer(
        module_file=trainer_module,
        examples=transform.outputs['transformed_examples'],
        transform_graph=transform.outputs['transform_graph'],
        schema=schema_gen.outputs['schema'],
        train_args=trainer_pb2.TrainArgs(num_steps=100),
        eval_args=trainer_pb2.EvalArgs(num_steps=20)
    )
    
    # 7. Evaluator - Evaluate model
    eval_config = tfma.EvalConfig(
        model_specs=[
            tfma.ModelSpec(
                label_key='churn',
                prediction_key='output'
            )
        ],
        slicing_specs=[
            tfma.SlicingSpec(),  # Overall
            tfma.SlicingSpec(feature_keys=['contract_type_indexed']),
            tfma.SlicingSpec(feature_keys=['internet_service_indexed'])
        ],
        metrics_specs=[
            tfma.MetricsSpec(metrics=[
                tfma.MetricConfig(class_name='AUC'),
                tfma.MetricConfig(class_name='Accuracy'),
                tfma.MetricConfig(class_name='Precision'),
                tfma.MetricConfig(class_name='Recall'),
                tfma.MetricConfig(class_name='ExampleCount')
            ])
        ]
    )
    
    evaluator = tfx.components.Evaluator(
        examples=example_gen.outputs['examples'],
        model=trainer.outputs['model'],
        eval_config=eval_config
    )
    
    # 8. Pusher - Deploy model
    pusher = tfx.components.Pusher(
        model=trainer.outputs['model'],
        model_blessing=evaluator.outputs['blessing'],
        push_destination=pusher_pb2.PushDestination(
            filesystem=pusher_pb2.PushDestination.Filesystem(
                base_directory=serving_path
            )
        )
    )
    
    # Create pipeline
    components = [
        example_gen,
        statistics_gen,
        schema_gen,
        example_validator,
        transform,
        trainer,
        evaluator,
        pusher
    ]
    
    return tfx.dsl.Pipeline(
        pipeline_name=pipeline_name,
        pipeline_root=pipeline_root,
        components=components,
        metadata_connection_config=sqlite_metadata_connection_config(METADATA_PATH)
    )

# Create the pipeline
print("Creating pipeline...")
pipeline = create_pipeline(
    PIPELINE_NAME,
    PIPELINE_ROOT,
    DATA_PATH,
    SERVING_PATH,
    TRANSFORM_MODULE,
    TRAINER_MODULE
)

   - Blesses model if metrics meet thresholds

8. Pusher
   - Copies blessed model to serving directory
   - Ready for TensorFlow Serving
""")

#!/usr/bin/env python3
"""
Continuous Training Script
Run periodically (e.g., via cron) to retrain model with new data.
"""

import os
import sys
from datetime import datetime

def run_pipeline():
    """Run the TFX pipeline."""
    
    # Add pipeline directory to path
    sys.path.insert(0, 'tfx_pipeline')
    
    # Import pipeline definition
    from pipeline import create_pipeline
    
    # Configuration
    PIPELINE_NAME = "end_to_end_pipeline"
    PIPELINE_ROOT = os.path.join(os.getcwd(), 'tfx_pipeline', PIPELINE_NAME)
    
    # Create and run pipeline
    pipeline = create_pipeline(
        pipeline_name=PIPELINE_NAME,
        pipeline_root=PIPELINE_ROOT,
        data_path=os.path.join(PIPELINE_ROOT, 'data'),
        serving_path=os.path.join(PIPELINE_ROOT, 'serving_model'),
        transform_module=os.path.join(PIPELINE_ROOT, 'modules', 'transform.py'),
        trainer_module=os.path.join(PIPELINE_ROOT, 'modules', 'trainer.py')
    )
    
    from tfx.orchestration.local.local_dag_runner import LocalDagRunner
    
    print(f"[{datetime.now()}] Starting pipeline execution...")
    LocalDagRunner().run(pipeline)
    print(f"[{datetime.now()}] Pipeline execution completed!")

if __name__ == '__main__':
    run_pipeline()
'''
