class DriftDetector:
    """Comprehensive drift detection class."""
    
    def __init__(self, reference_data, significance_level=0.05):
        """
        Initialize drift detector.
        
        Args:
            reference_data: Baseline/reference data
            significance_level: Threshold for statistical tests
        """
        self.reference = reference_data
        self.alpha = significance_level
        self.numeric_columns = reference_data.select_dtypes(include=[np.number]).columns
        self.categorical_columns = reference_data.select_dtypes(include=['object']).columns
    
    def ks_test(self, production_data, column):
        """
        Kolmogorov-Smirnov test for numeric features.
        Tests if two samples come from the same distribution.
        """
        statistic, p_value = stats.ks_2samp(
            self.reference[column].dropna(),
            production_data[column].dropna()
        )
        
        return {
            'test': 'Kolmogorov-Smirnov',
            'column': column,
            'statistic': statistic,
            'p_value': p_value,
            'drift_detected': p_value < self.alpha
        }
    
    def chi_square_test(self, production_data, column):
        """
        Chi-square test for categorical features.
        Tests if category distributions are different.
        """
        # Get value counts
        ref_counts = self.reference[column].value_counts()
        prod_counts = production_data[column].value_counts()
        
        # Align categories
        all_categories = set(ref_counts.index) | set(prod_counts.index)
        ref_aligned = [ref_counts.get(cat, 0) for cat in all_categories]
        prod_aligned = [prod_counts.get(cat, 0) for cat in all_categories]
        
        # Perform chi-square test
        statistic, p_value = stats.chisquare(prod_aligned, ref_aligned)
        
        return {
            'test': 'Chi-Square',
            'column': column,
            'statistic': statistic,
            'p_value': p_value,
            'drift_detected': p_value < self.alpha
        }
    
    def wasserstein_distance(self, production_data, column):
        """
        Wasserstein distance (Earth Mover's Distance).
        Measures distance between distributions.
        """
        from scipy.stats import wasserstein_distance as wd
        
        distance = wd(
            self.reference[column].dropna(),
            production_data[column].dropna()
        )
        
        # Normalize by reference std
        normalized_distance = distance / (self.reference[column].std() + 1e-10)
        
        return {
            'test': 'Wasserstein Distance',
            'column': column,
            'distance': distance,
            'normalized_distance': normalized_distance,
            'drift_detected': normalized_distance > 0.1  # Threshold
        }
    
    def jensen_shannon_divergence(self, production_data, column):
        """
        Jensen-Shannon divergence for categorical features.
        Symmetric and smoothed version of KL divergence.
        """
        # Get probability distributions
        ref_dist = self.reference[column].value_counts(normalize=True).sort_index()
        prod_dist = production_data[column].value_counts(normalize=True).sort_index()
        
        # Align indices
        all_categories = sorted(set(ref_dist.index) | set(prod_dist.index))
        ref_aligned = np.array([ref_dist.get(cat, 0) for cat in all_categories])
        prod_aligned = np.array([prod_dist.get(cat, 0) for cat in all_categories])
        
        # Calculate JS divergence
        js_div = jensenshannon(ref_aligned, prod_aligned)
        
        return {
            'test': 'Jensen-Shannon Divergence',
            'column': column,
            'divergence': js_div,
            'drift_detected': js_div > 0.1  # Threshold
        }
    
    def population_stability_index(self, production_data, column, bins=10):
        """
        Population Stability Index (PSI).
        Commonly used in credit risk modeling.
        PSI < 0.1: No change
        0.1 <= PSI < 0.25: Moderate change
        PSI >= 0.25: Significant change
        """
        # Create bins based on reference data
        ref_values = self.reference[column].dropna()
        prod_values = production_data[column].dropna()
        
        bin_edges = np.percentile(ref_values, np.linspace(0, 100, bins + 1))
        bin_edges[0] -= 0.001  # Adjust for edge cases
        bin_edges[-1] += 0.001
        
        # Calculate percentages
        ref_percents = np.histogram(ref_values, bins=bin_edges)[0] / len(ref_values)
        prod_percents = np.histogram(prod_values, bins=bin_edges)[0] / len(prod_values)
        
        # Avoid division by zero
        ref_percents = np.clip(ref_percents, 0.0001, 1)
        prod_percents = np.clip(prod_percents, 0.0001, 1)
        
        # Calculate PSI
        psi = np.sum((prod_percents - ref_percents) * np.log(prod_percents / ref_percents))
        
        return {
            'test': 'Population Stability Index',
            'column': column,
            'psi': psi,
            'drift_detected': psi > 0.25,
            'severity': 'High' if psi > 0.25 else ('Medium' if psi > 0.1 else 'Low')
        }
    
    def detect_all_drift(self, production_data):
        """Run all drift detection methods."""
        results = []
        
        # Numeric features
        for col in self.numeric_columns:
            if col == 'target':  # Skip target for data drift
                continue
            results.append(self.ks_test(production_data, col))
            results.append(self.wasserstein_distance(production_data, col))
            results.append(self.population_stability_index(production_data, col))
        
        # Categorical features
        for col in self.categorical_columns:
            results.append(self.chi_square_test(production_data, col))
            results.append(self.jensen_shannon_divergence(production_data, col))
        
        return pd.DataFrame(results)

"""
Concept drift occurs when the relationship between inputs and outputs changes.
This is detected by monitoring model performance and target distribution.
"""

def detect_concept_drift(reference_data, production_data, target_col='target'):
    """
    Detect concept drift by comparing target distributions and
    relationships with features.
    """
    results = {}
    
    # 1. Target distribution drift
    ref_target_dist = reference_data[target_col].value_counts(normalize=True)
    prod_target_dist = production_data[target_col].value_counts(normalize=True)
    
    # Chi-square test on target
    chi2, p_value = stats.chisquare(
        prod_target_dist.values,
        ref_target_dist.values
    )
    
    results['target_distribution_drift'] = {
        'p_value': p_value,
        'drift_detected': p_value < 0.05,
        'reference_distribution': ref_target_dist.to_dict(),
        'production_distribution': prod_target_dist.to_dict()
    }
    
    # 2. Feature-target correlation drift
    numeric_cols = reference_data.select_dtypes(include=[np.number]).columns
    numeric_cols = [c for c in numeric_cols if c != target_col]
    
    correlation_drift = {}
    for col in numeric_cols:
        ref_corr = reference_data[col].corr(reference_data[target_col])
        prod_corr = production_data[col].corr(production_data[target_col])
        
        # Check if correlation changed significantly
        correlation_diff = abs(ref_corr - prod_corr)
        
        correlation_drift[col] = {
            'reference_correlation': ref_corr,
            'production_correlation': prod_corr,
            'difference': correlation_diff,
            'drift_detected': correlation_diff > 0.1
        }
    
    results['correlation_drift'] = correlation_drift
    
    return results

class DriftMonitoringPipeline:
    """Production drift monitoring pipeline."""
    
    def __init__(self, reference_data, alert_threshold=0.25):
        self.detector = DriftDetector(reference_data)
        self.alert_threshold = alert_threshold
        self.history = []
    
    def monitor(self, production_data, timestamp=None):
        """
        Monitor production data for drift.
        
        Returns:
            dict with drift status and alerts
        """
        if timestamp is None:
            timestamp = pd.Timestamp.now()
        
        # Run drift detection
        results = self.detector.detect_all_drift(production_data)
        
        # Calculate overall drift score
        drift_detected_count = results['drift_detected'].sum()
        total_tests = len(results)
        drift_score = drift_detected_count / total_tests
        
        # Determine alert level
        if drift_score >= self.alert_threshold:
            alert_level = 'CRITICAL'
        elif drift_score >= self.alert_threshold / 2:
            alert_level = 'WARNING'
        else:
            alert_level = 'OK'
        
        # Create report
        report = {
            'timestamp': timestamp,
            'drift_score': drift_score,
            'drift_detected_count': int(drift_detected_count),
            'total_tests': total_tests,
            'alert_level': alert_level,
            'drifted_features': results[results['drift_detected']]['column'].unique().tolist()
        }
        
        self.history.append(report)
        
        return report
    
    def get_history(self):
        """Get monitoring history."""
        return pd.DataFrame(self.history)
    
    def plot_drift_trend(self):
        """Plot drift score over time."""
        if not self.history:
            print("No history to plot")
            return
        
        history_df = self.get_history()
        
        plt.figure(figsize=(12, 6))
        plt.plot(history_df['timestamp'], history_df['drift_score'], 
                 marker='o', linewidth=2)
        plt.axhline(y=self.alert_threshold, color='r', 
                    linestyle='--', label='Critical Threshold')
        plt.axhline(y=self.alert_threshold/2, color='orange', 
                    linestyle='--', label='Warning Threshold')
        plt.xlabel('Time')
        plt.ylabel('Drift Score')
        plt.title('Drift Score Trend')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.xticks(rotation=45)
        plt.tight_layout()
        plt.savefig('drift_trend.png', dpi=150)
        print("Drift trend saved to: drift_trend.png")
        plt.show()

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|---------------|-----|
| Using too small a production window for drift tests | KS test and PSI require sufficient samples to achieve statistical power; a window of <50 samples inflates false-negative rates and misses real drift | Use a minimum window size of 200–500 samples; calculate required sample size with a power analysis (e.g., `statsmodels.stats.power`) before setting thresholds |
| Setting a fixed drift threshold without calibrating for the feature | A universal p-value threshold (e.g., 0.05) is too sensitive for high-cardinality categorical features and too lenient for continuous features with known seasonal patterns | Calibrate per-feature thresholds on historical data using a rolling baseline; use Bonferroni correction when monitoring many features simultaneously |
| Conflating data drift with concept drift | Data drift (input distribution shift) does not always degrade model performance; concept drift (label-conditional shift) may occur even without data drift | Monitor both input feature distributions (data drift) and model output/error distributions (concept drift) independently; track prediction confidence histograms as a leading indicator of concept drift |
| Re-training on every detected drift event | Frequent retraining on minor drift consumes compute budget and may cause the model to overfit to transient distribution shifts (e.g., weekly seasonality) | Use a two-stage trigger: alert on drift detection, retrain only when drift is sustained over N consecutive windows or when model performance degrades below a threshold |
| Comparing against a stale reference distribution | If the reference is from initial deployment and months of gradual drift have accumulated, new batches compare against an outdated baseline and the drift magnitude is overestimated | Use a sliding reference window (e.g., the most recent 7-day baseline) or update the reference periodically; store reference distributions in a versioned artifact store |
