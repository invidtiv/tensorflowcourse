"""

Production deployment of TensorFlow Serving using Docker and Docker Compose.

import tensorflow as tf

print("\n" + "=" * 60)
print("PART 1: Creating Multiple Models")
print("=" * 60)

# Model 1: MNIST Classifier
def create_mnist_model():
    model = tf.keras.Sequential([
        tf.keras.layers.Conv2D(32, 3, activation='relu', input_shape=(28, 28, 1)),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Conv2D(64, 3, activation='relu'),
        tf.keras.layers.MaxPooling2D(),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    return model

# Model 2: Simple Regressor
def create_regressor_model():
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(64, activation='relu', input_shape=(5,)),
        tf.keras.layers.Dense(32, activation='relu'),
        tf.keras.layers.Dense(1)
    ])
    return model

# Train and export MNIST model
print("Creating MNIST classifier...")
mnist_model = create_mnist_model()
mnist_model.compile(optimizer='adam', loss='sparse_categorical_crossentropy')

(x_train, y_train), _ = tf.keras.datasets.mnist.load_data()
x_train = x_train[:1000].reshape(-1, 28, 28, 1).astype('float32') / 255.0
mnist_model.fit(x_train, y_train[:1000], epochs=2, verbose=0)

# Export MNIST model
mnist_path = "docker_models/mnist/1"
os.makedirs(mnist_path, exist_ok=True)
tf.saved_model.save(mnist_model, mnist_path)
print(f"MNIST model exported to: {mnist_path}")

# Train and export regressor
print("\nCreating regressor model...")
regressor = create_regressor_model()
regressor.compile(optimizer='adam', loss='mse')

x_reg = np.random.random((1000, 5))
y_reg = np.random.random((1000, 1))
regressor.fit(x_reg, y_reg, epochs=2, verbose=0)

# Export regressor

tf.saved_model.save(regressor, regressor_path)

  -e MODEL_NAME=mnist \\
  -t tensorflow/serving \\
  --monitoring_config_file=/models/monitoring.config \\

# 4. With Resource Limits
# -----------------------
docker run -p 8501:8501 \\
  --memory=2g \\
  --cpus=2 \\
  --mount type=bind,source=$(pwd)/docker_models,target=/models \\
  -e MODEL_NAME=mnist \\
  -t tensorflow/serving

# 5. Production Mode with Batching

docker run -p 8501:8501 \\
  --mount type=bind,source=$(pwd)/docker_models,target=/models \\
  --mount type=bind,source=$(pwd)/batch_config,target=/batch_config \\

  --enable_batching \\

# 6. With Custom Entry Point
# --------------------------

# Model Versioning and Rollback
# =============================

## Version Management

1. Directory Structure:
   ```
   models/
   └── mnist/
       ├── 1/          # Version 1
       │   └── ...
       ├── 2/          # Version 2
       │   └── ...
       └── labels/     # Version labels
           ├── stable -> ../1
           └── canary -> ../2
   ```

2. Version Policies in Config:
   - All: Serve all versions
   - Latest: Serve only latest N versions
   - Specific: Serve specific versions

3. Version Labels:
   ```protobuf
   version_labels: {
     key: "stable"
     value: "1"
   }
   version_labels: {
     key: "canary"
     value: "2"
   }
   ```

## Rollback Procedure

1. Identify problematic version:
   ```bash
   curl http://localhost:8501/v1/models/mnist
   ```

2. Update config to exclude version:
   ```protobuf
   model_version_policy: {
     specific: {
       versions: 1  # Only serve version 1
     }
   }
   ```

3. Reload config without restart:
   ```bash
   curl -X POST http://localhost:8501/v1/models/mnist/config/reload
   ```

## A/B Testing

Request specific version:
```bash
curl -X POST http://localhost:8501/v1/models/mnist/versions/1:predict
```

Or use labels:
```bash
curl -X POST http://localhost:8501/v1/models/mnist/labels/stable:predict
```
"""

deploy_script = """#!/bin/bash
# Production Deployment Script for TensorFlow Serving

set -e

echo "=== TensorFlow Serving Deployment ==="

# Configuration
MODELS_DIR="./docker_models"
COMPOSE_FILE="docker-compose.yml"

# Function: Check prerequisites
check_prerequisites() {
    echo "Checking prerequisites..."
    
    if ! command -v docker &> /dev/null; then
        echo "Error: Docker not installed"
        exit 1
    fi
    
    if ! command -v docker-compose &> /dev/null; then
        echo "Error: Docker Compose not installed"
        exit 1
    fi
    
    echo "Prerequisites OK"
}

# Function: Validate models
validate_models() {
    echo "Validating models..."
    
    for model_dir in $MODELS_DIR/*/; do
        if [ -d "$model_dir" ]; then
            model_name=$(basename "$model_dir")
            echo "  Checking model: $model_name"
            
            # Check for valid SavedModel
            if [ ! -f "$model_dir/1/saved_model.pb" ]; then
                echo "    Warning: No saved_model.pb found"
            fi
        fi
    done
    
    echo "Model validation complete"
}

# Function: Deploy
deploy() {
    echo "Deploying TensorFlow Serving..."
    
    # Pull latest image
    docker-compose -f $COMPOSE_FILE pull
    
    # Start services
    docker-compose -f $COMPOSE_FILE up -d
    
    # Wait for health check
    echo "Waiting for service to be healthy..."
    sleep 10
    
    # Verify deployment
    if curl -sf http://localhost:8501/v1/models/mnist > /dev/null; then
        echo "Deployment successful!"
        echo "REST API: http://localhost:8501"
        echo "gRPC: localhost:8500"
    else
        echo "Deployment may have failed. Check logs:"
        docker-compose -f $COMPOSE_FILE logs
    fi
}

# Function: View logs
view_logs() {
    docker-compose -f $COMPOSE_FILE logs -f
}

# Function: Stop
stop() {
    echo "Stopping TensorFlow Serving..."
    docker-compose -f $COMPOSE_FILE down
}

# Function: Update model
update_model() {
    local model_name=$1
    local version=$2
    
    echo "Updating model $model_name to version $version..."
    
    # Reload config
    curl -X POST http://localhost:8501/v1/models/$model_name/config/reload
    
    echo "Model updated"
}

# Main
case "$1" in
    deploy)
        check_prerequisites
        validate_models
        deploy
        ;;
    stop)
        stop
        ;;
    logs)
        view_logs
        ;;
    update)
        update_model $2 $3
        ;;
    *)
        echo "Usage: $0 {deploy|stop|logs|update <model> <version>}"
        exit 1
        ;;
esac
"""

with open('deploy.sh', 'w') as f:
    f.write(deploy_script)
