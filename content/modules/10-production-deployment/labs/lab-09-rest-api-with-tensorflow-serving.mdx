class TFServingClient:
    """
    TensorFlow Serving REST API Client.
    
    Handles all communication with TensorFlow Serving server.
    """
    
    def __init__(self, host='localhost', port=8501):
        self.base_url = f"http://{host}:{port}/v1/models"
    
    def get_model_status(self, model_name):
        """Get model status and available versions."""
        url = f"{self.base_url}/{model_name}"
        try:
            response = requests.get(url)
            return response.json()
        except requests.exceptions.ConnectionError:
            return {"error": "Could not connect to TensorFlow Serving"}
    
    def get_model_metadata(self, model_name, version=None):
        """Get model metadata including signatures."""
        if version:
            url = f"{self.base_url}/{model_name}/versions/{version}/metadata"
        else:
            url = f"{self.base_url}/{model_name}/metadata"
        
        try:
            response = requests.get(url)
            return response.json()
        except requests.exceptions.ConnectionError:
            return {"error": "Could not connect to TensorFlow Serving"}
    
    def predict(self, model_name, data, version=None, signature='serving_default'):
        """
        Make prediction request.
        
        Args:
            model_name: Name of the model
            data: Input data (numpy array or list)
            version: Model version (optional)
            signature: Signature name
        """
        if version:
            url = f"{self.base_url}/{model_name}/versions/{version}:predict"
        else:
            url = f"{self.base_url}/{model_name}:predict"
        
        # Prepare request
        if isinstance(data, np.ndarray):
            data = data.tolist()
        
        payload = {
            "signature_name": signature,
            "instances": data if isinstance(data[0], list) else [data]
        }
        
        try:
            response = requests.post(url, json=payload)
            return response.json()
        except requests.exceptions.ConnectionError:
            return {"error": "Could not connect to TensorFlow Serving"}
    
    def batch_predict(self, model_name, data_batch, version=None):
        """Make batch prediction request."""
        return self.predict(model_name, data_batch, version)
    
    def classify(self, model_name, data, version=None):
        """Make classification request (if model supports it)."""
        if version:
            url = f"{self.base_url}/{model_name}/versions/{version}:classify"
        else:
            url = f"{self.base_url}/{model_name}:classify"
        
        payload = {"instances": data if isinstance(data, list) else [data]}
        
        try:
            response = requests.post(url, json=payload)
            return response.json()
        except requests.exceptions.ConnectionError:
            return {"error": "Could not connect to TensorFlow Serving"}
    
    def regress(self, model_name, data, version=None):
        """Make regression request (if model supports it)."""
        if version:
            url = f"{self.base_url}/{model_name}/versions/{version}:regress"
        else:
            url = f"{self.base_url}/{model_name}:regress"
        
        payload = {"instances": data if isinstance(data, list) else [data]}
        
        try:
            response = requests.post(url, json=payload)
            return response.json()
        except requests.exceptions.ConnectionError:
            return {"error": "Could not connect to TensorFlow Serving"}

flask_app_code = '''
"""
Flask API Wrapper for TensorFlow Serving
Run with: python app.py
"""

from flask import Flask, request, jsonify
import requests
import numpy as np
import base64
import io
from PIL import Image

app = Flask(__name__)

# Configuration
TF_SERVING_HOST = 'localhost'
TF_SERVING_PORT = 8501
MODEL_NAME = 'mnist'

def preprocess_image(image_data):
    """Preprocess image for MNIST model."""
    # Decode base64 image
    image = Image.open(io.BytesIO(base64.b64decode(image_data)))
    
    # Convert to grayscale and resize
    image = image.convert('L').resize((28, 28))
    
    # Normalize
    image_array = np.array(image).astype('float32') / 255.0
    
    return image_array.reshape(28, 28).tolist()

@app.route('/health', methods=['GET'])
def health_check():
    """Health check endpoint."""
    return jsonify({"status": "healthy", "service": "MNIST API"})

@app.route('/predict', methods=['POST'])
def predict():
    """Single prediction endpoint."""
    try:
        data = request.json
        
        # Get image data
        if 'image' in data:
            image = preprocess_image(data['image'])
        elif 'data' in data:
            image = data['data']
        else:
            return jsonify({"error": "No image or data provided"}), 400
        
        # Forward to TensorFlow Serving
        tf_serving_url = f"http://{TF_SERVING_HOST}:{TF_SERVING_PORT}/v1/models/{MODEL_NAME}:predict"
        
        payload = {
            "signature_name": "serving_default",
            "instances": [image]
        }
        
        response = requests.post(tf_serving_url, json=payload)
        
        if response.status_code == 200:
            predictions = response.json()['predictions'][0]
            predicted_class = int(np.argmax(predictions))
            confidence = float(predictions[predicted_class])
            
            return jsonify({
                "predicted_class": predicted_class,
                "confidence": confidence,
                "probabilities": predictions
            })
        else:
            return jsonify({"error": "Prediction failed"}), 500
            
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@app.route('/batch_predict', methods=['POST'])
def batch_predict():
    """Batch prediction endpoint."""
    try:
        data = request.json
        
        if 'images' not in data:
            return jsonify({"error": "No images provided"}), 400
        
        images = [preprocess_image(img) for img in data['images']]
        
        # Forward to TensorFlow Serving
        tf_serving_url = f"http://{TF_SERVING_HOST}:{TF_SERVING_PORT}/v1/models/{MODEL_NAME}:predict"
        
        payload = {
            "signature_name": "serving_default",
            "instances": images
        }
        
        response = requests.post(tf_serving_url, json=payload)
        
        if response.status_code == 200:
            predictions = response.json()['predictions']
            results = [
                {
                    "predicted_class": int(np.argmax(pred)),
                    "confidence": float(max(pred))
                }
                for pred in predictions
            ]
            
            return jsonify({"predictions": results})
        else:
            return jsonify({"error": "Batch prediction failed"}), 500
            
    except Exception as e:
        return jsonify({"error": str(e)}), 500

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
'''

with open('flask_app.py', 'w') as f:

load_test_code = '''
"""
Load Testing Script for TensorFlow Serving API
Run with: python load_test.py
"""

import requests
import numpy as np
import time
import concurrent.futures
import statistics

# Configuration
API_URL = 'http://localhost:5000/predict'
NUM_REQUESTS = 100
CONCURRENT_WORKERS = 10

def make_request():
    """Make a single prediction request."""
    # Generate random MNIST-like input
    data = np.random.rand(28, 28).tolist()
    
    start = time.time()
    response = requests.post(API_URL, json={'data': data})
    elapsed = time.time() - start
    
    return {
        'status_code': response.status_code,
        'latency': elapsed * 1000  # Convert to ms
    }

def run_load_test():
    """Run concurrent load test."""
    print(f"Starting load test: {NUM_REQUESTS} requests, {CONCURRENT_WORKERS} workers")
    
    results = []
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=CONCURRENT_WORKERS) as executor:
        futures = [executor.submit(make_request) for _ in range(NUM_REQUESTS)]
        
        for future in concurrent.futures.as_completed(futures):
            results.append(future.result())
    
    # Analyze results
    latencies = [r['latency'] for r in results]
    success_count = sum(1 for r in results if r['status_code'] == 200)
    
    print(f"\\nResults:")
    print(f"  Successful requests: {success_count}/{NUM_REQUESTS}")
    print(f"  Success rate: {success_count/NUM_REQUESTS*100:.1f}%")
    print(f"  Average latency: {statistics.mean(latencies):.2f} ms")
    print(f"  Median latency: {statistics.median(latencies):.2f} ms")
    print(f"  95th percentile: {np.percentile(latencies, 95):.2f} ms")
    print(f"  99th percentile: {np.percentile(latencies, 99):.2f} ms")
    print(f"  Min/Max latency: {min(latencies):.2f}/{max(latencies):.2f} ms")
    print(f"  Throughput: {NUM_REQUESTS/sum(latencies)*1000:.1f} req/s")

if __name__ == '__main__':
    run_load_test()
'''
