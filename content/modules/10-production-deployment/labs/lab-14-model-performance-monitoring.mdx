class SimulatedModel:
    """Simulated model for demonstration."""
    
    def __init__(self, degradation_rate=0.0):
        self.degradation_rate = degradation_rate
        self.prediction_count = 0
    
    def predict(self, X):
        """Make predictions with optional degradation."""
        self.prediction_count += len(X)
        
        # Simulate predictions
        true_probs = 1 / (1 + np.exp(-X[:, 0] - X[:, 1] * 0.5))
        
        # Add degradation over time
        noise_level = 0.1 + self.degradation_rate * self.prediction_count / 1000
        noise = np.random.normal(0, noise_level, len(X))
        
        predictions = np.clip(true_probs + noise, 0, 1)
        return predictions

class ProductionSimulator:
    """Simulate production environment with varying conditions."""
    
    def __init__(self, model, n_features=5):
        self.model = model
        self.n_features = n_features
        self.time_step = 0
    
    def generate_batch(self, batch_size=100, scenario='normal'):
        """Generate a batch of production data."""
        self.time_step += 1
        
        if scenario == 'normal':
            # Normal operation
            X = np.random.randn(batch_size, self.n_features)
            y = (X[:, 0] + X[:, 1] * 0.5 + np.random.randn(batch_size) * 0.1) > 0
        
        elif scenario == 'shift':
            # Data distribution shift
            X = np.random.randn(batch_size, self.n_features) + 0.5
            y = (X[:, 0] + X[:, 1] * 0.5 + np.random.randn(batch_size) * 0.1) > 0
        
        elif scenario == 'noisy':
            # High noise period
            X = np.random.randn(batch_size, self.n_features)
            y = (X[:, 0] + X[:, 1] * 0.5 + np.random.randn(batch_size) * 0.5) > 0
        
        elif scenario == 'adversarial':
            # Adversarial inputs
            X = np.random.randn(batch_size, self.n_features) * 3
            y = (X[:, 0] + X[:, 1] * 0.5 + np.random.randn(batch_size) * 0.1) > 0
        
        else:
            raise ValueError(f"Unknown scenario: {scenario}")
        
        # Get predictions
        predictions = self.model.predict(X)
        
        return {
            'timestamp': datetime.now() + timedelta(hours=self.time_step),
            'features': X,
            'predictions': predictions,
            'labels': y.astype(int),
            'scenario': scenario
        }

class PerformanceMetrics:
    """Calculate various performance metrics."""
    
    @staticmethod
    def accuracy(y_true, y_pred):
        """Calculate accuracy."""
        return np.mean(y_true == (y_pred > 0.5).astype(int))
    
    @staticmethod
    def precision(y_true, y_pred):
        """Calculate precision."""
        pred_labels = (y_pred > 0.5).astype(int)
        tp = np.sum((pred_labels == 1) & (y_true == 1))
        fp = np.sum((pred_labels == 1) & (y_true == 0))
        return tp / (tp + fp) if (tp + fp) > 0 else 0
    
    @staticmethod
    def recall(y_true, y_pred):
        """Calculate recall."""
        pred_labels = (y_pred > 0.5).astype(int)
        tp = np.sum((pred_labels == 1) & (y_true == 1))
        fn = np.sum((pred_labels == 0) & (y_true == 1))
        return tp / (tp + fn) if (tp + fn) > 0 else 0
    
    @staticmethod
    def f1_score(y_true, y_pred):
        """Calculate F1 score."""
        prec = PerformanceMetrics.precision(y_true, y_pred)
        rec = PerformanceMetrics.recall(y_true, y_pred)
        return 2 * prec * rec / (prec + rec) if (prec + rec) > 0 else 0
    
    @staticmethod
    def auc_roc(y_true, y_pred):
        """Calculate AUC-ROC using simple approximation."""
        # Simple trapezoidal approximation
        sorted_indices = np.argsort(y_pred)[::-1]
        y_true_sorted = y_true[sorted_indices]
        
        n_pos = np.sum(y_true)
        n_neg = len(y_true) - n_pos
        
        if n_pos == 0 or n_neg == 0:
            return 0.5
        
        cum_pos = np.cumsum(y_true_sorted)
        cum_neg = np.arange(1, len(y_true) + 1) - cum_pos
        
        tpr = cum_pos / n_pos
        fpr = cum_neg / n_neg
        
        return np.trapz(tpr, fpr)
    
    @staticmethod
    def log_loss(y_true, y_pred):
        """Calculate binary cross-entropy loss."""
        eps = 1e-15
        y_pred = np.clip(y_pred, eps, 1 - eps)
        return -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))
    
    @staticmethod
    def calculate_all(y_true, y_pred):
        """Calculate all metrics."""
        return {
            'accuracy': PerformanceMetrics.accuracy(y_true, y_pred),
            'precision': PerformanceMetrics.precision(y_true, y_pred),
            'recall': PerformanceMetrics.recall(y_true, y_pred),
            'f1_score': PerformanceMetrics.f1_score(y_true, y_pred),
            'auc_roc': PerformanceMetrics.auc_roc(y_true, y_pred),
            'log_loss': PerformanceMetrics.log_loss(y_true, y_pred)
        }

class PredictionMonitor:
    """Monitor prediction distributions."""
    
    def __init__(self, window_size=1000):
        self.window_size = window_size
        self.predictions_history = deque(maxlen=window_size)
        self.timestamps = deque(maxlen=window_size)
    
    def add_predictions(self, predictions, timestamp):
        """Add new predictions to monitor."""
        self.predictions_history.extend(predictions)
        self.timestamps.extend([timestamp] * len(predictions))
    
    def get_distribution_stats(self):
        """Get statistics about prediction distribution."""
        if len(self.predictions_history) == 0:
            return {}
        
        preds = np.array(self.predictions_history)
        
        return {
            'mean': np.mean(preds),
            'std': np.std(preds),
            'min': np.min(preds),
            'max': np.max(preds),
            'median': np.median(preds),
            'percentile_5': np.percentile(preds, 5),
            'percentile_95': np.percentile(preds, 95),
            'fraction_positive': np.mean(preds > 0.5)
        }
    
    def detect_distribution_shift(self, reference_stats, threshold=0.1):
        """Detect if prediction distribution has shifted."""
        current_stats = self.get_distribution_stats()
        
        if not current_stats:
            return {'shift_detected': False}
        
        # Compare key statistics
        mean_diff = abs(current_stats['mean'] - reference_stats.get('mean', 0))
        std_diff = abs(current_stats['std'] - reference_stats.get('std', 0))
        
        shift_detected = mean_diff > threshold or std_diff > threshold
        
        return {
            'shift_detected': shift_detected,
            'mean_difference': mean_diff,
            'std_difference': std_diff,
            'current_stats': current_stats
        }

class ModelMonitor:
    """Comprehensive production model monitoring."""
    
    def __init__(self, model_name, alert_thresholds=None):
        self.model_name = model_name
        self.alert_thresholds = alert_thresholds or {
            'accuracy': 0.7,
            'precision': 0.6,
            'recall': 0.6,
            'f1_score': 0.65,
            'auc_roc': 0.7,
            'log_loss': 1.0
        }
        
        self.metrics_history = []
        self.prediction_monitor = PredictionMonitor()
        self.alert_history = []
    
    def process_batch(self, batch_data):
        """Process a batch of production data."""
        y_true = batch_data['labels']
        y_pred = batch_data['predictions']
        timestamp = batch_data['timestamp']
        
        # Calculate metrics
        metrics = PerformanceMetrics.calculate_all(y_true, y_pred)
        metrics['timestamp'] = timestamp
        metrics['batch_size'] = len(y_true)
        metrics['scenario'] = batch_data.get('scenario', 'unknown')
        
        self.metrics_history.append(metrics)
        
        # Update prediction monitor
        self.prediction_monitor.add_predictions(y_pred, timestamp)
        
        # Check for alerts
        alerts = self._check_alerts(metrics)
        if alerts:
            self.alert_history.extend(alerts)
        
        return metrics, alerts
    
    def _check_alerts(self, metrics):
        """Check if any metrics breach thresholds."""
        alerts = []
        
        for metric_name, threshold in self.alert_thresholds.items():
            if metric_name in metrics:
                value = metrics[metric_name]
                
                # For log_loss, higher is worse
                if metric_name == 'log_loss':
                    if value > threshold:
                        alerts.append({
                            'timestamp': metrics['timestamp'],
                            'metric': metric_name,
                            'value': value,
                            'threshold': threshold,
                            'severity': 'WARNING' if value < threshold * 1.5 else 'CRITICAL'
                        })
                else:
                    if value < threshold:
                        alerts.append({
                            'timestamp': metrics['timestamp'],
                            'metric': metric_name,
                            'value': value,
                            'threshold': threshold,
                            'severity': 'WARNING' if value > threshold * 0.9 else 'CRITICAL'
                        })
        
        return alerts
    
    def get_summary(self):
        """Get monitoring summary."""
        if not self.metrics_history:
            return {"error": "No data collected"}
        
        df = pd.DataFrame(self.metrics_history)
        
        summary = {
            'total_batches': len(self.metrics_history),
            'total_predictions': df['batch_size'].sum(),
            'time_range': {
                'start': df['timestamp'].min(),
                'end': df['timestamp'].max()
            },
            'average_metrics': {
                'accuracy': df['accuracy'].mean(),
                'precision': df['precision'].mean(),
                'recall': df['recall'].mean(),
                'f1_score': df['f1_score'].mean(),
                'auc_roc': df['auc_roc'].mean()
            },
            'alert_count': len(self.alert_history),
            'critical_alerts': len([a for a in self.alert_history if a['severity'] == 'CRITICAL'])
        }
        
        return summary
    
    def plot_metrics_trend(self):
        """Plot metrics over time."""
        if not self.metrics_history:
            print("No data to plot")
            return
        
        df = pd.DataFrame(self.metrics_history)
        
        fig, axes = plt.subplots(2, 3, figsize=(15, 10))
        metrics_to_plot = ['accuracy', 'precision', 'recall', 'f1_score', 'auc_roc', 'log_loss']
        
        for idx, metric in enumerate(metrics_to_plot):
            ax = axes[idx // 3, idx % 3]
            ax.plot(df['timestamp'], df[metric], marker='o', linewidth=2)
            
            # Add threshold line if applicable
            if metric in self.alert_thresholds:
                threshold = self.alert_thresholds[metric]
                ax.axhline(y=threshold, color='r', linestyle='--', alpha=0.5)
            
            ax.set_title(f'{metric.upper()} Over Time')
            ax.set_xlabel('Time')
            ax.set_ylabel(metric)
            ax.grid(True, alpha=0.3)
            ax.tick_params(axis='x', rotation=45)
        
        plt.tight_layout()
        plt.savefig('metrics_trend.png', dpi=150, bbox_inches='tight')
        print("\nMetrics trend saved to: metrics_trend.png")
        plt.show()

class AlertingSystem:
    """Production alerting system."""
    
    def __init__(self):
        self.alert_handlers = []
    
    def add_handler(self, handler):
        """Add an alert handler."""
        self.alert_handlers.append(handler)
    
    def send_alert(self, alert):
        """Send alert through all handlers."""
        for handler in self.alert_handlers:
            handler(alert)
    
    @staticmethod
    def console_handler(alert):
        """Print alert to console."""
        print(f"\n🚨 ALERT [{alert['severity']}]: {alert['metric']}")
        print(f"   Value: {alert['value']:.4f}")
        print(f"   Threshold: {alert['threshold']:.4f}")
        print(f"   Time: {alert['timestamp']}")
    
    @staticmethod
    def log_handler(alert):
        """Log alert to file."""
        log_entry = {
            'timestamp': alert['timestamp'].isoformat(),
            'severity': alert['severity'],
            'metric': alert['metric'],
            'value': alert['value'],
            'threshold': alert['threshold']
        }
        
        with open('alerts.log', 'a') as f:
            f.write(json.dumps(log_entry) + '\n')

- **Accuracy metric flat across all windows** → the production prediction generator is returning the same predictions each window; verify the drift injection coefficient is non-zero.

- **Monitoring chart blank** → `matplotlib` backend may be non-interactive in your environment; add `plt.switch_backend("Agg")` before the plot call.

<Callout type="pitfall" title="Common Pitfalls in Production Monitoring">
1. **Monitoring too infrequently** — hourly batch checks miss concept drift that degrades high-velocity models within minutes. Use sliding-window statistics with configurable granularity, and tune the window size to your data arrival rate.

2. **Single-threshold alerting** — one static threshold for PSI or accuracy misses gradual degradation. Use tiered severity: `INFO` (PSI 0.10–0.25), `WARNING` (PSI 0.25–0.40), `CRITICAL` (PSI > 0.40), with different escalation paths per tier.

3. **Alert fatigue** — over-alerting on noisy metrics causes on-call engineers to ignore pages. Apply rate limiting (max N alerts per hour), cooldown periods, and alert deduplication before triggering a PagerDuty/Slack notification.

4. **Confusing covariate shift with concept drift** — covariate shift is a change in P(X); concept drift is a change in P(Y|X). They need different detection strategies: statistical distribution tests (KS, PSI) for the former, and fresh labeled data accuracy for the latter. Treating them interchangeably leads to wrong remediation.
</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|---------------|-----|
| Monitoring too infrequently for the data velocity | Hourly batch checks miss concept drift that degrades high-velocity models within minutes; by the time the alert fires, thousands of bad predictions have been served | Use sliding-window monitoring with window size matched to data arrival rate; for real-time APIs, check every 100–500 requests |
| Single static alert threshold causing both false positives and missed drift | One threshold cannot simultaneously handle gradual drift and sudden spikes; too sensitive creates alert fatigue, too lenient misses real problems | Use tiered severity levels (INFO / WARNING / CRITICAL) with different escalation paths; calibrate each tier on historical drift data |
| Alert fatigue from correlated feature alarms | When multiple features drift simultaneously (e.g., seasonal shift), each one fires independently, generating dozens of alerts for a single root cause | Group correlated feature alerts with a 5-minute deduplication window; fire one root-cause alert instead of per-feature notifications |
| Confusing covariate shift (data drift) with concept drift | Data drift is P(X) changing; concept drift is P(Y|X) changing; treating them interchangeably leads to retraining on shifted inputs that does not fix degraded decision boundaries | Monitor input distributions with KS/PSI for data drift and monitor model error rate with ground-truth labels for concept drift; each requires a different remediation strategy |
| Using logged prediction confidence as a proxy for accuracy without ground-truth labels | High confidence on drifted inputs is a common failure mode; the model confidently predicts wrong classes on out-of-distribution data | Use confidence as a leading indicator only; always close the feedback loop by collecting delayed ground-truth labels for at least a sample (e.g., 5%) of production requests |
