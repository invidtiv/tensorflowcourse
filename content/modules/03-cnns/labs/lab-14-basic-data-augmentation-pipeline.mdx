Types of Augmentation:

2. Color/Intensity Transformations:

""")

def visualize_augmentations(image, augmentations, num_variations=5):
    """
    Visualize different augmentations on a single image.
    
    Args:
        image: Input image
        augmentations: Dictionary of augmentation functions
        num_variations: Number of variations to show
    """
    
    n_augs = len(augmentations)
    fig, axes = plt.subplots(n_augs, num_variations + 1, 
                            figsize=(3*(num_variations+1), 3*n_augs))
    
    for row_idx, (aug_name, aug_func) in enumerate(augmentations.items()):
        # Show original
        axes[row_idx, 0].imshow(image)
        axes[row_idx, 0].set_title('Original')
        axes[row_idx, 0].axis('off')
        
        # Show augmented versions
        for col_idx in range(num_variations):
            augmented = aug_func(image)
            axes[row_idx, col_idx + 1].imshow(augmented)
            axes[row_idx, col_idx + 1].set_title(f'{aug_name} {col_idx+1}')
            axes[row_idx, col_idx + 1].axis('off')
    
    plt.tight_layout()
    plt.savefig('/mnt/okcomputer/output/lab313_basic_augmentations.png', dpi=150)
    plt.show()
    print("\nBasic augmentations saved to: /mnt/okcomputer/output/lab313_basic_augmentations.png")


# Define augmentation functions
def random_rotation(img, max_angle=30):
    """Random rotation."""
    angle = np.random.uniform(-max_angle, max_angle)
    h, w = img.shape[:2]
    center = (w // 2, h // 2)
    M = cv2.getRotationMatrix2D(center, angle, 1.0)
    return cv2.warpAffine(img, M, (w, h))


def random_shift(img, max_shift=0.2):
    """Random horizontal and vertical shift."""
    h, w = img.shape[:2]
    tx = int(np.random.uniform(-max_shift, max_shift) * w)
    ty = int(np.random.uniform(-max_shift, max_shift) * h)
    M = np.float32([[1, 0, tx], [0, 1, ty]])
    return cv2.warpAffine(img, M, (w, h))


def random_zoom(img, zoom_range=(0.8, 1.2)):
    """Random zoom."""
    h, w = img.shape[:2]
    zoom = np.random.uniform(*zoom_range)
    new_h, new_w = int(h * zoom), int(w * zoom)
    
    resized = cv2.resize(img, (new_w, new_h))
    
    # Crop or pad to original size
    if zoom > 1:  # Zoom in - crop
        start_h = (new_h - h) // 2
        start_w = (new_w - w) // 2
        return resized[start_h:start_h+h, start_w:start_w+w]
    else:  # Zoom out - pad
        pad_h = (h - new_h) // 2
        pad_w = (w - new_w) // 2
        result = np.zeros_like(img)
        result[pad_h:pad_h+new_h, pad_w:pad_w+new_w] = resized
        return result


def random_flip(img):
    """Random horizontal flip."""
    if np.random.random() > 0.5:
        return cv2.flip(img, 1)
    return img


def random_brightness(img, max_delta=30):
    """Random brightness adjustment."""
    delta = np.random.randint(-max_delta, max_delta)
    return np.clip(img.astype(np.int16) + delta, 0, 255).astype(np.uint8)


# Create augmentation dictionary
augmentations = {

    'Shift': lambda img: random_shift(img, 0.2),
    'Zoom': lambda img: random_zoom(img, (0.8, 1.2)),
    'Flip': random_flip,
    'Brightness': lambda img: random_brightness(img, 30)
}

def demonstrate_keras_generator():

    return datagen

def train_with_augmentation(use_augmentation=True, epochs=10):
    """Train CNN with and without augmentation."""
    
    print("\n" + "=" * 60)
    print(f"TRAINING {'WITH' if use_augmentation else 'WITHOUT'} AUGMENTATION")
    print("=" * 60)
    
    # Load data
    (X_train, y_train), (X_test, y_test) = cifar10.load_data()
    X_train = X_train.astype('float32') / 255.0
    X_test = X_test.astype('float32') / 255.0
    y_train = to_categorical(y_train, 10)
    y_test = to_categorical(y_test, 10)
    
    # Use subset for speed
    X_train = X_train[:10000]
    y_train = y_train[:10000]
    
    # Build simple model
    model = models.Sequential([
        layers.Conv2D(32, 3, activation='relu', input_shape=(32, 32, 3)),
        layers.MaxPooling2D(2),
        layers.Conv2D(64, 3, activation='relu'),
        layers.MaxPooling2D(2),
        layers.Conv2D(128, 3, activation='relu'),
        layers.GlobalAveragePooling2D(),
        layers.Dense(128, activation='relu'),
        layers.Dropout(0.5),
        layers.Dense(10, activation='softmax')
    ])
    
    model.compile(optimizer='adam',
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    if use_augmentation:
        # Create generator
        train_datagen = ImageDataGenerator(
            rotation_range=15,
            width_shift_range=0.1,
            height_shift_range=0.1,
            horizontal_flip=True
        )
        
        # Fit generator
        train_datagen.fit(X_train)
        
        # Train with augmentation
        history = model.fit(
            train_datagen.flow(X_train, y_train, batch_size=64),
            steps_per_epoch=len(X_train) // 64,
            epochs=epochs,
            validation_data=(X_test[:2000], y_test[:2000]),
            verbose=1
        )
    else:
        # Train without augmentation
        history = model.fit(X_train, y_train,
                           batch_size=64,
                           epochs=epochs,
                           validation_data=(X_test[:2000], y_test[:2000]),
                           verbose=1)
    
    # Evaluate
    test_loss, test_acc = model.evaluate(X_test, y_test, verbose=0)
    print(f"\nTest accuracy: {test_acc:.4f}")
    
    return history, test_acc


# Train without augmentation
history_no_aug, acc_no_aug = train_with_augmentation(use_augmentation=False, epochs=10)

# Train with augmentation
history_aug, acc_aug = train_with_augmentation(use_augmentation=True, epochs=10)

def analyze_augmentation_intensity():
    """Analyze effect of different augmentation intensities."""
    
    print("\n" + "=" * 60)
    print("AUGMENTATION INTENSITY ANALYSIS")
    print("=" * 60)
    
    intensities = {
        'Light': {'rotation': 10, 'shift': 0.1, 'zoom': 0.1},
        'Medium': {'rotation': 20, 'shift': 0.2, 'zoom': 0.2},
        'Heavy': {'rotation': 40, 'shift': 0.3, 'zoom': 0.3}
    }
    
    results = {}
    
    for name, params in intensities.items():
        print(f"\n--- Testing {name} augmentation ---")
        
        # Load data
        (X_train, y_train), (X_test, y_test) = cifar10.load_data()
        X_train = X_train.astype('float32') / 255.0
        X_test = X_test.astype('float32') / 255.0
        y_train = to_categorical(y_train, 10)
        y_test = to_categorical(y_test, 10)
        
        X_train = X_train[:5000]
        y_train = y_train[:5000]
        
        # Create generator
        datagen = ImageDataGenerator(
            rotation_range=params['rotation'],
            width_shift_range=params['shift'],
            height_shift_range=params['shift'],
            zoom_range=params['zoom'],
            horizontal_flip=True
        )
        
        # Build model
        model = models.Sequential([
            layers.Conv2D(32, 3, activation='relu', input_shape=(32, 32, 3)),
            layers.MaxPooling2D(2),
            layers.Conv2D(64, 3, activation='relu'),
            layers.GlobalAveragePooling2D(),
            layers.Dense(64, activation='relu'),
            layers.Dense(10, activation='softmax')
        ])
        
        model.compile(optimizer='adam', loss='categorical_crossentropy', 
                     metrics=['accuracy'])
        
        # Train
        history = model.fit(
            datagen.flow(X_train, y_train, batch_size=64),
            steps_per_epoch=len(X_train) // 64,
            epochs=5,
            validation_data=(X_test[:2000], y_test[:2000]),
            verbose=0
        )
        
        final_acc = history.history['val_accuracy'][-1]
        results[name] = final_acc
        print(f"  Final accuracy: {final_acc:.4f}")
    
    return results


intensity_results = analyze_augmentation_intensity()

    ax.set_xlabel('Augmentation Intensity')
    ax.set_ylabel('Validation Accuracy')

    ax.grid(True, alpha=0.3, axis='y')
    
    plt.tight_layout()
    plt.savefig('/mnt/okcomputer/output/lab313_intensity_analysis.png', dpi=150)
    plt.show()
    print("\nIntensity analysis saved to: /mnt/okcomputer/output/lab313_intensity_analysis.png")
