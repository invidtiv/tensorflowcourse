class Conv2DLayer:
    """
    A complete convolutional layer with learnable parameters.
    """
    
    def __init__(self, in_channels, out_channels, kernel_size, stride=1, padding=0):
        """
        Initialize the convolutional layer.
        
        Args:
            in_channels: Number of input channels
            out_channels: Number of output channels (filters)
            kernel_size: Size of the convolutional kernel (assumes square)
            stride: Stride for convolution
            padding: Padding size
        """
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # Initialize weights using He initialization
        # He et al. (2015): sqrt(2 / fan_in)
        fan_in = kernel_size * kernel_size * in_channels
        self.weights = np.random.randn(kernel_size, kernel_size, 
                                       in_channels, out_channels) * np.sqrt(2.0 / fan_in)
        
        # Initialize bias to zeros
        self.bias = np.zeros(out_channels)
        
        # Cache for backward pass
        self.cache = None
    
    def forward(self, X):
        """
        Forward pass through the convolutional layer.
        
        Args:
            X: Input of shape (batch_size, H, W, C_in)
        
        Returns:
            Output of shape (batch_size, H_out, W_out, C_out)
        """
        batch_size, H, W, C_in = X.shape
        
        # Add padding
        if self.padding > 0:
            X_padded = np.pad(X, ((0, 0), (self.padding, self.padding), 
                                  (self.padding, self.padding), (0, 0)), 
                            mode='constant', constant_values=0)
        else:
            X_padded = X
        
        # Calculate output dimensions
        H_out = (H + 2 * self.padding - self.kernel_size) // self.stride + 1
        W_out = (W + 2 * self.padding - self.kernel_size) // self.stride + 1
        
        # Initialize output
        output = np.zeros((batch_size, H_out, W_out, self.out_channels))
        
        # Perform convolution for each sample in batch
        for b in range(batch_size):
            for c_out in range(self.out_channels):
                for i in range(H_out):
                    for j in range(W_out):
                        i_start = i * self.stride
                        j_start = j * self.stride
                        
                        # Extract region
                        region = X_padded[b, i_start:i_start + self.kernel_size,
                                         j_start:j_start + self.kernel_size, :]
                        
                        # Convolution operation
                        output[b, i, j, c_out] = np.sum(region * self.weights[:, :, :, c_out])
                
                # Add bias
                output[b, :, :, c_out] += self.bias[c_out]
        
        # Cache for backward pass
        self.cache = (X, X_padded)
        
        return output
    
    def backward(self, dout):
        """
        Backward pass through the convolutional layer.
        
        Args:
            dout: Gradient from next layer (batch_size, H_out, W_out, C_out)
        
        Returns:
            dX: Gradient w.r.t. input
            dW: Gradient w.r.t. weights
            db: Gradient w.r.t. bias
        """
        X, X_padded = self.cache
        batch_size, H, W, C_in = X.shape
        _, H_out, W_out, _ = dout.shape
        
        # Initialize gradients
        dX_padded = np.zeros_like(X_padded)
        dW = np.zeros_like(self.weights)
        db = np.zeros_like(self.bias)
        
        # Compute gradients
        for b in range(batch_size):
            for c_out in range(self.out_channels):
                # Gradient w.r.t. bias
                db[c_out] += np.sum(dout[b, :, :, c_out])
                
                for i in range(H_out):
                    for j in range(W_out):
                        i_start = i * self.stride
                        j_start = j * self.stride
                        
                        region = X_padded[b, i_start:i_start + self.kernel_size,
                                         j_start:j_start + self.kernel_size, :]
                        
                        # Gradient w.r.t. weights
                        dW[:, :, :, c_out] += region * dout[b, i, j, c_out]
                        
                        # Gradient w.r.t. input
                        dX_padded[b, i_start:i_start + self.kernel_size,
                                 j_start:j_start + self.kernel_size, :] += \
                            self.weights[:, :, :, c_out] * dout[b, i, j, c_out]
        
        # Remove padding from dX if needed
        if self.padding > 0:
            dX = dX_padded[:, self.padding:-self.padding, 
                          self.padding:-self.padding, :]
        else:
            dX = dX_padded
        
        return dX, dW, db

def compare_with_keras():
    """Compare our implementation with Keras Conv2D."""

    # Keras implementation

    _ = keras_conv(X)

    keras_output = keras_conv(X).numpy()
    
    print(f"\nOur output shape: {our_output.shape}")
    print(f"Keras output shape: {keras_output.shape}")
    
    diff = np.abs(our_output - keras_output)
    print(f"\nMax absolute difference: {np.max(diff):.10f}")
    print(f"Mean absolute difference: {np.mean(diff):.10f}")

        print("\n✓ Forward pass matches Keras!")

def calculate_receptive_field(layer_configs):
    """
    Calculate the receptive field size for a CNN architecture.
    
    Args:
        layer_configs: List of dicts with 'kernel_size', 'stride', 'dilation'
    
    Returns:
        receptive_field: Size of the receptive field
    """
    rf = 1  # Initial receptive field
    stride_product = 1
    
    for config in layer_configs:
        k = config.get('kernel_size', 3)
        s = config.get('stride', 1)
        d = config.get('dilation', 1)
        
        # Effective kernel size with dilation
        effective_k = k + (k - 1) * (d - 1)
        
        # Update receptive field
        rf = rf + (effective_k - 1) * stride_product
        stride_product *= s
    
    return rf

def count_parameters(in_channels, out_channels, kernel_size, use_bias=True):
    """Count parameters in a convolutional layer."""
    weights = kernel_size * kernel_size * in_channels * out_channels
    bias = out_channels if use_bias else 0
    return weights + bias

def grouped_convolution(X, weights, groups=1):
    """
    Perform grouped convolution.
    
    Args:
        X: Input (batch, H, W, C_in)
        weights: Weights (k, k, C_in/groups, C_out)
        groups: Number of groups
    
    Returns:
        Output (batch, H_out, W_out, C_out)
    """
    batch, H, W, C_in = X.shape
    k, _, _, C_out = weights.shape
    
    assert C_in % groups == 0
    assert C_out % groups == 0
    
    channels_per_group_in = C_in // groups
    channels_per_group_out = C_out // groups
    
    outputs = []
    
    for g in range(groups):
        # Extract group input
        X_group = X[:, :, :, g*channels_per_group_in:(g+1)*channels_per_group_in]
        
        # Extract group weights
        W_group = weights[:, :, :, g*channels_per_group_out:(g+1)*channels_per_group_out]
        
        # Perform convolution for this group
        # (Simplified - full implementation would use our conv2d function)
        group_output = np.zeros((batch, H-k+1, W-k+1, channels_per_group_out))
        
        for b in range(batch):
            for c in range(channels_per_group_out):
                for i in range(H-k+1):
                    for j in range(W-k+1):
                        region = X_group[b, i:i+k, j:j+k, :]
                        group_output[b, i, j, c] = np.sum(region * W_group[:, :, :, c])
        
        outputs.append(group_output)
    
    return np.concatenate(outputs, axis=-1)
