def load_cifar10():
    """Load and preprocess CIFAR-10 dataset."""
    
    print("\n" + "=" * 60)
    print("LOADING CIFAR-10")
    print("=" * 60)
    
    # Load data
    (X_train, y_train), (X_test, y_test) = cifar10.load_data()
    
    print(f"\nTraining data: {X_train.shape}")
    print(f"Test data: {X_test.shape}")
    
    # Normalize pixel values to [0, 1]
    X_train = X_train.astype('float32') / 255.0
    X_test = X_test.astype('float32') / 255.0
    
    # Convert labels to categorical
    y_train = to_categorical(y_train, 10)
    y_test = to_categorical(y_test, 10)
    
    # Class names
    class_names = ['airplane', 'automobile', 'bird', 'cat', 'deer',
                   'dog', 'frog', 'horse', 'ship', 'truck']
    
    print(f"\nClass names: {class_names}")
    print(f"Training labels shape: {y_train.shape}")
    
    return (X_train, y_train), (X_test, y_test), class_names


(X_train, y_train), (X_test, y_test), class_names = load_cifar10()

# Section 4: Build Custom CNN Architecture

def build_custom_cnn(input_shape=(32, 32, 3), num_classes=10):
    """
    Build a custom CNN optimized for CIFAR-10.
    
    Architecture principles:
    - Start with smaller filters, increase depth
    - Use batch normalization for stability
    - Add dropout for regularization
    - Global average pooling instead of flatten
    """
    
    print("\n" + "=" * 60)
    print("BUILDING CUSTOM CNN")
    print("=" * 60)
    
    model = models.Sequential([
        # Block 1
        layers.Conv2D(32, 3, padding='same', input_shape=input_shape),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(32, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        layers.Dropout(0.25),
        
        # Block 2
        layers.Conv2D(64, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(64, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        layers.Dropout(0.25),
        
        # Block 3
        layers.Conv2D(128, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(128, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        layers.Dropout(0.25),
        
        # Classifier
        layers.GlobalAveragePooling2D(),
        layers.Dense(256),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Dropout(0.5),
        layers.Dense(num_classes, activation='softmax')
    ])
    
    return model


# Build model
cnn_model = build_custom_cnn()
cnn_model.summary()

def create_data_generator():
    """Create data augmentation generator."""
    
    print("\n" + "=" * 60)
    print("DATA AUGMENTATION SETUP")
    print("=" * 60)
    
    datagen = ImageDataGenerator(
        rotation_range=15,      # Random rotation
        width_shift_range=0.1,  # Horizontal shift
        height_shift_range=0.1, # Vertical shift
        horizontal_flip=True,   # Random horizontal flip
        zoom_range=0.1,         # Random zoom
        fill_mode='nearest'     # Fill mode for new pixels
    )
    
    print("\nAugmentation parameters:")
    print("  - Rotation range: ±15 degrees")
    print("  - Width shift: ±10%")
    print("  - Height shift: ±10%")
    print("  - Horizontal flip: True")
    print("  - Zoom range: ±10%")
    
    return datagen


datagen = create_data_generator()

# Fit generator on training data
datagen.fit(X_train)

# Section 6: Compile and Train
# =============================================================================

def compile_and_train(model, X_train, y_train, X_test, y_test, 
                      datagen, epochs=50):
    """Compile and train the model with callbacks."""
    
    print("\n" + "=" * 60)
    print("COMPILING AND TRAINING")
    print("=" * 60)
    
    # Compile
    model.compile(optimizer=optimizers.Adam(learning_rate=0.001),
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    # Callbacks
    cb = [
        callbacks.EarlyStopping(patience=10, restore_best_weights=True),
        callbacks.ReduceLROnPlateau(factor=0.5, patience=5, min_lr=1e-6),
        callbacks.ModelCheckpoint('/mnt/okcomputer/output/cifar10_best_model.h5',
                                  save_best_only=True)
    ]
    
    # Train with augmentation
    start_time = time.time()
    
    history = model.fit(
        datagen.flow(X_train, y_train, batch_size=64),
        steps_per_epoch=len(X_train) // 64,
        epochs=epochs,
        validation_data=(X_test, y_test),
        callbacks=cb,
        verbose=1
    )
    
    training_time = time.time() - start_time
    print(f"\nTraining completed in {training_time:.2f} seconds")
    
    return history


# Train model (reduced epochs for demo)

def evaluate_model(model, X_test, y_test, class_names):
    """Evaluate model and show detailed metrics."""
    
    print("\n" + "=" * 60)
    print("MODEL EVALUATION")
    print("=" * 60)
    
    # Overall metrics
    loss, accuracy = model.evaluate(X_test, y_test, verbose=0)
    print(f"\nTest Loss: {loss:.4f}")
    print(f"Test Accuracy: {accuracy:.4f}")
    
    # Per-class accuracy
    predictions = model.predict(X_test)
    pred_labels = np.argmax(predictions, axis=1)
    true_labels = np.argmax(y_test, axis=1)
    
    print("\nPer-class accuracy:")
    for i, name in enumerate(class_names):
        mask = true_labels == i
        class_acc = np.mean(pred_labels[mask] == i)
        print(f"  {name:12s}: {class_acc:.4f}")
    
    return pred_labels, true_labels

<Callout type="tip" title="Expected Output">
After ~30 epochs on CIFAR-10 with a modest custom CNN (3 conv blocks + GAP + Dense(10)):

- **Test accuracy**: 72–78% without augmentation, 80–86% with `RandomFlip` + `RandomTranslation`, 87–90% with Mixup/CutMix + cosine LR.
- **Train accuracy**: 92–97% (higher than test — a ~10–15% gap is normal for CIFAR-10 without aggressive regularization).
- **Parameter count**: 200k–600k for a reasonable hand-rolled CNN. If `model.summary()` reports >2M params, you are almost certainly using `Flatten() → Dense(512)` after the last conv — swap for `GlobalAveragePooling2D`.
- **Loss curves**: smooth decay for both train and val in the first 5–10 epochs; val loss should plateau around epoch 15–20 before train loss does. A *rising* val loss with falling train loss after epoch ~8 is the signature of overfitting — add Dropout(0.3–0.5) in the dense block.
- **Per-class accuracy**: expect lower numbers (55–70%) for visually similar classes — cat vs. dog, deer vs. horse, car vs. truck. Classes like "ship" and "frog" usually come in 10+ points higher.
- **Training time**: 8–15 min for 30 epochs on a free Colab T4; 40–80 min on a modern laptop CPU.

**Diagnostic tree if numbers are off:**

- **Test acc < 50%** → pixel normalization missing (`/255.0`) or labels not one-hot for `categorical_crossentropy` (use `sparse_categorical_crossentropy` with integer labels instead).
- **Train acc ~100%, test acc < 65%** → severe overfitting — add augmentation and Dropout before blaming the architecture.
- **Loss goes to NaN** → lr too high for Adam (try 1e-3 → 1e-4) or un-clipped gradients after a bad batch.
- **One class at ~0% accuracy** → label encoding mismatch (check `np.unique(y_train)` returns 0–9).
</Callout>

## Common Pitfalls

<Callout type="warning" title="Common Pitfalls">
- **Not normalizing pixel values** — CIFAR-10 images are uint8 [0, 255]. Forgetting `x_train = x_train.astype('float32') / 255.0` causes training to diverge or converge very slowly because gradients scale with input magnitude.
- **Too many FC parameters after flattening** — A 32×32 image after several conv/pool layers can still produce a large flatten vector. Use Global Average Pooling instead of Flatten → Dense(512) to reduce parameters by 10–50×.
- **No data augmentation** — CIFAR-10 has only 50K training images. Without augmentation (random flips, shifts, rotations), even a small CNN will overfit. `tf.keras.layers.RandomFlip` and `RandomTranslation` are the minimum.
- **Learning rate too high for deep models** — Starting at lr=0.01 with Adam often works for shallow CNNs but diverges for deeper architectures. Try lr=0.001 or use a warmup schedule.
</Callout>
