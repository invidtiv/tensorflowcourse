print("=" * 60)

print("""
Grad-CAM (Selvaraju et al., 2017):
==================================

Purpose: Visualize where the CNN "looks" to make predictions.

How it works:
1. Forward pass to get class prediction
2. Compute gradient of class score w.r.t. feature maps
3. Global average pooling of gradients → importance weights
4. Weighted combination of feature maps
5. ReLU to keep only positive contributions

Formula:
  L_gradcam = ReLU(Σ_k α_k * A^k)
  
  where α_k = (1/Z) * Σ_i Σ_j (∂y^c / ∂A^k_ij)
  
  A^k = k-th feature map
  y^c = class score for class c

Applications:
- Model interpretability
- Debugging predictions
- Weakly-supervised localization
""")


# =============================================================================
# Section 2: Grad-CAM Implementation
# =============================================================================

class GradCAM:
    """Grad-CAM implementation for CNN visualization."""
    
    def __init__(self, model, layer_name):
        """
        Initialize GradCAM.
        
        Args:
            model: CNN model
            layer_name: Name of the last convolutional layer
        """
        self.model = model
        self.layer_name = layer_name
        
        # Create model that outputs both predictions and conv layer output
        self.grad_model = models.Model(
            inputs=model.inputs,
            outputs=[model.get_layer(layer_name).output, model.output]
        )
    
    def compute_heatmap(self, img_array, class_index=None):
        """
        Compute Grad-CAM heatmap.
        
        Args:
            img_array: Input image array (batch_size, H, W, C)
            class_index: Target class index (None for predicted class)
        
        Returns:
            heatmap: Grad-CAM heatmap
        """
        # Record operations for gradient computation
        with tf.GradientTape() as tape:
            conv_outputs, predictions = self.grad_model(img_array)
            
            if class_index is None:
                class_index = tf.argmax(predictions[0])
            
            # Get the score for the target class
            class_score = predictions[:, class_index]
        
        # Compute gradients
        grads = tape.gradient(class_score, conv_outputs)
        
        # Global average pooling of gradients
        pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
        
        # Weight feature maps by gradients
        conv_outputs = conv_outputs[0]
        heatmap = tf.reduce_sum(tf.multiply(pooled_grads, conv_outputs), axis=-1)
        
        # Apply ReLU
        heatmap = np.maximum(heatmap, 0)
        
        # Normalize
        heatmap = heatmap / (np.max(heatmap) + 1e-8)
        
        return heatmap.numpy()
    
    def overlay_heatmap(self, img, heatmap, alpha=0.4, colormap=cv2.COLORMAP_JET):
        """
        Overlay heatmap on original image.
        
        Args:
            img: Original image (H, W, C)
            heatmap: Grad-CAM heatmap
            alpha: Transparency
            colormap: OpenCV colormap
        
        Returns:
            Overlayed image
        """
        # Resize heatmap to match image
        heatmap = cv2.resize(heatmap, (img.shape[1], img.shape[0]))
        
        # Convert to uint8
        heatmap = np.uint8(255 * heatmap)
        
        # Apply colormap
        heatmap = cv2.applyColorMap(heatmap, colormap)
        
        # Convert image to uint8 if needed
        if img.max() <= 1.0:
            img = np.uint8(255 * img)
        
        # Overlay
        overlayed = cv2.addWeighted(heatmap, alpha, img, 1 - alpha, 0)
        
        return overlayed

# Section 8: Guided Grad-CAM
# =============================================================================

def guided_backprop(model, img_array, layer_name):
    """
    Compute guided backpropagation.
    
    Guided backprop only propagates positive gradients through ReLU.
    """
    
    grad_model = models.Model(
        inputs=model.inputs,
        outputs=model.get_layer(layer_name).output
    )
    
    with tf.GradientTape() as tape:
        tape.watch(img_array)
        conv_output = grad_model(img_array)
        
        # Get the max activation
        max_activation = tf.reduce_max(conv_output, axis=(1, 2, 3))
    
    grads = tape.gradient(max_activation, img_array)
    
    # Guided backprop: only positive gradients
    guided_grads = tf.nn.relu(grads)
    
    return guided_grads[0].numpy()


def create_guided_gradcam(grad_cam, model, img_array, original_img):
    """Create Guided Grad-CAM (Guided Backprop * Grad-CAM)."""
    
    # Get Grad-CAM
    heatmap = grad_cam.compute_heatmap(img_array)
    heatmap_resized = cv2.resize(heatmap, (original_img.shape[1], original_img.shape[0]))
    
    # Get guided backprop
    guided_bp = guided_backprop(model, img_array, 'block5_conv3')
    
    # Convert to grayscale
    guided_bp_gray = np.mean(guided_bp, axis=-1)
    
    # Multiply with Grad-CAM
    guided_gradcam = guided_bp_gray * heatmap_resized
    
    # Normalize
    guided_gradcam = (guided_gradcam - guided_gradcam.min()) / (guided_gradcam.max() - guided_gradcam.min() + 1e-8)
    
    # Visualize
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))
    
    axes[0].imshow(heatmap, cmap='jet')
    axes[0].set_title('Grad-CAM')
    axes[0].axis('off')
    
    axes[1].imshow(guided_bp_gray, cmap='gray')
    axes[1].set_title('Guided Backprop')
    axes[1].axis('off')
    
    axes[2].imshow(guided_gradcam, cmap='jet')
    axes[2].set_title('Guided Grad-CAM')
    axes[2].axis('off')
    
    plt.tight_layout()
    plt.savefig('/mnt/okcomputer/output/lab39_guided_gradcam.png', dpi=150)
    plt.show()
    print("\nGuided Grad-CAM saved to: /mnt/okcomputer/output/lab39_guided_gradcam.png")

        f"Raw heatmap shape {raw_hm.shape[:2]} != conv layer {(expected_h, expected_w)}"

## Expected Output

When you run this lab, you should observe the following results:

**For VGG16 with `block5_conv3`:**
- Raw heatmap shape: **(14, 14)** — matching the spatial dimensions of VGG16's last convolutional layer.
- Top-class prediction confidence: **> 0.8** for recognizable objects (or > 0.5 for synthetic test images).
- Overlay visualization: hot regions (red/yellow in JET colormap) concentrated on the **main object body/head**, NOT scattered across the background.
- Heatmap is relatively coarse due to the layer's receptive field, but semantically coherent.

**For ResNet50 with `conv5_block3_out`:**
- Raw heatmap shape: **(7, 7)** — smaller spatial dimensions due to ResNet50's deeper stride architecture.
- Heatmap is **sharper and more localized** than VGG16's, with tighter focus on discriminative regions.
- The smaller spatial size trades resolution for computational efficiency.

**Validation Assertions:**
- ✅ Heatmap values in range [0, 1] after ReLU and normalization.
- ✅ Heatmap sum > 1e-3 (not all zeros — indicates gradients are flowing).
- ✅ Raw heatmap spatial shape matches the target convolutional layer's output shape.
- ✅ Different class indices produce meaningfully different heatmaps (cosine similarity < 0.98).

**Common Failure Modes & How to Spot Them:**

- **Heatmap is all zeros or uniform:** Gradient is `None` (you forgot `tape.watch(conv_output)` on a non-model output), OR you're using softmax probabilities instead of logits (gradients vanish after saturation).
- **Heatmap highlights background instead of the object:** Wrong target layer selected — you picked an early conv layer (e.g., `block1_conv2`) which lacks semantic information. Always use the *last* convolutional layer before pooling.
- **Heatmap shape mismatches image size before overlay:** This is expected — you must resize the heatmap to match the original image dimensions before visualization. The raw heatmap should match the conv layer's output shape.

## Common Pitfalls

1. **Picking the wrong target layer.** Grad-CAM is designed for the *last* convolutional layer (just before global pooling / flatten). Using an early conv layer gives a high-resolution but semantically meaningless heatmap; using a dense layer breaks the spatial interpretation entirely. For VGG16 use `block5_conv3`, for ResNet50 use `conv5_block3_out`.
2. **Feeding unpreprocessed images.** VGG16/ResNet50 expect `preprocess_input` (BGR channel order and mean-subtraction for VGG). If you skip this, gradients are computed against out-of-distribution activations and heatmaps are noise.
3. **Forgetting `tape.watch(conv_output)`.** When the feature map is a non-trainable intermediate tensor, `GradientTape` will silently return `None` for its gradient unless you either (a) capture it from a model whose output *is* that tensor, or (b) call `tape.watch(conv_output)` explicitly. A `None` gradient → `TypeError` or all-zero heatmap.
4. **Using softmax probabilities instead of logits for the target score.** Once softmax saturates, gradients vanish and Grad-CAM produces flat heatmaps. Either extract the pre-softmax logit, or use the top-class probability from a model whose last layer has `activation=None` for visualization.
5. **Resizing with the wrong interpolation.** Using `cv2.INTER_NEAREST` on a 14×14 heatmap upscaled to 224×224 produces blocky visualizations that misrepresent the underlying attention. Always use `cv2.INTER_CUBIC` or `INTER_LINEAR` for Grad-CAM overlays.

## Stretch Goals
