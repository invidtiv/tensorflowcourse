ResNet (Residual Networks) - He et al. (2015):
==============================================

Key Innovation: Skip Connections (Residual Connections)

Standard block:    x → Conv → BN → ReLU → Conv → BN → F(x)
Residual block:    x → Conv → BN → ReLU → Conv → BN → + → ReLU
                     ↓______________________________↑

The residual mapping: F(x) = H(x) - x
Learning: F(x) + x instead of H(x)

Benefits:
1. Solves vanishing gradient problem
2. Enables training of very deep networks (1000+ layers)
3. Easier optimization (learning identity is easier)

Types of Blocks:
- Basic Block: Two 3x3 convolutions (ResNet-18, ResNet-34)
- Bottleneck Block: 1x1 → 3x3 → 1x1 (ResNet-50, ResNet-101, ResNet-152)
""")


# =============================================================================
# Section 2: Basic Residual Block
# =============================================================================

def basic_residual_block(x, filters, stride=1, name='res_block'):
    """
    Basic residual block with two 3x3 convolutions.
    
    Args:
        x: Input tensor
        filters: Number of output filters
        stride: Stride for first convolution
        name: Block name
    
    Returns:
        Output tensor
    """
    shortcut = x
    
    # First convolution
    x = layers.Conv2D(filters, 3, strides=stride, padding='same',
                     use_bias=False, name=f'{name}_conv1')(x)
    x = layers.BatchNormalization(name=f'{name}_bn1')(x)
    x = layers.Activation('relu', name=f'{name}_relu1')(x)
    
    # Second convolution
    x = layers.Conv2D(filters, 3, strides=1, padding='same',
                     use_bias=False, name=f'{name}_conv2')(x)
    x = layers.BatchNormalization(name=f'{name}_bn2')(x)
    
    # Shortcut connection
    if stride != 1 or shortcut.shape[-1] != filters:
        # Projection shortcut
        shortcut = layers.Conv2D(filters, 1, strides=stride, padding='same',
                                use_bias=False, name=f'{name}_shortcut')(shortcut)
        shortcut = layers.BatchNormalization(name=f'{name}_shortcut_bn')(shortcut)
    
    # Add shortcut
    x = layers.Add(name=f'{name}_add')([x, shortcut])
    x = layers.Activation('relu', name=f'{name}_relu2')(x)
    
    return x


# =============================================================================
# Section 3: Bottleneck Block
# =============================================================================

def bottleneck_block(x, filters, stride=1, name='bottleneck'):
    """
    Bottleneck residual block: 1x1 → 3x3 → 1x1
    
    Args:
        x: Input tensor
        filters: Number of output filters
        stride: Stride for middle convolution
        name: Block name
    
    Returns:
        Output tensor
    """
    shortcut = x
    
    # 1x1 convolution (reduce)
    x = layers.Conv2D(filters // 4, 1, strides=1, padding='same',
                     use_bias=False, name=f'{name}_conv1')(x)
    x = layers.BatchNormalization(name=f'{name}_bn1')(x)
    x = layers.Activation('relu', name=f'{name}_relu1')(x)
    
    # 3x3 convolution (process)
    x = layers.Conv2D(filters // 4, 3, strides=stride, padding='same',
                     use_bias=False, name=f'{name}_conv2')(x)
    x = layers.BatchNormalization(name=f'{name}_bn2')(x)
    x = layers.Activation('relu', name=f'{name}_relu2')(x)
    
    # 1x1 convolution (expand)
    x = layers.Conv2D(filters, 1, strides=1, padding='same',
                     use_bias=False, name=f'{name}_conv3')(x)
    x = layers.BatchNormalization(name=f'{name}_bn3')(x)
    
    # Shortcut connection
    if stride != 1 or shortcut.shape[-1] != filters:
        shortcut = layers.Conv2D(filters, 1, strides=stride, padding='same',
                                use_bias=False, name=f'{name}_shortcut')(shortcut)
        shortcut = layers.BatchNormalization(name=f'{name}_shortcut_bn')(shortcut)
    
    # Add shortcut
    x = layers.Add(name=f'{name}_add')([x, shortcut])
    x = layers.Activation('relu', name=f'{name}_relu3')(x)
    
    return x


# =============================================================================
# Section 4: Build ResNet-18
# =============================================================================

def build_resnet18(input_shape=(32, 32, 3), num_classes=10):
    """
    Build ResNet-18 architecture.
    
    Architecture:
    - Conv1: 7x7, 64, stride 2
    - Pool: 3x3, stride 2
    - Conv2_x: 2 basic blocks, 64 filters
    - Conv3_x: 2 basic blocks, 128 filters, stride 2
    - Conv4_x: 2 basic blocks, 256 filters, stride 2
    - Conv5_x: 2 basic blocks, 512 filters, stride 2
    - Global Average Pooling
    - FC: num_classes
    """
    
    print("\n" + "=" * 60)
    print("BUILDING ResNet-18")
    print("=" * 60)
    
    inputs = layers.Input(shape=input_shape, name='input')
    
    # Initial convolution
    x = layers.Conv2D(64, 3, strides=1, padding='same',
                     use_bias=False, name='conv1')(inputs)
    x = layers.BatchNormalization(name='bn1')(x)
    x = layers.Activation('relu', name='relu1')(x)
    # Note: No maxpool for CIFAR-10 (small images)
    
    # Conv2_x - 2 blocks, 64 filters
    x = basic_residual_block(x, 64, stride=1, name='conv2_1')
    x = basic_residual_block(x, 64, stride=1, name='conv2_2')
    
    # Conv3_x - 2 blocks, 128 filters
    x = basic_residual_block(x, 128, stride=2, name='conv3_1')
    x = basic_residual_block(x, 128, stride=1, name='conv3_2')
    
    # Conv4_x - 2 blocks, 256 filters
    x = basic_residual_block(x, 256, stride=2, name='conv4_1')
    x = basic_residual_block(x, 256, stride=1, name='conv4_2')
    
    # Conv5_x - 2 blocks, 512 filters
    x = basic_residual_block(x, 512, stride=2, name='conv5_1')
    x = basic_residual_block(x, 512, stride=1, name='conv5_2')
    
    # Global average pooling
    x = layers.GlobalAveragePooling2D(name='global_avg_pool')(x)
    
    # Fully connected layer
    outputs = layers.Dense(num_classes, activation='softmax', name='output')(x)
    
    model = models.Model(inputs, outputs, name='ResNet18')
    
    return model


# Build ResNet-18
resnet18 = build_resnet18()
resnet18.summary()

def build_resnet50(input_shape=(32, 32, 3), num_classes=10):
    """
    Build ResNet-50 architecture with bottleneck blocks.
    
    Architecture:
    - Conv1: 7x7, 64, stride 2
    - Conv2_x: 3 bottleneck blocks, 256 filters
    - Conv3_x: 4 bottleneck blocks, 512 filters, stride 2
    - Conv4_x: 6 bottleneck blocks, 1024 filters, stride 2
    - Conv5_x: 3 bottleneck blocks, 2048 filters, stride 2
    """
    
    print("\n" + "=" * 60)
    print("BUILDING ResNet-50")
    print("=" * 60)
    
    inputs = layers.Input(shape=input_shape, name='input')
    
    # Initial convolution
    x = layers.Conv2D(64, 3, strides=1, padding='same',
                     use_bias=False, name='conv1')(inputs)
    x = layers.BatchNormalization(name='bn1')(x)
    x = layers.Activation('relu', name='relu1')(x)
    
    # Conv2_x - 3 bottleneck blocks, 256 filters
    x = bottleneck_block(x, 256, stride=1, name='conv2_1')
    x = bottleneck_block(x, 256, stride=1, name='conv2_2')
    x = bottleneck_block(x, 256, stride=1, name='conv2_3')
    
    # Conv3_x - 4 bottleneck blocks, 512 filters
    x = bottleneck_block(x, 512, stride=2, name='conv3_1')
    x = bottleneck_block(x, 512, stride=1, name='conv3_2')
    x = bottleneck_block(x, 512, stride=1, name='conv3_3')
    x = bottleneck_block(x, 512, stride=1, name='conv3_4')
    
    # Conv4_x - 6 bottleneck blocks, 1024 filters
    x = bottleneck_block(x, 1024, stride=2, name='conv4_1')
    x = bottleneck_block(x, 1024, stride=1, name='conv4_2')
    x = bottleneck_block(x, 1024, stride=1, name='conv4_3')
    x = bottleneck_block(x, 1024, stride=1, name='conv4_4')
    x = bottleneck_block(x, 1024, stride=1, name='conv4_5')
    x = bottleneck_block(x, 1024, stride=1, name='conv4_6')
    
    # Conv5_x - 3 bottleneck blocks, 2048 filters
    x = bottleneck_block(x, 2048, stride=2, name='conv5_1')
    x = bottleneck_block(x, 2048, stride=1, name='conv5_2')
    x = bottleneck_block(x, 2048, stride=1, name='conv5_3')
    
    # Global average pooling
    x = layers.GlobalAveragePooling2D(name='global_avg_pool')(x)
    
    # Fully connected layer
    outputs = layers.Dense(num_classes, activation='softmax', name='output')(x)
    
    model = models.Model(inputs, outputs, name='ResNet50')
    
    return model


# Build ResNet-50
resnet50 = build_resnet50()
print(f"\nResNet-50 total parameters: {resnet50.count_params():,}")

def train_resnet(model, X_train, y_train, X_test, y_test, epochs=20):
    """Train ResNet model."""
    
    print("\n" + "=" * 60)
    print(f"TRAINING {model.name}")
    print("=" * 60)
    
    # Compile
    model.compile(optimizer=optimizers.Adam(learning_rate=0.001),
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    # Callbacks
    cb = [
        keras.callbacks.EarlyStopping(patience=10, restore_best_weights=True),
        keras.callbacks.ReduceLROnPlateau(factor=0.5, patience=5, min_lr=1e-6)
    ]
    
    # Train
    start_time = time.time()
    
    history = model.fit(X_train, y_train,
                       batch_size=128,
                       epochs=epochs,
                       validation_data=(X_test, y_test),
                       callbacks=cb,
                       verbose=1)
    
    training_time = time.time() - start_time
    print(f"\nTraining completed in {training_time:.2f} seconds")
    
    # Evaluate
    test_loss, test_acc = model.evaluate(X_test, y_test, verbose=0)
    print(f"Test accuracy: {test_acc:.4f}")
    
    return history


# Load CIFAR-10

def build_plain_cnn(input_shape=(32, 32, 3), num_classes=10):
    """Build a plain CNN without residual connections."""
    
    model = models.Sequential([
        layers.Conv2D(64, 3, padding='same', input_shape=input_shape),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(64, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        
        layers.Conv2D(128, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(128, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        
        layers.Conv2D(256, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(256, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        
        layers.Conv2D(512, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.Conv2D(512, 3, padding='same'),
        layers.BatchNormalization(),
        layers.Activation('relu'),
        layers.MaxPooling2D(2),
        
        layers.GlobalAveragePooling2D(),
        layers.Dense(num_classes, activation='softmax')
    ])
    
    return model
