def create_feature_extractor(model, layer_names):
    """
    Create a model that outputs features from specified layers.
    
    Args:
        model: Base model
        layer_names: List of layer names to extract features from
    
    Returns:
        Feature extraction model
    """
    
    outputs = [model.get_layer(name).output for name in layer_names]
    feature_model = models.Model(inputs=model.input, outputs=outputs)
    
    return feature_model


# Select layers for visualization
layer_names = ['block1_conv1', 'block1_conv2', 
               'block2_conv1', 'block2_conv2',
               'block3_conv1', 'block3_conv3',
               'block4_conv1', 'block4_conv3',

def prepare_sample_images():
    """Prepare sample images for visualization."""
    
    # Load CIFAR-10

    # Select one image per class
    class_names = ['airplane', 'car', 'bird', 'cat', 'deer',
                   'dog', 'frog', 'horse', 'ship', 'truck']
    
    sample_images = []
    for class_idx in range(10):
        idx = np.where(y_train == class_idx)[0][0]

    return np.array(sample_images), class_names


sample_images, class_names = prepare_sample_images()

def visualize_layer_features(feature_model, image, layer_names, max_features=64):
    """
    Visualize features from all layers for a single image.
    
    Args:
        feature_model: Feature extraction model
        image: Input image
        layer_names: Names of layers to visualize
        max_features: Maximum number of features to show per layer
    """
    
    # Extract features
    features = feature_model.predict(np.expand_dims(image, axis=0))
    
    # Create figure with subplots for each layer
    n_layers = len(layer_names)
    fig, axes = plt.subplots(n_layers, 1, figsize=(20, 4*n_layers))
    
    if n_layers == 1:
        axes = [axes]
    
    for idx, (layer_name, feature) in enumerate(zip(layer_names, features)):
        feature = feature[0]  # Remove batch dimension
        n_features = min(max_features, feature.shape[-1])
        
        # Create grid of feature maps
        size = feature.shape[0]
        n_cols = int(np.ceil(np.sqrt(n_features)))
        n_rows = int(np.ceil(n_features / n_cols))
        
        display_grid = np.zeros((n_rows * size, n_cols * size))
        
        for i in range(n_features):
            row = i // n_cols
            col = i % n_cols
            
            feature_map = feature[:, :, i]
            # Normalize
            feature_map = (feature_map - feature_map.min()) / (feature_map.max() - feature_map.min() + 1e-8)
            
            display_grid[row*size:(row+1)*size, col*size:(col+1)*size] = feature_map
        
        axes[idx].imshow(display_grid, aspect='auto', cmap='viridis')
        axes[idx].set_title(f'{layer_name} - Shape: {feature.shape}', fontsize=12)
        axes[idx].axis('off')
    
    plt.tight_layout()
    plt.savefig('/mnt/okcomputer/output/lab38_all_layers_features.png', dpi=150)
    plt.show()
    print("\nAll layers features saved to: /mnt/okcomputer/output/lab38_all_layers_features.png")

def visualize_feature_evolution(feature_model, image, selected_layers):
    """
    Show how features evolve through selected layers.
    """
    
    features = feature_model.predict(np.expand_dims(image, axis=0))
    
    fig, axes = plt.subplots(2, len(selected_layers), figsize=(4*len(selected_layers), 8))
    
    for idx, (layer_name, feature) in enumerate(zip(selected_layers, features)):
        feature = feature[0]
        
        # Show average activation
        avg_activation = np.mean(feature, axis=-1)
        axes[0, idx].imshow(avg_activation, cmap='hot')
        axes[0, idx].set_title(f'{layer_name}\n(Avg)', fontsize=10)
        axes[0, idx].axis('off')
        
        # Show max activation
        max_activation = np.max(feature, axis=-1)
        axes[1, idx].imshow(max_activation, cmap='hot')
        axes[1, idx].set_title(f'{layer_name}\n(Max)', fontsize=10)
        axes[1, idx].axis('off')
    
    plt.tight_layout()
    plt.savefig('/mnt/okcomputer/output/lab38_feature_evolution.png', dpi=150)
    plt.show()
    print("\nFeature evolution saved to: /mnt/okcomputer/output/lab38_feature_evolution.png")

def analyze_feature_statistics(feature_model, images, layer_names):
    """
    Analyze statistics of features across layers.
    """
    
    print("\n" + "=" * 60)
    print("FEATURE STATISTICS ANALYSIS")
    print("=" * 60)
    
    features_list = feature_model.predict(images)
    
    stats_data = []
    
    for layer_name, features in zip(layer_names, features_list):
        # Calculate statistics
        mean_activation = np.mean(features)
        std_activation = np.std(features)
        sparsity = np.mean(features < 0.01)  # Percentage of near-zero activations
        max_activation = np.max(features)
        
        stats_data.append({
            'layer': layer_name,
            'mean': mean_activation,
            'std': std_activation,
            'sparsity': sparsity,
            'max': max_activation
        })
        
        print(f"\n{layer_name}:")
        print(f"  Mean activation: {mean_activation:.4f}")
        print(f"  Std activation: {std_activation:.4f}")
        print(f"  Sparsity (% near-zero): {sparsity*100:.2f}%")
        print(f"  Max activation: {max_activation:.4f}")
    
    return stats_data


stats = analyze_feature_statistics(feature_model, sample_images[:5], layer_names)

    axes[0, 1].set_xlabel('Layer')

def analyze_class_specific_features(feature_model, images, labels, layer_name):

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| `InvalidArgumentError` when building multi-output model | Sharing a layer output node that was already used in the base model graph causes duplicate node references | Use `tf.keras.Model(inputs=base.input, outputs=[...])` on the *base* model directly, not a re-instantiated copy |
| Feature maps look identical for early and late layers | All channels displayed at the same global color scale — small late-layer values appear blank | Normalize each feature map to `[0, 1]` independently before display; use `vmin/vmax=None` in `imshow` |
| Memory error when extracting all 13 VGG16 conv layers at once | 13 output tensors × batch size × spatial dims × 512 channels exceeds RAM for large batches | Extract one image at a time (`batch_size=1`) or extract one layer group at a time |
| VGG16 feature maps on 32×32 CIFAR images collapse to 1×1 after block5 | VGG16 was designed for 224×224 inputs; at 32×32 the spatial resolution reaches 1×1 after three pooling layers | Either up-sample input to ≥96×96, or stop extracting after block3 for 32×32 inputs |
| `model.predict` returns all-zero activations for custom layer names | The layer name in the extraction list doesn't match what `model.layers` actually contains | Print `{l.name: l.output_shape for l in model.layers}` to get the exact layer name → shape mapping |
