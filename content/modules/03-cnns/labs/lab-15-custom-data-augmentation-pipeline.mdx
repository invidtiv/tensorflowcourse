class Cutout(layers.Layer):
    """
    Cutout augmentation layer.
    Randomly masks out square regions.
    """
    
    def __init__(self, mask_size=8, n_masks=1, **kwargs):
        super().__init__(**kwargs)
        self.mask_size = mask_size
        self.n_masks = n_masks
    
    def call(self, inputs, training=None):
        if not training:
            return inputs
        
        batch_size = tf.shape(inputs)[0]
        height = tf.shape(inputs)[1]
        width = tf.shape(inputs)[2]
        channels = tf.shape(inputs)[3]
        
        # Create mask
        mask = tf.ones_like(inputs)
        
        for _ in range(self.n_masks):
            # Random center position
            y = tf.random.uniform([], 0, height, dtype=tf.int32)
            x = tf.random.uniform([], 0, width, dtype=tf.int32)
            
            # Calculate mask boundaries
            y1 = tf.maximum(0, y - self.mask_size // 2)
            y2 = tf.minimum(height, y + self.mask_size // 2)
            x1 = tf.maximum(0, x - self.mask_size // 2)
            x2 = tf.minimum(width, x + self.mask_size // 2)
            
            # Create mask for this region
            y_indices = tf.range(height)
            x_indices = tf.range(width)
            
            y_mask = tf.logical_and(y_indices >= y1, y_indices < y2)
            x_mask = tf.logical_and(x_indices >= x1, x_indices < x2)
            
            y_mask = tf.reshape(y_mask, [height, 1])
            x_mask = tf.reshape(x_mask, [1, width])
            
            region_mask = tf.logical_and(y_mask, x_mask)
            region_mask = tf.cast(region_mask, tf.float32)
            region_mask = tf.expand_dims(region_mask, -1)
            
            mask = mask * (1 - region_mask)
        
        return inputs * mask
    
    def get_config(self):
        config = super().get_config()
        config.update({
            'mask_size': self.mask_size,
            'n_masks': self.n_masks
        })
        return config


class RandomColorJitter(layers.Layer):
    """Random color jittering layer."""
    
    def __init__(self, brightness=0.2, contrast=0.2, saturation=0.2, **kwargs):
        super().__init__(**kwargs)
        self.brightness = brightness
        self.contrast = contrast
        self.saturation = saturation
    
    def call(self, inputs, training=None):
        if not training:
            return inputs
        
        x = inputs
        
        # Random brightness
        if self.brightness > 0:
            delta = tf.random.uniform([], -self.brightness, self.brightness)
            x = tf.image.adjust_brightness(x, delta)
        
        # Random contrast
        if self.contrast > 0:
            contrast_factor = tf.random.uniform([], 1 - self.contrast, 1 + self.contrast)
            x = tf.image.adjust_contrast(x, contrast_factor)
        
        # Random saturation
        if self.saturation > 0:
            saturation_factor = tf.random.uniform([], 1 - self.saturation, 1 + self.saturation)
            x = tf.image.adjust_saturation(x, saturation_factor)
        
        return tf.clip_by_value(x, 0, 1)


class MixUp(layers.Layer):
    """MixUp augmentation layer."""
    
    def __init__(self, alpha=0.2, **kwargs):
        super().__init__(**kwargs)
        self.alpha = alpha
    
    def call(self, inputs, training=None):
        if not training:
            return inputs
        
        batch_size = tf.shape(inputs)[0]
        
        # Sample lambda from Beta distribution
        lam = tf.random.uniform([], 0, self.alpha)
        
        # Shuffle batch
        indices = tf.random.shuffle(tf.range(batch_size))
        shuffled = tf.gather(inputs, indices)
        
        # Mix
        mixed = lam * inputs + (1 - lam) * shuffled
        
        return mixed

def create_tfdata_pipeline(X, y, batch_size=64, augment=True, shuffle=True):
    """
    Create efficient tf.data pipeline with augmentation.
    
    Args:
        X: Images
        y: Labels
        batch_size: Batch size
        augment: Whether to apply augmentation
        shuffle: Whether to shuffle data
    
    Returns:
        tf.data.Dataset
    """
    
    print("\n" + "=" * 60)
    print("CREATING TF.DATA PIPELINE")
    print("=" * 60)
    
    # Create dataset from tensors
    dataset = tf.data.Dataset.from_tensor_slices((X, y))
    
    # Shuffle
    if shuffle:
        dataset = dataset.shuffle(buffer_size=1000)
    
    # Apply augmentation
    if augment:
        augmentation = build_augmentation_pipeline()
        
        def augment_fn(image, label):
            image = augmentation(image, training=True)
            return image, label
        
        dataset = dataset.map(augment_fn, num_parallel_calls=tf.data.AUTOTUNE)
    
    # Batch and prefetch
    dataset = dataset.batch(batch_size)
    dataset = dataset.prefetch(tf.data.AUTOTUNE)
    
    print(f"Pipeline created:")
    print(f"  Batch size: {batch_size}")
    print(f"  Augmentation: {augment}")
    print(f"  Prefetch: AUTOTUNE")
    
    return dataset

def train_with_tfdataset(train_ds, test_ds, epochs=10):

def compare_strategies():

            layers.Dense(64, activation='relu'),

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| MixUp labels not mixed — only hard labels used | MixUp requires `y_mixed = λ*y1 + (1-λ)*y2` but the code mixes only images and keeps hard labels | Apply the same λ to labels: `y_mixed = lam * y_train_oh[idx1] + (1 - lam) * y_train_oh[idx2]`; use `categorical_crossentropy`, not `sparse_categorical_crossentropy` |
| CutMix bounding box coordinates exceed image dimensions | Random `rx, ry` center point + half-width `rw/2` extends past the 0–31 boundary | Clip box: `x1=max(0, rx-rw//2)`, `x2=min(W, rx+rw//2)` before slicing |
| `tensorflow_addons` import fails in newer TF environments | TFA is deprecated in TF 2.14+ and not compatible with TF 2.16+ | Replace `tfa.image.cutout` with a custom NumPy/TF implementation; replace `tfa.optimizers.SGDW` with `tf.keras.optimizers.SGD` + weight decay via `kernel_regularizer` |
| RandAugment magnitude parameter too high (M > 15) | Large magnitudes produce unrealistic transformations (>45° rotations, extreme color shifts) that confuse the model | Keep `M ≤ 10` for CIFAR-10; search M ∈ {5, 7, 9, 11} via validation accuracy |
| tf.data pipeline slower than expected — GPU starved | Too few `num_parallel_calls` or missing `.prefetch(tf.data.AUTOTUNE)` causes the CPU augmentation pipeline to become a bottleneck | Add `.map(augment_fn, num_parallel_calls=tf.data.AUTOTUNE).prefetch(tf.data.AUTOTUNE)` and profile with `tf.data.experimental.enable_debug_mode()` |
