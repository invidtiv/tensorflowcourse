# Transfer Learning with ResNet50

## Objectives
- Use pretrained ResNet50 for transfer learning
- Implement bottleneck feature extraction
- Compare with VGG16 transfer learning

## Prerequisites
- Completion of Lab 3.10
- Understanding of ResNet architecture

```python
"""
Lab 3.11: Transfer Learning with ResNet50
Using ResNet50 for transfer learning tasks
"""

import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers, models, optimizers
from tensorflow.keras.applications import ResNet50, VGG16
from tensorflow.keras.datasets import cifar10
from tensorflow.keras.utils import to_categorical
import time

# =============================================================================
# Section 1: ResNet50 Overview
# =============================================================================

print("=" * 60)
print("TRANSFER LEARNING WITH ResNet50")
print("=" * 60)

print("""
ResNet50 Architecture:
======================
- 50 weight layers
- Residual connections for gradient flow
- Bottleneck blocks (1x1 → 3x3 → 1x1)
- Pretrained on ImageNet
- Better gradient flow than VGG

Advantages over VGG:
- Fewer parameters (25.6M vs 138M)
- Better performance on many tasks
- Easier to fine-tune (skip connections)
- Batch normalization for stability

Transfer Learning Strategy:
1. Use as feature extractor
2. Fine-tune with small learning rate
3. Progressive unfreezing from end to start
""")


# =============================================================================
# Section 2: Load Pretrained ResNet50
# =============================================================================

print("\n" + "=" * 60)
print("LOADING PRETRAINED ResNet50")
print("=" * 60)

# Load ResNet50 without top layers
resnet_base = ResNet50(weights='imagenet',
                       include_top=False,
                       input_shape=(32, 32, 3))

print(f"\nBase model loaded: {len(resnet_base.layers)} layers")
print(f"Total parameters: {resnet_base.count_params():,}")

# Display layer groups
print("\nLayer Groups:")
layer_groups = {}
for layer in resnet_base.layers:
    group = layer.name.split('_')[0] if '_' in layer.name else 'other'
    if group not in layer_groups:
        layer_groups[group] = 0
    layer_groups[group] += 1

for group, count in sorted(layer_groups.items()):
    print(f"  {group:15s}: {count:3d} layers")


# =============================================================================
# Section 3: Build Transfer Learning Model
# =============================================================================

def build_resnet_transfer_model(base_model, num_classes=10, 
                                fine_tune_at=None, dropout_rate=0.5):
    """
    Build transfer learning model with ResNet50.
    
    Args:
        base_model: Pretrained ResNet50
        num_classes: Number of output classes
        fine_tune_at: Layer name to start fine-tuning from
        dropout_rate: Dropout rate for classifier
    """
    
    print("\n" + "=" * 60)
    print("BUILDING ResNet50 TRANSFER MODEL")
    print("=" * 60)
    
    # Set trainable layers
    if fine_tune_at is not None:
        base_model.trainable = True
        for layer in base_model.layers:
            if layer.name == fine_tune_at:
                break
            layer.trainable = False
        print(f"\nFine-tuning from: {fine_tune_at}")
    else:
        base_model.trainable = False
        print("\nFeature extraction mode (all layers frozen)")
    
    # Count trainable parameters
    trainable_count = sum([keras.backend.count_params(w) 
                          for w in base_model.trainable_weights])
    print(f"Trainable parameters in base: {trainable_count:,}")
    
    # Build model
    inputs = keras.Input(shape=(32, 32, 3))
    x = base_model(inputs, training=(fine_tune_at is not None))
    
    # Global pooling
    x = layers.GlobalAveragePooling2D()(x)
    
    # Classifier
    x = layers.BatchNormalization()(x)
    x = layers.Dense(512, activation='relu')(x)
    x = layers.Dropout(dropout_rate)(x)
    x = layers.Dense(256, activation='relu')(x)
    x = layers.Dropout(dropout_rate)(x)
    outputs = layers.Dense(num_classes, activation='softmax')(x)
    
    model = models.Model(inputs, outputs)
    
    total_trainable = sum([keras.backend.count_params(w) 
                          for w in model.trainable_weights])
    print(f"Total trainable parameters: {total_trainable:,}")
    
    return model


# Build feature extraction model
resnet_feature = build_resnet_transfer_model(resnet_base, fine_tune_at=None)


# =============================================================================
# Section 4: Prepare Data
# =============================================================================

def prepare_data():
    """Prepare CIFAR-10 data with ResNet preprocessing."""
    
    (X_train, y_train), (X_test, y_test) = cifar10.load_data()
    
    # Convert and preprocess
    X_train = X_train.astype('float32')
    X_test = X_test.astype('float32')
    
    # ResNet preprocessing
    X_train = keras.applications.resnet50.preprocess_input(X_train)
    X_test = keras.applications.resnet50.preprocess_input(X_test)
    
    y_train = to_categorical(y_train, 10)
    y_test = to_categorical(y_test, 10)
    
    return (X_train, y_train), (X_test, y_test)


(X_train, y_train), (X_test, y_test) = prepare_data()


# =============================================================================
# Section 5: Train Feature Extraction
# =============================================================================

def train_model(model, X_train, y_train, X_test, y_test, 
                epochs=10, lr=0.001, model_name="Model"):
    """Generic training function."""
    
    print(f"\n" + "=" * 60)
    print(f"TRAINING {model_name}")
    print("=" * 60)
    
    model.compile(optimizer=optimizers.Adam(learning_rate=lr),
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    callbacks = [
        keras.callbacks.EarlyStopping(patience=5, restore_best_weights=True),
        keras.callbacks.ReduceLROnPlateau(factor=0.5, patience=3)
    ]
    
    start = time.time()
    
    history = model.fit(X_train[:10000], y_train[:10000],
                       batch_size=64,
                       epochs=epochs,
                       validation_data=(X_test[:2000], y_test[:2000]),
                       callbacks=callbacks,
                       verbose=1)
    
    train_time = time.time() - start
    test_loss, test_acc = model.evaluate(X_test, y_test, verbose=0)
    
    print(f"\nTraining time: {train_time:.2f}s")
    print(f"Test accuracy: {test_acc:.4f}")
    
    return history, test_acc, train_time


# Train feature extraction
history_resnet_feat, acc_resnet_feat, time_resnet_feat = train_model(
    resnet_feature, X_train, y_train, X_test, y_test,
    epochs=10, model_name="ResNet50 Feature Extraction"
)


# =============================================================================
# Section 6: Fine-tuning ResNet50
# =============================================================================

# Reload base model for fine-tuning
resnet_base_ft = ResNet50(weights='imagenet',
                          include_top=False,
                          input_shape=(32, 32, 3))

# Fine-tune from conv5_block1
resnet_finetune = build_resnet_transfer_model(resnet_base_ft, 
                                              fine_tune_at='conv5_block1_0_conv')

# Train with fine-tuning
history_resnet_ft, acc_resnet_ft, time_resnet_ft = train_model(
    resnet_finetune, X_train, y_train, X_test, y_test,
    epochs=10, lr=0.0001, model_name="ResNet50 Fine-tuning"
)


# =============================================================================
# Section 7: Compare VGG16 vs ResNet50
# =============================================================================

def compare_architectures():
    """Compare transfer learning performance of VGG16 vs ResNet50."""
    
    print("\n" + "=" * 60)
    print("ARCHITECTURE COMPARISON")
    print("=" * 60)
    
    # Load VGG16
    vgg_base = VGG16(weights='imagenet', include_top=False, input_shape=(32, 32, 3))
    vgg_base.trainable = False
    
    inputs = keras.Input(shape=(32, 32, 3))
    x = vgg_base(inputs, training=False)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.Dense(256, activation='relu')(x)
    outputs = layers.Dense(10, activation='softmax')(x)
    vgg_model = models.Model(inputs, outputs)
    
    # Train VGG16
    history_vgg, acc_vgg, time_vgg = train_model(
        vgg_model, X_train, y_train, X_test, y_test,
        epochs=10, model_name="VGG16 Feature Extraction"
    )
    
    # Compare
    results = {
        'VGG16': {'accuracy': acc_vgg, 'time': time_vgg, 'params': vgg_base.count_params()},
        'ResNet50': {'accuracy': acc_resnet_feat, 'time': time_resnet_feat, 
                    'params': resnet_base.count_params()}
    }
    
    print("\n" + "-" * 60)
    print(f"{'Model':<15} {'Accuracy':>10} {'Time (s)':>10} {'Parameters':>15}")
    print("-" * 60)
    for name, data in results.items():
        print(f"{name:<15} {data['accuracy']:>10.4f} {data['time']:>10.2f} {data['params']:>15,}")
    print("-" * 60)
    
    return history_vgg, results


history_vgg, comparison_results = compare_architectures()


# =============================================================================
# Section 8: Visualize Comparison
# =============================================================================

def plot_comparison(history_vgg, history_resnet, history_resnet_ft):

    axes[1].plot(history_vgg.history['val_loss'],

    axes[1].plot(history_resnet.history['val_loss'],

plot_comparison(history_vgg, history_resnet_feat, history_resnet_ft)

# Section 9: Bottleneck Feature Extraction
# =============================================================================

def extract_bottleneck_features(base_model, X):
    """
    Extract bottleneck features using pretrained model.
    
    This is useful when you want to cache features and train
    multiple classifiers without re-running the CNN.
    """
    
    print("\n" + "=" * 60)
    print("BOTTLENECK FEATURE EXTRACTION")
    print("=" * 60)
    
    # Create feature extraction model
    feature_model = models.Model(inputs=base_model.input,
                                 outputs=base_model.output)
    
    # Extract features in batches
    batch_size = 256
    features = []
    
    for i in range(0, len(X), batch_size):
        batch = X[i:i+batch_size]
        batch_features = feature_model.predict(batch, verbose=0)
        features.append(batch_features)
    
    features = np.concatenate(features, axis=0)
    
    # Flatten spatial dimensions
    features = features.reshape(features.shape[0], -1)
    
    print(f"\nInput shape: {X.shape}")
    print(f"Bottleneck features shape: {features.shape}")
    print(f"Compression ratio: {np.prod(X.shape[1:]) / features.shape[1]:.2f}x")
    
    return features


# Extract bottleneck features
bottleneck_features = extract_bottleneck_features(resnet_base, X_train[:5000])


# =============================================================================
# Section 10: Train Classifier on Bottleneck Features
# =============================================================================

def train_bottleneck_classifier(features, y, epochs=20):
    """Train a simple classifier on bottleneck features."""
    
    print("\n" + "=" * 60)
    print("TRAINING BOTTLENECK CLASSIFIER")
    print("=" * 60)
    
    # Simple classifier
    model = models.Sequential([
        layers.Dense(256, activation='relu', input_shape=(features.shape[1],)),
        layers.Dropout(0.5),
        layers.Dense(128, activation='relu'),
        layers.Dropout(0.5),
        layers.Dense(10, activation='softmax')
    ])
    
    model.compile(optimizer='adam',
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    history = model.fit(features, y[:len(features)],
                       batch_size=256,
                       epochs=epochs,
                       validation_split=0.1,
                       verbose=1)
    
    return history, model


history_bottleneck, bottleneck_model = train_bottleneck_classifier(
    bottleneck_features, y_train, epochs=10
)
```
