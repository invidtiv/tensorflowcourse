Fine-tuning Best Practices:
===========================

1. Differential Learning Rates:
   - Lower LR for early layers (general features)
   - Higher LR for later layers (task-specific)
   - Formula: lr_layer = base_lr * decay_factor^(depth - layer_depth)

2. Progressive Unfreezing:
   - Start with classifier only
   - Gradually unfreeze layers from end to start
   - Train each stage with appropriate LR

3. Discriminative Fine-tuning (ULMFiT):
   - Different LRs for different layer groups
   - Layer-wise learning rate decay

4. Learning Rate Warmup:
   - Start with very small LR
   - Gradually increase to target LR
   - Helps stabilize early training
""")

class LayerWiseOptimizer:
    """Custom optimizer wrapper for layer-wise learning rates."""
    
    def __init__(self, base_lr=0.001, decay_factor=0.9):
        self.base_lr = base_lr
        self.decay_factor = decay_factor
    
    def create_optimizer(self, model, layer_groups):
        """
        Create optimizer with different LRs for each layer group.
        
        Args:
            model: Keras model
            layer_groups: List of (layer_names, lr_multiplier) tuples
        """
        
        # Build learning rate dictionary
        lr_dict = {}
        
        for group_idx, (names, multiplier) in enumerate(layer_groups):
            lr = self.base_lr * (self.decay_factor ** group_idx) * multiplier
            for name in names:
                layer = model.get_layer(name)
                if layer is not None and layer.trainable:
                    lr_dict[id(layer)] = lr
        
        # Use Adam with different LRs
        # Note: This is a simplified version
        # Full implementation would require custom training loop
        
        return optimizers.Adam(learning_rate=self.base_lr)

def progressive_finetuning_strategy(X_train, y_train, X_test, y_test):
    """
    Implement progressive fine-tuning with multiple stages.
    """
    
    print("\n" + "=" * 60)
    print("PROGRESSIVE FINE-TUNING IMPLEMENTATION")
    print("=" * 60)
    
    # Load base model
    base_model = ResNet50(weights='imagenet', include_top=False, 
                          input_shape=(32, 32, 3))
    
    histories = []
    
    # Stage 1: Train only classifier
    print("\n--- Stage 1: Classifier Only ---")
    base_model.trainable = False
    
    inputs = keras.Input(shape=(32, 32, 3))
    x = base_model(inputs, training=False)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.Dense(256, activation='relu')(x)
    x = layers.Dropout(0.5)(x)
    outputs = layers.Dense(10, activation='softmax')(x)
    
    model = models.Model(inputs, outputs)
    model.compile(optimizer=optimizers.Adam(0.001),
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    history1 = model.fit(X_train[:5000], y_train[:5000],
                        batch_size=64, epochs=5, verbose=1,
                        validation_split=0.1)
    histories.append(('Stage 1: Classifier', history1))
    
    # Stage 2: Fine-tune last block
    print("\n--- Stage 2: Fine-tune conv5 ---")
    base_model.trainable = True
    
    for layer in base_model.layers:
        if 'conv5' in layer.name:
            layer.trainable = True
        else:
            layer.trainable = False
    
    model.compile(optimizer=optimizers.Adam(0.0001),
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    history2 = model.fit(X_train[:5000], y_train[:5000],
                        batch_size=64, epochs=5, verbose=1,
                        validation_split=0.1)
    histories.append(('Stage 2: conv5', history2))
    
    # Stage 3: Fine-tune more layers
    print("\n--- Stage 3: Fine-tune conv4+conv5 ---")
    for layer in base_model.layers:
        if 'conv4' in layer.name or 'conv5' in layer.name:
            layer.trainable = True
        else:
            layer.trainable = False
    
    model.compile(optimizer=optimizers.Adam(0.00001),
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])
    
    history3 = model.fit(X_train[:5000], y_train[:5000],
                        batch_size=64, epochs=5, verbose=1,
                        validation_split=0.1)
    histories.append(('Stage 3: conv4+conv5', history3))
    
    return model, histories

class WarmUpLearningRateScheduler(keras.callbacks.Callback):
    """Learning rate warmup scheduler."""
    
    def __init__(self, warmup_epochs, target_lr):
        super().__init__()
        self.warmup_epochs = warmup_epochs
        self.target_lr = target_lr
        self.current_epoch = 0
    
    def on_epoch_begin(self, epoch, logs=None):
        self.current_epoch = epoch
        if epoch < self.warmup_epochs:
            # Linear warmup
            new_lr = self.target_lr * (epoch + 1) / self.warmup_epochs
            self.model.optimizer.learning_rate.assign(new_lr)
            print(f"\nWarmup LR: {new_lr:.6f}")
    
    def on_epoch_end(self, epoch, logs=None):
        if epoch == self.warmup_epochs - 1:
            self.model.optimizer.learning_rate.assign(self.target_lr)
            print(f"\nWarmup complete. LR set to: {self.target_lr:.6f}")

def compare_strategies():
    """Compare different fine-tuning strategies."""
    
    print("\n" + "=" * 60)
    print("STRATEGY COMPARISON")
    print("=" * 60)
    
    strategies = {
        'Feature Extraction': {'trainable': 0, 'lr': 0.001},
        'Fine-tune Last Block': {'trainable': 10, 'lr': 0.0001},
        'Fine-tune Half': {'trainable': 50, 'lr': 0.00001},
        'Full Fine-tuning': {'trainable': 100, 'lr': 0.00001}
    }
    
    results = {}
    
    for name, config in strategies.items():
        print(f"\n--- Testing: {name} ---")
        
        base = ResNet50(weights='imagenet', include_top=False, 
                       input_shape=(32, 32, 3))
        
        # Set trainable layers
        if config['trainable'] == 0:
            base.trainable = False
        elif config['trainable'] == 100:
            base.trainable = True
        else:
            # Partial unfreezing
            n_layers = len(base.layers)
            n_trainable = int(n_layers * config['trainable'] / 100)
            for i, layer in enumerate(base.layers):
                layer.trainable = (i >= n_layers - n_trainable)
        
        # Build and train
        inputs = keras.Input(shape=(32, 32, 3))
        x = base(inputs, training=True)
        x = layers.GlobalAveragePooling2D()(x)
        x = layers.Dense(256, activation='relu')(x)
        outputs = layers.Dense(10, activation='softmax')(x)
        
        model = models.Model(inputs, outputs)
        model.compile(optimizer=optimizers.Adam(config['lr']),
                      loss='categorical_crossentropy',
                      metrics=['accuracy'])
        
        start = time.time()
        history = model.fit(X_train[:3000], y_train[:3000],
                           batch_size=64, epochs=5, verbose=0,
                           validation_split=0.1)
        train_time = time.time() - start
        
        results[name] = {
            'final_acc': history.history['val_accuracy'][-1],
            'time': train_time,
            'history': history
        }
        
        print(f"  Final accuracy: {results[name]['final_acc']:.4f}")
        print(f"  Training time: {train_time:.2f}s")
    
    return results


strategy_results = compare_strategies()

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Fine-tuning decreases accuracy below frozen baseline | LR too high when unfreezing — even one batch at LR=1e-3 can irreversibly corrupt pretrained features | Reduce fine-tuning LR to ≤ 1e-5 for already-trained backbone weights; use `ReduceLROnPlateau` as a safety net |
| Progressive unfreezing shows no accuracy gain over single-step unfreezing | Same total epochs used for both; progressive unfreezing needs more epochs per stage | Give each stage ≥5 epochs before unfreezing the next block; use early stopping per stage |
| BatchNorm layers running mean drifts during fine-tuning | When only a few layers are unfrozen, BN in frozen layers still updates its running statistics if `training=True` | Set `base.trainable = False` and pass `training=False` to the base call; frozen BN should use its saved ImageNet statistics |
| Cosine LR restart conflicts with progressive unfreezing schedule | The LR restarts to `lr_max` right when a new block is unfrozen, which can be too aggressive for the newly thawed weights | Synchronize restart boundaries with unfreeze events, or use a separate small LR for newly unfrozen blocks |
| Model diverges after knowledge distillation teacher comparison | Distillation loss mixed with cross-entropy loss at equal weight causes the soft-label term to dominate | Scale distillation loss by `(T^2 * α)` where `T` is temperature and `α ∈ [0.1, 0.5]`; start with `α=0.2, T=3` |
