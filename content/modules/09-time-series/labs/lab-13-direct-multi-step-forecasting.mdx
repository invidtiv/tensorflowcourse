# Direct Multi-step Forecasting

### Complete Implementation

```python
"""
Lab 5.1: Direct Multi-step Forecasting
Topic: Predict Multiple Future Steps at Once
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error, mean_absolute_error

np.random.seed(42)
tf.random.set_seed(42)

# Generate Data
print("=" * 60)
print("STEP 1: Generating Data")
print("=" * 60)

def generate_data(n_points=1000):
    t = np.arange(n_points)
    trend = 0.01 * t
    seasonal = 10 * np.sin(2 * np.pi * t / 50)
    noise = np.random.normal(0, 2, n_points)
    return 100 + trend + seasonal + noise

data = generate_data(1000)
scaler = MinMaxScaler()
scaled_data = scaler.fit_transform(data.reshape(-1, 1)).flatten()

# Create Multi-step Windows
print("\n" + "=" * 60)
print("STEP 2: Creating Multi-step Windows")
print("=" * 60)

def create_multistep_windows(data, window_size, horizon):
    X, y = [], []
    for i in range(len(data) - window_size - horizon + 1):
        X.append(data[i:(i + window_size)])
        y.append(data[(i + window_size):(i + window_size + horizon)])
    return np.array(X), np.array(y)

window_size = 60
horizon = 7  # Predict 7 steps ahead
X, y = create_multistep_windows(scaled_data, window_size, horizon)
X = X.reshape(X.shape[0], X.shape[1], 1)

print(f"X shape: {X.shape}")
print(f"y shape: {y.shape} (samples, horizon)")

train_size = int(0.7 * len(X))
X_train, y_train = X[:train_size], y[:train_size]
X_val, y_val = X[train_size:int(0.85*len(X))], y[train_size:int(0.85*len(X))]
X_test, y_test = X[int(0.85*len(X)):], y[int(0.85*len(X)):]

# Build Direct Multi-step Model
print("\n" + "=" * 60)
print("STEP 3: Building Direct Multi-step Model")
print("=" * 60)

def build_direct_multistep_model(window_size, horizon, n_features=1):
    model = keras.Sequential([
        layers.LSTM(64, return_sequences=True, input_shape=(window_size, n_features)),
        layers.Dropout(0.2),
        layers.LSTM(32, return_sequences=False),
        layers.Dropout(0.2),
        layers.Dense(32, activation='relu'),
        layers.Dense(horizon)  # Output horizon predictions
    ])
    model.compile(optimizer='adam', loss='mse', metrics=['mae'])
    return model

model_direct = build_direct_multistep_model(window_size, horizon)
model_direct.summary()

# Train Model
print("\n" + "=" * 60)
print("STEP 4: Training Direct Multi-step Model")
print("=" * 60)

early_stop = keras.callbacks.EarlyStopping(
    monitor='val_loss', patience=10, restore_best_weights=True
)

history = model_direct.fit(
    X_train, y_train,
    validation_data=(X_val, y_val),
    epochs=50,
    batch_size=32,
    callbacks=[early_stop],
    verbose=1
)

# Plot training history
plt.figure(figsize=(10, 4))
plt.plot(history.history['loss'], label='Train')
plt.plot(history.history['val_loss'], label='Validation')
plt.title('Training Loss')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab5_1_direct_history.png', dpi=150)
plt.show()

# Evaluate Multi-step Predictions
print("\n" + "=" * 60)
print("STEP 5: Evaluating Multi-step Predictions")
print("=" * 60)

y_pred = model_direct.predict(X_test)
y_test_inv = scaler.inverse_transform(y_test)
y_pred_inv = scaler.inverse_transform(y_pred)

mse_per_step = []
mae_per_step = []

for i in range(horizon):
    mse = mean_squared_error(y_test_inv[:, i], y_pred_inv[:, i])
    mae = mean_absolute_error(y_test_inv[:, i], y_pred_inv[:, i])
    mse_per_step.append(mse)
    mae_per_step.append(mae)
    print(f"Step {i+1}: MSE={mse:.4f}, MAE={mae:.4f}")

# Visualize predictions for different horizons
fig, axes = plt.subplots(2, 2, figsize=(15, 10))
axes = axes.flatten()

for idx, step in enumerate([0, 2, 4, 6]):
    axes[idx].plot(y_test_inv[:50, step], label='Actual', linewidth=2)
    axes[idx].plot(y_pred_inv[:50, step], label='Predicted', linewidth=2, alpha=0.8)
    axes[idx].set_title(f'Horizon Step {step+1}')
    axes[idx].legend()
    axes[idx].grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('/mnt/okcomputer/output/lab5_1_direct_predictions.png', dpi=150)
plt.show()

# Save model
model_direct.save('/mnt/okcomputer/output/lab5_1_direct_multistep.h5')

print("\n" + "=" * 60)
print("Lab 5.1 Complete!")
print("=" * 60)
```
