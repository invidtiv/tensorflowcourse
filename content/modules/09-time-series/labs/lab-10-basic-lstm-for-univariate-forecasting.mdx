# Basic LSTM for Univariate Forecasting

```python
"""
Lab 4.1: Basic LSTM for Univariate Forecasting
Topic: Stock Price Prediction with LSTM
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.metrics import mean_squared_error, mean_absolute_error
from sklearn.preprocessing import MinMaxScaler
import joblib

np.random.seed(42)
tf.random.set_seed(42)

# Generate and Prepare Data
print("=" * 60)
print("STEP 1: Generating Stock Data")
print("=" * 60)

def generate_stock_data(n_days=1000):
    dates = pd.bdate_range(start='2020-01-01', periods=n_days)
    trend = np.cumsum(np.random.normal(0.001, 0.01, n_days))
    seasonal = 5 * np.sin(2 * np.pi * np.arange(n_days) / 252)
    prices = 100 + trend + seasonal + np.cumsum(np.random.normal(0, 0.5, n_days))
    return pd.DataFrame({'Close': prices}, index=dates)

stock_data = generate_stock_data(1000)
print(f"Generated {len(stock_data)} data points")

scaler = MinMaxScaler()
scaled_data = scaler.fit_transform(stock_data[['Close']]).flatten()

# Create Windows
print("\n" + "=" * 60)
print("STEP 2: Creating Windows")
print("=" * 60)

def create_windows(data, window_size, horizon=1):
    X, y = [], []
    for i in range(len(data) - window_size - horizon + 1):
        X.append(data[i:(i + window_size)])
        y.append(data[i + window_size])
    return np.array(X), np.array(y)

window_size = 60
X, y = create_windows(scaled_data, window_size, horizon=1)
X = X.reshape(X.shape[0], X.shape[1], 1)

train_size = int(0.7 * len(X))
val_size = int(0.15 * len(X))
X_train, y_train = X[:train_size], y[:train_size]
X_val, y_val = X[train_size:train_size + val_size], y[train_size:train_size + val_size]
X_test, y_test = X[train_size + val_size:], y[train_size + val_size:]

print(f"X_train: {X_train.shape}")
print(f"X_val: {X_val.shape}")
print(f"X_test: {X_test.shape}")

# Build LSTM Model
print("\n" + "=" * 60)
print("STEP 3: Building LSTM Model")
print("=" * 60)

def build_lstm_model(window_size, n_features=1):
    model = keras.Sequential([
        layers.LSTM(50, return_sequences=True, input_shape=(window_size, n_features)),
        layers.Dropout(0.2),
        layers.LSTM(50, return_sequences=False),
        layers.Dropout(0.2),
        layers.Dense(25, activation='relu'),
        layers.Dense(1)
    ])
    model.compile(optimizer='adam', loss='mse', metrics=['mae'])
    return model

model_lstm = build_lstm_model(window_size)
model_lstm.summary()

# Train LSTM
print("\n" + "=" * 60)
print("STEP 4: Training LSTM")
print("=" * 60)

early_stop = keras.callbacks.EarlyStopping(
    monitor='val_loss', patience=10, restore_best_weights=True
)
reduce_lr = keras.callbacks.ReduceLROnPlateau(
    monitor='val_loss', factor=0.5, patience=5, min_lr=1e-6
)

history = model_lstm.fit(
    X_train, y_train,
    validation_data=(X_val, y_val),
    epochs=50,
    batch_size=32,
    callbacks=[early_stop, reduce_lr],
    verbose=1
)

# Plot training history
fig, axes = plt.subplots(1, 2, figsize=(15, 5))
axes[0].plot(history.history['loss'], label='Train')
axes[0].plot(history.history['val_loss'], label='Validation')
axes[0].set_title('Loss')
axes[0].legend()
axes[0].grid(True, alpha=0.3)
axes[1].plot(history.history['mae'], label='Train')
axes[1].plot(history.history['val_mae'], label='Validation')
axes[1].set_title('MAE')
axes[1].legend()
axes[1].grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab4_1_lstm_history.png', dpi=150)
plt.show()

# Evaluate LSTM
print("\n" + "=" * 60)
print("STEP 5: Evaluating LSTM")
print("=" * 60)

y_pred = model_lstm.predict(X_test)
y_test_inv = scaler.inverse_transform(y_test.reshape(-1, 1)).flatten()
y_pred_inv = scaler.inverse_transform(y_pred).flatten()

mse = mean_squared_error(y_test_inv, y_pred_inv)
mae = mean_absolute_error(y_test_inv, y_pred_inv)
rmse = np.sqrt(mse)

print(f"Test MSE: {mse:.4f}")
print(f"Test MAE: {mae:.4f}")
print(f"Test RMSE: {rmse:.4f}")

# Visualize predictions
plt.figure(figsize=(15, 5))
plt.plot(y_test_inv[:100], label='Actual', color='blue', linewidth=1.5)
plt.plot(y_pred_inv[:100], label='Predicted', color='red', linewidth=1.5, alpha=0.8)
plt.title('LSTM Predictions vs Actual')
plt.xlabel('Time Step')
plt.ylabel('Price')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab4_1_lstm_predictions.png', dpi=150)
plt.show()

# Save model
model_lstm.save('/mnt/okcomputer/output/lab4_1_lstm_model.h5')
joblib.dump(scaler, '/mnt/okcomputer/output/lab4_1_scaler.pkl')

print("\n" + "=" * 60)
print("Lab 4.1 Complete!")
print("=" * 60)
```
