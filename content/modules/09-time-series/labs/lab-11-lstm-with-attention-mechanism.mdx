# LSTM with Attention Mechanism

```python
"""
Lab 4.2: LSTM with Attention Mechanism
Topic: Enhanced Time Series Prediction
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error, mean_absolute_error

np.random.seed(42)
tf.random.set_seed(42)

# Prepare Data
print("=" * 60)
print("STEP 1: Preparing Data")
print("=" * 60)

def generate_data(n_points=1000):
    t = np.arange(n_points)
    trend = 0.01 * t
    seasonal = 10 * np.sin(2 * np.pi * t / 50)
    noise = np.random.normal(0, 2, n_points)
    return 100 + trend + seasonal + noise

data = generate_data(1000)
scaler = MinMaxScaler()
scaled_data = scaler.fit_transform(data.reshape(-1, 1)).flatten()

def create_windows(data, window_size, horizon=1):
    X, y = [], []
    for i in range(len(data) - window_size - horizon + 1):
        X.append(data[i:(i + window_size)])
        y.append(data[i + window_size])
    return np.array(X), np.array(y)

window_size = 60
X, y = create_windows(scaled_data, window_size, horizon=1)
X = X.reshape(X.shape[0], X.shape[1], 1)

train_size = int(0.7 * len(X))
X_train, y_train = X[:train_size], y[:train_size]
X_val, y_val = X[train_size:int(0.85*len(X))], y[train_size:int(0.85*len(X))]
X_test, y_test = X[int(0.85*len(X)):], y[int(0.85*len(X)):]
print(f"X_train: {X_train.shape}")

# Build Attention Layer
print("\n" + "=" * 60)
print("STEP 2: Building Attention Mechanism")
print("=" * 60)

class AttentionLayer(layers.Layer):
    def __init__(self, units):
        super(AttentionLayer, self).__init__()
        self.W1 = layers.Dense(units)
        self.W2 = layers.Dense(units)
        self.V = layers.Dense(1)
    
    def call(self, query, values):
        query_with_time_axis = tf.expand_dims(query, 1)
        score = self.V(tf.nn.tanh(
            self.W1(query_with_time_axis) + self.W2(values)
        ))
        attention_weights = tf.nn.softmax(score, axis=1)
        context_vector = attention_weights * values
        context_vector = tf.reduce_sum(context_vector, axis=1)
        return context_vector, attention_weights

# Build LSTM with Attention Model
print("\n" + "=" * 60)
print("STEP 3: Building LSTM with Attention Model")
print("=" * 60)

def build_attention_lstm(window_size, n_features=1):
    inputs = layers.Input(shape=(window_size, n_features))
    lstm_out, state_h, state_c = layers.LSTM(64, return_sequences=True, return_state=True)(inputs)
    attention_layer = AttentionLayer(64)
    context_vector, attention_weights = attention_layer(state_h, lstm_out)
    concat = layers.Concatenate()([context_vector, state_h])
    x = layers.Dense(32, activation='relu')(concat)
    x = layers.Dropout(0.2)(x)
    outputs = layers.Dense(1)(x)
    model = keras.Model(inputs=inputs, outputs=outputs)
    model.compile(optimizer='adam', loss='mse', metrics=['mae'])
    return model

model_attention = build_attention_lstm(window_size)
model_attention.summary()

# Train Model
print("\n" + "=" * 60)
print("STEP 4: Training Attention LSTM")
print("=" * 60)

early_stop = keras.callbacks.EarlyStopping(
    monitor='val_loss', patience=10, restore_best_weights=True
)

history = model_attention.fit(
    X_train, y_train,
    validation_data=(X_val, y_val),
    epochs=50,
    batch_size=32,
    callbacks=[early_stop],
    verbose=1
)

# Plot training history
plt.figure(figsize=(10, 4))
plt.plot(history.history['loss'], label='Train')
plt.plot(history.history['val_loss'], label='Validation')
plt.title('Training Loss')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab4_2_attention_history.png', dpi=150)
plt.show()

# Evaluate
print("\n" + "=" * 60)
print("STEP 5: Evaluation")
print("=" * 60)

y_pred = model_attention.predict(X_test)
y_test_inv = scaler.inverse_transform(y_test.reshape(-1, 1)).flatten()
y_pred_inv = scaler.inverse_transform(y_pred).flatten()

mse = mean_squared_error(y_test_inv, y_pred_inv)
mae = mean_absolute_error(y_test_inv, y_pred_inv)

print(f"Test MSE: {mse:.4f}")
print(f"Test MAE: {mae:.4f}")

# Save model
model_attention.save('/mnt/okcomputer/output/lab4_2_attention_lstm.h5')

print("\n" + "=" * 60)
print("Lab 4.2 Complete!")
print("=" * 60)
```
