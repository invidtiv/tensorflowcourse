"""
Lab 2.3: Advanced Windowing Strategies
Topic: Expanding Windows and Rolling Statistics
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf

np.random.seed(42)
tf.random.set_seed(42)

# Generate Sample Data
print("=" * 60)
print("STEP 1: Generating Sample Data")
print("=" * 60)
n_points = 500
t = np.arange(n_points)
trend = 0.01 * t
seasonal = 10 * np.sin(2 * np.pi * t / 50)
noise = np.random.normal(0, 2, n_points)
data = 100 + trend + seasonal + noise
print(f"Generated {len(data)} data points")

# Expanding Windows
print("\n" + "=" * 60)
print("STEP 2: Expanding Windows")
print("=" * 60)

def create_expanding_windows(data, min_window=10, horizon=1):
    X, y = [], []
    for i in range(min_window, len(data) - horizon + 1):
        X.append(data[:i])
        y.append(data[i:i + horizon])
    return X, np.array(y)

X_expanding, y_expanding = create_expanding_windows(data, min_window=20, horizon=1)
print(f"Created {len(X_expanding)} expanding windows")
print(f"First window size: {len(X_expanding[0])}")
print(f"Last window size: {len(X_expanding[-1])}")

# Rolling Window Statistics as Features
print("\n" + "=" * 60)
print("STEP 3: Rolling Window Statistics as Features")
print("=" * 60)

def create_rolling_feature_windows(data, window_size=20, horizon=1):
    X, y = [], []
    for i in range(window_size, len(data) - horizon + 1):
        window = data[i - window_size:i]
        features = [
            window[-1], np.mean(window), np.std(window), np.min(window),
            np.max(window), np.median(window), np.percentile(window, 25),
            np.percentile(window, 75),
        ]
        x = np.arange(len(window))
        slope = np.polyfit(x, window, 1)[0]
        features.append(slope)
        X.append(features)
        y.append(data[i:i + horizon])
    return np.array(X), np.array(y)

X_rolling, y_rolling = create_rolling_feature_windows(data, window_size=20, horizon=1)
feature_names = ['Last', 'Mean', 'Std', 'Min', 'Max', 'Median', 'Q1', 'Q3', 'Slope']
print(f"X shape: {X_rolling.shape}")
print(f"Features: {feature_names}")

# Combined Windowing Strategy
print("\n" + "=" * 60)
print("STEP 4: Combined Windowing Strategy")
print("=" * 60)

def create_combined_windows(data, raw_window_size=10, stat_window_size=20, horizon=1):
    X, y = [], []
    for i in range(stat_window_size, len(data) - horizon + 1):
        raw_window = data[i - raw_window_size:i]
        stat_window = data[i - stat_window_size:i]
        stats = [np.mean(stat_window), np.std(stat_window), np.min(stat_window), np.max(stat_window)]
        features = np.concatenate([raw_window, stats])
        X.append(features)
        y.append(data[i:i + horizon])
    return np.array(X), np.array(y)

X_combined, y_combined = create_combined_windows(data, raw_window_size=10, 
                                                  stat_window_size=20, horizon=1)
print(f"Combined windows shape: {X_combined.shape}")
print(f"  - Raw values: 10")
print(f"  - Statistics: 4")
print(f"  - Total features: 14")

# Save data
np.save('/mnt/okcomputer/output/lab2_3_X_rolling.npy', X_rolling)
np.save('/mnt/okcomputer/output/lab2_3_y_rolling.npy', y_rolling)
np.save('/mnt/okcomputer/output/lab2_3_X_combined.npy', X_combined)
np.save('/mnt/okcomputer/output/lab2_3_y_combined.npy', y_combined)
print("\nSaved advanced windowed data")

print("\n" + "=" * 60)
print("Lab 2.3 Complete!")
print("=" * 60)
