# Basic Windowing for Univariate Time Series

```python
"""
Lab 2.1: Basic Windowing for Univariate Time Series
Topic: Creating Training Windows
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from sklearn.preprocessing import MinMaxScaler

np.random.seed(42)
tf.random.set_seed(42)

# Generate Sample Time Series Data
def generate_sine_wave(n_points=1000, freq=0.1, noise_level=0.1):
    t = np.arange(n_points)
    signal = np.sin(2 * np.pi * freq * t)
    noise = np.random.normal(0, noise_level, n_points)
    return signal + noise

print("=" * 60)
print("STEP 1: Generating Sample Time Series")
print("=" * 60)
data = generate_sine_wave(n_points=1000, freq=0.02, noise_level=0.1)
print(f"Generated {len(data)} data points")

# Implement Sliding Window (NumPy)
print("\n" + "=" * 60)
print("STEP 2: Sliding Window Implementation (NumPy)")
print("=" * 60)

def create_windows_numpy(data, window_size, horizon=1):
    X, y = [], []
    for i in range(len(data) - window_size - horizon + 1):
        X.append(data[i:(i + window_size)])
        y.append(data[(i + window_size):(i + window_size + horizon)])
    return np.array(X), np.array(y)

window_size = 20
horizon = 1
X, y = create_windows_numpy(data, window_size, horizon)
print(f"Window size: {window_size}")
print(f"Horizon: {horizon}")
print(f"X shape: {X.shape}")
print(f"y shape: {y.shape}")

# TensorFlow Dataset Windowing
print("\n" + "=" * 60)
print("STEP 3: TensorFlow Dataset Windowing")
print("=" * 60)

def create_windows_tf(data, window_size, horizon=1, batch_size=32, shuffle=True):
    dataset = tf.data.Dataset.from_tensor_slices(data)
    dataset = dataset.window(window_size + horizon, shift=1, drop_remainder=True)
    dataset = dataset.flat_map(lambda x: x.batch(window_size + horizon))
    dataset = dataset.map(lambda x: (x[:window_size], x[window_size:]))
    if shuffle:
        dataset = dataset.shuffle(buffer_size=1000)
    dataset = dataset.batch(batch_size).prefetch(tf.data.AUTOTUNE)
    return dataset

tf_dataset = create_windows_tf(data, window_size=20, horizon=1, batch_size=32)
print("TensorFlow Dataset structure:")
for batch_x, batch_y in tf_dataset.take(1):
    print(f"Batch X shape: {batch_x.shape}")
    print(f"Batch y shape: {batch_y.shape}")

# Multi-Horizon Windowing
print("\n" + "=" * 60)
print("STEP 4: Multi-Horizon Windowing")
print("=" * 60)

horizons = [1, 3, 7, 14]
for horizon in horizons:
    X_multi, y_multi = create_windows_numpy(data, window_size=20, horizon=horizon)
    print(f"Horizon {horizon}: X shape {X_multi.shape}, y shape {y_multi.shape}")

# Train-Validation-Test Split
print("\n" + "=" * 60)
print("STEP 5: Train-Validation-Test Split")
print("=" * 60)

def split_time_series(data, train_ratio=0.7, val_ratio=0.15):
    n = len(data)
    train_end = int(n * train_ratio)
    val_end = int(n * (train_ratio + val_ratio))
    train = data[:train_end]
    val = data[train_end:val_end]
    test = data[val_end:]
    return train, val, test

train_data, val_data, test_data = split_time_series(data)
scaler = MinMaxScaler()
train_scaled = scaler.fit_transform(train_data.reshape(-1, 1)).flatten()
val_scaled = scaler.transform(val_data.reshape(-1, 1)).flatten()
test_scaled = scaler.transform(test_data.reshape(-1, 1)).flatten()

X_train, y_train = create_windows_numpy(train_scaled, window_size, horizon)
X_val, y_val = create_windows_numpy(val_scaled, window_size, horizon)
X_test, y_test = create_windows_numpy(test_scaled, window_size, horizon)

print(f"X_train: {X_train.shape}, y_train: {y_train.shape}")
print(f"X_val: {X_val.shape}, y_val: {y_val.shape}")
print(f"X_test: {X_test.shape}, y_test: {y_test.shape}")

# Save data
np.save('/mnt/okcomputer/output/lab2_1_X_train.npy', X_train)
np.save('/mnt/okcomputer/output/lab2_1_y_train.npy', y_train)
np.save('/mnt/okcomputer/output/lab2_1_X_val.npy', X_val)
np.save('/mnt/okcomputer/output/lab2_1_y_val.npy', y_val)
np.save('/mnt/okcomputer/output/lab2_1_X_test.npy', X_test)
np.save('/mnt/okcomputer/output/lab2_1_y_test.npy', y_test)
import joblib
joblib.dump(scaler, '/mnt/okcomputer/output/lab2_1_scaler.pkl')
print("\nSaved windowed data for next labs")

print("\n" + "=" * 60)
print("Lab 2.1 Complete!")
print("=" * 60)
```
