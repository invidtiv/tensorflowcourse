# Bidirectional LSTM for Forecasting

### Complete Implementation

```python
"""
Lab 4.3: Bidirectional LSTM for Forecasting
Topic: Weather Forecasting with BiLSTM
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error, mean_absolute_error

np.random.seed(42)
tf.random.set_seed(42)

# Generate Weather Data
print("=" * 60)
print("STEP 1: Generating Weather Data")
print("=" * 60)

def generate_weather_data(n_days=365*2):
    dates = pd.date_range(start='2020-01-01', periods=n_days, freq='D')
    day_of_year = np.arange(n_days) % 365
    temp = 15 + 10 * np.sin(2 * np.pi * day_of_year / 365 - np.pi/2)
    temp += np.random.normal(0, 3, n_days)
    return pd.DataFrame({'temperature': temp}, index=dates)

weather_data = generate_weather_data(365*2)
print(f"Generated {len(weather_data)} days of weather data")

scaler = MinMaxScaler()
scaled_data = scaler.fit_transform(weather_data[['temperature']]).flatten()

# Create Windows
print("\n" + "=" * 60)
print("STEP 2: Creating Windows")
print("=" * 60)

def create_windows(data, window_size, horizon=1):
    X, y = [], []
    for i in range(len(data) - window_size - horizon + 1):
        X.append(data[i:(i + window_size)])
        y.append(data[i + window_size])
    return np.array(X), np.array(y)

window_size = 30
X, y = create_windows(scaled_data, window_size, horizon=1)
X = X.reshape(X.shape[0], X.shape[1], 1)

train_size = int(0.7 * len(X))
X_train, y_train = X[:train_size], y[:train_size]
X_val, y_val = X[train_size:int(0.85*len(X))], y[train_size:int(0.85*len(X))]
X_test, y_test = X[int(0.85*len(X)):], y[int(0.85*len(X)):]
print(f"X_train: {X_train.shape}")

# Build Bidirectional LSTM
print("\n" + "=" * 60)
print("STEP 3: Building Bidirectional LSTM")
print("=" * 60)

def build_bidirectional_lstm(window_size, n_features=1):
    model = keras.Sequential([
        layers.Bidirectional(
            layers.LSTM(64, return_sequences=True),
            input_shape=(window_size, n_features)
        ),
        layers.Dropout(0.2),
        layers.Bidirectional(layers.LSTM(32, return_sequences=False)),
        layers.Dropout(0.2),
        layers.Dense(16, activation='relu'),
        layers.Dense(1)
    ])
    model.compile(optimizer='adam', loss='mse', metrics=['mae'])
    return model

model_bilstm = build_bidirectional_lstm(window_size)
model_bilstm.summary()

# Train Model
print("\n" + "=" * 60)
print("STEP 4: Training Bidirectional LSTM")
print("=" * 60)

early_stop = keras.callbacks.EarlyStopping(
    monitor='val_loss', patience=10, restore_best_weights=True
)

history = model_bilstm.fit(
    X_train, y_train,
    validation_data=(X_val, y_val),
    epochs=50,
    batch_size=32,
    callbacks=[early_stop],
    verbose=1
)

# Plot training history
plt.figure(figsize=(10, 4))
plt.plot(history.history['loss'], label='Train')
plt.plot(history.history['val_loss'], label='Validation')
plt.title('Training Loss')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab4_3_bilstm_history.png', dpi=150)
plt.show()

# Evaluate
print("\n" + "=" * 60)
print("STEP 5: Evaluation")
print("=" * 60)

y_pred = model_bilstm.predict(X_test)
y_test_inv = scaler.inverse_transform(y_test.reshape(-1, 1)).flatten()
y_pred_inv = scaler.inverse_transform(y_pred).flatten()

mse = mean_squared_error(y_test_inv, y_pred_inv)
mae = mean_absolute_error(y_test_inv, y_pred_inv)
rmse = np.sqrt(mse)

print(f"Test MSE: {mse:.4f}")
print(f"Test MAE: {mae:.4f}")
print(f"Test RMSE: {rmse:.4f}")

# Visualize predictions
plt.figure(figsize=(15, 5))
plt.plot(y_test_inv[:60], label='Actual', linewidth=2)
plt.plot(y_pred_inv[:60], label='Predicted', linewidth=2, alpha=0.8)
plt.title('Bidirectional LSTM Predictions')
plt.xlabel('Day')
plt.ylabel('Temperature (C)')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab4_3_bilstm_predictions.png', dpi=150)
plt.show()

# Save model
model_bilstm.save('/mnt/okcomputer/output/lab4_3_bilstm_model.h5')

print("\n" + "=" * 60)
print("Lab 4.3 Complete!")
print("=" * 60)
```
