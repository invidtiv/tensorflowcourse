import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from sklearn.preprocessing import MinMaxScaler

np.random.seed(42)
tf.random.set_seed(42)

# Generate Multivariate Time Series
def generate_multivariate_series(n_points=1000):
    t = np.arange(n_points)
    base = np.sin(2 * np.pi * 0.02 * t)
    var1 = base + np.random.normal(0, 0.1, n_points)
    var2 = np.roll(base, 5) + np.random.normal(0, 0.15, n_points)
    var3 = -0.5 * base + np.random.normal(0, 0.2, n_points)
    trend = 0.001 * t
    var4 = trend + 0.5 * base + np.random.normal(0, 0.1, n_points)
    return np.column_stack([var1, var2, var3, var4])

print("=" * 60)
print("STEP 1: Generating Multivariate Time Series")
print("=" * 60)
data = generate_multivariate_series(n_points=1000)
var_names = ['Signal_A', 'Signal_B', 'Signal_C', 'Signal_D']
print(f"Generated data shape: {data.shape}")

# Basic Multivariate Windowing
print("\n" + "=" * 60)
print("STEP 2: Basic Multivariate Windowing")
print("=" * 60)

def create_multivariate_windows(data, window_size, target_col=0, horizon=1):
    X, y = [], []
    for i in range(len(data) - window_size - horizon + 1):
        X.append(data[i:(i + window_size)])
        y.append(data[i + window_size:i + window_size + horizon, target_col])
    return np.array(X), np.array(y)

window_size = 20
X_multi, y_multi = create_multivariate_windows(data, window_size, target_col=0, horizon=1)
print(f"X shape: {X_multi.shape} (samples, timesteps, features)")
print(f"y shape: {y_multi.shape}")

# TensorFlow Dataset for Multivariate Data
print("\n" + "=" * 60)
print("STEP 3: TensorFlow Dataset for Multivariate Data")
print("=" * 60)

def create_multivariate_tf_dataset(data, window_size, target_col=0, horizon=1, 
                                    batch_size=32, shuffle=True):
    data_tensor = tf.constant(data, dtype=tf.float32)
    dataset = tf.data.Dataset.from_tensor_slices(data_tensor)
    dataset = dataset.window(window_size + horizon, shift=1, drop_remainder=True)
    dataset = dataset.flat_map(lambda x: x.batch(window_size + horizon))
    def split_window(window):
        inputs = window[:window_size]
        labels = window[window_size:, target_col]
        return inputs, labels
    dataset = dataset.map(split_window)
    if shuffle:
        dataset = dataset.shuffle(buffer_size=1000)
    dataset = dataset.batch(batch_size).prefetch(tf.data.AUTOTUNE)
    return dataset

tf_dataset_multi = create_multivariate_tf_dataset(data, window_size=20, target_col=0, horizon=1)
print("TensorFlow Dataset structure:")
for batch_x, batch_y in tf_dataset_multi.take(1):

    return np.array(X), np.array(y)

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Fitting one shared scaler across all features | A single `MinMaxScaler.fit()` on a 2D array of shape `(N, F)` scales each feature by the *global* min/max across all features, distorting relative magnitudes | Use `scaler.fit(data)` where `data` is the 2D `(N, F)` array; `MinMaxScaler` operates per-column by default when the input is 2D |
| Target channel included in input scaler | If the target variable is in both `X` and `y`, scaling `X` after filtering it out but fitting on the full frame leaks target statistics into feature scaling | Separate input features from the target column before fitting any scaler |
| Temporal split applied after windowing | Splitting `(X, y)` arrays with random indices instead of the raw series means windows near the boundary appear in both train and val | Split the raw series first, then window each split independently |
| Returning 2D `(N, W×F)` instead of 3D `(N, W, F)` | A common mistake is `X.append(data[i:i+W].flatten())` which collapses the time and feature dimensions | Use `X.append(data[i:i+W])` (no flatten); final array should be `np.array(X)` with shape `(N, W, F)` |
| Ignoring lag structure between features | Feeding all features at the same timestep ignores that some channels lead others | Inspect cross-correlation (`df.shift(k).corrwith(target)` for k=1…W) and add optimal-lag copies as extra channels |
