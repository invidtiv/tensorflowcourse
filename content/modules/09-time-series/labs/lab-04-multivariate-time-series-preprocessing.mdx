# Generate Synthetic Energy Data
def generate_energy_data(n_hours=24*365*2):
    dates = pd.date_range(start='2020-01-01', periods=n_hours, freq='H')
    hour = np.arange(n_hours) % 24
    day_of_week = (np.arange(n_hours) // 24) % 7
    month = ((np.arange(n_hours) // 24) % 365) / 30.44 % 12
    temp_daily = 5 * np.sin(2 * np.pi * hour / 24 - np.pi/2)
    temp_seasonal = 15 * np.sin(2 * np.pi * ((np.arange(n_hours) // 24) % 365) / 365 - np.pi/2)
    temperature = 20 + temp_daily + temp_seasonal + np.random.normal(0, 2, n_hours)
    humidity = 70 - 0.5 * temperature + 10 * np.sin(2 * np.pi * hour / 24) + np.random.normal(0, 5, n_hours)
    humidity = np.clip(humidity, 20, 100)
    wind_speed = 10 + 5 * np.sin(2 * np.pi * ((np.arange(n_hours) // 24) % 365) / 365) + np.random.exponential(3, n_hours)
    wind_speed = np.clip(wind_speed, 0, 50)
    base_demand = 500
    hourly_pattern = 100 * (np.sin(2 * np.pi * hour / 24 - np.pi/3) + 0.5 * np.sin(4 * np.pi * hour / 24))
    weekly_pattern = np.where(day_of_week < 5, 50, -30)
    temp_effect = 2 * (temperature - 20)**2
    seasonal_effect = 30 * np.sin(2 * np.pi * ((np.arange(n_hours) // 24) % 365) / 365)
    demand = base_demand + hourly_pattern + weekly_pattern + temp_effect + seasonal_effect + np.random.normal(0, 20, n_hours)
    demand = np.maximum(demand, 200)
    df = pd.DataFrame({
        'demand': demand,
        'temperature': temperature,
        'humidity': humidity,
        'wind_speed': wind_speed,
        'hour': hour,
        'day_of_week': day_of_week,
        'month': month.astype(int)
    }, index=dates)
    return df

print("=" * 60)
print("STEP 1: Generating Multivariate Energy Data")

energy_data = generate_energy_data(n_hours=24*365*2)

def create_multivariate_features(df):
    data = df.copy()
    data['hour_sin'] = np.sin(2 * np.pi * data['hour'] / 24)
    data['hour_cos'] = np.cos(2 * np.pi * data['hour'] / 24)
    data['dow_sin'] = np.sin(2 * np.pi * data['day_of_week'] / 7)
    data['dow_cos'] = np.cos(2 * np.pi * data['day_of_week'] / 7)
    data['month_sin'] = np.sin(2 * np.pi * data['month'] / 12)
    data['month_cos'] = np.cos(2 * np.pi * data['month'] / 12)
    for lag in [1, 2, 3, 6, 12, 24, 48]:
        data[f'demand_lag_{lag}'] = data['demand'].shift(lag)
    for lag in [1, 6, 24]:
        data[f'temp_lag_{lag}'] = data['temperature'].shift(lag)
    for window in [6, 12, 24]:
        data[f'demand_ma_{window}'] = data['demand'].rolling(window=window).mean()
        data[f'demand_std_{window}'] = data['demand'].rolling(window=window).std()
    data['demand_diff_1'] = data['demand'].diff(1)
    data['demand_diff_24'] = data['demand'].diff(24)
    data['temp_humidity'] = data['temperature'] * data['humidity']
    data['temp_squared'] = data['temperature'] ** 2
    data['is_weekend'] = (data['day_of_week'] >= 5).astype(int)
    data['is_peak_hour'] = ((data['hour'] >= 8) & (data['hour'] <= 10) | 
                            (data['hour'] >= 18) & (data['hour'] <= 21)).astype(int)
    return data

print(f"Test set: X={X_test_scaled.shape}, y={y_test_scaled.shape}")

np.save('/mnt/okcomputer/output/lab1_3_y_test.npy', y_test_scaled)
joblib.dump(scaler_X, '/mnt/okcomputer/output/lab1_3_scaler_X.pkl')
joblib.dump(scaler_y, '/mnt/okcomputer/output/lab1_3_scaler_y.pkl')
pd.Series(X_cols).to_csv('/mnt/okcomputer/output/lab1_3_feature_names.csv', index=False)
print("\nSaved all preprocessed data")

print("\n" + "=" * 60)

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Fitting one scaler for all features jointly | Each feature has its own range; joint scaling distorts relative magnitudes | Fit a separate `MinMaxScaler` (or `StandardScaler`) per feature, or use `feature_range` per column via a `ColumnTransformer` |
| Scaler target leakage | Fitting the feature scaler on `y` values or including target in `scaler_X.fit()` inflates apparent training accuracy | Fit `scaler_X` on `X_train` only; fit `scaler_y` on `y_train` only; keep them separate objects |
| Temporal split ignoring autocorrelation horizon | If the gap between train end and val start is smaller than the forecast horizon, the last train windows overlap with first val windows | Add a `gap = horizon` buffer between splits: `val_start = train_end + horizon` |
| High-cardinality calendar features without encoding | Hour-of-day (0–23) passed as a raw integer looks like a continuous variable; the model treats 23 and 0 as far apart | Encode cyclical features as `sin`/`cos` pairs: `sin(2π·hour/24)`, `cos(2π·hour/24)` |
| NaN propagation from lag features | Lag features create NaN in the first `max_lag` rows, which silently produce 0s after scaling | Drop or fill NaN before scaling; use `.dropna()` after constructing lag columns and before the train/val split |
