# LSTM Autoencoder for Anomaly Detection

"""
Lab 6.1: LSTM Autoencoder for Anomaly Detection
Topic: Sensor Data Anomaly Detection
"""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import precision_recall_curve, roc_curve, auc

np.random.seed(42)
tf.random.set_seed(42)

# Generate Normal and Anomalous Sensor Data
print("=" * 60)
print("STEP 1: Generating Sensor Data")
print("=" * 60)

def generate_sensor_data(n_points=2000, anomaly_ratio=0.05):
    t = np.arange(n_points)
    normal_signal = 50 + 10 * np.sin(2 * np.pi * t / 100) + np.random.normal(0, 2, n_points)
    n_anomalies = int(n_points * anomaly_ratio)
    anomaly_indices = np.random.choice(n_points, n_anomalies, replace=False)
    anomalous_signal = normal_signal.copy()
    labels = np.zeros(n_points)
    for idx in anomaly_indices:
        if np.random.rand() < 0.5:
            anomalous_signal[idx] += np.random.choice([-1, 1]) * np.random.uniform(15, 25)
        else:
            anomalous_signal[idx:min(idx+5, n_points)] += np.random.choice([-1, 1]) * np.random.uniform(10, 15)
        labels[idx] = 1
    return anomalous_signal, labels

data, labels = generate_sensor_data(2000, anomaly_ratio=0.05)
print(f"Generated {len(data)} data points")
print(f"Anomalies: {int(labels.sum())} ({labels.mean()*100:.1f}%)")

# Visualize
plt.figure(figsize=(15, 5))
plt.plot(data, color='blue', alpha=0.7, label='Sensor Data')
anomaly_indices = np.where(labels == 1)[0]
plt.scatter(anomaly_indices, data[anomaly_indices], color='red', s=30, label='Anomalies', zorder=5)
plt.title('Sensor Data with Anomalies')
plt.xlabel('Time Step')
plt.ylabel('Sensor Value')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab6_1_sensor_data.png', dpi=150)
plt.show()

# Create Windows for Autoencoder
print("\n" + "=" * 60)
print("STEP 2: Creating Windows")
print("=" * 60)

def create_windows(data, window_size):
    X = []
    for i in range(len(data) - window_size + 1):
        X.append(data[i:(i + window_size)])
    return np.array(X)

window_size = 60
X = create_windows(data, window_size)
X = X.reshape(X.shape[0], X.shape[1], 1)
windowed_labels = labels[window_size-1:]

print(f"X shape: {X.shape}")
print(f"Windowed labels shape: {windowed_labels.shape}")

# Split into train (normal only), val, test
train_size = int(0.6 * len(X))
val_size = int(0.2 * len(X))
X_train = X[:train_size]
X_val = X[train_size:train_size + val_size]
X_test = X[train_size + val_size:]
y_val = windowed_labels[train_size:train_size + val_size]
y_test = windowed_labels[train_size + val_size:]

# Filter training data to include only normal samples
normal_train_mask = windowed_labels[:train_size] == 0
X_train_normal = X_train[normal_train_mask]
print(f"X_train_normal (normal only): {X_train_normal.shape}")
print(f"X_val: {X_val.shape}")
print(f"X_test: {X_test.shape}")

# Build LSTM Autoencoder
print("\n" + "=" * 60)
print("STEP 3: Building LSTM Autoencoder")
print("=" * 60)

def build_lstm_autoencoder(window_size, n_features=1):
    inputs = layers.Input(shape=(window_size, n_features))
    encoded = layers.LSTM(32, return_sequences=True)(inputs)
    encoded = layers.LSTM(16, return_sequences=False)(encoded)
    bottleneck = layers.RepeatVector(window_size)(encoded)
    decoded = layers.LSTM(16, return_sequences=True)(bottleneck)
    decoded = layers.LSTM(32, return_sequences=True)(decoded)
    outputs = layers.TimeDistributed(layers.Dense(n_features))(decoded)
    model = keras.Model(inputs, outputs)
    model.compile(optimizer='adam', loss='mse')
    return model

autoencoder = build_lstm_autoencoder(window_size)
autoencoder.summary()

# Train Autoencoder
print("\n" + "=" * 60)
print("STEP 4: Training Autoencoder")
print("=" * 60)

early_stop = keras.callbacks.EarlyStopping(
    monitor='val_loss', patience=10, restore_best_weights=True
)

history = autoencoder.fit(
    X_train_normal, X_train_normal,
    validation_data=(X_val, X_val),
    epochs=50,
    batch_size=32,
    callbacks=[early_stop],
    verbose=1
)

# Plot training history
plt.figure(figsize=(10, 4))
plt.plot(history.history['loss'], label='Train')
plt.plot(history.history['val_loss'], label='Validation')
plt.title('Autoencoder Training Loss')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('/mnt/okcomputer/output/lab6_1_autoencoder_history.png', dpi=150)
plt.show()

# Calculate Reconstruction Error
print("\n" + "=" * 60)
print("STEP 5: Calculating Reconstruction Error")
print("=" * 60)

X_test_pred = autoencoder.predict(X_test)
reconstruction_error = np.mean(np.abs(X_test - X_test_pred), axis=(1, 2))

print(f"Reconstruction error shape: {reconstruction_error.shape}")
print(f"Mean error: {reconstruction_error.mean():.4f}")
print(f"Std error: {reconstruction_error.std():.4f}")

# Visualize reconstruction error distribution
fig, axes = plt.subplots(1, 2, figsize=(15, 5))
axes[0].hist(reconstruction_error[y_test == 0], bins=50, alpha=0.7, label='Normal', color='blue')
axes[0].hist(reconstruction_error[y_test == 1], bins=50, alpha=0.7, label='Anomaly', color='red')
axes[0].set_title('Reconstruction Error Distribution')
axes[0].set_xlabel('Reconstruction Error')
axes[0].set_ylabel('Frequency')
axes[0].legend()
axes[0].grid(True, alpha=0.3)

axes[1].plot(reconstruction_error, color='blue', alpha=0.7)
axes[1].scatter(np.where(y_test == 1)[0], reconstruction_error[y_test == 1], 
                color='red', s=30, label='Anomalies', zorder=5)
axes[1].set_title('Reconstruction Error Over Time')
axes[1].set_xlabel('Time Step')
axes[1].set_ylabel('Reconstruction Error')
axes[1].legend()
axes[1].grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('/mnt/okcomputer/output/lab6_1_reconstruction_error.png', dpi=150)
plt.show()

# Determine Threshold and Evaluate
print("\n" + "=" * 60)
print("STEP 6: Determining Threshold and Evaluating")
print("=" * 60)

X_val_pred = autoencoder.predict(X_val)
val_error = np.mean(np.abs(X_val - X_val_pred), axis=(1, 2))
normal_val_error = val_error[y_val == 0]
threshold = np.percentile(normal_val_error, 95)
print(f"Threshold (95th percentile): {threshold:.4f}")

# Predict anomalies
y_pred = (reconstruction_error > threshold).astype(int)

# Calculate metrics
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix

accuracy = accuracy_score(y_test, y_pred)
precision = precision_score(y_test, y_pred)
recall = recall_score(y_test, y_pred)
f1 = f1_score(y_test, y_pred)

print(f"\nMetrics:")
print(f"Accuracy: {accuracy:.4f}")
print(f"Precision: {precision:.4f}")
print(f"Recall: {recall:.4f}")
print(f"F1 Score: {f1:.4f}")

cm = confusion_matrix(y_test, y_pred)
print(f"\nConfusion Matrix:")
print(cm)

# ROC Curve
fpr, tpr, _ = roc_curve(y_test, reconstruction_error)
roc_auc = auc(fpr, tpr)

plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
plt.plot(fpr, tpr, color='blue', lw=2, label=f'ROC Curve (AUC = {roc_auc:.3f})')
plt.plot([0, 1], [0, 1], color='gray', linestyle='--')
plt.xlabel('False Positive Rate')
plt.ylabel('True Positive Rate')
plt.title('ROC Curve')
plt.legend()
plt.grid(True, alpha=0.3)

plt.subplot(1, 2, 2)
plt.imshow(cm, interpolation='nearest', cmap=plt.cm.Blues)
plt.title('Confusion Matrix')
plt.colorbar()
plt.xlabel('Predicted')
plt.ylabel('Actual')
plt.tight_layout()
plt.savefig('/mnt/okcomputer/output/lab6_1_evaluation.png', dpi=150)
plt.show()

# Save model
autoencoder.save('/mnt/okcomputer/output/lab6_1_autoencoder.h5')

# ──────────────────────────────────────────────────────────
# VALIDATION CHECKPOINT
# ──────────────────────────────────────────────────────────
print("\n" + "=" * 60)
print("VALIDATION CHECKPOINT")
print("=" * 60)

# 1. Reconstruction error shape matches test set
assert reconstruction_error.shape[0] == X_test.shape[0], \
    f"Expected {X_test.shape[0]} reconstruction errors, got {reconstruction_error.shape[0]}"

# 2. Threshold is positive (model learned something)
assert threshold > 0, \
    f"Threshold {threshold:.6f} <= 0 — autoencoder may not have trained properly"

# 3. AUC-ROC above random baseline (> 0.55)
assert roc_auc > 0.55, \
    f"AUC-ROC {roc_auc:.3f} is near random (0.5) — model is not detecting anomalies"

# 4. Recall is above 0 (at least some anomalies detected)
assert recall > 0.0, \
    f"Recall is 0.0 — no anomalies detected at all; check threshold or model capacity"

print("✅ All validation checks passed!")
print(f"   Reconstruction errors: {reconstruction_error.shape[0]}")
print(f"   Threshold (95th pctl): {threshold:.4f}")
print(f"   AUC-ROC:              {roc_auc:.3f} (> 0.55)")
print(f"   Recall:               {recall:.3f} (> 0.0)")

print("\n" + "=" * 60)
print("Lab 6.1 Complete!")
print("=" * 60)

**Diagnostic tree — if something looks wrong:**

- `AUC-ROC ≈ 0.5` → autoencoder is reconstructing anomalies as well as normal data; likely anomalies leaked into training. Re-check that `X_train_normal` is filtered to normal-only windows.
- `Recall = 0.0` → threshold is too aggressive; the 95th percentile on val may be lower than the maximum reconstruction error on normal test windows. Try the 90th percentile, or inspect the reconstruction error histogram to see if normal and anomalous distributions overlap.
- `Threshold = 0` → autoencoder never trained (optimizer or data issue); inspect training loss curve — if it starts at `0.25` and does not decrease the model is receiving constant-zero input.
- Poor F1 despite reasonable AUC → class imbalance (very few anomalies). Use `precision_recall_curve` and find the threshold that maximises F1 explicitly, rather than relying on the 95th percentile heuristic.

</Callout>

<Callout type="pitfall" title="Common Pitfalls">

- **Training on anomalous data** — The autoencoder must be trained on *normal data only*. If anomalies leak into the training set, the model learns to reconstruct them, defeating the purpose. Always filter training data with known labels or use the earliest, cleanest segment of your data.
- **Window-label alignment** — When creating sliding windows, the label for a window should correspond to the *last* time step (or all steps) in that window. Off-by-one errors here silently corrupt evaluation: your confusion matrix looks fine but the model is evaluated on misaligned labels.
- **Threshold sensitivity** — The 95th-percentile threshold is a common default but is arbitrary. In production, tune it using a labeled validation set and optimize for F1 or a business-specific cost function (e.g., cost of false alarms vs. missed anomalies).
- **Reconstruction error aggregation** — `np.mean(np.abs(...), axis=(1,2))` averages across the entire window. Point anomalies buried in a 60-step window get diluted. Consider using `np.max(...)` along the time axis to catch single-step spikes, or report per-step errors.
- **Scaling before splitting** — Fit `MinMaxScaler` on training data only, then `.transform()` val and test. Fitting on the full dataset leaks future distribution information into training, inflating apparent model performance.

</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Training on anomalous windows | If anomalies leak into the training set, the autoencoder learns to reconstruct them, destroying the anomaly signal | Filter training data to normal-labeled windows only; if labels are unavailable, use the earliest chronological segment |
| Window-label alignment off by one | The label for a window should correspond to the *last* timestep; an off-by-one error silently misaligns labels with windows | Align: `label[i] = labels[i + window_size - 1]`; unit-test with a toy array where anomaly position is known |
| Threshold set arbitrarily at 95th percentile | The 95th percentile on the validation error distribution is a heuristic; in imbalanced datasets it can produce very high false-alarm rates | Tune threshold using a labeled validation set, optimising for the business cost function (F1, or weighted precision/recall) |
| Reconstruction error dilution by window averaging | `np.mean(errors, axis=(1,2))` averages a point anomaly across the entire window, making it nearly invisible | Use `np.max(errors, axis=1)` along the time axis to detect single-step spikes; report both mean and max per-window errors |
| Scaler fitted on full dataset including test anomalies | Future anomalous values (extreme spikes) inflate the scaler's range, compressing normal variation into a narrow band | Fit `MinMaxScaler` on training (normal-only) data; apply `.transform()` to val and test |
