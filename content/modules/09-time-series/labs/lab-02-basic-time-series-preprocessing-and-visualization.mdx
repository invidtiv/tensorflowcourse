import numpy as np

# Generate Synthetic Stock Price Data
def generate_stock_data(n_days=1000, trend=0.001, volatility=0.02, 
                        seasonality_period=252, seasonality_amplitude=5):
    dates = pd.bdate_range(start='2020-01-01', periods=n_days)
    trend_component = np.cumsum(np.ones(n_days) * trend)
    seasonal_component = seasonality_amplitude * np.sin(
        2 * np.pi * np.arange(n_days) / seasonality_period
    )
    random_returns = np.random.normal(0, volatility, n_days)
    price_changes = np.cumsum(random_returns)
    base_price = 100
    prices = base_price + trend_component + seasonal_component + price_changes
    prices = np.maximum(prices, 10)
    df = pd.DataFrame({'Close': prices}, index=dates)
    return df

print("=" * 60)

stock_data = generate_stock_data(n_days=1000)

# Feature Engineering - Technical Indicators
print("\n" + "=" * 60)
print("STEP 4: Feature Engineering - Technical Indicators")
print("=" * 60)

def add_technical_indicators(df):
    data = df.copy()
    data['MA_5'] = data['Close'].rolling(window=5).mean()
    data['MA_10'] = data['Close'].rolling(window=10).mean()
    data['MA_20'] = data['Close'].rolling(window=20).mean()
    data['MA_50'] = data['Close'].rolling(window=50).mean()
    data['EMA_12'] = data['Close'].ewm(span=12, adjust=False).mean()
    data['EMA_26'] = data['Close'].ewm(span=26, adjust=False).mean()
    data['MACD'] = data['EMA_12'] - data['EMA_26']
    data['MACD_Signal'] = data['MACD'].ewm(span=9, adjust=False).mean()
    data['BB_Middle'] = data['Close'].rolling(window=20).mean()
    bb_std = data['Close'].rolling(window=20).std()
    data['BB_Upper'] = data['BB_Middle'] + (bb_std * 2)
    data['BB_Lower'] = data['BB_Middle'] - (bb_std * 2)
    delta = data['Close'].diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
    rs = gain / loss
    data['RSI'] = 100 - (100 / (1 + rs))
    data['Returns'] = data['Close'].pct_change()
    data['Log_Returns'] = np.log(data['Close'] / data['Close'].shift(1))
    data['Volatility'] = data['Returns'].rolling(window=20).std()
    for lag in [1, 2, 3, 5, 10]:
        data[f'Close_Lag_{lag}'] = data['Close'].shift(lag)
    return data

stock_features = add_technical_indicators(stock_data)

standard_scaler = StandardScaler()
data_standard = standard_scaler.fit_transform(data_clean)
data_standard_df = pd.DataFrame(data_standard, columns=features_for_scaling, index=data_clean.index)

print(f"Original data shape: {data_clean.shape}")
print(f"\nMin-Max Scaled data range:")
print(data_minmax_df.describe().loc[['min', 'max']])

print("Lab 1.1 Complete!")

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Scaler fitted before train/test split | Global min/max from the full series leaks future price extremes into training statistics | Split first, then call `scaler.fit()` on the training slice only; `transform` val and test |
| Forward-fill applied after split | NaN values in val/test get filled with values from the end of the training window, injecting lookahead | Apply `ffill()` only within each split, or use `.fillna(method='ffill', limit=5)` on the full series before splitting |
| Missing stationarity check | Raw prices have a unit root (non-stationary); many models assume stationarity | Run `adfuller(series)` and report the p-value; difference or log-return the series until `p < 0.05` |
| RSI/MACD computed over entire series including test | Rolling indicator windows peek into the future for test-set rows near the split boundary | Compute indicators on training data, then roll forward using only past values for val/test |
| Confusing MinMaxScaler with StandardScaler use cases | Min-Max is bounded [0,1] but sensitive to outliers; Standard is unbounded but centers data | Use MinMaxScaler for sigmoid/output-bounded models; use StandardScaler for LSTM/dense networks with no bounded activation |
