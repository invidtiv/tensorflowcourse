# 3: Sequential vs Functional API Comparison

```python
"""
Lab 2.2.3: Sequential vs Functional API Comparison
Direct comparison of both APIs on the same task.
"""

import tensorflow as tf
import numpy as np
import time

print("=" * 60)
print("LAB 2.2.3: Sequential vs Functional API Comparison")
print("=" * 60)

# ============================================================
# STEP 1: Define Common Architectures
# ============================================================
print("\n[STEP 1] Defining equivalent architectures...")

# Architecture parameters
INPUT_DIM = 100
HIDDEN_UNITS = [128, 64, 32]
NUM_CLASSES = 10
DROPOUT_RATE = 0.3

# Sequential Version
def create_sequential_model():
    """Create model using Sequential API."""
    model = tf.keras.Sequential(name='sequential_version')
    
    # Input layer
    model.add(tf.keras.layers.Dense(HIDDEN_UNITS[0], activation='relu', 
                                    input_shape=(INPUT_DIM,)))
    model.add(tf.keras.layers.Dropout(DROPOUT_RATE))
    model.add(tf.keras.layers.BatchNormalization())
    
    # Hidden layers
    for units in HIDDEN_UNITS[1:]:
        model.add(tf.keras.layers.Dense(units, activation='relu'))
        model.add(tf.keras.layers.Dropout(DROPOUT_RATE))
        model.add(tf.keras.layers.BatchNormalization())
    
    # Output layer
    model.add(tf.keras.layers.Dense(NUM_CLASSES, activation='softmax'))
    
    return model

# Functional Version
def create_functional_model():
    """Create equivalent model using Functional API."""
    inputs = tf.keras.Input(shape=(INPUT_DIM,), name='input')
    
    x = tf.keras.layers.Dense(HIDDEN_UNITS[0], activation='relu')(inputs)
    x = tf.keras.layers.Dropout(DROPOUT_RATE)(x)
    x = tf.keras.layers.BatchNormalization()(x)
    
    for units in HIDDEN_UNITS[1:]:
        x = tf.keras.layers.Dense(units, activation='relu')(x)
        x = tf.keras.layers.Dropout(DROPOUT_RATE)(x)
        x = tf.keras.layers.BatchNormalization()(x)
    
    outputs = tf.keras.layers.Dense(NUM_CLASSES, activation='softmax')(x)
    
    return tf.keras.Model(inputs=inputs, outputs=outputs, name='functional_version')

# Create both models
seq_model = create_sequential_model()
func_model = create_functional_model()

print("\nSequential Model:")
seq_model.summary()

print("\nFunctional Model:")
func_model.summary()

# ============================================================
# STEP 2: Compare Model Properties
# ============================================================
print("\n[STEP 2] Comparing model properties...")

# Count parameters
seq_params = seq_model.count_params()
func_params = func_model.count_params()

print(f"\nParameter count:")
print(f"  Sequential: {seq_params:,}")
print(f"  Functional: {func_params:,}")
print(f"  Match: {seq_params == func_params}")

# Compare layer structure
print(f"\nLayer structure:")
print(f"  Sequential layers: {len(seq_model.layers)}")
print(f"  Functional layers: {len(func_model.layers)}")

# ============================================================
# STEP 3: Training Performance Comparison
# ============================================================
print("\n[STEP 3] Comparing training performance...")

# Generate data
np.random.seed(42)
X = np.random.randn(5000, INPUT_DIM)
y = np.random.randint(0, NUM_CLASSES, 5000)
y_cat = tf.keras.utils.to_categorical(y, NUM_CLASSES)

# Compile both models
seq_model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])
func_model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

# Train Sequential
print("\nTraining Sequential model...")
start_time = time.time()
seq_history = seq_model.fit(
    X, y_cat,
    validation_split=0.2,
    epochs=10,
    batch_size=64,
    verbose=0
)
seq_time = time.time() - start_time

# Train Functional
print("Training Functional model...")
start_time = time.time()
func_history = func_model.fit(
    X, y_cat,
    validation_split=0.2,
    epochs=10,
    batch_size=64,
    verbose=0
)
func_time = time.time() - start_time

print(f"\nTraining time:")
print(f"  Sequential: {seq_time:.2f}s")
print(f"  Functional: {func_time:.2f}s")
print(f"  Difference: {abs(seq_time - func_time):.2f}s")

print(f"\nFinal validation accuracy:")
print(f"  Sequential: {seq_history.history['val_accuracy'][-1]:.4f}")
print(f"  Functional: {func_history.history['val_accuracy'][-1]:.4f}")

# ============================================================
# STEP 4: When to Use Each API
# ============================================================
print("\n[STEP 4] When to use each API...")

print("""
SEQUENTIAL API - Use when:
- Simple stack of layers (input -> layer1 -> layer2 -> output)
- Single input, single output
- No layer sharing or branching
- Quick prototyping
- Beginners learning Keras

Advantages:
- Simpler syntax
- Less code to write
- Easier to read for simple models

FUNCTIONAL API - Use when:
- Multiple inputs or outputs
- Shared layers (same layer called multiple times)
- Non-linear topology (residual connections, multi-branch)
- Need to inspect/manipulate intermediate layers
- Converting models to other formats

Advantages:
- More flexible
- Handles complex architectures
- Better for production code
- Easier model visualization
""")

# ============================================================
# STEP 5: Complex Architecture Example
# ============================================================
print("\n[STEP 5] Complex architecture - Only possible with Functional API...")

# Multi-modal model (image + text + metadata)
# This cannot be done with Sequential API

# Image input branch
image_input = tf.keras.Input(shape=(224, 224, 3), name='image')
x1 = tf.keras.layers.Conv2D(32, 3, activation='relu')(image_input)
x1 = tf.keras.layers.MaxPooling2D()(x1)
x1 = tf.keras.layers.Conv2D(64, 3, activation='relu')(x1)
x1 = tf.keras.layers.MaxPooling2D()(x1)
x1 = tf.keras.layers.Flatten()(x1)
x1 = tf.keras.layers.Dense(128, activation='relu')(x1)

# Text input branch
text_input = tf.keras.Input(shape=(100,), name='text')
x2 = tf.keras.layers.Embedding(10000, 128)(text_input)
x2 = tf.keras.layers.LSTM(64)(x2)

# Metadata input branch
meta_input = tf.keras.Input(shape=(20,), name='metadata')
x3 = tf.keras.layers.Dense(32, activation='relu')(meta_input)

# Merge all branches
merged = tf.keras.layers.Concatenate(name='merge')([x1, x2, x3])

# Shared processing
x = tf.keras.layers.Dense(256, activation='relu')(merged)
x = tf.keras.layers.Dropout(0.5)(x)

# Multiple outputs
class_output = tf.keras.layers.Dense(10, activation='softmax', name='class')(x)
reg_output = tf.keras.layers.Dense(1, name='regression')(x)

multi_modal_model = tf.keras.Model(
    inputs=[image_input, text_input, meta_input],
    outputs=[class_output, reg_output],
    name='multi_modal'
)

print("\nMulti-modal Model (only possible with Functional API):")
multi_modal_model.summary()

# Visualize
tf.keras.utils.plot_model(
    multi_modal_model,
    to_file='/mnt/okcomputer/output/multi_modal_model.png',
    show_shapes=True,
    show_layer_names=True
)
print("\nModel diagram saved!")

print("\n" + "=" * 60)
print("LAB 2.2.3 COMPLETED SUCCESSFULLY!")
print("=" * 60)
```

## Expected Output

| Check | Expected result |
|-------|----------------|
| Sequential model `model.summary()` | All layers listed in order; input shape shown at top as `InputLayer`. |
| Functional model `model.summary()` | Connected-to column shows the directed graph of tensor flow. Each layer shows its upstream connection. |
| `model.predict()` equivalence | Given the same random seed for weights initialization, Sequential and Functional models with identical layer configs should produce the same output: `np.allclose(seq_out, func_out, atol=1e-5)`. |
| Multi-input Functional model | `model.input` returns a list of tensors. `len(model.inputs)` equals the number of input branches. |
| `tf.keras.utils.plot_model(model, show_shapes=True)` | Generates a PNG showing the computational graph. Functional models show branching; Sequential models show a straight chain. |

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Calling `Sequential.add()` with a Functional sub-model** | Keras allows nesting, but the Sequential `.summary()` collapses the sub-model into one block, hiding its internal shape details. | Use Functional API end-to-end when you need full visibility into a complex sub-graph. |
| **Forgetting `Model(inputs=..., outputs=...)` in Functional API** | Defining layers and calling them does not create a trainable model — you must call `tf.keras.Model(inputs=x_in, outputs=x_out)`. Forgetting this gives a plain function, not a Keras model. | Always close the Functional definition with `model = tf.keras.Model(inputs=..., outputs=...)`. |
| **Using the same layer object in two separate branches** | Keras layers are stateful — calling a shared layer on two branches shares its weights. If you intend independent weights (not shared), create two separate layer instances. | For independent branches: `dense_a = Dense(64)`, `dense_b = Dense(64)`. For weight sharing: use the same instance on both branches. |
| **Dynamic shapes crashing `plot_model`** | `plot_model` requires `pydot` and `graphviz`. On a fresh environment these may not be installed. | Run `pip install pydot graphviz` — or skip plotting and use `model.summary()` for shape inspection. |
| **Sequential model not accepting `input_shape` in later TF versions** | In TF ≥ 2.13, specifying `input_shape` in the first layer constructor (e.g., `Dense(64, input_shape=(784,))`) is deprecated. | Use an explicit `InputLayer`: `model.add(tf.keras.layers.Input(shape=(784,)))` as the first call. |

## Stretch Goals

1. **Shared encoder**: Build a Functional model where two input branches share the same `Dense(64)` encoder weights (Siamese network pattern). Verify weight sharing by checking `len(model.trainable_weights)` — it should be the same as a single-branch model, not double.
2. **Multi-output model**: Add a second output head to the Functional model that predicts an auxiliary task (e.g., classification AND regression). Compile with `loss={'main': 'categorical_crossentropy', 'aux': 'mse'}` and `loss_weights={'main': 1.0, 'aux': 0.1}`. Check that both losses appear in the training log.
3. **Subclassed model equivalent**: Re-implement the same architecture as a `tf.keras.Model` subclass using `__init__` and `call()`. Compare its `model.summary()` output — subclassed models show a less detailed summary because the graph is only traced at call time.
4. **Intermediate layer outputs**: Using the Functional model, create an "extractor" model: `extractor = tf.keras.Model(inputs=model.input, outputs=model.get_layer('dense_1').output)`. Run a batch through it to get intermediate activations — the backbone of feature visualization and transfer learning.
