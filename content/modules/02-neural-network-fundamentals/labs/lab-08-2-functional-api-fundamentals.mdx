```python
print("\n[STEP 1] Creating basic Functional API model...")

# Define inputs
inputs = tf.keras.Input(shape=(784,), name='input_layer')

# Build the model
x = tf.keras.layers.Dense(256, activation='relu', name='dense_1')(inputs)
x = tf.keras.layers.Dropout(0.3, name='dropout_1')(x)

# STEP 2: Multi-Input Model
# ============================================================

# Input 1: Image features
image_input = tf.keras.Input(shape=(784,), name='image_input')
x1 = tf.keras.layers.Dense(256, activation='relu')(image_input)
x1 = tf.keras.layers.Dropout(0.3)(x1)
x1 = tf.keras.layers.Dense(128, activation='relu')(x1)

# Input 2: Metadata features
meta_input = tf.keras.Input(shape=(10,), name='metadata_input')
x2 = tf.keras.layers.Dense(32, activation='relu')(meta_input)
x2 = tf.keras.layers.Dense(16, activation='relu')(x2)

# Concatenate inputs
combined = tf.keras.layers.Concatenate(name='concatenate')([x1, x2])

# Shared layers
x = tf.keras.layers.Dense(128, activation='relu')(combined)
x = tf.keras.layers.Dropout(0.3)(x)
x = tf.keras.layers.Dense(64, activation='relu')(x)

# Output
output = tf.keras.layers.Dense(10, activation='softmax', name='prediction')(x)

# Create multi-input model
multi_input_model = tf.keras.Model(

# ============================================================
# STEP 5: Residual Connections (ResNet-style)
# ============================================================
print("\n[STEP 5] Creating model with residual connections...")

def residual_block(x, units, dropout_rate=0.3):
    """Create a residual block."""
    # Save input for skip connection
    shortcut = x
    
    # Main path
    x = tf.keras.layers.Dense(units, activation='relu')(x)
    x = tf.keras.layers.BatchNormalization()(x)
    x = tf.keras.layers.Dropout(dropout_rate)(x)
    x = tf.keras.layers.Dense(units)(x)
    x = tf.keras.layers.BatchNormalization()(x)
    
    # Skip connection (with projection if needed)
    if shortcut.shape[-1] != units:
        shortcut = tf.keras.layers.Dense(units)(shortcut)
    
    # Add skip connection
    x = tf.keras.layers.Add()([x, shortcut])
    x = tf.keras.layers.Activation('relu')(x)
    
    return x

# Build residual model
inputs = tf.keras.Input(shape=(784,), name='input')
x = tf.keras.layers.Dense(256, activation='relu')(inputs)

# Stack residual blocks
x = residual_block(x, 256)
x = residual_block(x, 256)
x = residual_block(x, 128)
x = residual_block(x, 128)
x = residual_block(x, 64)

# Output
outputs = tf.keras.layers.Dense(10, activation='softmax')(x)

residual_model = tf.keras.Model(inputs=inputs, outputs=outputs, name='residual_model')

print("\nResidual Connection Model:")
residual_model.summary()

# STEP 7: Model Subclassing from Functional
# ============================================================
print("\n[STEP 8] Converting Functional to reusable class...")

class MultiInputClassifier(tf.keras.Model):
    """Reusable multi-input classifier."""
    
    def __init__(self, image_dim, meta_dim, num_classes, **kwargs):
        super().__init__(**kwargs)
        
        # Image branch
        self.image_dense1 = tf.keras.layers.Dense(256, activation='relu')
        self.image_dropout1 = tf.keras.layers.Dropout(0.3)
        self.image_dense2 = tf.keras.layers.Dense(128, activation='relu')
        
        # Meta branch
        self.meta_dense1 = tf.keras.layers.Dense(32, activation='relu')
        self.meta_dense2 = tf.keras.layers.Dense(16, activation='relu')
        
        # Combined
        self.concat = tf.keras.layers.Concatenate()
        self.combined_dense1 = tf.keras.layers.Dense(128, activation='relu')
        self.combined_dropout = tf.keras.layers.Dropout(0.3)
        self.combined_dense2 = tf.keras.layers.Dense(64, activation='relu')
        
        # Output
        self.output_layer = tf.keras.layers.Dense(num_classes, activation='softmax')
    
    def call(self, inputs, training=False):
        image_input, meta_input = inputs
        
        # Image branch
        x1 = self.image_dense1(image_input)
        x1 = self.image_dropout1(x1, training=training)
        x1 = self.image_dense2(x1)
        
        # Meta branch
        x2 = self.meta_dense1(meta_input)
        x2 = self.meta_dense2(x2)
        
        # Combine
        combined = self.concat([x1, x2])
        x = self.combined_dense1(combined)
        x = self.combined_dropout(x, training=training)
        x = self.combined_dense2(x)
        
        return self.output_layer(x)

# Create and test
subclassed_model = MultiInputClassifier(784, 10, 10, name='subclassed_multi_input')

# Build by calling

subclassed_model.summary()
```

## Expected Output

A successful run prints a sequence of `model.summary()` blocks. The key structural checkpoints to verify:

```
Model: "multi_input_classifier"
__________________________________________________________________________________________________
 Layer (type)                    Output Shape         Param #     Connected to
==================================================================================================
 image_input (InputLayer)        [(None, 784)]        0           []
 meta_input (InputLayer)         [(None, 10)]         0           []
 dense_img_1 (Dense)             (None, 128)          100480      ['image_input[0][0]']
 dense_meta_1 (Dense)            (None, 32)           352         ['meta_input[0][0]']
 concatenate (Concatenate)       (None, 160)          0           ['dense_img_1[0][0]',
                                                                   'dense_meta_1[0][0]']
 dense_combined_1 (Dense)        (None, 64)           10304       ['concatenate[0][0]']
 output (Dense)                  (None, 10)           650         ['dense_combined_1[0][0]']
==================================================================================================
Total params: 111,786
```

Signals to confirm:

- **"Connected to" column is populated** for every non-Input layer. If it is empty you accidentally built a Sequential model.
- **Parameter count is deterministic** from shapes: a Dense(128) on a 784-vector input produces 784·128 + 128 = 100,480 params. Mismatches here almost always mean a wrong `Input(shape=...)`.
- **`plot_model(model, show_shapes=True)`** renders a proper DAG with the two `InputLayer` nodes on top merging at the `Concatenate` node. If you see a linear chain, the second input is not wired in.
- **Shared-layer example** (if included) shows the same Dense layer's `Param #` counted **once** in the total, while `.output` is reachable from two different call sites — the Functional API's canonical weight-sharing proof.
- **Subclassed vs Functional equivalence check**: `subclassed_model.count_params() == functional_model.count_params()` should be exactly equal when both are built with the same shapes and no bias tricks.

<Callout type="warning" title="Common Pitfalls">

- **Passing a `KerasTensor` through a Python operation that isn't a layer** (`x + 1`, `x * mask`) is allowed in Keras 3 but silently fails in older TF 2.x: wrap arithmetic in `tf.keras.layers.Lambda` or use `layers.Add()` / `layers.Multiply()` for portable code.
- **Reusing the *same instance* of an Input layer across two models** ties both models' graphs together and causes `ValueError: Graph disconnected` on save. Create a fresh `Input(...)` per model.
- **Calling `Model(inputs, outputs)` with a *list of outputs* that share a loss** — Keras expects a matching list of loss functions (or a dict keyed by output name). Passing a single string loss is interpreted as "apply to output 0 only" and silently ignores the other heads.
- **Forgetting that shared layers share weights *including the bias*.** If you want shared weights but independent biases, build the weight-sharing manually via `layers.Dense(..., use_bias=False)` + a separate bias-only layer per branch.
- **Naming collisions** — two layers with `name='dense_1'` in the same model raise a confusing error about the functional graph. Always name layers explicitly in non-trivial Functional models so errors point to the right place.
- **Calling `.predict()` on a multi-input model with a single NumPy array** silently broadcasts it across inputs, producing garbage. Pass a list `[X_image, X_meta]` or a dict `{'image_input': ..., 'meta_input': ...}`.
- **Mixing `tf.keras` and standalone `keras` 3.x imports** produces two incompatible layer classes with the same name; stick to one import style per project (the course standardizes on `from tensorflow import keras`).

</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|---------------|-----|
| Calling `Model(inputs, outputs)` before all layers are connected | Incomplete DAG raises `ValueError: Graph disconnected` at model construction time | Trace every layer from an `Input(...)` node to an output tensor; the final tensor passed as `outputs` must be reachable from `inputs` |
| Reusing the same `Input` instance across two separate models | Both models share internal graph state, causing `AttributeError` on `model.save()` and unpredictable prediction behavior | Create a fresh `tf.keras.Input(shape=...)` for each model; `Input` objects are cheap to construct |
| Passing a Python list of outputs with a single string loss | Keras applies the loss only to the first output head and silently ignores the rest, giving misleadingly low total loss | Pass a matching list of loss functions `['sparse_categorical_crossentropy', 'mse']` or a dict keyed by output layer name |
| Forgetting that weight-shared layers share biases too | Independent-bias branches initialized with the same weights produce identical activations, defeating the purpose of branching | Use `layers.Dense(units, use_bias=False)` for the shared part and add a separate `layers.Bias()` or `layers.Dense(units)` per branch |
| Mixing `tf.keras` and standalone `keras` imports in the same file | Both provide a class named `Dense` but they are different Python objects; layers from one cannot be composed with layers from the other | Standardize on one import style per project; the course uses `from tensorflow import keras` throughout |

```python
# Functional API structural assertions
import tensorflow as tf

# Build the multi-input model from the lab
image_input = tf.keras.Input(shape=(784,), name='image_input')
meta_input  = tf.keras.Input(shape=(10,),  name='meta_input')
x_img  = tf.keras.layers.Dense(128, activation='relu', name='dense_img_1')(image_input)
x_meta = tf.keras.layers.Dense(32,  activation='relu', name='dense_meta_1')(meta_input)
merged = tf.keras.layers.Concatenate()([x_img, x_meta])
x      = tf.keras.layers.Dense(64, activation='relu', name='dense_combined_1')(merged)
output = tf.keras.layers.Dense(10, activation='softmax', name='output')(x)
model  = tf.keras.Model(inputs=[image_input, meta_input], outputs=output,
                        name='multi_input_classifier')

assert len(model.inputs) == 2, "Model must have exactly 2 input nodes"
assert model.count_params() == 111786, f"Expected 111 786 params, got {model.count_params()}"
assert model.output_shape == (None, 10), f"Expected output shape (None, 10), got {model.output_shape}"
layer_names = [l.name for l in model.layers]
assert 'dense_img_1'      in layer_names, "dense_img_1 layer missing"
assert 'dense_meta_1'     in layer_names, "dense_meta_1 layer missing"
assert 'dense_combined_1' in layer_names, "dense_combined_1 layer missing"
print("✅ All Functional API structural assertions passed!")
```
