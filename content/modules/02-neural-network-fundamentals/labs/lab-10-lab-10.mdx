# Method 1: Function-based custom loss
def custom_mse(y_true, y_pred):
    """Custom MSE with additional penalty for large errors."""
    squared_error = tf.square(y_true - y_pred)
    # Add extra penalty for errors > 1
    large_error_penalty = tf.where(
        tf.abs(y_true - y_pred) > 1.0,
        squared_error * 0.5,  # Extra penalty
        tf.zeros_like(squared_error)
    )
    return tf.reduce_mean(squared_error + large_error_penalty)

class WeightedMSE(tf.keras.losses.Loss):
    """
    MSE with per-sample weights and feature weights.
    
    Args:
        sample_weights: Weight for each sample
        feature_weights: Weight for each feature dimension
    """
    
    def __init__(self, sample_weights=None, feature_weights=None, name='weighted_mse'):
        super().__init__(name=name)
        self.sample_weights = sample_weights
        self.feature_weights = feature_weights
    
    def call(self, y_true, y_pred):
        # Calculate squared error
        squared_error = tf.square(y_true - y_pred)
        
        # Apply feature weights if provided
        if self.feature_weights is not None:
            squared_error = squared_error * self.feature_weights
        
        # Mean across features
        sample_loss = tf.reduce_mean(squared_error, axis=-1)
        
        # Apply sample weights if provided
        if self.sample_weights is not None:
            sample_loss = sample_loss * self.sample_weights
        
        # Return mean across samples
        return tf.reduce_mean(sample_loss)

class FocalLoss(tf.keras.losses.Loss):
    """
    Focal Loss for addressing class imbalance.
    
    Down-weights easy examples and focuses on hard examples.
    
    Reference: Lin et al., "Focal Loss for Dense Object Detection", ICCV 2017
    
    Args:
        alpha: Weighting factor for rare class
        gamma: Focusing parameter (higher = more focus on hard examples)
    """
    
    def __init__(self, alpha=0.25, gamma=2.0, name='focal_loss'):
        super().__init__(name=name)
        self.alpha = alpha
        self.gamma = gamma
    
    def call(self, y_true, y_pred):
        # Clip predictions to prevent log(0)
        epsilon = tf.keras.backend.epsilon()
        y_pred = tf.clip_by_value(y_pred, epsilon, 1.0 - epsilon)
        
        # Cross entropy
        cross_entropy = -y_true * tf.math.log(y_pred)
        
        # Focal weight: (1 - p_t)^gamma
        p_t = (y_true * y_pred) + ((1 - y_true) * (1 - y_pred))
        focal_weight = tf.pow(1.0 - p_t, self.gamma)
        
        # Apply alpha weighting
        alpha_t = y_true * self.alpha + (1 - y_true) * (1 - self.alpha)
        
        # Focal loss
        loss = alpha_t * focal_weight * cross_entropy
        
        return tf.reduce_mean(loss)

class TverskyLoss(tf.keras.losses.Loss):
    """
    Tversky Loss for image segmentation.
    
    Generalization of Dice loss that allows weighting of false positives
    and false negatives differently.
    
    Args:
        alpha: Weight for false negatives
        beta: Weight for false positives
        smooth: Smoothing factor to avoid division by zero
    """
    
    def __init__(self, alpha=0.5, beta=0.5, smooth=1.0, name='tversky_loss'):
        super().__init__(name=name)
        self.alpha = alpha
        self.beta = beta
        self.smooth = smooth
    
    def call(self, y_true, y_pred):
        # Flatten
        y_true_f = tf.reshape(y_true, [-1])
        y_pred_f = tf.reshape(y_pred, [-1])
        
        # True positives, false positives, false negatives
        tp = tf.reduce_sum(y_true_f * y_pred_f)
        fp = tf.reduce_sum((1 - y_true_f) * y_pred_f)
        fn = tf.reduce_sum(y_true_f * (1 - y_pred_f))
        
        # Tversky index
        tversky = (tp + self.smooth) / (tp + self.alpha * fn + self.beta * fp + self.smooth)
        
        # Tversky loss
        return 1 - tversky

# Method 1: Function-based metric
def r_squared(y_true, y_pred):
    """R-squared (coefficient of determination) metric."""
    ss_res = tf.reduce_sum(tf.square(y_true - y_pred))
    ss_tot = tf.reduce_sum(tf.square(y_true - tf.reduce_mean(y_true)))
    return 1 - ss_res / (ss_tot + tf.keras.backend.epsilon())

# Method 2: Class-based metric
class PrecisionAtRecall(tf.keras.metrics.Metric):
    """
    Precision at a specific recall threshold.
    
    Useful when you need to guarantee a minimum recall.
    """
    
    def __init__(self, recall_threshold=0.8, name='precision_at_recall', **kwargs):
        super().__init__(name=name, **kwargs)
        self.recall_threshold = recall_threshold
        
        # Variables to track true positives and false positives at threshold
        self.true_positives = self.add_weight(name='tp', initializer='zeros')
        self.false_positives = self.add_weight(name='fp', initializer='zeros')
        self.false_negatives = self.add_weight(name='fn', initializer='zeros')
    
    def update_state(self, y_true, y_pred, sample_weight=None):
        # Convert predictions to binary at recall threshold
        # This is a simplified version - in practice you'd use precision_recall_curve
        y_pred_binary = tf.cast(y_pred > 0.5, tf.float32)
        y_true = tf.cast(y_true, tf.float32)
        
        # Calculate TP, FP, FN
        tp = tf.reduce_sum(y_true * y_pred_binary)
        fp = tf.reduce_sum((1 - y_true) * y_pred_binary)
        fn = tf.reduce_sum(y_true * (1 - y_pred_binary))
        
        self.true_positives.assign_add(tp)
        self.false_positives.assign_add(fp)
        self.false_negatives.assign_add(fn)
    
    def result(self):
        precision = self.true_positives / (self.true_positives + self.false_positives + 
                                           tf.keras.backend.epsilon())
        return precision
    
    def reset_state(self):
        self.true_positives.assign(0.0)
        self.false_positives.assign(0.0)
        self.false_negatives.assign(0.0)

class CompositeLoss(tf.keras.losses.Loss):
    """
    Combines multiple losses with different weights.
    
    Useful for multi-task learning.
    """
    
    def __init__(self, losses, weights, name='composite_loss'):
        super().__init__(name=name)
        self.losses = losses
        self.weights = weights
    
    def call(self, y_true, y_pred):
        total_loss = 0.0
        for loss_fn, weight in zip(self.losses, self.weights):
            total_loss += weight * loss_fn(y_true, y_pred)
        return total_loss

    ],
    weights=[0.5, 0.3, 0.2]
)
