"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

print("LAB 2.11: Building with Functional API")
print("=" * 60)

# ============================================================
# STEP 1: Basic Functional API Model
# ============================================================
print("\n[STEP 1] Creating basic Functional API model...")

# Define inputs
inputs = tf.keras.Input(shape=(784,), name='input')

# Build model
x = tf.keras.layers.Dense(256, activation='relu', name='dense1')(inputs)
x = tf.keras.layers.Dropout(0.3, name='dropout1')(x)
x = tf.keras.layers.Dense(128, activation='relu', name='dense2')(x)
x = tf.keras.layers.Dropout(0.3, name='dropout2')(x)
outputs = tf.keras.layers.Dense(10, activation='softmax', name='output')(x)

# Create model
model = tf.keras.Model(inputs=inputs, outputs=outputs, name='basic_functional')

print("\nBasic Functional Model:")
model.summary()

# ============================================================
# STEP 2: Multi-Input Model
# ============================================================
print("\n[STEP 2] Creating multi-input model...")

# Input 1: Image features
image_input = tf.keras.Input(shape=(784,), name='image_input')
x1 = tf.keras.layers.Dense(256, activation='relu')(image_input)
x1 = tf.keras.layers.Dropout(0.3)(x1)
x1 = tf.keras.layers.Dense(128, activation='relu')(x1)

# Input 2: Metadata features
meta_input = tf.keras.Input(shape=(10,), name='meta_input')
x2 = tf.keras.layers.Dense(32, activation='relu')(meta_input)
x2 = tf.keras.layers.Dense(16, activation='relu')(x2)

# Concatenate
combined = tf.keras.layers.Concatenate(name='concat')([x1, x2])

# Shared layers
x = tf.keras.layers.Dense(128, activation='relu')(combined)
x = tf.keras.layers.Dropout(0.3)(x)
x = tf.keras.layers.Dense(64, activation='relu')(x)

# Output
output = tf.keras.layers.Dense(10, activation='softmax', name='prediction')(x)

# Create multi-input model
multi_input_model = tf.keras.Model(
    inputs=[image_input, meta_input],
    outputs=output,
    name='multi_input_model'
)

print("\nMulti-Input Model:")
multi_input_model.summary()

# ============================================================
# STEP 3: ResNet-style Skip Connections
# ============================================================
print("\n[STEP 3] Creating ResNet-style model with skip connections...")

def residual_block(x, units, dropout_rate=0.3):
    """Create a residual block with skip connection."""
    shortcut = x
    
    # Main path
    x = tf.keras.layers.Dense(units, activation='relu')(x)
    x = tf.keras.layers.BatchNormalization()(x)
    x = tf.keras.layers.Dropout(dropout_rate)(x)
    x = tf.keras.layers.Dense(units)(x)
    x = tf.keras.layers.BatchNormalization()(x)
    
    # Skip connection (with projection if needed)
    if shortcut.shape[-1] != units:
        shortcut = tf.keras.layers.Dense(units)(shortcut)
    
    # Add skip connection
    x = tf.keras.layers.Add()([x, shortcut])
    x = tf.keras.layers.Activation('relu')(x)
    
    return x

# Build ResNet-style model
inputs = tf.keras.Input(shape=(784,), name='input')
x = tf.keras.layers.Dense(256, activation='relu')(inputs)

# Stack residual blocks
x = residual_block(x, 256)
x = residual_block(x, 256)
x = residual_block(x, 128)
x = residual_block(x, 128)
x = residual_block(x, 64)

# Output
outputs = tf.keras.layers.Dense(10, activation='softmax')(x)

resnet_model = tf.keras.Model(inputs=inputs, outputs=outputs, name='resnet_model')

print("\nResNet-style Model:")
resnet_model.summary()

# ============================================================
# STEP 4: Multi-Output Model
# ============================================================
print("\n[STEP 4] Creating multi-output model...")

# Shared base
base_input = tf.keras.Input(shape=(784,), name='input')
x = tf.keras.layers.Dense(256, activation='relu')(base_input)
x = tf.keras.layers.Dropout(0.3)(x)
x = tf.keras.layers.Dense(128, activation='relu')(x)
shared = tf.keras.layers.Dense(64, activation='relu')(x)

# Output 1: Classification
class_output = tf.keras.layers.Dense(10, activation='softmax', name='class_output')(shared)

# Output 2: Regression
reg_output = tf.keras.layers.Dense(1, name='reg_output')(shared)

# Output 3: Binary classification
binary_output = tf.keras.layers.Dense(1, activation='sigmoid', name='binary_output')(shared)

# Create multi-output model
multi_output_model = tf.keras.Model(
    inputs=base_input,
    outputs=[class_output, reg_output, binary_output],
    name='multi_output_model'
)

# Compile with different losses
multi_output_model.compile(
    optimizer='adam',
    loss={
        'class_output': 'categorical_crossentropy',
        'reg_output': 'mse',
        'binary_output': 'binary_crossentropy'
    },
    loss_weights={
        'class_output': 1.0,
        'reg_output': 0.5,
        'binary_output': 0.3
    },
    metrics={
        'class_output': 'accuracy',
        'reg_output': 'mae',
        'binary_output': 'accuracy'
    }
)

print("\nMulti-Output Model:")
multi_output_model.summary()

# ============================================================
# STEP 5: Shared Layers (Siamese Network)
# ============================================================
print("\n[STEP 5] Creating Siamese network with shared layers...")

# Shared encoder
shared_encoder = tf.keras.Sequential([
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(64, activation='relu'),
    tf.keras.layers.Dense(32, activation='relu')
], name='shared_encoder')

# Two inputs
input_a = tf.keras.Input(shape=(784,), name='input_a')
input_b = tf.keras.Input(shape=(784,), name='input_b')

# Encode both inputs with shared weights
encoded_a = shared_encoder(input_a)
encoded_b = shared_encoder(input_b)

# Compute distance
distance = tf.keras.layers.Lambda(
    lambda x: tf.abs(x[0] - x[1]),
    name='distance'
)([encoded_a, encoded_b])

# Output similarity
output = tf.keras.layers.Dense(1, activation='sigmoid', name='similarity')(distance)

# Create Siamese model
siamese_model = tf.keras.Model(
    inputs=[input_a, input_b],
    outputs=output,
    name='siamese_model'
)

print("\nSiamese Model:")
siamese_model.summary()

# ============================================================
# STEP 6: Training Multi-Input Model

# Generate synthetic data

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Mismatched tensor shapes at merge layers** | `Add()` and `Concatenate()` require identical shapes (for Add) or matching all dims except the merge axis (for Concat). A skip connection from a 784-unit Dense to a 64-unit Dense cannot be added directly. | Insert a `Dense(64)` projection layer on the skip path before the `Add()` to match the shape of the main path. |
| **Calling `tf.keras.Input()` inside a model** | `tf.keras.Input()` creates a symbolic tensor for the Functional API. Calling it inside a `model.call()` or a custom `Layer.call()` instead of at graph-build time raises `ValueError: A KerasTensor cannot be used as input`. | Define all `Input()` tensors at the top level before chaining layers. Never use `Input()` inside a layer's `call()` method. |
| **Shared layer weights not actually shared** | Instantiating two `Dense(64)` layers with the same config but different Python objects gives them separate weights — they are not shared. True weight sharing requires calling the **same layer object** on multiple inputs. | `shared_layer = Dense(64); out_a = shared_layer(input_a); out_b = shared_layer(input_b)` — one object, two call sites. |
| **Multi-output model loss not weighted** | When a model has multiple outputs (e.g., classification head + regression head), passing a single loss function applies it equally to all outputs. This rarely matches your actual training objective. | Pass a dictionary of losses: `model.compile(loss={'class_head': 'sparse_categorical_crossentropy', 'reg_head': 'mse'}, loss_weights={'class_head': 1.0, 'reg_head': 0.1})`. |
| **Graph disconnected error when slicing layers** | Attempting to take intermediate layer outputs (`model.layers[3].output`) from a compiled `tf.keras.Model` that was built via subclassing (not Functional API) raises `AttributeError: Layer has no inbound nodes`. | The `.output` shortcut only works in Functional API models. For subclassed models, use a separate `tf.keras.Model(inputs=..., outputs=intermediate_layer.output)` built from the Functional graph. |
