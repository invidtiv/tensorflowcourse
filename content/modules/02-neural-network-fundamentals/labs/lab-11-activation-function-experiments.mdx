class ActivationFunctions:
    """Collection of activation functions with derivatives."""
    
    @staticmethod
    def sigmoid(x):
        """Sigmoid: sigma(x) = 1 / (1 + e^(-x))"""
        return 1 / (1 + np.exp(-x))
    
    @staticmethod
    def sigmoid_derivative(x):
        """Derivative of sigmoid: sigma'(x) = sigma(x) * (1 - sigma(x))"""
        s = ActivationFunctions.sigmoid(x)
        return s * (1 - s)
    
    @staticmethod
    def tanh(x):
        """Hyperbolic tangent: tanh(x) = (e^x - e^(-x)) / (e^x + e^(-x))"""
        return np.tanh(x)
    
    @staticmethod
    def tanh_derivative(x):
        """Derivative of tanh: tanh'(x) = 1 - tanh^2(x)"""
        return 1 - np.tanh(x) ** 2
    
    @staticmethod
    def relu(x):
        """ReLU: f(x) = max(0, x)"""
        return np.maximum(0, x)
    
    @staticmethod
    def relu_derivative(x):
        """Derivative of ReLU: f'(x) = 1 if x > 0, else 0"""
        return np.where(x > 0, 1, 0)
    
    @staticmethod
    def leaky_relu(x, alpha=0.01):
        """Leaky ReLU: f(x) = x if x > 0, else alpha*x"""
        return np.where(x > 0, x, alpha * x)
    
    @staticmethod
    def leaky_relu_derivative(x, alpha=0.01):
        """Derivative of Leaky ReLU"""
        return np.where(x > 0, 1, alpha)
    
    @staticmethod
    def elu(x, alpha=1.0):
        """ELU: f(x) = x if x > 0, else alpha*(e^x - 1)"""
        return np.where(x > 0, x, alpha * (np.exp(x) - 1))
    
    @staticmethod
    def swish(x, beta=1.0):
        """Swish: f(x) = x * sigma(beta*x)"""
        return x * ActivationFunctions.sigmoid(beta * x)
    
    @staticmethod
    def gelu(x):
        """GELU: f(x) = x * Phi(x) where Phi is CDF of standard normal"""
        return 0.5 * x * (1 + np.tanh(np.sqrt(2 / np.pi) * (x + 0.044715 * x ** 3)))
    
    @staticmethod
    def softmax(x):
        """Softmax: sigma(x)_i = e^(x_i) / sum_j e^(x_j)"""
        exp_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
        return exp_x / np.sum(exp_x, axis=-1, keepdims=True)
    
    @staticmethod
    def mish(x):
        """Mish: f(x) = x * tanh(softplus(x))"""
        return x * np.tanh(np.log(1 + np.exp(x)))

def demonstrate_vanishing_gradient(activation_name, depth=20):
    """Show how gradients vanish with depth for different activations."""
    
    # Initialize weights with small random values
    np.random.seed(42)
    weights = [np.random.randn(10, 10) * 0.01 for _ in range(depth)]
    
    # Input
    x = np.random.randn(1, 10)
    
    # Forward pass
    activations_dict = {'sigmoid': ActivationFunctions.sigmoid,
                   'tanh': ActivationFunctions.tanh,
                   'relu': ActivationFunctions.relu}
    
    activation = activations_dict.get(activation_name, ActivationFunctions.sigmoid)
    
    layer_outputs = [x]
    current = x
    for w in weights:
        z = current @ w
        current = activation(z)
        layer_outputs.append(current)
    
    # Calculate gradient magnitudes (simplified)
    gradient_magnitudes = [np.mean(np.abs(out)) for out in layer_outputs]
    
    return gradient_magnitudes

# Compare gradient flow
depths = range(21)

class CustomActivation(tf.keras.layers.Layer):
    """
    Custom activation: Smooth Leaky ReLU variant.
    
    f(x) = x if x >= 0
    f(x) = alpha * x / (1 - x/beta) if x < 0
    
    This provides smooth transition while maintaining negative gradients.
    """
    
    def __init__(self, alpha=0.1, beta=5.0, **kwargs):
        super().__init__(**kwargs)
        self.alpha = alpha
        self.beta = beta
    
    def call(self, inputs):
        # Positive part
        positive = tf.nn.relu(inputs)
        
        # Negative part (smooth)
        negative_mask = tf.cast(inputs < 0, tf.float32)
        negative = self.alpha * inputs / (1 - inputs / self.beta) * negative_mask
        
        return positive + negative
    
    def get_config(self):
        config = super().get_config()
        config.update({'alpha': self.alpha, 'beta': self.beta})
        return config
