# 1: Sequential API Deep Dive

```python
"""
Lab 2.2.1: Sequential API Deep Dive
Comprehensive guide to building models with Keras Sequential API.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

print("=" * 60)
print("LAB 2.2.1: Sequential API Deep Dive")
print("=" * 60)

# ============================================================
# STEP 1: Basic Sequential Model
# ============================================================
print("\n[STEP 1] Creating basic Sequential model...")

# Method 1: List of layers
model1 = tf.keras.Sequential([
    tf.keras.layers.Dense(64, activation='relu', input_shape=(10,)),
    tf.keras.layers.Dense(32, activation='relu'),
    tf.keras.layers.Dense(1, activation='sigmoid')
], name='basic_sequential')

print("\nModel 1 (List method):")
model1.summary()

# Method 2: Adding layers incrementally
model2 = tf.keras.Sequential(name='incremental_sequential')
model2.add(tf.keras.layers.Dense(64, activation='relu', input_shape=(10,)))
model2.add(tf.keras.layers.Dense(32, activation='relu'))
model2.add(tf.keras.layers.Dense(1, activation='sigmoid'))

print("\nModel 2 (Add method):")
model2.summary()

# ============================================================
# STEP 2: Advanced Sequential Patterns
# ============================================================
print("\n[STEP 2] Advanced Sequential patterns...")

# Pattern 1: With regularization
def create_regularized_model(input_dim, num_classes):
    """Create model with L1/L2 regularization."""
    return tf.keras.Sequential([
        tf.keras.layers.Dense(128, activation='relu', 
                             input_shape=(input_dim,),
                             kernel_regularizer=tf.keras.regularizers.l2(0.001),
                             bias_regularizer=tf.keras.regularizers.l1(0.001)),
        tf.keras.layers.Dropout(0.3),
        tf.keras.layers.Dense(64, activation='relu',
                             kernel_regularizer=tf.keras.regularizers.l2(0.001)),
        tf.keras.layers.Dropout(0.3),
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ], name='regularized_model')

# Pattern 2: With batch normalization
def create_bn_model(input_dim, num_classes):
    """Create model with batch normalization."""
    return tf.keras.Sequential([
        tf.keras.layers.Dense(128, input_shape=(input_dim,)),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Dropout(0.3),
        
        tf.keras.layers.Dense(64),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Dropout(0.3),
        
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ], name='batch_norm_model')

# Pattern 3: With different initialization
def create_initialized_model(input_dim, num_classes):
    """Create model with custom weight initialization."""
    return tf.keras.Sequential([
        tf.keras.layers.Dense(128, activation='relu', 
                             input_shape=(input_dim,),
                             kernel_initializer='he_normal'),
        tf.keras.layers.Dense(64, activation='relu',
                             kernel_initializer='he_normal'),
        tf.keras.layers.Dense(num_classes, activation='softmax',
                             kernel_initializer='glorot_uniform')
    ], name='initialized_model')

# Create and compare models
regularized_model = create_regularized_model(20, 3)
bn_model = create_bn_model(20, 3)
initialized_model = create_initialized_model(20, 3)

print("\nRegularized Model:")
regularized_model.summary()

print("\nBatch Normalization Model:")
bn_model.summary()

# ============================================================
# STEP 3: Convolutional Sequential Model
# ============================================================
print("\n[STEP 3] Convolutional Sequential model...")

def create_cnn_model(input_shape, num_classes):
    """Create CNN using Sequential API."""
    return tf.keras.Sequential([
        # First conv block
        tf.keras.layers.Conv2D(32, (3, 3), padding='same', input_shape=input_shape),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Conv2D(32, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.MaxPooling2D((2, 2)),
        tf.keras.layers.Dropout(0.25),
        
        # Second conv block
        tf.keras.layers.Conv2D(64, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Conv2D(64, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.MaxPooling2D((2, 2)),
        tf.keras.layers.Dropout(0.25),
        
        # Classifier
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(128),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ], name='cnn_sequential')

cnn_model = create_cnn_model((28, 28, 1), 10)
print("\nCNN Model:")
cnn_model.summary()

# ============================================================
# STEP 4: Recurrent Sequential Model
# ============================================================
print("\n[STEP 4] Recurrent Sequential model...")

def create_rnn_model(input_shape, num_classes):
    """Create RNN using Sequential API."""
    return tf.keras.Sequential([
        # First LSTM layer
        tf.keras.layers.LSTM(128, return_sequences=True, 
                            input_shape=input_shape,
                            dropout=0.2,
                            recurrent_dropout=0.2),
        
        # Second LSTM layer
        tf.keras.layers.LSTM(64, return_sequences=False,
                            dropout=0.2,
                            recurrent_dropout=0.2),
        
        # Classifier
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.3),
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ], name='rnn_sequential')

rnn_model = create_rnn_model((100, 10), 5)
print("\nRNN Model:")
rnn_model.summary()

# ============================================================
# STEP 5: Training with Sequential Models
# ============================================================
print("\n[STEP 5] Training demonstration with Sequential model...")

# Generate synthetic data
np.random.seed(42)
X_train = np.random.randn(1000, 20)
y_train = np.random.randint(0, 3, 1000)

X_test = np.random.randn(200, 20)
y_test = np.random.randint(0, 3, 200)

# One-hot encode labels
y_train_cat = tf.keras.utils.to_categorical(y_train, 3)
y_test_cat = tf.keras.utils.to_categorical(y_test, 3)

# Create and compile model
model = create_bn_model(20, 3)
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

# Train
print("\nTraining model...")
history = model.fit(
    X_train, y_train_cat,
    validation_split=0.2,
    epochs=20,
    batch_size=32,
    verbose=1
)

# Evaluate
test_loss, test_acc = model.evaluate(X_test, y_test_cat, verbose=0)
print(f"\nTest accuracy: {test_acc:.4f}")

# ============================================================
# STEP 6: Model Saving and Loading
# ============================================================
print("\n[STEP 6] Saving and loading Sequential models...")

# Save entire model
model.save('/mnt/okcomputer/output/sequential_model.keras')
print("Model saved to: /mnt/okcomputer/output/sequential_model.keras")

# Load model
loaded_model = tf.keras.models.load_model('/mnt/okcomputer/output/sequential_model.keras')
print("Model loaded successfully!")

# Verify
loaded_loss, loaded_acc = loaded_model.evaluate(X_test, y_test_cat, verbose=0)
print(f"Loaded model test accuracy: {loaded_acc:.4f}")

print("\n" + "=" * 60)
print("LAB 2.2.1 COMPLETED SUCCESSFULLY!")
print("=" * 60)
print("\nKEY TAKEAWAYS:")
print("1. Sequential API is simple and intuitive for linear stacks")
print("2. Two ways to create: list of layers or .add() method")
print("3. Easy to add regularization, dropout, batch normalization")
print("4. Works well for most common architectures")
print("5. Limitations: single input/output, no shared layers, no branching")
```

**Diagnostic tree if something looks off:**

1. **Test accuracy stuck at chance (~0.5 binary, ~1/n for n classes):** check that labels are one-hot when using `categorical_crossentropy` (use `sparse_categorical_crossentropy` for integer labels). Also verify `activation='softmax'` on the final Dense (or `sigmoid` + 1 unit for binary).
2. **Train accuracy ~1.0 but val/test drops to ~0.5:** severe overfit — dataset too small or no regularization. Add `Dropout(0.2–0.5)` between Dense layers or increase training data.
3. **Loss = NaN after a few epochs:** learning rate too high (try `1e-4`), or un-normalized inputs. Confirm inputs are standardized (mean≈0, std≈1) before fit.
4. **`loaded_acc` differs from `test_acc` by > 1%:** save/load broken — confirm the `.keras` extension (not legacy `.h5`), and that custom objects/layers are registered if used.
</Callout>

<Callout type="warning" title="Common Pitfalls">
- **Wrong input_shape:** In Keras 3, pass `input_shape` to the **first layer**, not to `model.add(tf.keras.layers.InputLayer(...))`. The old `input_dim` shorthand still works for 1D inputs only.
- **Saving to a bad path:** `/mnt/okcomputer/output/` is a lab-runner mount point. If running locally, substitute a real path (e.g. `./output/sequential_model.keras`). Create the directory first with `os.makedirs("output", exist_ok=True)`.
- **Dropout at inference time:** If you add `Dropout` layers and call `model.predict()`, Keras automatically uses `training=False` (no dropout). Calling `model(x, training=True)` at inference is a common mistake that leads to non-deterministic predictions.
- **`model.summary()` before `build()`:** The Sequential API delays building until `compile()` or the first call. If `summary()` shows `None` for output shapes, call `model.build((None, input_dim))` first.
</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Adding layers after `model.compile()`** | Keras allows it, but the optimizer state and gradient graph are not rebuilt. The new layer trains without its gradients being tracked. | Always finalize the layer stack before calling `model.compile()`. |
| **Wrong `input_shape` argument** | Passing `input_shape=(784,)` is correct for a flat vector. Passing `input_shape=(28, 28)` (without Flatten) feeds a 2D tensor to a Dense layer and raises a shape error at training time. | Add `tf.keras.layers.Flatten()` as the first layer, or explicitly reshape inputs. |
| **`model.add()` after calling `model.build()`** | Building freezes the layer stack for graph compilation. Adding after build is unsupported and may silently not add the layer in some TF versions. | Construct all layers before the first call or `model.compile()`. |
| **Not compiling before `model.fit()`** | Forgetting `model.compile()` raises `RuntimeError: You must compile your model before training`. Loss and optimizer are mandatory. | Always call `model.compile(optimizer=..., loss=..., metrics=[...])` before `model.fit()`. |
| **Using `model.weights` to check trainability** | `len(model.weights)` counts all weight tensors including biases. `len(model.trainable_weights)` tells you what actually updates during training — more useful for debugging frozen layers. | Use `model.trainable_weights` and `model.non_trainable_weights` for diagnostic checks. |
