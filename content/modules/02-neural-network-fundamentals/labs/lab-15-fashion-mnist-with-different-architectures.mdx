```python
"""
Lab 2.8: Fashion-MNIST with Different Architectures
Comparing different CNN architectures on Fashion-MNIST.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

print("=" * 60)
print("LAB 2.8: Fashion-MNIST with Different Architectures")
print("=" * 60)

# ============================================================
# STEP 1: Load and Prepare Data
# ============================================================
print("\n[STEP 1] Loading Fashion-MNIST dataset...")

(X_train, y_train), (X_test, y_test) = tf.keras.datasets.fashion_mnist.load_data()

# Normalize and reshape
X_train = X_train.astype('float32') / 255.0
X_test = X_test.astype('float32') / 255.0
X_train = X_train.reshape(-1, 28, 28, 1)
X_test = X_test.reshape(-1, 28, 28, 1)

# Use subset for faster training
X_train_sub = X_train[:10000]
y_train_sub = y_train[:10000]
X_test_sub = X_test[:2000]
y_test_sub = y_test[:2000]

print(f"Training samples: {len(X_train_sub)}")
print(f"Test samples: {len(X_test_sub)}")

# ============================================================
# STEP 2: Define Different Architectures
# ============================================================
print("\n[STEP 2] Defining different architectures...")

def create_shallow_cnn():
    """Shallow CNN with fewer layers."""
    model = tf.keras.Sequential([
        tf.keras.layers.Conv2D(32, (3, 3), activation='relu', input_shape=(28, 28, 1)),
        tf.keras.layers.MaxPooling2D((2, 2)),
        tf.keras.layers.Conv2D(64, (3, 3), activation='relu'),
        tf.keras.layers.MaxPooling2D((2, 2)),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ], name='shallow_cnn')
    return model

def create_deep_cnn():
    """Deep CNN with more layers."""
    model = tf.keras.Sequential([
        tf.keras.layers.Conv2D(32, (3, 3), padding='same', input_shape=(28, 28, 1)),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Conv2D(32, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.MaxPooling2D((2, 2)),
        tf.keras.layers.Dropout(0.25),
        
        tf.keras.layers.Conv2D(64, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.Conv2D(64, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.MaxPooling2D((2, 2)),
        tf.keras.layers.Dropout(0.25),
        
        tf.keras.layers.Conv2D(128, (3, 3), padding='same'),
        tf.keras.layers.BatchNormalization(),
        tf.keras.layers.Activation('relu'),
        tf.keras.layers.GlobalAveragePooling2D(),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(10, activation='softmax')
    ], name='deep_cnn')
    return model

def create_lenet_style():
    """LeNet-style architecture."""
    model = tf.keras.Sequential([
        tf.keras.layers.Conv2D(6, (5, 5), activation='relu', input_shape=(28, 28, 1)),
        tf.keras.layers.AveragePooling2D((2, 2)),
        tf.keras.layers.Conv2D(16, (5, 5), activation='relu'),
        tf.keras.layers.AveragePooling2D((2, 2)),
        tf.keras.layers.Flatten(),
        tf.keras.layers.Dense(120, activation='relu'),
        tf.keras.layers.Dense(84, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ], name='lenet_style')
    return model

# ============================================================
# STEP 3: Data Augmentation
# ============================================================
print("\n[STEP 3] Setting up data augmentation...")

data_augmentation = tf.keras.Sequential([
    tf.keras.layers.RandomRotation(0.1),
    tf.keras.layers.RandomZoom(0.1),
    tf.keras.layers.RandomTranslation(0.1, 0.1)
])

# Visualize augmented images
fig, axes = plt.subplots(2, 5, figsize=(15, 6))
sample_image = X_train_sub[0:1]

for i in range(5):
    axes[0, i].imshow(sample_image[0].reshape(28, 28), cmap='gray')
    axes[0, i].set_title('Original')
    axes[0, i].axis('off')
    
    augmented = data_augmentation(sample_image)
    axes[1, i].imshow(augmented[0].numpy().reshape(28, 28), cmap='gray')
    axes[1, i].set_title('Augmented')
    axes[1, i].axis('off')

plt.suptitle('Data Augmentation Examples', fontsize=14, fontweight='bold')
plt.tight_layout()
plt.savefig('/mnt/okcomputer/output/data_augmentation.png', dpi=150, bbox_inches='tight')
plt.show()

# ============================================================
# STEP 4: Train Models with Callbacks
# ============================================================
print("\n[STEP 4] Training models with callbacks...")

# Callbacks
early_stopping = tf.keras.callbacks.EarlyStopping(
    monitor='val_loss',
    patience=5,
    restore_best_weights=True
)

lr_scheduler = tf.keras.callbacks.ReduceLROnPlateau(
    monitor='val_loss',
    factor=0.5,
    patience=3,
    min_lr=1e-6
)

models_dict = {
    'Shallow CNN': create_shallow_cnn(),
    'Deep CNN': create_deep_cnn(),
    'LeNet Style': create_lenet_style()
}

results = {}

for name, model in models_dict.items():
    print(f"\nTraining {name}...")
    
    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    history = model.fit(
        X_train_sub, y_train_sub,
        validation_split=0.2,
        epochs=30,
        batch_size=64,
        callbacks=[early_stopping, lr_scheduler],
        verbose=0
    )
    
    # Evaluate
    test_loss, test_acc = model.evaluate(X_test_sub, y_test_sub, verbose=0)
    
    results[name] = {
        'history': history,
        'test_accuracy': test_acc,
        'epochs_trained': len(history.history['loss'])
    }
    
    print(f"  Test accuracy: {test_acc:.4f}")
    print(f"  Epochs trained: {len(history.history['loss'])}")

# ============================================================
# STEP 5: Compare Results
# ============================================================
print("\n[STEP 5] Comparing results...")

fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Validation accuracy
ax1 = axes[0]
for name, result in results.items():
    ax1.plot(result['history'].history['val_accuracy'], 
             label=f"{name} ({result['test_accuracy']:.3f})", linewidth=2)
ax1.set_xlabel('Epoch')
ax1.set_ylabel('Validation Accuracy')
ax1.set_title('Validation Accuracy Comparison', fontweight='bold')
ax1.legend()
ax1.grid(True, alpha=0.3)

# Test accuracy comparison
ax2 = axes[1]
names = list(results.keys())
accuracies = [results[n]['test_accuracy'] for n in names]
colors = ['skyblue', 'lightgreen', 'salmon']
bars = ax2.bar(names, accuracies, color=colors, alpha=0.7)
ax2.set_ylabel('Test Accuracy')
ax2.set_title('Final Test Accuracy', fontweight='bold')
ax2.set_ylim([0.7, 0.9])
ax2.grid(True, alpha=0.3, axis='y')

for bar in bars:
    height = bar.get_height()
    ax2.text(bar.get_x() + bar.get_width()/2., height,
             f'{height:.3f}', ha='center', va='bottom')

plt.suptitle('Fashion-MNIST Architecture Comparison', fontsize=14, fontweight='bold')
plt.tight_layout()
plt.savefig('/mnt/okcomputer/output/architecture_comparison.png', dpi=150, bbox_inches='tight')
plt.show()

print("\n" + "=" * 60)
print("LAB 2.8 COMPLETED SUCCESSFULLY!")
print("=" * 60)
print("\nKEY TAKEAWAYS:")
print("1. Deeper CNNs can achieve better performance")
print("2. Batch normalization and dropout improve generalization")
print("3. Early stopping prevents overfitting")
print("4. Learning rate scheduling helps convergence")
print("5. Data augmentation increases effective dataset size")
```

<Callout type="note" title="Expected Output">
Fashion-MNIST has 10 classes, 28×28 grayscale, 60k train / 10k test. A no-preprocessing MLP baseline sits around **~0.88 test accuracy**; a well-tuned CNN reliably clears **~0.92**, and with BN + dropout + augmentation you should see **0.93–0.94** by epoch ~20.

Typical per-architecture test-accuracy bands (seed-dependent, ±0.005):

- **Simple MLP (Flatten → 128 → 10):** 0.86–0.89. Overfits by epoch ~10 without dropout.
- **Small CNN (2× Conv32/64 → Dense128):** 0.90–0.92. ~200k params, <2 min/epoch on CPU, <10 s on a T4.
- **Deep CNN + BN + Dropout:** 0.92–0.94. Training/test gap should stay under ~4pp; larger gaps mean augmentation is under-applied.
- **LR-scheduled run (ReduceLROnPlateau, factor=0.5, patience=2):** val-loss "knee" appears around epoch 8–12 with a second smaller drop after the first LR cut.

The final bar chart (`architecture_comparison.png`) should show monotonic improvement MLP → CNN → CNN+BN. If the deep CNN is **worse** than the small CNN, that's almost always a learning-rate or init issue, not architectural.

**Diagnostic checks at the end of the run:**

1. `history.history['val_accuracy'][-5:]` within ±0.005 of each other (stable plateau).
2. `model.count_params()` matches printed "Total params" in `summary()`.
3. The saved `.keras` reloads and matches test accuracy within ±0.005.
</Callout>

<Callout type="warning" title="Common Pitfalls">
- **Forgetting to scale pixels to `[0, 1]`:** Fashion-MNIST `x_train` is `uint8` in `[0, 255]`. Divide by `255.0` (as float32) before training — skipping this makes loss explode or stall at chance.
- **Channel dimension missing:** Conv2D expects `(H, W, C)`. Reshape grayscale to `(28, 28, 1)` via `x_train[..., None]`. Otherwise `ValueError: Input 0 of layer conv2d is incompatible`.
- **Integer vs one-hot labels:** `sparse_categorical_crossentropy` wants integer class indices. `categorical_crossentropy` needs `tf.keras.utils.to_categorical(y, 10)`. Mixing them silently gives ~0.1 accuracy.
- **Over-aggressive augmentation:** `RandomFlip('horizontal')` is fine for most classes, but watch Sandal/Ankle-boot for a 1–2pp accuracy dip — inspect the per-class confusion matrix.
- **BN + small batch size:** BatchNormalization needs batch ≥ 32 for stable running statistics. At batch 8–16, prefer GroupNormalization or LayerNormalization.
- **Saving to `/mnt/okcomputer/output/`:** lab-runner mount path. Locally, replace with `./output/` and `os.makedirs("output", exist_ok=True)` first.
</Callout>

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Forgetting pixel normalization** | Fashion-MNIST `x_train` is `uint8` in [0, 255]. Un-normalized inputs cause loss to explode or stall at chance-level accuracy (~10%). | Divide by `255.0` (cast to float32): `X_train = X_train.astype('float32') / 255.0`. |
| **Missing channel dimension for Conv2D** | Conv2D expects input shape `(H, W, C)`. Grayscale images have shape `(28, 28)` after `load_data()`. Passing them without reshaping raises `ValueError: Input 0 of layer conv2d is incompatible`. | Reshape: `X_train = X_train[..., np.newaxis]` or `X_train = X_train.reshape(-1, 28, 28, 1)`. |
| **BatchNormalization with tiny batch size** | BN accumulates running mean/variance estimates from the current batch. With batch_size < 16 these estimates are noisy, causing unstable training and poor test-time performance. | Keep batch_size ≥ 32 when using BN. For smaller batches, consider `LayerNormalization` instead. |
| **Over-aggressive horizontal flip augmentation** | `RandomFlip('horizontal')` flips "7" (Sneaker) to look like "0" (T-shirt) for asymmetric items. Sandals and ankle boots can lose ~1–2pp accuracy. | Inspect per-class confusion matrices before and after augmentation. Restrict augmentation to rotation/zoom for footwear classes if needed. |
| **Learning rate not reduced early enough** | A fixed learning rate of 1e-3 tends to plateau after 5–8 epochs on Fashion-MNIST. Without `ReduceLROnPlateau` or a schedule, the model never escapes the flat region of the loss landscape. | Add `callbacks=[tf.keras.callbacks.ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=2)]`. |
