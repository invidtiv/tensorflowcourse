class CustomSegmentationLoss:

    2. Output: scalar loss value
    3. Differentiable for backpropagation
    """
    
    def __init__(self, name='custom_loss'):
        self.name = name
    
    def __call__(self, y_true, y_pred):
        """
        Compute loss.
        
        Args:
            y_true: Ground truth (batch, H, W, C)
            y_pred: Predictions (batch, H, W, C)
        
        Returns:
            Scalar loss value
        """
        raise NotImplementedError
    
    def get_config(self):
        """Get loss configuration for serialization."""
        return {'name': self.name}

class BoundaryAwareLoss(CustomSegmentationLoss):
    """
    Loss that focuses more on boundary regions.
    
    Combines:
    1. Standard segmentation loss
    2. Boundary-specific loss with higher weight
    """
    
    def __init__(self, boundary_weight=2.0, base_loss='dice'):
        super().__init__('boundary_aware_loss')
        self.boundary_weight = boundary_weight
        self.base_loss = base_loss
    
    def extract_boundary(self, mask):
        """Extract boundary from binary mask using morphological gradient."""
        # Use Sobel operators to detect edges
        sobel_x = tf.constant([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]], dtype=tf.float32)
        sobel_y = tf.constant([[-1, -2, -1], [0, 0, 0], [1, 2, 1]], dtype=tf.float32)
        
        sobel_x = tf.reshape(sobel_x, [3, 3, 1, 1])
        sobel_y = tf.reshape(sobel_y, [3, 3, 1, 1])
        
        # Apply to each channel
        channels = []
        for c in range(mask.shape[-1]):
            mask_c = mask[..., c:c+1]
            gx = tf.nn.conv2d(mask_c, sobel_x, strides=[1, 1, 1, 1], padding='SAME')
            gy = tf.nn.conv2d(mask_c, sobel_y, strides=[1, 1, 1, 1], padding='SAME')
            boundary_c = tf.sqrt(tf.square(gx) + tf.square(gy) + 1e-6)
            channels.append(boundary_c)
        
        return tf.concat(channels, axis=-1)
    
    def dice_loss(self, y_true, y_pred, smooth=1e-6):
        """Standard Dice loss."""
        y_true_f = tf.reshape(y_true, [-1])
        y_pred_f = tf.reshape(y_pred, [-1])
        
        intersection = tf.reduce_sum(y_true_f * y_pred_f)
        union = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f)
        
        dice = (2. * intersection + smooth) / (union + smooth)
        return 1 - dice
    
    def __call__(self, y_true, y_pred):
        # Base loss on full image
        base_loss = self.dice_loss(y_true, y_pred)
        
        # Extract boundaries
        true_boundary = self.extract_boundary(y_true)
        pred_boundary = self.extract_boundary(y_pred)
        
        # Boundary loss
        boundary_loss = self.dice_loss(true_boundary, pred_boundary)
        
        # Combined loss
        total_loss = base_loss + self.boundary_weight * boundary_loss
        
        return total_loss

class MultiScaleLoss(CustomSegmentationLoss):
    """
    Loss computed at multiple scales.
    
    Helps the model learn both fine details and global structure.
    """
    
    def __init__(self, scales=[1, 0.5, 0.25], weights=[1.0, 0.5, 0.25]):
        super().__init__('multi_scale_loss')
        self.scales = scales
        self.weights = weights
    
    def dice_loss(self, y_true, y_pred, smooth=1e-6):
        y_true_f = tf.reshape(y_true, [-1])
        y_pred_f = tf.reshape(y_pred, [-1])
        
        intersection = tf.reduce_sum(y_true_f * y_pred_f)
        union = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f)
        
        dice = (2. * intersection + smooth) / (union + smooth)
        return 1 - dice
    
    def __call__(self, y_true, y_pred):
        total_loss = 0.0
        
        for scale, weight in zip(self.scales, self.weights):
            if scale == 1.0:
                # Original scale
                scaled_true = y_true
                scaled_pred = y_pred
            else:
                # Downsample
                h = tf.cast(tf.cast(tf.shape(y_true)[1], tf.float32) * scale, tf.int32)
                w = tf.cast(tf.cast(tf.shape(y_true)[2], tf.float32) * scale, tf.int32)
                
                scaled_true = tf.image.resize(y_true, [h, w], method='nearest')
                scaled_pred = tf.image.resize(y_pred, [h, w], method='bilinear')
            
            loss = self.dice_loss(scaled_true, scaled_pred)
            total_loss += weight * loss
        
        return total_loss

class LearnableWeightedLoss(keras.layers.Layer):
    """
    Loss with learnable weights for different terms.
    
    Automatically learns the optimal balance between loss components.
    """
    
    def __init__(self, num_terms=2, **kwargs):
        super().__init__(**kwargs)
        self.num_terms = num_terms
    
    def build(self, input_shape):
        # Initialize log weights (will be exponentiated to ensure positivity)
        self.log_weights = self.add_weight(
            name='log_weights',
            shape=(self.num_terms,),
            initializer='zeros',
            trainable=True
        )
        super().build(input_shape)
    
    def call(self, losses):
        """
        Compute weighted sum of losses.
        
        Args:
            losses: List of loss tensors
        
        Returns:
            Weighted loss and individual weights
        """
        # Exponentiate to ensure positive weights
        weights = tf.exp(self.log_weights)
        
        # Normalize weights
        weights = weights / tf.reduce_sum(weights)
        
        # Weighted sum
        total_loss = sum(w * l for w, l in zip(weights, losses))
        
        return total_loss, weights

class MedicalSegmentationLoss(CustomSegmentationLoss):
    """
    Loss optimized for medical image segmentation.
    
    Characteristics:
    - High sensitivity (don't miss pathology)
    - Reasonable specificity (avoid false alarms)
    - Good boundary definition
    """
    
    def __init__(self, sensitivity_weight=2.0, specificity_weight=1.0, 
                 boundary_weight=1.0):
        super().__init__('medical_segmentation_loss')
        self.sensitivity_weight = sensitivity_weight
        self.specificity_weight = specificity_weight
        self.boundary_weight = boundary_weight
    
    def sensitivity_loss(self, y_true, y_pred):
        """Loss based on false negatives (missing pathology)."""
        # Penalize false negatives heavily
        fn = tf.reduce_sum(y_true * (1 - y_pred))
        tp = tf.reduce_sum(y_true * y_pred)
        
        # Sensitivity = TP / (TP + FN)
        sensitivity = tp / (tp + fn + 1e-6)
        return 1 - sensitivity
    
    def specificity_loss(self, y_true, y_pred):
        """Loss based on false positives (false alarms)."""
        fp = tf.reduce_sum((1 - y_true) * y_pred)
        tn = tf.reduce_sum((1 - y_true) * (1 - y_pred))
        
        # Specificity = TN / (TN + FP)
        specificity = tn / (tn + fp + 1e-6)
        return 1 - specificity
    
    def boundary_loss(self, y_true, y_pred):
        """Boundary quality loss."""
        # Simple boundary loss using gradient
        sobel_x = tf.constant([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]], dtype=tf.float32)
        sobel_x = tf.reshape(sobel_x, [3, 3, 1, 1])
        
        true_gx = tf.nn.conv2d(y_true, sobel_x, strides=[1, 1, 1, 1], padding='SAME')
        pred_gx = tf.nn.conv2d(y_pred, sobel_x, strides=[1, 1, 1, 1], padding='SAME')
        
        return tf.reduce_mean(tf.square(true_gx - pred_gx))
    
    def __call__(self, y_true, y_pred):
        sens_loss = self.sensitivity_loss(y_true, y_pred)
        spec_loss = self.specificity_loss(y_true, y_pred)
        bound_loss = self.boundary_loss(y_true, y_pred)
        
        total = (self.sensitivity_weight * sens_loss + 
                 self.specificity_weight * spec_loss +
                 self.boundary_weight * bound_loss)
        
        return total

def debug_loss(loss_fn, y_true, y_pred, name='Loss'):
    """
    Debug a loss function.
    
    Checks:
    1. Output is scalar
    2. Loss is non-negative
    3. Loss is differentiable
    """
    print(f"\nDebugging {name}:")
    
    try:
        # Compute loss
        with tf.GradientTape() as tape:
            tape.watch(y_pred)
            loss_value = loss_fn(y_true, y_pred)
        
        # Check output shape
        print(f"  Output shape: {loss_value.shape}")
        assert loss_value.shape == (), f"Expected scalar, got {loss_value.shape}"
        
        # Check non-negative
        loss_np = loss_value.numpy()
        print(f"  Loss value: {loss_np:.4f}")
        assert loss_np >= 0, f"Loss should be non-negative, got {loss_np}"
        
        # Check gradients
        grads = tape.gradient(loss_value, y_pred)
        print(f"  Gradient shape: {grads.shape}")
        print(f"  Gradient range: [{grads.numpy().min():.4f}, {grads.numpy().max():.4f}]")
        assert grads is not None, "Gradients are None!"
        
        print(f"  ✓ {name} passed all checks!")
        return True
        
    except Exception as e:
        print(f"  ✗ {name} failed: {e}")
        return False

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Custom loss not returning a scalar** | Returning a per-pixel or per-sample loss tensor (e.g., shape `(B,H,W)`) causes Keras to call `tf.reduce_mean` automatically, which sometimes works but silently ignores sample weights and custom reduction strategies. | Always explicitly `tf.reduce_mean` the per-sample losses inside the loss function and return a scalar tensor of shape `()`; use `GradientTape` to confirm the output shape before training. |
| **Non-differentiable operations in the loss** | Using `tf.math.argmax`, `tf.round`, or `np` operations inside the loss function creates graph nodes with zero or undefined gradients; the optimizer receives zero-gradient updates and training stalls with constant loss. | Replace hard threshold operations with soft approximations: use softmax probabilities directly instead of argmax; use `sigmoid` instead of step functions; validate with `tf.GradientTape` that gradients are non-zero. |
| **Boundary mask computed from predicted output instead of ground truth** | Computing the edge weight map from the model's own prediction creates a circular dependency — the loss changes shape as the prediction improves, destabilizing training (loss may initially decrease then oscillate). | Always compute boundary/edge masks from the ground-truth label map only; cache them as part of data preprocessing if possible to avoid recomputing per step. |
| **Multi-scale loss weights not balanced relative to scale resolution** | Assigning equal weight `[1, 1, 1]` to losses at 128×128, 64×64, and 32×32 means the coarse-scale loss dominates total gradient magnitude simply because it uses lower-resolution features with larger spatial averages. | Normalize multi-scale weights so their sum is 1 (`[0.5, 0.3, 0.2]` for fine-to-coarse); or normalize each scale's loss by the number of pixels at that scale before combining. |
| **Gradient explosion when combining multiple loss terms** | Summing three loss terms (Dice + boundary + BCE) without checking their relative magnitudes can produce total gradient norms 10–100× larger than any single loss, causing `NaN` weights after 1–3 steps. | Log gradient norms for each loss component separately using `GradientTape`; apply `tf.keras.optimizers.Adam(clipnorm=1.0)` as a safety net; normalize each loss term to similar magnitude before combining. |
