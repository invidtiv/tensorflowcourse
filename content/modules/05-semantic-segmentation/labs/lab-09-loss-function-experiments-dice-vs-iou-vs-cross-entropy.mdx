class SegmentationLosses:
    """Collection of segmentation loss functions."""
    
    @staticmethod
    def dice_loss(y_true, y_pred, smooth=1e-6):
        """
        Dice loss for binary segmentation.
        
        Dice = 2*|X∩Y| / (|X| + |Y|)
        Dice Loss = 1 - Dice
        
        Good for: Imbalanced classes, medical segmentation
        """
        y_true_f = tf.reshape(y_true, [-1])
        y_pred_f = tf.reshape(y_pred, [-1])
        
        intersection = tf.reduce_sum(y_true_f * y_pred_f)
        union = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f)
        
        dice = (2. * intersection + smooth) / (union + smooth)
        return 1 - dice
    
    @staticmethod
    def dice_coefficient(y_true, y_pred, smooth=1e-6):
        """Dice coefficient (metric)."""
        return 1 - SegmentationLosses.dice_loss(y_true, y_pred, smooth)
    
    @staticmethod
    def iou_loss(y_true, y_pred, smooth=1e-6):
        """
        IoU (Jaccard) loss for binary segmentation.
        
        IoU = |X∩Y| / |X∪Y|
        IoU Loss = 1 - IoU
        
        Good for: General segmentation, object detection
        """
        y_true_f = tf.reshape(y_true, [-1])
        y_pred_f = tf.reshape(y_pred, [-1])
        
        intersection = tf.reduce_sum(y_true_f * y_pred_f)
        union = tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f) - intersection
        
        iou = (intersection + smooth) / (union + smooth)
        return 1 - iou
    
    @staticmethod
    def iou_coefficient(y_true, y_pred, smooth=1e-6):
        """IoU coefficient (metric)."""
        return 1 - SegmentationLosses.iou_loss(y_true, y_pred, smooth)
    
    @staticmethod
    def tversky_loss(y_true, y_pred, alpha=0.3, beta=0.7, smooth=1e-6):
        """
        Tversky loss - generalization of Dice and IoU.
        
        Tversky = |TP| / (|TP| + alpha*|FN| + beta*|FP|)
        
        Args:
            alpha: Weight for false negatives (higher = more penalty for missing positives)
            beta: Weight for false positives (higher = more penalty for false alarms)
        
        Good for: When you need to balance precision/recall
        """
        y_true_f = tf.reshape(y_true, [-1])
        y_pred_f = tf.reshape(y_pred, [-1])
        
        tp = tf.reduce_sum(y_true_f * y_pred_f)
        fn = tf.reduce_sum(y_true_f * (1 - y_pred_f))
        fp = tf.reduce_sum((1 - y_true_f) * y_pred_f)
        
        tversky = (tp + smooth) / (tp + alpha * fn + beta * fp + smooth)
        return 1 - tversky
    
    @staticmethod
    def focal_loss(y_true, y_pred, gamma=2.0, alpha=0.25):
        """
        Focal loss for addressing class imbalance.
        
        FL = -alpha * (1-p)^gamma * log(p)
        
        Args:
            gamma: Focusing parameter (higher = more focus on hard examples)
            alpha: Weighting factor for positive class
        
        Good for: Highly imbalanced datasets
        """
        epsilon = tf.keras.backend.epsilon()
        y_pred = tf.clip_by_value(y_pred, epsilon, 1 - epsilon)
        
        # Binary cross-entropy
        bce = -y_true * tf.math.log(y_pred) - (1 - y_true) * tf.math.log(1 - y_pred)
        
        # Focal weighting
        p_t = (y_true * y_pred) + ((1 - y_true) * (1 - y_pred))
        alpha_t = y_true * alpha + (1 - y_true) * (1 - alpha)
        
        loss = tf.reduce_mean(alpha_t * tf.pow(1.0 - p_t, gamma) * bce)
        return loss
    
    @staticmethod
    def boundary_loss(y_true, y_pred):
        """
        Boundary loss that focuses on segmentation boundaries.
        
        Good for: Improving edge quality
        """
        # Compute boundary using Sobel operator
        sobel_x = tf.constant([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]], dtype=tf.float32)
        sobel_y = tf.constant([[-1, -2, -1], [0, 0, 0], [1, 2, 1]], dtype=tf.float32)
        
        sobel_x = tf.reshape(sobel_x, [3, 3, 1, 1])
        sobel_y = tf.reshape(sobel_y, [3, 3, 1, 1])
        
        # Apply to predictions
        pred_x = tf.nn.conv2d(y_pred, sobel_x, strides=[1, 1, 1, 1], padding='SAME')
        pred_y = tf.nn.conv2d(y_pred, sobel_y, strides=[1, 1, 1, 1], padding='SAME')
        pred_boundary = tf.sqrt(tf.square(pred_x) + tf.square(pred_y) + 1e-6)
        
        # Apply to ground truth
        true_x = tf.nn.conv2d(y_true, sobel_x, strides=[1, 1, 1, 1], padding='SAME')
        true_y = tf.nn.conv2d(y_true, sobel_y, strides=[1, 1, 1, 1], padding='SAME')
        true_boundary = tf.sqrt(tf.square(true_x) + tf.square(true_y) + 1e-6)
        
        # Boundary loss
        loss = tf.reduce_mean(tf.square(pred_boundary - true_boundary))
        return loss
    
    @staticmethod
    def combined_loss(y_true, y_pred, weights={'bce': 1.0, 'dice': 1.0}):
        """Combined BCE + Dice loss."""
        bce = keras.losses.binary_crossentropy(y_true, y_pred)
        dice = SegmentationLosses.dice_loss(y_true, y_pred)
        return weights['bce'] * bce + weights['dice'] * dice

def generate_segmentation_data(num_samples=100, image_size=(128, 128)):
    """Generate synthetic segmentation data."""
    images = []
    masks = []
    
    for _ in range(num_samples):
        # Image
        img = np.random.rand(image_size[0], image_size[1], 3).astype(np.float32)
        
        # Mask with circles
        mask = np.zeros((image_size[0], image_size[1], 1), dtype=np.float32)
        center = (np.random.randint(20, image_size[0]-20), 
                  np.random.randint(20, image_size[1]-20))
        radius = np.random.randint(10, 30)
        
        y, x = np.ogrid[:image_size[0], :image_size[1]]
        dist = np.sqrt((x - center[1])**2 + (y - center[0])**2)
        mask[dist <= radius] = 1.0
        
        images.append(img)
        masks.append(mask)
    
    return np.array(images), np.array(masks)


def build_simple_unet(input_shape=(128, 128, 3)):
    """Build simple U-Net for comparison."""
    inputs = layers.Input(shape=input_shape)
    
    # Encoder
    c1 = layers.Conv2D(32, 3, activation='relu', padding='same')(inputs)
    c1 = layers.Conv2D(32, 3, activation='relu', padding='same')(c1)
    p1 = layers.MaxPooling2D(2)(c1)
    
    c2 = layers.Conv2D(64, 3, activation='relu', padding='same')(p1)
    c2 = layers.Conv2D(64, 3, activation='relu', padding='same')(c2)
    p2 = layers.MaxPooling2D(2)(c2)
    
    # Bottleneck
    c3 = layers.Conv2D(128, 3, activation='relu', padding='same')(p2)
    c3 = layers.Conv2D(128, 3, activation='relu', padding='same')(c3)
    
    # Decoder
    u4 = layers.Conv2DTranspose(64, 2, strides=2, padding='same')(c3)
    u4 = layers.Concatenate()([u4, c2])
    c4 = layers.Conv2D(64, 3, activation='relu', padding='same')(u4)
    
    u5 = layers.Conv2DTranspose(32, 2, strides=2, padding='same')(c4)
    u5 = layers.Concatenate()([u5, c1])
    c5 = layers.Conv2D(32, 3, activation='relu', padding='same')(u5)
    
    outputs = layers.Conv2D(1, 1, activation='sigmoid')(c5)
    
    return Model(inputs, outputs)


def compare_losses():
    """Compare different loss functions on same model."""
    print("\n" + "="*60)
    print("LOSS COMPARISON EXPERIMENT")
    print("="*60)
    
    # Generate data
    X_train, y_train = generate_segmentation_data(100)
    X_val, y_val = generate_segmentation_data(20)
    
    losses = SegmentationLosses()
    
    loss_functions = {
        'BCE': keras.losses.binary_crossentropy,
        'Dice': losses.dice_loss,
        'IoU': losses.iou_loss,
        'BCE+Dice': lambda yt, yp: losses.combined_loss(yt, yp, {'bce': 1.0, 'dice': 1.0})
    }
    
    results = {}
    
    for loss_name, loss_fn in loss_functions.items():
        print(f"\nTraining with {loss_name} loss...")
        
        # Build fresh model
        model = build_simple_unet()
        model.compile(
            optimizer=keras.optimizers.Adam(1e-3),
            loss=loss_fn,
            metrics=[losses.dice_coefficient, losses.iou_coefficient]
        )
        
        # Train
        history = model.fit(
            X_train, y_train,
            validation_data=(X_val, y_val),
            batch_size=8,
            epochs=20,
            verbose=0
        )
        
        # Evaluate
        val_dice = max(history.history['val_dice_coefficient'])
        val_iou = max(history.history['val_iou_coefficient'])
        
        results[loss_name] = {
            'val_dice': val_dice,
            'val_iou': val_iou,
            'history': history
        }
        
        print(f"  Best Val Dice: {val_dice:.4f}, Best Val IoU: {val_iou:.4f}")
    
    # Visualize results
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    for loss_name, result in results.items():
        axes[0].plot(result['history'].history['val_dice_coefficient'], 
                     label=f'{loss_name} (max={result["val_dice"]:.3f})')
    axes[0].set_xlabel('Epoch')
    axes[0].set_ylabel('Dice Coefficient')
    axes[0].set_title('Validation Dice Coefficient')
    axes[0].legend()
    axes[0].grid(True)
    
    for loss_name, result in results.items():
        axes[1].plot(result['history'].history['val_iou_coefficient'],
                     label=f'{loss_name} (max={result["val_iou"]:.3f})')
    axes[1].set_xlabel('Epoch')
    axes[1].set_ylabel('IoU')
    axes[1].set_title('Validation IoU')
    axes[1].legend()
    axes[1].grid(True)
    
    plt.tight_layout()
    plt.savefig('/mnt/okcomputer/output/loss_comparison.png', dpi=150)
    plt.show()
    print("\nComparison saved to /mnt/okcomputer/output/loss_comparison.png")
    
    return results
