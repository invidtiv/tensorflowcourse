        noisy_masks.append(noisy_mask)
    
    return (np.array(images), np.array(clean_masks),

def apply_morphological_operations(mask, kernel_size=5):
    """
    Apply morphological operations to clean up segmentation.
    
    Operations:
    1. Closing: Fill small holes
    2. Opening: Remove small noise
    3. Dilation/Erosion: Adjust boundary
    
    Args:
        mask: Binary mask (0-1 or 0-255)
        kernel_size: Size of morphological kernel
    
    Returns:
        Processed mask
    """
    # Convert to uint8
    if mask.max() <= 1.0:
        mask_uint8 = (mask * 255).astype(np.uint8)
    else:
        mask_uint8 = mask.astype(np.uint8)
    
    # Create kernel
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (kernel_size, kernel_size))
    
    # Closing: Dilation followed by Erosion (fills holes)
    mask_closed = cv2.morphologyEx(mask_uint8, cv2.MORPH_CLOSE, kernel)
    
    # Opening: Erosion followed by Dilation (removes noise)
    mask_opened = cv2.morphologyEx(mask_closed, cv2.MORPH_OPEN, kernel)
    
    return mask_opened / 255.0

def connected_component_filtering(mask, min_area=100, max_components=None):
    """
    Filter connected components by area.
    
    Args:
        mask: Binary mask
        min_area: Minimum area to keep
        max_components: Maximum number of components to keep (None = all)
    
    Returns:
        Filtered mask
    """
    # Convert to uint8
    if mask.max() <= 1.0:
        mask_uint8 = (mask * 255).astype(np.uint8)
    else:
        mask_uint8 = mask.astype(np.uint8)
    
    # Find connected components
    num_labels, labels, stats, centroids = cv2.connectedComponentsWithStats(
        mask_uint8, connectivity=8
    )
    
    # Create output mask
    filtered_mask = np.zeros_like(mask_uint8)
    
    # Filter by area (skip background label 0)
    component_areas = [(i, stats[i, cv2.CC_STAT_AREA]) for i in range(1, num_labels)]
    component_areas.sort(key=lambda x: x[1], reverse=True)
    
    kept_components = 0
    for label, area in component_areas:
        if area >= min_area:
            filtered_mask[labels == label] = 255
            kept_components += 1
            
            if max_components and kept_components >= max_components:
                break
    
    return filtered_mask / 255.0, kept_components

def crf_refinement(image, mask, num_iterations=5, sxy=3, srgb=10, compat=3):
    """
    Simplified CRF refinement for segmentation.
    
    CRF models pixel relationships to refine boundaries.
    This is a simplified version for demonstration.
    
    Args:
        image: Input image (H, W, 3)
        mask: Initial segmentation mask (H, W)
        num_iterations: Number of CRF iterations
        sxy: Spatial standard deviation
        srgb: Color standard deviation
        compat: Compatibility factor
    
    Returns:
        Refined mask
    """
    from scipy.ndimage import gaussian_filter
    
    h, w = mask.shape
    
    # Initialize with mask probabilities
    prob_fg = mask.astype(np.float32)
    prob_bg = 1 - prob_fg
    
    for _ in range(num_iterations):
        # Compute pairwise potentials (simplified bilateral filter)
        # Spatial term
        spatial_weight = np.exp(-((np.arange(h)[:, None] - h//2)**2 + 
                                   (np.arange(w) - w//2)**2) / (2 * sxy**2))
        
        # Color term
        if len(image.shape) == 3:
            color_diff = np.zeros((h, w))
            for c in range(3):
                color_diff += (image[:, :, c] - image[h//2, w//2, c])**2
            color_weight = np.exp(-color_diff / (2 * srgb**2))
        else:
            color_weight = 1.0
        
        # Combine weights
        pairwise = spatial_weight * color_weight
        
        # Message passing (simplified)
        msg_fg = gaussian_filter(prob_fg, sigma=sxy/2)
        msg_bg = gaussian_filter(prob_bg, sigma=sxy/2)
        
        # Update beliefs
        unary_fg = np.log(prob_fg + 1e-6)
        unary_bg = np.log(prob_bg + 1e-6)
        
        # Combine unary and pairwise
        belief_fg = np.exp(unary_fg + compat * msg_fg)
        belief_bg = np.exp(unary_bg + compat * msg_bg)
        
        # Normalize
        Z = belief_fg + belief_bg
        prob_fg = belief_fg / Z
        prob_bg = belief_bg / Z
    
    return (prob_fg > 0.5).astype(np.float32)

def postprocess_segmentation(image, mask, config=None):
    """
    Complete post-processing pipeline.
    
    Args:
        image: Input image
        mask: Raw prediction mask
        config: Post-processing configuration
    
    Returns:
        Processed mask
    """
    if config is None:
        config = {
            'morphology': {'enabled': True, 'kernel_size': 5},
            'connected_components': {'enabled': True, 'min_area': 50},
            'crf': {'enabled': False, 'iterations': 5}
        }
    
    result = mask.copy()
    
    # Morphological operations
    if config['morphology']['enabled']:
        result = apply_morphological_operations(
            result, 
            kernel_size=config['morphology']['kernel_size']
        )
    
    # Connected component filtering
    if config['connected_components']['enabled']:
        result, _ = connected_component_filtering(
            result,
            min_area=config['connected_components']['min_area']
        )
    
    # CRF refinement
    if config['crf']['enabled']:
        result = crf_refinement(
            image, 
            result,
            num_iterations=config['crf']['iterations']
        )
    
    return result

def evaluate_segmentation(y_true, y_pred, smooth=1e-6):
    """Compute segmentation metrics."""
    y_true_f = y_true.flatten()
    y_pred_f = y_pred.flatten()
    
    # Dice coefficient
    intersection = np.sum(y_true_f * y_pred_f)
    dice = (2. * intersection + smooth) / (np.sum(y_true_f) + np.sum(y_pred_f) + smooth)
    
    # IoU
    union = np.sum(y_true_f) + np.sum(y_pred_f) - intersection
    iou = (intersection + smooth) / (union + smooth)
    
    # Accuracy
    accuracy = np.mean(y_true_f == y_pred_f)
    
    return {'dice': dice, 'iou': iou, 'accuracy': accuracy}

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Morphological kernel too large for thin structures** | An erosion kernel of size 7×7 on thin road or vessel segments (3–5 px wide) removes the entire structure during the erosion step of `MORPH_OPEN`, causing Dice to drop below the raw model output. | Set kernel size ≤ `structure_width / 2`; for thin structures, use a 3×3 kernel; always compare post-processing Dice against the raw baseline before deploying. |
| **Connected component filtering using pixel count on varying-resolution images** | A fixed `min_area=500` threshold removes all foreground on 128×128 images but keeps noise on 1024×1024 images; the same threshold scales with image resolution differently than object scale. | Use relative thresholds: `min_area = max(50, 0.001 * H * W)` where `H×W` is the image size; or express `min_area` in physical units (e.g., mm²) if the pixel spacing is known. |
| **DenseCRF run in Python loop over batch instead of vectorized** | Calling `pydensecrf.densecrf.DenseCRF2D` per image inside a Python loop processes images serially; for a batch of 32 at 512×512, CRF post-processing takes 7–10 seconds, making it impractical for real-time or online evaluation. | Run CRF post-processing only on the test/evaluation set, not during training validation; parallelize with `multiprocessing.Pool` if CRF must be applied to many images. |
| **Applying CRF to multi-class softmax outputs without proper unary setup** | DenseCRF expects negative log-probabilities (unaries) as input. Passing raw softmax probabilities directly without taking `-log` produces the wrong energy function; the CRF output will look identical to the softmax argmax with no boundary refinement. | Convert softmax probabilities to unaries: `unary = -np.log(softmax_prob + 1e-8)` before passing to DenseCRF; ensure the array is C-contiguous (`np.ascontiguousarray`). |
| **Post-processing evaluated only on overall mIoU, missing per-class degradation** | Morphological operations may improve background IoU (removing isolated foreground noise) while degrading foreground IoU (eroding small objects). Reporting only mIoU hides this class-level tradeoff. | Report per-class IoU before and after each post-processing step; accept a step only if it does not decrease any class IoU by more than 1% relative to its value before post-processing. |
