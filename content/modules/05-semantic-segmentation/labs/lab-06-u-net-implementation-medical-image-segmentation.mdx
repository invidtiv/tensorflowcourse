def generate_medical_dataset(num_samples=200, image_size=(256, 256), num_classes=3):
    """
    Generate synthetic medical segmentation dataset.
    
    Simulates:
    - Background (class 0)
    - Organ/Structure 1 (class 1)
    - Organ/Structure 2 (class 2)
    
    Args:
        num_samples: Number of samples
        image_size: Image dimensions
        num_classes: Number of segmentation classes
    
    Returns:
        images, masks
    """

    print("GENERATING SYNTHETIC MEDICAL DATASET")
    print("="*60)
    
    images = []
    masks = []
    
    for i in range(num_samples):
        # Create ultrasound-like image
        img = np.zeros((*image_size, 3), dtype=np.float32)
        
        # Background texture (speckle noise)
        speckle = np.random.randn(*image_size) * 0.1 + 0.5
        img[:, :, 0] = speckle
        img[:, :, 1] = speckle
        img[:, :, 2] = speckle
        
        # Create multi-class mask
        mask = np.zeros((*image_size, num_classes), dtype=np.float32)
        mask[:, :, 0] = 1  # Background
        
        # Generate organ 1 (elliptical shape)
        center1 = (np.random.randint(80, 176), np.random.randint(80, 176))
        axes1 = (np.random.randint(30, 60), np.random.randint(20, 40))
        angle1 = np.random.randint(0, 180)
        
        organ1_mask = np.zeros(image_size, dtype=np.uint8)
        cv2.ellipse(organ1_mask, center1, axes1, angle1, 0, 360, 1, -1)
        
        # Update mask
        mask[:, :, 0] = 1 - organ1_mask
        mask[:, :, 1] = organ1_mask
        
        # Add intensity variation for organ 1
        img[:, :, 0] = np.where(organ1_mask, 0.7 + np.random.randn(*image_size)*0.05, img[:, :, 0])
        img[:, :, 1] = np.where(organ1_mask, 0.6 + np.random.randn(*image_size)*0.05, img[:, :, 1])
        img[:, :, 2] = np.where(organ1_mask, 0.5 + np.random.randn(*image_size)*0.05, img[:, :, 2])
        
        # Generate organ 2 (smaller, circular)
        if num_classes > 2:
            center2 = (np.random.randint(60, 196), np.random.randint(60, 196))
            radius2 = np.random.randint(10, 25)
            
            organ2_mask = np.zeros(image_size, dtype=np.uint8)
            cv2.circle(organ2_mask, center2, radius2, 1, -1)
            
            # Ensure no overlap with organ 1
            organ2_mask = organ2_mask * (1 - organ1_mask)
            
            mask[:, :, 0] = mask[:, :, 0] * (1 - organ2_mask)
            mask[:, :, 2] = organ2_mask
            
            # Add intensity variation for organ 2
            img[:, :, 0] = np.where(organ2_mask, 0.3 + np.random.randn(*image_size)*0.05, img[:, :, 0])
            img[:, :, 1] = np.where(organ2_mask, 0.4 + np.random.randn(*image_size)*0.05, img[:, :, 1])
            img[:, :, 2] = np.where(organ2_mask, 0.8 + np.random.randn(*image_size)*0.05, img[:, :, 2])
        
        # Add global noise
        img += np.random.randn(*img.shape) * 0.02
        img = np.clip(img, 0, 1)
        
        images.append(img)
        masks.append(mask)
    
    images = np.array(images)
    masks = np.array(masks)
    
    print(f"Generated {num_samples} samples")
    print(f"  Images shape: {images.shape}")
    print(f"  Masks shape: {masks.shape}")

    class_pixels = masks.sum(axis=(0, 1, 2))
    total_pixels = masks.shape[0] * masks.shape[1] * masks.shape[2]
    
    print("\nClass distribution:")
    for c in range(num_classes):
        pct = class_pixels[c] / total_pixels * 100
        print(f"  Class {c}: {pct:.2f}%")
    
    return images, masks

def sensitivity(y_true, y_pred):
    """Sensitivity (Recall) - True Positive Rate."""
    true_positives = tf.reduce_sum(tf.round(tf.clip_by_value(y_true * y_pred, 0, 1)))
    possible_positives = tf.reduce_sum(tf.round(tf.clip_by_value(y_true, 0, 1)))
    return true_positives / (possible_positives + tf.keras.backend.epsilon())


def specificity(y_true, y_pred):
    """Specificity - True Negative Rate."""
    true_negatives = tf.reduce_sum(tf.round(tf.clip_by_value((1-y_true) * (1-y_pred), 0, 1)))
    possible_negatives = tf.reduce_sum(tf.round(tf.clip_by_value(1-y_true, 0, 1)))
    return true_negatives / (possible_negatives + tf.keras.backend.epsilon())


def dice_coef_multiclass(y_true, y_pred, smooth=1e-6):
    """Dice coefficient for multi-class segmentation."""
    y_true_f = tf.reshape(y_true, [-1, y_true.shape[-1]])
    y_pred_f = tf.reshape(y_pred, [-1, y_pred.shape[-1]])
    
    intersection = tf.reduce_sum(y_true_f * y_pred_f, axis=0)
    union = tf.reduce_sum(y_true_f, axis=0) + tf.reduce_sum(y_pred_f, axis=0)
    
    dice = tf.reduce_mean((2. * intersection + smooth) / (union + smooth))
    return dice


def iou_multiclass(y_true, y_pred, smooth=1e-6):
    """IoU (Jaccard Index) for multi-class segmentation."""
    y_true_f = tf.reshape(y_true, [-1, y_true.shape[-1]])
    y_pred_f = tf.reshape(y_pred, [-1, y_pred.shape[-1]])
    
    intersection = tf.reduce_sum(y_true_f * y_pred_f, axis=0)
    union = tf.reduce_sum(y_true_f, axis=0) + tf.reduce_sum(y_pred_f, axis=0) - intersection
    
    iou = tf.reduce_mean((intersection + smooth) / (union + smooth))
    return iou

def compute_class_weights(masks):
    """
    Compute class weights for imbalanced dataset.
    
    Uses inverse frequency weighting.
    """
    class_pixels = masks.sum(axis=(0, 1, 2))
    total_pixels = masks.shape[0] * masks.shape[1] * masks.shape[2]
    
    # Inverse frequency
    weights = total_pixels / (len(class_pixels) * class_pixels + 1e-6)
    
    # Normalize
    weights = weights / weights.sum() * len(class_pixels)
    
    return weights

def weighted_categorical_crossentropy(weights):
    """
    Create weighted categorical crossentropy loss.
    
    Args:
        weights: Class weights array
    
    Returns:
        Loss function
    """
    weights_tensor = tf.constant(weights, dtype=tf.float32)
    
    def loss(y_true, y_pred):
        # Clip predictions to avoid log(0)
        y_pred = tf.clip_by_value(y_pred, 1e-7, 1 - 1e-7)
        
        # Compute weighted loss
        loss = y_true * tf.math.log(y_pred)
        loss = loss * weights_tensor
        loss = -tf.reduce_sum(loss, axis=-1)
        
        return tf.reduce_mean(loss)
    
    return loss


# Combined weighted loss
def combined_weighted_loss(y_true, y_pred):
    """Combine weighted CE with Dice loss."""
    wce = weighted_categorical_crossentropy(class_weights)(y_true, y_pred)
    dice = 1 - dice_coef_multiclass(y_true, y_pred)
    return wce + dice

def build_medical_unet(input_shape=(256, 256, 3), num_classes=3):
    """Build U-Net for multi-class medical segmentation."""
    print("\n" + "="*60)
    print("BUILDING MEDICAL U-NET")
    print("="*60)
    
    def conv_block(inputs, num_filters):
        x = layers.Conv2D(num_filters, 3, padding='same', use_bias=False)(inputs)
        x = layers.BatchNormalization()(x)
        x = layers.ReLU()(x)
        x = layers.Conv2D(num_filters, 3, padding='same', use_bias=False)(x)
        x = layers.BatchNormalization()(x)
        x = layers.ReLU()(x)
        return x
    
    inputs = layers.Input(shape=input_shape)
    
    # Encoder
    filters = [64, 128, 256, 512, 1024]
    skips = []
    x = inputs
    
    for i, f in enumerate(filters[:-1]):
        x = conv_block(x, f)
        skips.append(x)
        x = layers.MaxPooling2D(2)(x)
    
    # Bottleneck
    x = conv_block(x, filters[-1])
    x = layers.Dropout(0.5)(x)
    
    # Decoder
    for i, f in enumerate(reversed(filters[:-1])):
        x = layers.Conv2DTranspose(f, 2, strides=2, padding='same')(x)
        x = layers.Concatenate()([x, skips[-(i+1)]])
        x = conv_block(x, f)
    
    # Output - softmax for multi-class
    outputs = layers.Conv2D(num_classes, 1, activation='softmax')(x)
    
    model = Model(inputs, outputs, name='Medical-U-Net')
    
    print(f"Model input shape: {model.input_shape}")
    print(f"Model output shape: {model.output_shape}")
    print(f"Total parameters: {model.count_params():,}")
    
    return model

def compute_per_class_metrics(y_true, y_pred, num_classes=3):
    """Compute per-class segmentation metrics."""
    print("\n" + "="*60)
    print("PER-CLASS METRICS")
    print("="*60)
    
    y_true_classes = np.argmax(y_true, axis=-1)
    y_pred_classes = np.argmax(y_pred, axis=-1)
    
    print(f"{'Class':<10} {'Dice':<10} {'IoU':<10} {'Precision':<12} {'Recall':<10}")
    print("-" * 60)
    
    for c in range(num_classes):
        true_c = (y_true_classes == c).astype(float)
        pred_c = (y_pred_classes == c).astype(float)
        
        # Flatten
        true_c = true_c.flatten()
        pred_c = pred_c.flatten()
        
        # Compute metrics
        intersection = np.sum(true_c * pred_c)
        union = np.sum(true_c) + np.sum(pred_c) - intersection
        
        dice = (2 * intersection) / (np.sum(true_c) + np.sum(pred_c) + 1e-6)
        iou = intersection / (union + 1e-6)
        
        tp = np.sum((true_c == 1) & (pred_c == 1))
        fp = np.sum((true_c == 0) & (pred_c == 1))
        fn = np.sum((true_c == 1) & (pred_c == 0))
        
        precision = tp / (tp + fp + 1e-6)
        recall = tp / (tp + fn + 1e-6)
        
        print(f"{c:<10} {dice:<10.4f} {iou:<10.4f} {precision:<12.4f} {recall:<10.4f}")

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Class weight vector wrong length or on wrong axis** | Passing `class_weights=[1, 10, 5]` to `SparseCategoricalCrossentropy` does not work directly; the built-in loss doesn't accept a vector. Attempting to use `sample_weight` instead of pixel-wise weights silently applies the same weight to entire images. | Implement pixel-wise weighting manually: compute a weight map tensor of shape `(B,H,W)` assigning each pixel its class weight, then multiply the per-pixel loss tensor element-wise before reducing. |
| **Evaluating Dice on the argmax mask instead of soft probabilities** | Taking `argmax` before computing Dice introduces rounding that makes Dice non-differentiable and alters metric values at evaluation time compared to training (where soft probabilities are used). | At evaluation, compute Dice on one-hot encoded argmax predictions to get the hard metric; at training, compute soft Dice directly on softmax probabilities to keep gradients smooth. |
| **Preprocessing mismatch: training on normalized images, inferring on raw scans** | Medical images (CT, MRI) often have vastly different intensity ranges between scanners. Training on z-score normalized images but inferring on unnormalized HU values degrades Dice from 0.75 to near zero. | Apply identical normalization (clip to `[p1, p99]` then z-score) in both the training pipeline and the inference preprocessing function; export the computed statistics alongside the model weights. |
| **Comparing sensitivity without fixing specificity** | Reporting sensitivity in isolation allows trivially optimizing it by predicting the entire image as tumor; sensitivity reaches 1.0 while generating massive false positives (specificity → 0). | Always report sensitivity and specificity as a pair, or use the Dice/F1 score which intrinsically balances precision and recall. Report confidence intervals over the test set. |
| **Thin structure (vessel) loss due to max-pool resolution collapse** | Vessels are often 1–3 pixels wide; the standard U-Net bottleneck at 16×16 loses fine vascular topology entirely. Reconstructed vessels appear as blobs, not tubular structures. | Use deep supervision at the 64×64 and 32×32 decoder levels with auxiliary Dice losses; or switch to a topology-preserving loss (clDice, Shit et al. 2021) that penalizes centerline connectivity errors. |
