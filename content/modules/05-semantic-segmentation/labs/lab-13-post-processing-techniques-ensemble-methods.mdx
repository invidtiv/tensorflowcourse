def build_simple_unet(input_shape=(128, 128, 3), name='unet'):
    """Build simple U-Net variant."""
    inputs = layers.Input(shape=input_shape, name=f'{name}_input')
    
    # Encoder
    c1 = layers.Conv2D(32, 3, activation='relu', padding='same')(inputs)
    c1 = layers.Conv2D(32, 3, activation='relu', padding='same')(c1)
    p1 = layers.MaxPooling2D(2)(c1)
    
    c2 = layers.Conv2D(64, 3, activation='relu', padding='same')(p1)
    c2 = layers.Conv2D(64, 3, activation='relu', padding='same')(c2)
    p2 = layers.MaxPooling2D(2)(c2)
    
    # Bottleneck
    c3 = layers.Conv2D(128, 3, activation='relu', padding='same')(p2)
    
    # Decoder
    u4 = layers.Conv2DTranspose(64, 2, strides=2, activation='relu', padding='same')(c3)
    u4 = layers.Concatenate()([u4, c2])
    c4 = layers.Conv2D(64, 3, activation='relu', padding='same')(u4)
    
    u5 = layers.Conv2DTranspose(32, 2, strides=2, activation='relu', padding='same')(c4)
    u5 = layers.Concatenate()([u5, c1])
    c5 = layers.Conv2D(32, 3, activation='relu', padding='same')(u5)
    
    outputs = layers.Conv2D(1, 1, activation='sigmoid')(c5)
    
    return Model(inputs, outputs, name=name)


def build_simple_fcn(input_shape=(128, 128, 3), name='fcn'):
    """Build simple FCN variant."""
    inputs = layers.Input(shape=input_shape, name=f'{name}_input')
    
    # Encoder
    x = layers.Conv2D(32, 3, activation='relu', padding='same')(inputs)
    x = layers.Conv2D(32, 3, activation='relu', padding='same')(x)
    x = layers.MaxPooling2D(2)(x)
    
    x = layers.Conv2D(64, 3, activation='relu', padding='same')(x)
    x = layers.Conv2D(64, 3, activation='relu', padding='same')(x)
    x = layers.MaxPooling2D(2)(x)
    
    x = layers.Conv2D(128, 3, activation='relu', padding='same')(x)
    
    # Decoder
    x = layers.Conv2DTranspose(64, 2, strides=2, activation='relu', padding='same')(x)
    x = layers.Conv2DTranspose(32, 2, strides=2, activation='relu', padding='same')(x)
    
    outputs = layers.Conv2D(1, 1, activation='sigmoid')(x)
    
    return Model(inputs, outputs, name=name)

def dice_loss(y_true, y_pred, smooth=1e-6):
    y_true_f = tf.reshape(y_true, [-1])
    y_pred_f = tf.reshape(y_pred, [-1])
    intersection = tf.reduce_sum(y_true_f * y_pred_f)
    return 1 - (2. * intersection + smooth) / (tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f) + smooth)


def train_model(model, X_train, y_train, X_val, y_val, epochs=20):
    """Train a model."""

        optimizer=keras.optimizers.Adam(1e-3),
        loss=dice_loss,
        metrics=['accuracy']
    )
    
    history = model.fit(
        X_train, y_train,
        validation_data=(X_val, y_val),
        batch_size=8,
        epochs=epochs,
        verbose=0
    )
    
    return history

def ensemble_average(models, X):
    """
    Simple averaging ensemble.
    
    Args:
        models: List of trained models
        X: Input data
    
    Returns:
        Averaged predictions
    """
    predictions = []
    for model in models:
        pred = model.predict(X, verbose=0)
        predictions.append(pred)
    
    # Average predictions
    ensemble_pred = np.mean(predictions, axis=0)
    
    return ensemble_pred


def ensemble_weighted(models, X, weights=None):
    """
    Weighted averaging ensemble.
    
    Args:
        models: List of trained models
        X: Input data
        weights: Model weights (default: equal weights)
    
    Returns:
        Weighted averaged predictions
    """
    if weights is None:
        weights = [1.0 / len(models)] * len(models)
    
    predictions = []
    for model in models:
        pred = model.predict(X, verbose=0)
        predictions.append(pred)
    
    # Weighted average
    ensemble_pred = sum(w * p for w, p in zip(weights, predictions))
    
    return ensemble_pred

def tta_predict(model, X, augmentations=['flip_h', 'flip_v', 'rot90']):
    """
    Test-time augmentation for more robust predictions.
    
    Args:
        model: Trained model
        X: Input data (batch, H, W, C)
        augmentations: List of augmentations to apply
    
    Returns:
        Averaged predictions over augmentations
    """
    predictions = []
    
    # Original
    predictions.append(model.predict(X, verbose=0))
    
    # Horizontal flip
    if 'flip_h' in augmentations:
        X_flipped = np.flip(X, axis=2)
        pred_flipped = model.predict(X_flipped, verbose=0)
        pred_restored = np.flip(pred_flipped, axis=2)
        predictions.append(pred_restored)
    
    # Vertical flip
    if 'flip_v' in augmentations:
        X_flipped = np.flip(X, axis=1)
        pred_flipped = model.predict(X_flipped, verbose=0)
        pred_restored = np.flip(pred_flipped, axis=1)
        predictions.append(pred_restored)
    
    # 90 degree rotation
    if 'rot90' in augmentations:
        X_rotated = np.rot90(X, k=1, axes=(1, 2))
        pred_rotated = model.predict(X_rotated, verbose=0)
        pred_restored = np.rot90(pred_rotated, k=-1, axes=(1, 2))
        predictions.append(pred_restored)
    
    # Average all predictions
    return np.mean(predictions, axis=0)

def multi_scale_predict(model, X, scales=[0.75, 1.0, 1.25]):
    """
    Multi-scale inference for better predictions.
    
    Args:
        model: Trained model
        X: Input data
        scales: List of scales to test
    
    Returns:
        Averaged predictions over scales
    """
    original_shape = X.shape[1:3]
    predictions = []
    
    for scale in scales:
        if scale == 1.0:
            # Original scale
            pred = model.predict(X, verbose=0)
        else:
            # Resize input
            new_h = int(original_shape[0] * scale)
            new_w = int(original_shape[1] * scale)
            
            X_scaled = tf.image.resize(X, [new_h, new_w], method='bilinear')
            pred_scaled = model.predict(X_scaled, verbose=0)
            
            # Resize prediction back
            pred = tf.image.resize(pred_scaled, original_shape, method='bilinear')
            pred = pred.numpy()
        
        predictions.append(pred)
    
    return np.mean(predictions, axis=0)

def full_ensemble_predict(models, X, use_tta=True, use_multiscale=True):
    """
    Full ensemble with TTA and multi-scale.
    
    Args:
        models: List of models
        X: Input data
        use_tta: Whether to use test-time augmentation
        use_multiscale: Whether to use multi-scale inference
    
    Returns:
        Final predictions
    """
    all_predictions = []
    
    for model in models:
        if use_tta and use_multiscale:
            # Apply both TTA and multi-scale
            scales = [0.75, 1.0, 1.25]
            for scale in scales:
                if scale == 1.0:
                    pred = tta_predict(model, X)
                else:
                    original_shape = X.shape[1:3]
                    new_h = int(original_shape[0] * scale)
                    new_w = int(original_shape[1] * scale)
                    
                    X_scaled = tf.image.resize(X, [new_h, new_w], method='bilinear')
                    pred_scaled = tta_predict(model, X_scaled.numpy())
                    pred = tf.image.resize(pred_scaled, original_shape, method='bilinear')
                    pred = pred.numpy()
                
                all_predictions.append(pred)
        
        elif use_tta:
            all_predictions.append(tta_predict(model, X))
        
        elif use_multiscale:
            all_predictions.append(multi_scale_predict(model, X))
        
        else:
            all_predictions.append(model.predict(X, verbose=0))
    
    return np.mean(all_predictions, axis=0)

def evaluate_predictions(y_true, y_pred, threshold=0.5):
    """Evaluate predictions."""
    y_pred_binary = (y_pred > threshold).astype(float)
    
    y_true_f = y_true.flatten()
    y_pred_f = y_pred_binary.flatten()
    
    # Dice
    intersection = np.sum(y_true_f * y_pred_f)
    dice = (2. * intersection) / (np.sum(y_true_f) + np.sum(y_pred_f) + 1e-6)
    
    # IoU
    union = np.sum(y_true_f) + np.sum(y_pred_f) - intersection
    iou = intersection / (union + 1e-6)
    
    # Accuracy
    accuracy = np.mean(y_true_f == y_pred_f)
    
    return {'dice': dice, 'iou': iou, 'accuracy': accuracy}

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Averaging logits instead of probabilities across ensemble members** | Models trained with different loss scales or architectures produce logits on different numerical ranges; averaging raw logits gives disproportionate weight to the model with larger magnitude outputs, effectively reducing the ensemble to a single dominant model. | Always apply `softmax` to convert each model's logits to probabilities before averaging; verify that ensemble probabilities sum to 1.0 per pixel with `np.allclose(avg.sum(-1), 1)`. |
| **Applying TTA flip without reversing the prediction mask** | Horizontal-flipping the input image for TTA then averaging the predicted mask directly (without flipping the mask back) produces an averaged mask that is half-flipped, degrading boundary localization especially for asymmetric objects. | After inference on the flipped input, reverse-flip the predicted mask (`np.flip(mask, axis=2)`) before averaging with the original-orientation prediction. |
| **Multi-scale TTA without resizing predictions back to original resolution** | Running inference at 0.75× produces a smaller output mask (e.g., 96×96 for a 128×128 image); averaging this with the 128×128 full-scale prediction requires upsampling first; skipping this step causes shape mismatch errors or silent zero-padding that corrupts the ensemble. | After inference at each scale, resize the output mask back to the original resolution using bilinear interpolation before accumulating into the ensemble average. |
| **Ensemble models with identical architecture and initialization** | Ensembling three copies of U-Net trained with the same random seed and data order adds no diversity; the models make the same errors on the same samples, and the ensemble Dice is nearly identical to the single-model baseline. | Ensure diversity: use different architectures (FCN, U-Net, DeepLab), different random seeds, different data augmentation sequences, or different backbone weights (ImageNet vs random init). |
| **Measuring ensemble inference time incorrectly** | Timing the ensemble on a single image in a Python loop includes Python overhead per call; reported latency may be 2–3× higher than the true model execution time, making the ensemble seem slower than it is. | Batch all ensemble forward passes together (model 1, 2, 3 each on the full batch), measure total wall time, then divide by batch size; use `tf.function` to reduce per-call overhead. |
