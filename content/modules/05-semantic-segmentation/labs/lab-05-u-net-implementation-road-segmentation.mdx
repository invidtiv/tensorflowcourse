def generate_synthetic_road_data(num_samples=100, image_size=(256, 256)):
    """
    Generate synthetic road segmentation dataset.
    
    Creates images with roads and corresponding segmentation masks.
    This simulates a real road dataset for training.
    
    Args:
        num_samples: Number of samples to generate
        image_size: Size of generated images
    
    Returns:
        images, masks as numpy arrays
    """
    print("\n" + "="*60)
    print("GENERATING SYNTHETIC ROAD DATASET")
    print("="*60)
    
    images = []
    masks = []
    
    for i in range(num_samples):
        # Create background (grass/ground)
        img = np.ones((*image_size, 3), dtype=np.float32)
        img[:, :, 0] = 0.3 + np.random.randn(*image_size) * 0.05  # R: brownish
        img[:, :, 1] = 0.5 + np.random.randn(*image_size) * 0.05  # G: greenish
        img[:, :, 2] = 0.2 + np.random.randn(*image_size) * 0.05  # B: dark
        
        # Create mask
        mask = np.zeros(image_size, dtype=np.float32)
        
        # Generate random road shape
        num_points = np.random.randint(3, 6)
        points = []
        for j in range(num_points):
            x = np.random.randint(20, image_size[1] - 20)
            y = np.random.randint(20, image_size[0] - 20)
            points.append((x, y))
        
        # Draw road on image and mask
        road_width = np.random.randint(15, 30)
        
        for j in range(len(points) - 1):
            # Draw line on image (gray road)
            cv2.line(img, points[j], points[j+1], (0.4, 0.4, 0.4), road_width)
            # Draw line on mask
            cv2.line(mask, points[j], points[j+1], 1.0, road_width)
        
        # Add some noise
        img += np.random.randn(*img.shape) * 0.02
        img = np.clip(img, 0, 1)
        
        images.append(img)
        masks.append(mask)
    
    images = np.array(images)
    masks = np.array(masks)
    masks = masks[..., np.newaxis]  # Add channel dimension
    
    print(f"Generated {num_samples} samples")
    print(f"  Images shape: {images.shape}")
    print(f"  Masks shape: {masks.shape}")
    
    return images, masks

def build_road_unet(input_shape=(256, 256, 3)):
    """
    Build U-Net optimized for road segmentation.
    
    Uses binary classification (road vs background).
    """
    print("\n" + "="*60)
    print("BUILDING ROAD SEGMENTATION U-NET")
    print("="*60)
    
    def conv_block(inputs, num_filters):
        x = layers.Conv2D(num_filters, 3, padding='same', use_bias=False)(inputs)
        x = layers.BatchNormalization()(x)
        x = layers.ReLU()(x)
        x = layers.Conv2D(num_filters, 3, padding='same', use_bias=False)(x)
        x = layers.BatchNormalization()(x)
        x = layers.ReLU()(x)
        return x
    
    inputs = layers.Input(shape=input_shape)
    
    # Encoder
    filters = [64, 128, 256, 512, 1024]
    skips = []
    x = inputs
    
    for i, f in enumerate(filters[:-1]):
        x = conv_block(x, f)
        skips.append(x)
        x = layers.MaxPooling2D(2)(x)
    
    # Bottleneck
    x = conv_block(x, filters[-1])
    x = layers.Dropout(0.5)(x)
    
    # Decoder
    for i, f in enumerate(reversed(filters[:-1])):
        x = layers.Conv2DTranspose(f, 2, strides=2, padding='same')(x)
        x = layers.Concatenate()([x, skips[-(i+1)]])
        x = conv_block(x, f)
    
    # Output
    outputs = layers.Conv2D(1, 1, activation='sigmoid')(x)
    
    model = Model(inputs, outputs, name='Road-U-Net')
    
    print(f"Model input shape: {model.input_shape}")
    print(f"Model output shape: {model.output_shape}")
    print(f"Total parameters: {model.count_params():,}")
    
    return model


road_unet = build_road_unet()

def dice_coefficient(y_true, y_pred, smooth=1e-6):
    """
    Dice coefficient metric.
    
    Measures overlap between predicted and true masks.
    Range: 0 (no overlap) to 1 (perfect overlap)
    """
    y_true_f = tf.reshape(y_true, [-1])
    y_pred_f = tf.reshape(y_pred, [-1])
    
    intersection = tf.reduce_sum(y_true_f * y_pred_f)
    return (2. * intersection + smooth) / (tf.reduce_sum(y_true_f) + tf.reduce_sum(y_pred_f) + smooth)


def dice_loss(y_true, y_pred):
    """Dice loss = 1 - Dice coefficient."""
    return 1 - dice_coefficient(y_true, y_pred)


def combined_loss(y_true, y_pred):
    """
    Combined BCE + Dice loss.
    
    Balances pixel-wise accuracy (BCE) with overlap (Dice).
    """
    bce = keras.losses.binary_crossentropy(y_true, y_pred)
    dice = dice_loss(y_true, y_pred)
    return bce + dice

def post_process_mask(mask, min_area=100, kernel_size=5):
    """
    Apply post-processing to segmentation mask.
    
    Args:
        mask: Binary mask
        min_area: Minimum contour area to keep
        kernel_size: Morphological kernel size
    
    Returns:
        Post-processed mask
    """
    # Convert to uint8
    mask_uint8 = (mask * 255).astype(np.uint8)
    
    # Morphological closing (fill small holes)
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (kernel_size, kernel_size))
    mask_closed = cv2.morphologyEx(mask_uint8, cv2.MORPH_CLOSE, kernel)
    
    # Morphological opening (remove small noise)
    mask_opened = cv2.morphologyEx(mask_closed, cv2.MORPH_OPEN, kernel)
    
    # Remove small contours
    contours, _ = cv2.findContours(mask_opened, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    mask_filtered = np.zeros_like(mask_opened)
    
    for contour in contours:
        area = cv2.contourArea(contour)
        if area >= min_area:
            cv2.drawContours(mask_filtered, [contour], -1, 255, -1)
    
    return mask_filtered / 255.0

<Callout type="pitfall" title="Common Pitfalls">

**1. Forgetting to normalize images to [0, 1].** U-Net with sigmoid output expects inputs scaled to [0, 1] or standardized. Feeding uint8 [0, 255] images causes gradient explosion and NaN losses in the first epoch.

**2. Using standard BCE on heavily imbalanced masks.** Road pixels may be only 15-30% of the image. Standard BCE optimizes for pixel accuracy, so the model learns to predict "all background" and still gets 70%+ accuracy. Always use Dice loss or weighted BCE.

**3. Not applying the same augmentation to image and mask.** Geometric transforms (rotation, flip, crop) must be applied identically to both. Using separate random augmentation pipelines for image and mask produces misaligned training pairs, causing the model to learn noise.

**4. Setting `min_area` too aggressively in post-processing.** Removing small connected components helps clean noise, but setting the threshold too high (e.g., 1000 pixels on 256×256 images) can delete valid narrow road segments. Always visualize before choosing the threshold.

</Callout>
