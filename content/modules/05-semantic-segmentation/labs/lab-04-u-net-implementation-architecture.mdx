# Section 1: U-Net Building Blocks

def conv_block(inputs, num_filters, kernel_size=3, activation='relu', name=None):
    """
    Convolutional block with two conv layers and batch normalization.
    
    This is the basic building block of U-Net.
    
    Args:
        inputs: Input tensor
        num_filters: Number of output filters
        kernel_size: Convolution kernel size
        activation: Activation function
        name: Block name prefix
    
    Returns:
        Output tensor
    """
    x = inputs
    
    # First conv
    x = layers.Conv2D(
        num_filters, 
        kernel_size, 
        padding='same',
        use_bias=False,
        name=f'{name}_conv1' if name else None
    )(x)
    x = layers.BatchNormalization(name=f'{name}_bn1' if name else None)(x)
    x = layers.Activation(activation, name=f'{name}_act1' if name else None)(x)
    
    # Second conv
    x = layers.Conv2D(
        num_filters, 
        kernel_size, 
        padding='same',
        use_bias=False,
        name=f'{name}_conv2' if name else None
    )(x)
    x = layers.BatchNormalization(name=f'{name}_bn2' if name else None)(x)
    x = layers.Activation(activation, name=f'{name}_act2' if name else None)(x)
    
    return x


def encoder_block(inputs, num_filters, dropout_rate=0.0, name=None):
    """
    Encoder block: Conv block + Max pooling.
    
    Args:
        inputs: Input tensor
        num_filters: Number of filters for conv block
        dropout_rate: Dropout rate (0 to disable)
        name: Block name prefix
    
    Returns:
        (pooled_output, skip_connection) tuple
    """
    # Conv block
    x = conv_block(inputs, num_filters, name=name)
    
    # Save for skip connection
    skip = x
    
    # Max pooling
    x = layers.MaxPooling2D(pool_size=(2, 2), name=f'{name}_pool' if name else None)(x)
    
    # Optional dropout
    if dropout_rate > 0:
        x = layers.Dropout(dropout_rate, name=f'{name}_dropout' if name else None)(x)
    
    return x, skip


def decoder_block(inputs, skip_features, num_filters, dropout_rate=0.0, name=None):
    """
    Decoder block: Upsampling + Concatenation + Conv block.
    
    Args:
        inputs: Input tensor from deeper layer
        skip_features: Skip connection features from encoder
        num_filters: Number of filters for conv block
        dropout_rate: Dropout rate
        name: Block name prefix
    
    Returns:
        Output tensor
    """
    # Upsample
    x = layers.Conv2DTranspose(
        num_filters,
        kernel_size=(2, 2),
        strides=(2, 2),
        padding='same',
        name=f'{name}_upconv' if name else None
    )(inputs)
    
    # Concatenate with skip connection
    x = layers.Concatenate(name=f'{name}_concat' if name else None)([x, skip_features])
    
    # Optional dropout
    if dropout_rate > 0:
        x = layers.Dropout(dropout_rate, name=f'{name}_dropout' if name else None)(x)
    
    # Conv block
    x = conv_block(x, num_filters, name=name)
    
    return x

# Section 2: Complete U-Net Architecture

def build_unet(
    input_shape=(256, 256, 3),
    num_classes=1,
    filters=[64, 128, 256, 512, 1024],
    dropout_rates=[0.0, 0.0, 0.0, 0.0, 0.5],
    output_activation='sigmoid'
):
    """
    Build complete U-Net architecture.
    
    Architecture:
    - Contracting path (encoder): 4 downsampling blocks
    - Bottleneck: 1 conv block
    - Expanding path (decoder): 4 upsampling blocks with skip connections
    - Output: 1x1 conv for classification
    
    Args:
        input_shape: Input image shape (H, W, C)
        num_classes: Number of output classes (1 for binary)
        filters: List of filter counts for each level
        dropout_rates: Dropout rates for each level
        output_activation: 'sigmoid' for binary, 'softmax' for multi-class
    
    Returns:
        U-Net model
    """
    print("\n" + "="*60)
    print("BUILDING U-NET")
    print("="*60)
    
    # Input
    inputs = layers.Input(shape=input_shape, name='input')
    
    # ============ CONTRACTING PATH (ENCODER) ============
    
    skips = []
    x = inputs
    
    for i, (num_filters, dropout) in enumerate(zip(filters[:-1], dropout_rates[:-1])):
        x, skip = encoder_block(
            x, 
            num_filters, 
            dropout_rate=dropout,
            name=f'encoder_{i+1}'
        )
        skips.append(skip)
        print(f"  Encoder {i+1}: output={x.shape}, skip={skip.shape}")
    
    # ============ BOTTLENECK ============
    
    x = conv_block(x, filters[-1], name='bottleneck')
    if dropout_rates[-1] > 0:
        x = layers.Dropout(dropout_rates[-1], name='bottleneck_dropout')(x)
    print(f"  Bottleneck: {x.shape}")
    
    # ============ EXPANDING PATH (DECODER) ============
    
    for i, (num_filters, dropout) in enumerate(zip(reversed(filters[:-1]), reversed(dropout_rates[:-1]))):
        skip = skips[-(i+1)]  # Get skip connection in reverse order
        x = decoder_block(
            x,
            skip,
            num_filters,
            dropout_rate=dropout,
            name=f'decoder_{len(filters)-i-1}'
        )
        print(f"  Decoder {len(filters)-i-1}: {x.shape}")
    
    # ============ OUTPUT ============
    
    if num_classes == 1:
        # Binary segmentation
        outputs = layers.Conv2D(
            1,
            kernel_size=1,
            activation=output_activation,
            name='output'
        )(x)
    else:
        # Multi-class segmentation
        outputs = layers.Conv2D(
            num_classes,
            kernel_size=1,
            activation=output_activation,
            name='output'
        )(x)
    
    model = Model(inputs, outputs, name='U-Net')
    
    print(f"\nU-Net input shape: {model.input_shape}")
    print(f"U-Net output shape: {model.output_shape}")
    print(f"U-Net parameters: {model.count_params():,}")
    
    return model


# Build U-Net for binary segmentation
unet_binary = build_unet(

def build_attention_unet(input_shape=(256, 256, 3), num_classes=1):
    """
    Build Attention U-Net with attention gates in skip connections.
    
    Attention gates help the model focus on relevant regions
    when combining skip connections.
    """
    print("\n" + "="*60)
    print("BUILDING ATTENTION U-NET")
    print("="*60)
    
    def attention_gate(x, g, num_filters):
        """
        Attention gate for skip connections.
        
        Args:
            x: Skip connection feature map
            g: Gating signal from decoder
            num_filters: Number of filters
        
        Returns:
            Attended feature map
        """
        # Transform gating signal
        g_conv = layers.Conv2D(num_filters, 1, padding='same')(g)
        
        # Transform skip connection
        x_conv = layers.Conv2D(num_filters, 1, padding='same')(x)
        
        # Add and activate
        psi = layers.Add()([g_conv, x_conv])
        psi = layers.ReLU()(psi)
        
        # Attention coefficients
        psi = layers.Conv2D(1, 1, padding='same', activation='sigmoid')(psi)
        
        # Apply attention
        return layers.Multiply()([x, psi])
    
    inputs = layers.Input(shape=input_shape)
    
    # Encoder
    filters = [64, 128, 256, 512, 1024]
    x = inputs
    skips = []
    
    for i, f in enumerate(filters[:-1]):
        x, skip = encoder_block(x, f, name=f'enc{i+1}')
        skips.append(skip)
    
    # Bottleneck
    x = conv_block(x, filters[-1], name='bottleneck')
    
    # Decoder with attention gates
    for i, f in enumerate(reversed(filters[:-1])):
        skip = skips[-(i+1)]
        
        # Upsample
        up = layers.Conv2DTranspose(f, 2, strides=2, padding='same')(x)
        
        # Apply attention gate to skip connection
        attended_skip = attention_gate(skip, up, f)
        
        # Concatenate
        x = layers.Concatenate()([up, attended_skip])
        x = conv_block(x, f, name=f'dec{len(filters)-i-1}')
    
    outputs = layers.Conv2D(num_classes, 1, activation='sigmoid')(x)
    
    model = Model(inputs, outputs, name='Attention-U-Net')
    print(f"Attention U-Net parameters: {model.count_params():,}")
    
    return model

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Skip-connection shape mismatch at `Concatenate()`** | When input spatial dimensions are not powers of 2, or when `padding='valid'` is used in any encoder conv, the encoder and decoder feature maps can differ by 1–2 pixels, causing a shape error at concatenation. | Enforce `padding='same'` everywhere; ensure input is padded to a power-of-2 size; or add `tf.keras.layers.CenterCrop` / `Resizing` before concatenation. |
| **Uninitialized BatchNorm running statistics** | At inference (`training=False`), BatchNorm uses running mean/variance accumulated during training. Evaluating immediately after `model.build()` without training returns near-zero running mean and all-ones variance, producing misleading outputs. | Always run at least one forward pass with `training=True` (or call `model.fit`) before trusting evaluation metrics; the loss may look reasonable while predictions are random. |
| **Bottleneck dropout rate too high for small datasets** | The original paper's `dropout=0.5` was designed for 572×572 histology images with thousands of annotated samples. Applying it to small synthetic datasets (< 500 images) causes the network to fail to learn any structure for the first 10–20 epochs. | Start at `dropout=0.2–0.3` for small datasets; only increase to 0.5 if the model clearly overfits on the training Dice score. |
| **`Conv2DTranspose` checkerboard artifacts** | Transposed convolution with `strides=2` and `kernel_size=2` creates overlapping output regions that produce a visible grid pattern in the predicted mask. | Replace `Conv2DTranspose(f, 2, strides=2)` with `UpSampling2D(2) + Conv2D(f, 3, padding='same')`; bilinear upsampling eliminates the artifact at the cost of slightly fewer learnable decoder parameters. |
| **Multi-class sigmoid + `categorical_crossentropy` mismatch** | For `num_classes > 1`, using `activation='sigmoid'` on the output produces independent per-class probabilities that sum to more than 1, which breaks `CategoricalCrossentropy` and makes Dice loss undefined. | Use `activation='softmax'` for multi-class output; pass one-hot masks (`tf.one_hot(mask, num_classes)`) with `CategoricalCrossentropy`, or integer masks with `SparseCategoricalCrossentropy`. |

## Stretch Goals

1. **Build a Residual U-Net.** Replace each `conv_block` with a residual block that adds the block input to its output via a 1×1 projection conv when channel counts differ. Improved gradient flow typically yields 0.5–1.5% higher mIoU over 50 epochs on the same parameter budget, particularly on deeper (5-level) architectures.
2. **Implement U-Net++ (nested dense skip connections).** Add intermediate dense nodes between encoder and decoder paths at each resolution level. Start with the `d=2` variant (two intermediate nodes per level, as in Zhou et al. 2018) and compare Dice score and parameter count vs. standard U-Net.
3. **Swap in an EfficientNetB0 encoder backbone.** Replace the custom contracting path with `tf.keras.applications.EfficientNetB0(include_top=False, weights='imagenet')`, freeze the first 3 stages, and attach the U-Net decoder to the last 4 stage outputs. ImageNet pretraining typically boosts mIoU by 5–10% on real segmentation benchmarks.
4. **Enable dynamic (any-resolution) inference.** Modify `build_unet` to accept `input_shape=(None, None, C)` and verify that `model.predict` correctly handles both a 256×256 and a 512×384 input in a single forward pass, producing output spatial dimensions that match the input.
