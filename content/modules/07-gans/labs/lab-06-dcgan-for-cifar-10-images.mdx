## Complete Implementation

```python
"""
Lab 7.5: DCGAN for CIFAR-10 Images
===================================
DCGAN implementation for generating CIFAR-10 color images.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from tensorflow import keras
from tensorflow.keras import layers
import time
import os

os.makedirs('dcgan_cifar_output', exist_ok=True)

# =============================================================================
# Part 1: Load and Preprocess CIFAR-10
# =============================================================================

def load_cifar10_data():
    """
    Load and preprocess CIFAR-10 dataset.
    
    Returns:
        Preprocessed training images
    """
    # Load CIFAR-10
    (train_images, train_labels), (_, _) = keras.datasets.cifar10.load_data()
    
    # Normalize to [-1, 1]
    train_images = train_images.astype('float32')
    train_images = (train_images - 127.5) / 127.5
    
    print(f"Training images shape: {train_images.shape}")
    print(f"Value range: [{train_images.min():.2f}, {train_images.max():.2f}]")
    
    # Class names
    class_names = ['airplane', 'automobile', 'bird', 'cat', 'deer',
                   'dog', 'frog', 'horse', 'ship', 'truck']
    
    return train_images, train_labels, class_names

# Load data
train_images, train_labels, class_names = load_cifar10_data()

# Visualize sample images
plt.figure(figsize=(12, 6))
for i in range(20):
    plt.subplot(4, 5, i+1)
    plt.imshow((train_images[i] + 1) / 2)  # Denormalize for display
    plt.title(class_names[train_labels[i][0]])
    plt.axis('off')
plt.suptitle('Sample CIFAR-10 Images')
plt.tight_layout()
plt.savefig('dcgan_cifar_output/real_cifar_samples.png', dpi=150)
plt.show()

# =============================================================================
# Part 2: Build CIFAR-10 Generator
# =============================================================================

def build_cifar_generator(latent_dim=100):
    """
    Build generator for CIFAR-10 (32x32x3 images).
    
    Architecture:
    latent_dim -> 4x4x512 -> 8x8x256 -> 16x16x128 -> 32x32x3
    """
    model = keras.Sequential(name='CIFAR_Generator')
    
    # Foundation: 4x4x512
    model.add(layers.Dense(4 * 4 * 512, input_shape=(latent_dim,)))
    model.add(layers.Reshape((4, 4, 512)))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 4x4 -> 8x8
    model.add(layers.Conv2DTranspose(
        256, (4, 4), strides=(2, 2), padding='same', use_bias=False
    ))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 8x8 -> 16x16
    model.add(layers.Conv2DTranspose(
        128, (4, 4), strides=(2, 2), padding='same', use_bias=False
    ))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 16x16 -> 32x32
    model.add(layers.Conv2DTranspose(
        64, (4, 4), strides=(2, 2), padding='same', use_bias=False
    ))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # Output: 32x32x3
    model.add(layers.Conv2D(
        3, (3, 3), padding='same', activation='tanh'
    ))
    
    return model

# Create generator
generator = build_cifar_generator(latent_dim=100)
print("\n=== CIFAR-10 Generator Architecture ===")
generator.summary()

# =============================================================================
# Part 3: Build CIFAR-10 Discriminator
# =============================================================================

def build_cifar_discriminator():
    """
    Build discriminator for CIFAR-10.
    
    Architecture:
    32x32x3 -> 16x16x64 -> 8x8x128 -> 4x4x256 -> 1
    """
    model = keras.Sequential(name='CIFAR_Discriminator')
    
    # 32x32 -> 16x16
    model.add(layers.Conv2D(
        64, (3, 3), strides=(2, 2), padding='same', input_shape=(32, 32, 3)
    ))
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # 16x16 -> 8x8
    model.add(layers.Conv2D(
        128, (3, 3), strides=(2, 2), padding='same', use_bias=False
    ))
    model.add(layers.BatchNormalization())
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # 8x8 -> 4x4
    model.add(layers.Conv2D(
        256, (3, 3), strides=(2, 2), padding='same', use_bias=False
    ))
    model.add(layers.BatchNormalization())
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # 4x4 -> 2x2
    model.add(layers.Conv2D(
        512, (3, 3), strides=(2, 2), padding='same', use_bias=False
    ))
    model.add(layers.BatchNormalization())
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # Output
    model.add(layers.Flatten())
    model.add(layers.Dense(1, activation='sigmoid'))
    
    return model

# Create discriminator
discriminator = build_cifar_discriminator()
print("\n=== CIFAR-10 Discriminator Architecture ===")
discriminator.summary()

# =============================================================================
# Part 4: Training Configuration
# =============================================================================

# Loss and optimizers
cross_entropy = keras.losses.BinaryCrossentropy()

def discriminator_loss(real_output, fake_output):
    real_loss = cross_entropy(tf.ones_like(real_output) * 0.9, real_output)
    fake_loss = cross_entropy(tf.zeros_like(fake_output), fake_output)
    return real_loss + fake_loss

def generator_loss(fake_output):
    return cross_entropy(tf.ones_like(fake_output), fake_output)

# Use slightly different learning rates for stability
generator_optimizer = keras.optimizers.Adam(0.0002, beta_1=0.5)
discriminator_optimizer = keras.optimizers.Adam(0.0002, beta_1=0.5)

# =============================================================================
# Part 5: Training Loop
# =============================================================================

@tf.function
def train_step_cifar(images, latent_dim):
    """Training step for CIFAR-10 DCGAN."""
    batch_size = tf.shape(images)[0]
    noise = tf.random.normal([batch_size, latent_dim])
    
    # Train discriminator
    with tf.GradientTape() as disc_tape:
        generated_images = generator(noise, training=True)
        real_output = discriminator(images, training=True)
        fake_output = discriminator(generated_images, training=True)
        disc_loss = discriminator_loss(real_output, fake_output)
    
    disc_grads = disc_tape.gradient(disc_loss, discriminator.trainable_variables)
    discriminator_optimizer.apply_gradients(
        zip(disc_grads, discriminator.trainable_variables)
    )
    
    # Train generator
    noise = tf.random.normal([batch_size, latent_dim])
    with tf.GradientTape() as gen_tape:
        generated_images = generator(noise, training=True)
        fake_output = discriminator(generated_images, training=True)
        gen_loss = generator_loss(fake_output)
    
    gen_grads = gen_tape.gradient(gen_loss, generator.trainable_variables)
    generator_optimizer.apply_gradients(
        zip(gen_grads, generator.trainable_variables)
    )
    
    return gen_loss, disc_loss

def generate_and_save_cifar(model, epoch, test_input, save_dir='dcgan_cifar_output'):
    """Generate and save CIFAR-10 images."""
    predictions = model(test_input, training=False)
    
    fig = plt.figure(figsize=(10, 10))
    for i in range(25):
        plt.subplot(5, 5, i+1)
        plt.imshow((predictions[i] + 1) / 2)
        plt.axis('off')
    
    plt.suptitle(f'Generated CIFAR-10 Images at Epoch {epoch}')
    plt.tight_layout()
    plt.savefig(f'{save_dir}/cifar_epoch_{epoch:04d}.png', dpi=150)
    plt.close()

def train_cifar_dcgan(dataset, epochs=100, latent_dim=100, save_interval=10):
    """Train DCGAN on CIFAR-10."""
    seed = tf.random.normal([25, latent_dim])
    gen_losses, disc_losses = [], []
    
    print(f"\nStarting CIFAR-10 DCGAN training for {epochs} epochs...")
    print("=" * 60)
    
    for epoch in range(epochs):
        start = time.time()
        
        epoch_gen_loss, epoch_disc_loss = 0, 0
        num_batches = 0
        
        for batch in dataset:
            gen_loss, disc_loss = train_step_cifar(batch, latent_dim)
            epoch_gen_loss += gen_loss
            epoch_disc_loss += disc_loss
            num_batches += 1
        
        gen_losses.append(float(epoch_gen_loss / num_batches))
        disc_losses.append(float(epoch_disc_loss / num_batches))
        
        elapsed = time.time() - start
        print(f"Epoch {epoch+1}/{epochs} - {elapsed:.2f}s - "
              f"G: {gen_losses[-1]:.4f}, D: {disc_losses[-1]:.4f}")
        
        if (epoch + 1) % save_interval == 0 or epoch == 0:
            generate_and_save_cifar(generator, epoch + 1, seed)
    
    generate_and_save_cifar(generator, epochs, seed)
    print("\nTraining completed!")
    
    return gen_losses, disc_losses

# =============================================================================
# Part 6: Train Model
# =============================================================================

LATENT_DIM = 100
BATCH_SIZE = 128
EPOCHS = 100

# Prepare dataset
train_dataset = tf.data.Dataset.from_tensor_slices(train_images)
train_dataset = train_dataset.shuffle(50000)
train_dataset = train_dataset.batch(BATCH_SIZE, drop_remainder=True)

# Train
gen_losses, disc_losses = train_cifar_dcgan(
    train_dataset, epochs=EPOCHS, latent_dim=LATENT_DIM, save_interval=10
)

# =============================================================================
# Part 7: Visualize and Save
# =============================================================================

# Plot losses
plt.figure(figsize=(10, 5))
plt.plot(gen_losses, label='Generator', color='red')
plt.plot(disc_losses, label='Discriminator', color='blue')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.title('CIFAR-10 DCGAN Training Losses')
plt.legend()
plt.grid(True, alpha=0.3)
plt.savefig('dcgan_cifar_output/training_losses.png', dpi=150)
plt.show()

# Generate final samples
noise = tf.random.normal([64, LATENT_DIM])
generated = generator(noise, training=False)

fig, axes = plt.subplots(8, 8, figsize=(12, 12))
for i, ax in enumerate(axes.flat):
    ax.imshow((generated[i] + 1) / 2)
    ax.axis('off')

print(f"Final Discriminator Loss: {disc_losses[-1]:.4f}")
```

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|---------------|-----|
| Generated images are correct grey scale but have wrong colour | Generator outputs shape `(N,32,32,1)` because last `Conv2DTranspose` uses `filters=1` instead of `filters=3` | Verify the last transposed-conv has `filters=3` and matches 3-channel RGB; add `assert generated.shape[-1] == 3` |
| Training diverges (G loss spikes to >5 after epoch 5) | CIFAR-10 has more texture variance than MNIST; larger feature maps require more stable gradient flow | Add spectral normalization to D (preview of lab-13) or reduce D LR to 1e-4 while keeping G at 2e-4 |
| Generated images lose colour saturation (all washed out) | Data normalised to [0, 1] but generator uses `tanh` output in [-1, 1]; discriminator sees range mismatch | Normalise CIFAR images to [-1, 1]: `(x / 127.5) - 1.0`; denormalise when saving: `(img + 1) / 2` |
| Mode collapse to one dominant class (e.g., all horses) | Discriminator overpowers on the most distinguishable class; generator finds a single mode that fools D | Apply label smoothing (0.9 for real), use unequal G:D update ratio (2 G steps per D step), or switch to WGAN-GP |
| Checkerboard artefacts in generated images | `Conv2DTranspose` with stride=2 produces overlap artefacts; common in DCGAN on RGB | Replace `Conv2DTranspose(stride=2)` with `UpSampling2D(2)` followed by `Conv2D(same padding)` (resize-conv pattern) |
