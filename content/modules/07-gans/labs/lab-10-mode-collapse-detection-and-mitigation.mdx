# Mode Collapse Detection and Mitigation

## Objectives
- Understand mode collapse in GANs
- Implement techniques to detect mode collapse
- Apply mitigation strategies
- Compare different approaches

## Theory Overview

Mode collapse occurs when the generator produces limited variety, ignoring parts of the true data distribution.

Detection methods:
1. Visual inspection of samples
2. Cluster analysis of generated data
3. Inception Score analysis
4. FID score monitoring

Mitigation strategies:
1. Mini-batch discrimination
2. Unrolled GANs
3. Wasserstein GAN
4. Feature matching

def create_multimodal_data(n_samples=10000, n_modes=8):
    """
    Create a dataset with multiple distinct modes.
    This makes mode collapse easy to detect.
    """
    samples_per_mode = n_samples // n_modes
    data = []
    labels = []
    
    for mode in range(n_modes):
        angle = 2 * np.pi * mode / n_modes
        center_x = 3 * np.cos(angle)
        center_y = 3 * np.sin(angle)
        
        # Generate cluster
        cluster = np.random.randn(samples_per_mode, 2) * 0.3
        cluster[:, 0] += center_x
        cluster[:, 1] += center_y
        
        data.append(cluster)
        labels.extend([mode] * samples_per_mode)
    
    return np.vstack(data).astype('float32'), np.array(labels)

def build_simple_generator(latent_dim=2):
    """Simple generator."""
    return keras.Sequential([
        layers.Dense(128, input_shape=(latent_dim,)),
        layers.ReLU(),
        layers.Dense(256),
        layers.ReLU(),
        layers.Dense(2, activation='tanh')
    ], name='Simple_Generator')

def build_simple_discriminator():
    """Simple discriminator."""
    return keras.Sequential([
        layers.Dense(256, input_shape=(2,)),
        layers.LeakyReLU(0.2),
        layers.Dense(128),
        layers.LeakyReLU(0.2),
        layers.Dense(1, activation='sigmoid')
    ], name='Simple_Discriminator')

def detect_mode_collapsed_samples(generated_data, n_modes=8, threshold=0.5):
    """
    Detect mode collapse using clustering.
    
    Args:
        generated_data: Generated samples
        n_modes: Expected number of modes
        threshold: Minimum samples per mode to consider it "covered"
    
    Returns:
        coverage_ratio: Ratio of modes covered
        cluster_sizes: Number of samples in each cluster
    """
    # Cluster generated data
    kmeans = KMeans(n_clusters=n_modes, random_state=42, n_init=10)
    cluster_labels = kmeans.fit_predict(generated_data)
    
    # Count samples per cluster
    cluster_sizes = np.bincount(cluster_labels, minlength=n_modes)
    
    # Calculate coverage (clusters with sufficient samples)
    expected_per_mode = len(generated_data) / n_modes
    covered_modes = np.sum(cluster_sizes > threshold * expected_per_mode)
    coverage_ratio = covered_modes / n_modes
    
    return coverage_ratio, cluster_sizes, kmeans

def calculate_mode_coverage_score(real_data, generated_data, n_modes=8):
    """
    Calculate how well the generated data covers the real data modes.
    """
    # Cluster real data to find true modes
    kmeans_real = KMeans(n_clusters=n_modes, random_state=42, n_init=10)
    real_clusters = kmeans_real.fit_predict(real_data)
    
    # Assign generated data to real clusters
    generated_clusters = kmeans_real.predict(generated_data)
    
    # Count coverage
    real_mode_counts = np.bincount(real_clusters, minlength=n_modes)
    gen_mode_counts = np.bincount(generated_clusters, minlength=n_modes)
    
    # Calculate coverage ratio for each mode
    coverage = gen_mode_counts / (real_mode_counts + 1e-8)
    avg_coverage = np.mean(coverage)
    
    return avg_coverage, coverage, gen_mode_counts

class StandardGAN(keras.Model):
    """Standard GAN (prone to mode collapse)."""
    
    def __init__(self, latent_dim=2):
        super().__init__()
        self.latent_dim = latent_dim
        self.generator = build_simple_generator(latent_dim)
        self.discriminator = build_simple_discriminator()
        
        self.g_opt = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.d_opt = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.cross_entropy = keras.losses.BinaryCrossentropy()
    
    @tf.function
    def train_step(self, real_data):
        batch_size = tf.shape(real_data)[0]
        noise = tf.random.normal([batch_size, self.latent_dim])
        
        # Train discriminator
        with tf.GradientTape() as tape_d:
            fake_data = self.generator(noise, training=True)
            real_pred = self.discriminator(real_data, training=True)
            fake_pred = self.discriminator(fake_data, training=True)
            
            d_loss = (self.cross_entropy(tf.ones_like(real_pred), real_pred) +
                     self.cross_entropy(tf.zeros_like(fake_pred), fake_pred))
        
        grads_d = tape_d.gradient(d_loss, self.discriminator.trainable_variables)
        self.d_opt.apply_gradients(zip(grads_d, self.discriminator.trainable_variables))
        
        # Train generator
        noise = tf.random.normal([batch_size, self.latent_dim])
        with tf.GradientTape() as tape_g:
            fake_data = self.generator(noise, training=True)
            fake_pred = self.discriminator(fake_data, training=True)
            g_loss = self.cross_entropy(tf.ones_like(fake_pred), fake_pred)
        
        grads_g = tape_g.gradient(g_loss, self.generator.trainable_variables)
        self.g_opt.apply_gradients(zip(grads_g, self.generator.trainable_variables))
        
        return {'g_loss': g_loss, 'd_loss': d_loss}

class MinibatchDiscrimination(layers.Layer):
    """
    Mini-batch discrimination layer to encourage diversity.
    """
    
    def __init__(self, num_kernels=5, kernel_dim=3):
        super().__init__()
        self.num_kernels = num_kernels
        self.kernel_dim = kernel_dim
    
    def build(self, input_shape):
        self.T = self.add_weight(
            shape=(input_shape[-1], self.num_kernels * self.kernel_dim),
            initializer='glorot_uniform',
            trainable=True,
            name='T'
        )
    
    def call(self, x):
        # x: (batch_size, features)
        batch_size = tf.shape(x)[0]
        
        # Compute M = x * T (batch_size, num_kernels * kernel_dim)
        M = tf.matmul(x, self.T)
        M = tf.reshape(M, [batch_size, self.num_kernels, self.kernel_dim])
        
        # Compute L1 distances between samples
        M_i = tf.expand_dims(M, 0)  # (1, batch, kernels, dim)
        M_j = tf.expand_dims(M, 1)  # (batch, 1, kernels, dim)
        
        # L1 distance
        dist = tf.reduce_sum(tf.abs(M_i - M_j), axis=-1)  # (batch, batch, kernels)
        
        # Apply negative exponential
        dist = tf.exp(-dist)
        
        # Sum over samples (excluding self)
        dist = tf.reduce_sum(dist, axis=1) - 1  # Subtract self-distance
        
        # Concatenate with original features
        return tf.concat([x, dist], axis=-1)

def build_mb_discriminator():
    """Build discriminator with mini-batch discrimination."""
    inputs = keras.Input(shape=(2,))
    
    x = layers.Dense(256)(inputs)
    x = layers.LeakyReLU(0.2)(x)
    
    x = layers.Dense(128)(x)
    x = layers.LeakyReLU(0.2)(x)
    
    # Mini-batch discrimination
    x = MinibatchDiscrimination(num_kernels=5, kernel_dim=3)(x)
    
    x = layers.Dense(64)(x)
    x = layers.LeakyReLU(0.2)(x)
    
    outputs = layers.Dense(1, activation='sigmoid')(x)
    
    return keras.Model(inputs, outputs)

class MBGAN(keras.Model):
    """GAN with Mini-Batch Discrimination."""
    
    def __init__(self, latent_dim=2):
        super().__init__()
        self.latent_dim = latent_dim
        self.generator = build_simple_generator(latent_dim)
        self.discriminator = build_mb_discriminator()
        
        self.g_opt = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.d_opt = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.cross_entropy = keras.losses.BinaryCrossentropy()
    
    @tf.function
    def train_step(self, real_data):
        batch_size = tf.shape(real_data)[0]
        noise = tf.random.normal([batch_size, self.latent_dim])
        
        # Train discriminator
        with tf.GradientTape() as tape_d:
            fake_data = self.generator(noise, training=True)
            real_pred = self.discriminator(real_data, training=True)
            fake_pred = self.discriminator(fake_data, training=True)
            
            d_loss = (self.cross_entropy(tf.ones_like(real_pred), real_pred) +
                     self.cross_entropy(tf.zeros_like(fake_pred), fake_pred))
        
        grads_d = tape_d.gradient(d_loss, self.discriminator.trainable_variables)
        self.d_opt.apply_gradients(zip(grads_d, self.discriminator.trainable_variables))
        
        # Train generator
        noise = tf.random.normal([batch_size, self.latent_dim])
        with tf.GradientTape() as tape_g:
            fake_data = self.generator(noise, training=True)
            fake_pred = self.discriminator(fake_data, training=True)
            g_loss = self.cross_entropy(tf.ones_like(fake_pred), fake_pred)
        
        grads_g = tape_g.gradient(g_loss, self.generator.trainable_variables)
        self.g_opt.apply_gradients(zip(grads_g, self.generator.trainable_variables))
        
        return {'g_loss': g_loss, 'd_loss': d_loss}
