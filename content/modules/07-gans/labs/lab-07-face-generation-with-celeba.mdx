# Face Generation with CelebA

## Objectives
- Generate realistic human faces using DCGAN

## Complete Implementation

"""
Lab 7.6: Face Generation with CelebA
======================================
DCGAN for generating realistic human faces.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from tensorflow import keras
from tensorflow.keras import layers
import time
import os
import glob
from PIL import Image

os.makedirs('celeb_gan_output', exist_ok=True)

# =============================================================================
# Part 1: Data Loading and Preprocessing
# =============================================================================

def load_celeba_data(img_dir, img_size=64, max_images=None):
    """
    Load and preprocess CelebA images.
    
    Args:
        img_dir: Directory containing CelebA images
        img_size: Target image size (64 or 128)
        max_images: Maximum number of images to load (for testing)
    
    Returns:
        Preprocessed image dataset
    """
    # Get image paths
    img_paths = glob.glob(os.path.join(img_dir, '*.jpg'))
    
    if max_images:
        img_paths = img_paths[:max_images]
    
    print(f"Found {len(img_paths)} images")
    
    def preprocess_image(img_path):
        """Load and preprocess a single image."""
        img = tf.io.read_file(img_path)
        img = tf.image.decode_jpeg(img, channels=3)
        img = tf.image.resize(img, [img_size, img_size])
        img = (tf.cast(img, tf.float32) - 127.5) / 127.5  # Normalize to [-1, 1]
        return img
    
    # Create dataset
    dataset = tf.data.Dataset.from_tensor_slices(img_paths)
    dataset = dataset.map(preprocess_image, num_parallel_calls=tf.data.AUTOTUNE)
    
    return dataset

# =============================================================================
# Part 2: CelebA Generator (64x64)
# =============================================================================

def build_celeb_generator_64(latent_dim=100):
    """
    Build generator for 64x64 face images.
    
    Architecture: latent_dim -> 4x4x1024 -> 8x8x512 -> 16x16x256 -> 32x32x128 -> 64x64x3
    """
    model = keras.Sequential(name='CelebA_Generator_64')
    
    # Foundation: 4x4x1024
    model.add(layers.Dense(4 * 4 * 1024, input_shape=(latent_dim,)))
    model.add(layers.Reshape((4, 4, 1024)))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 4x4 -> 8x8
    model.add(layers.Conv2DTranspose(512, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 8x8 -> 16x16
    model.add(layers.Conv2DTranspose(256, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 16x16 -> 32x32
    model.add(layers.Conv2DTranspose(128, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 32x32 -> 64x64
    model.add(layers.Conv2DTranspose(64, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # Output: 64x64x3
    model.add(layers.Conv2D(3, 3, padding='same', activation='tanh'))
    
    return model

# =============================================================================
# Part 3: CelebA Discriminator (64x64)
# =============================================================================

def build_celeb_discriminator_64():
    """
    Build discriminator for 64x64 face images.
    
    Architecture: 64x64x3 -> 32x32x64 -> 16x16x128 -> 8x8x256 -> 4x4x512 -> 1
    """
    model = keras.Sequential(name='CelebA_Discriminator_64')
    
    # 64x64 -> 32x32
    model.add(layers.Conv2D(64, 4, strides=2, padding='same', input_shape=(64, 64, 3)))
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # 32x32 -> 16x16
    model.add(layers.Conv2D(128, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # 16x16 -> 8x8
    model.add(layers.Conv2D(256, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # 8x8 -> 4x4
    model.add(layers.Conv2D(512, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.LeakyReLU(0.2))
    model.add(layers.Dropout(0.25))
    
    # Output
    model.add(layers.Flatten())
    model.add(layers.Dense(1, activation='sigmoid'))
    
    return model

# =============================================================================
# Part 4: CelebA Generator (128x128) - Higher Resolution
# =============================================================================

def build_celeb_generator_128(latent_dim=100):
    """
    Build generator for 128x128 face images.
    Adds one more upsampling layer.
    """
    model = keras.Sequential(name='CelebA_Generator_128')
    
    # Foundation: 4x4x2048
    model.add(layers.Dense(4 * 4 * 2048, input_shape=(latent_dim,)))
    model.add(layers.Reshape((4, 4, 2048)))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 4x4 -> 8x8
    model.add(layers.Conv2DTranspose(1024, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 8x8 -> 16x16
    model.add(layers.Conv2DTranspose(512, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 16x16 -> 32x32
    model.add(layers.Conv2DTranspose(256, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 32x32 -> 64x64
    model.add(layers.Conv2DTranspose(128, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # 64x64 -> 128x128
    model.add(layers.Conv2DTranspose(64, 4, strides=2, padding='same', use_bias=False))
    model.add(layers.BatchNormalization())
    model.add(layers.ReLU())
    
    # Output: 128x128x3
    model.add(layers.Conv2D(3, 3, padding='same', activation='tanh'))
    
    return model

# =============================================================================
# Part 5: Training Class
# =============================================================================

class CelebADCGAN(keras.Model):
    """DCGAN for CelebA face generation."""
    
    def __init__(self, latent_dim=100, img_size=64):
        super().__init__()
        self.latent_dim = latent_dim
        self.img_size = img_size
        
        if img_size == 64:
            self.generator = build_celeb_generator_64(latent_dim)
            self.discriminator = build_celeb_discriminator_64()
        else:
            self.generator = build_celeb_generator_128(latent_dim)
            # Build corresponding discriminator for 128x128
            
        self.g_optimizer = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.d_optimizer = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.cross_entropy = keras.losses.BinaryCrossentropy()
    
    def compile(self):
        super().compile()
    
    @tf.function
    def train_step(self, real_images):
        batch_size = tf.shape(real_images)[0]
        noise = tf.random.normal([batch_size, self.latent_dim])
        
        # Train discriminator
        with tf.GradientTape() as tape_d:
            fake_images = self.generator(noise, training=True)
            real_pred = self.discriminator(real_images, training=True)
            fake_pred = self.discriminator(fake_images, training=True)
            
            d_loss_real = self.cross_entropy(tf.ones_like(real_pred) * 0.9, real_pred)
            d_loss_fake = self.cross_entropy(tf.zeros_like(fake_pred), fake_pred)
            d_loss = d_loss_real + d_loss_fake
        
        grads_d = tape_d.gradient(d_loss, self.discriminator.trainable_variables)
        self.d_optimizer.apply_gradients(zip(grads_d, self.discriminator.trainable_variables))
        
        # Train generator
        noise = tf.random.normal([batch_size, self.latent_dim])
        with tf.GradientTape() as tape_g:
            fake_images = self.generator(noise, training=True)
            fake_pred = self.discriminator(fake_images, training=True)
            g_loss = self.cross_entropy(tf.ones_like(fake_pred), fake_pred)
        
        grads_g = tape_g.gradient(g_loss, self.generator.trainable_variables)
        self.g_optimizer.apply_gradients(zip(grads_g, self.generator.trainable_variables))
        
        return {'d_loss': d_loss, 'g_loss': g_loss}

# =============================================================================
# Part 6: Training and Visualization
# =============================================================================

def train_celeba_gan(img_dir=None, epochs=50, batch_size=64, img_size=64, 
                     latent_dim=100, max_images=10000):
    """
    Train CelebA DCGAN.
    
    Args:
        img_dir: Directory with CelebA images (if None, uses synthetic data)
        epochs: Number of training epochs
        batch_size: Batch size
        img_size: Image size (64 or 128)
        latent_dim: Latent dimension
        max_images: Maximum images to load
    """
    # Create model
    gan = CelebADCGAN(latent_dim=latent_dim, img_size=img_size)
    gan.compile()
    
    print("\n=== CelebA DCGAN Architecture ===")
    print(f"Generator:")
    gan.generator.summary()
    print(f"\nDiscriminator:")
    gan.discriminator.summary()
    
    # Create or load dataset
    if img_dir and os.path.exists(img_dir):
        dataset = load_celeba_data(img_dir, img_size, max_images)
    else:
        # Use synthetic face-like data for demonstration
        print("\nUsing synthetic data for demonstration...")
        synthetic_images = np.random.randn(max_images, img_size, img_size, 3).astype('float32') * 0.5
        dataset = tf.data.Dataset.from_tensor_slices(synthetic_images)
    
    dataset = dataset.shuffle(10000).batch(batch_size, drop_remainder=True)
    dataset = dataset.prefetch(tf.data.AUTOTUNE)
    
    # Callbacks
    seed = tf.random.normal([16, latent_dim])
    
    class ImageCallback(keras.callbacks.Callback):
        def on_epoch_end(self, epoch, logs=None):
            if (epoch + 1) % 5 == 0:
                generated = self.model.generator(seed, training=False)
                fig, axes = plt.subplots(4, 4, figsize=(10, 10))
                for i, ax in enumerate(axes.flat):
                    ax.imshow((generated[i] + 1) / 2)
                    ax.axis('off')
                plt.suptitle(f'Generated Faces - Epoch {epoch+1}')
                plt.tight_layout()
                plt.savefig(f'celeb_gan_output/faces_epoch_{epoch+1:03d}.png', dpi=150)
                plt.close()
    
    # Train
    print(f"\nStarting training for {epochs} epochs...")
    history = gan.fit(
        dataset,
        epochs=epochs,
        callbacks=[ImageCallback()],
        verbose=1
    )
    
    # Save models
    gan.generator.save('celeb_gan_output/celeb_generator.h5')
    gan.discriminator.save('celeb_gan_output/celeb_discriminator.h5')
    
    # Plot losses
    plt.figure(figsize=(10, 5))
    plt.plot(history.history['g_loss'], label='Generator', color='red')
    plt.plot(history.history['d_loss'], label='Discriminator', color='blue')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.title('CelebA DCGAN Training Losses')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.savefig('celeb_gan_output/training_losses.png', dpi=150)
    plt.show()
    
    return gan, history

# Run training (with synthetic data for demo)
gan, history = train_celeba_gan(

def interpolate_faces(generator, n_steps=10):
    """
    Create smooth transitions between two random faces.
    """
    z1 = tf.random.normal([1, 100])
    z2 = tf.random.normal([1, 100])
    
    alphas = np.linspace(0, 1, n_steps)
    interpolated = []
    
    for alpha in alphas:
        z = (1 - alpha) * z1 + alpha * z2
        face = generator(z, training=False)
        interpolated.append(face[0])
    
    # Visualize
    fig, axes = plt.subplots(1, n_steps, figsize=(20, 2))
    for i, ax in enumerate(axes):
        ax.imshow((interpolated[i] + 1) / 2)
        ax.axis('off')
    plt.suptitle('Face Interpolation in Latent Space')
    plt.tight_layout()
    plt.savefig('celeb_gan_output/face_interpolation.png', dpi=150)
    plt.show()

interpolate_faces(gan.generator)

# Generate random faces

<Callout type="pitfall" title="Common Pitfalls">
1. **CelebA center-crop matters** — CelebA images are 178×218. If you resize directly without center-cropping to 178×178 first, faces are vertically squished, and the discriminator learns the distortion instead of face features.
2. **tanh output + wrong normalization** — The generator uses `tanh` (output in [-1, 1]), so input images must be normalized to [-1, 1] not [0, 1]. Mismatch causes the discriminator to trivially distinguish real from fake by pixel range alone.
3. **Learning rate too high for 64×64** — Default Adam LR of 1e-3 is too aggressive for face generation. Use 2e-4 with β₁=0.5 (the DCGAN recipe) or the training oscillates wildly.
4. **Batch size too small for diversity** — Batch sizes below 32 give the discriminator very noisy gradient estimates, leading to mode collapse. Use 64–128 for CelebA.
</Callout>
