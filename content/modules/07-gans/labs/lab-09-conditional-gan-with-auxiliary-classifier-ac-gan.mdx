# Conditional GAN with Auxiliary Classifier (AC-GAN)

## Objectives
- Implement AC-GAN architecture
- Add auxiliary classification task to discriminator
- Improve generation quality with multi-task learning
- Evaluate both generation and classification accuracy

## Complete Implementation

```python
"""
Lab 7.8: AC-GAN (Auxiliary Classifier GAN)
============================================
AC-GAN adds a classification task to the discriminator,
improving both generation quality and class consistency.
"""

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from tensorflow import keras
from tensorflow.keras import layers
import os

os.makedirs('acgan_output', exist_ok=True)

# =============================================================================
# Part 1: Build AC-GAN Generator
# =============================================================================

def build_acgan_generator(latent_dim, num_classes, img_shape):
    """
    Build AC-GAN generator.
    Similar to CGAN generator.
    """
    noise_input = keras.Input(shape=(latent_dim,), name='noise')
    label_input = keras.Input(shape=(1,), dtype='int32', name='label')
    
    # Embed label
    label_embedding = layers.Embedding(num_classes, latent_dim)(label_input)
    label_embedding = layers.Flatten()(label_embedding)
    
    # Merge
    merged = layers.Multiply()([noise_input, label_embedding])
    
    # Foundation
    x = layers.Dense(7 * 7 * 256)(merged)
    x = layers.Reshape((7, 7, 256))(x)
    x = layers.BatchNormalization()(x)
    x = layers.ReLU()(x)
    
    # Upsample
    x = layers.Conv2DTranspose(128, 5, strides=2, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.ReLU()(x)
    
    x = layers.Conv2DTranspose(64, 5, strides=2, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.ReLU()(x)
    
    # Output image
    img_output = layers.Conv2D(1, 5, padding='same', activation='tanh', name='image')(x)
    
    model = keras.Model([noise_input, label_input], img_output, name='ACGAN_Generator')
    return model

# =============================================================================
# Part 2: Build AC-GAN Discriminator
# =============================================================================

def build_acgan_discriminator(num_classes, img_shape):
    """
    Build AC-GAN discriminator with two outputs:
    1. Real/Fake probability (adversarial)
    2. Class prediction (auxiliary)
    """
    img_input = keras.Input(shape=img_shape, name='image')
    
    # Shared convolutional layers
    x = layers.Conv2D(64, 3, strides=2, padding='same')(img_input)
    x = layers.LeakyReLU(0.2)(x)
    x = layers.Dropout(0.25)(x)
    
    x = layers.Conv2D(128, 3, strides=2, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.LeakyReLU(0.2)(x)
    x = layers.Dropout(0.25)(x)
    
    x = layers.Conv2D(256, 3, strides=2, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.LeakyReLU(0.2)(x)
    x = layers.Dropout(0.25)(x)
    
    x = layers.Conv2D(512, 3, strides=2, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.LeakyReLU(0.2)(x)
    x = layers.Dropout(0.25)(x)
    
    # Flatten for outputs
    x = layers.Flatten()(x)
    
    # Output 1: Real/Fake (adversarial)
    validity = layers.Dense(1, activation='sigmoid', name='validity')(x)
    
    # Output 2: Class label (auxiliary)
    label = layers.Dense(num_classes, activation='softmax', name='label')(x)
    
    model = keras.Model(img_input, [validity, label], name='ACGAN_Discriminator')
    return model

# =============================================================================
# Part 3: AC-GAN Training Class
# =============================================================================

class ACGAN(keras.Model):
    """Auxiliary Classifier GAN."""
    
    def __init__(self, generator, discriminator, latent_dim, num_classes):
        super().__init__()
        self.generator = generator
        self.discriminator = discriminator
        self.latent_dim = latent_dim
        self.num_classes = num_classes
        
        self.g_optimizer = keras.optimizers.Adam(0.0002, beta_1=0.5)
        self.d_optimizer = keras.optimizers.Adam(0.0002, beta_1=0.5)
        
        self.cross_entropy = keras.losses.BinaryCrossentropy()
        self.sparse_categorical = keras.losses.SparseCategoricalCrossentropy()
    
    def compile(self):
        super().compile()
    
    @tf.function
    def train_step(self, data):
        images, labels = data
        batch_size = tf.shape(images)[0]
        
        # Create one-hot labels for generator conditioning
        noise = tf.random.normal([batch_size, self.latent_dim])
        
        # Generate fake images
        generated_images = self.generator([noise, labels], training=True)
        
        # Combined labels for discriminator
        fake_labels = tf.random.uniform([batch_size], 0, self.num_classes, dtype='int32')
        
        # ======================================================================
        # Train Discriminator
        # ======================================================================
        with tf.GradientTape() as tape_d:
            # Real images
            real_validity, real_class = self.discriminator(images, training=True)
            
            # Fake images
            fake_validity, fake_class = self.discriminator(generated_images, training=True)
            
            # Adversarial loss
            d_loss_real = self.cross_entropy(tf.ones_like(real_validity) * 0.9, real_validity)
            d_loss_fake = self.cross_entropy(tf.zeros_like(fake_validity), fake_validity)
            d_loss_adv = d_loss_real + d_loss_fake
            
            # Auxiliary classification loss (on real images)
            d_loss_aux = self.sparse_categorical(labels, real_class)
            
            # Total discriminator loss
            d_loss = d_loss_adv + d_loss_aux
        
        grads_d = tape_d.gradient(d_loss, self.discriminator.trainable_variables)
        self.d_optimizer.apply_gradients(zip(grads_d, self.discriminator.trainable_variables))
        
        # ======================================================================
        # Train Generator
        # ======================================================================
        noise = tf.random.normal([batch_size, self.latent_dim])
        
        with tf.GradientTape() as tape_g:
            generated_images = self.generator([noise, labels], training=True)
            fake_validity, fake_class = self.discriminator(generated_images, training=True)
            
            # Adversarial loss (want discriminator to think fake is real)
            g_loss_adv = self.cross_entropy(tf.ones_like(fake_validity), fake_validity)
            
            # Auxiliary classification loss (want correct class prediction)
            g_loss_aux = self.sparse_categorical(labels, fake_class)
            
            # Total generator loss
            g_loss = g_loss_adv + g_loss_aux
        
        grads_g = tape_g.gradient(g_loss, self.generator.trainable_variables)
        self.g_optimizer.apply_gradients(zip(grads_g, self.generator.trainable_variables))
        
        # Calculate accuracy
        class_pred = tf.argmax(fake_class, axis=1)
        accuracy = tf.reduce_mean(tf.cast(tf.equal(tf.cast(class_pred, 'int32'), labels), 'float32'))
        
        return {
            'd_loss': d_loss,
            'g_loss': g_loss,
            'd_loss_adv': d_loss_adv,
            'd_loss_aux': d_loss_aux,
            'class_accuracy': accuracy
        }

# =============================================================================
# Part 4: Training
# =============================================================================

# Load MNIST
(train_images, train_labels), (_, _) = keras.datasets.mnist.load_data()
train_images = train_images.reshape(-1, 28, 28, 1).astype('float32')
train_images = (train_images - 127.5) / 127.5
train_labels = train_labels.astype('int32')

# Build models
LATENT_DIM = 100
NUM_CLASSES = 10
generator = build_acgan_generator(LATENT_DIM, NUM_CLASSES, (28, 28, 1))
discriminator = build_acgan_discriminator(NUM_CLASSES, (28, 28, 1))

print("=== AC-GAN Generator ===")
generator.summary()
print("\n=== AC-GAN Discriminator ===")
discriminator.summary()

# Create AC-GAN
acgan = ACGAN(generator, discriminator, LATENT_DIM, NUM_CLASSES)
acgan.compile()

# Prepare dataset
dataset = tf.data.Dataset.from_tensor_slices((train_images, train_labels))
dataset = dataset.shuffle(60000).batch(256, drop_remainder=True)

# Callback
class ACGANCallback(keras.callbacks.Callback):
    def __init__(self):
        self.seed_noise = tf.random.normal([10, LATENT_DIM])
        self.seed_labels = tf.constant([[i] for i in range(10)], dtype='int32')
    
    def on_epoch_end(self, epoch, logs=None):
        if (epoch + 1) % 10 == 0:
            generated = self.model.generator(
                [self.seed_noise, self.seed_labels], training=False
            )
            
            # Get discriminator predictions
            validity, class_pred = self.model.discriminator(generated, training=False)
            predicted_labels = tf.argmax(class_pred, axis=1).numpy()
            
            fig, axes = plt.subplots(2, 5, figsize=(12, 5))
            for i, ax in enumerate(axes.flat):
                ax.imshow(generated[i, :, :, 0] * 0.5 + 0.5, cmap='gray')
                ax.set_title(f'Target: {i}, Pred: {predicted_labels[i]}')
                ax.axis('off')
            plt.suptitle(f'AC-GAN Generation - Epoch {epoch+1}')
            plt.tight_layout()
            plt.savefig(f'acgan_output/acgan_epoch_{epoch+1:03d}.png', dpi=150)
            plt.close()

# Train
print("\nStarting AC-GAN training...")
history = acgan.fit(
    dataset,
    epochs=50,
    callbacks=[ACGANCallback()],
    verbose=1
)

# =============================================================================
# Part 5: Evaluate Classification Accuracy
# =============================================================================

# Generate test images and check classification accuracy
test_noise = tf.random.normal([1000, LATENT_DIM])
test_labels = tf.random.uniform([1000], 0, 10, dtype='int32')
test_generated = generator([test_noise, test_labels], training=False)

_, class_predictions = discriminator(test_generated, training=False)
predicted_classes = tf.argmax(class_predictions, axis=1).numpy()

accuracy = np.mean(predicted_classes == test_labels.numpy())
print(f"\nClassification Accuracy on Generated Images: {accuracy * 100:.2f}%")

# Plot losses
plt.figure(figsize=(15, 5))

plt.subplot(1, 3, 1)
plt.plot(history.history['g_loss'], label='Generator', color='red')
plt.plot(history.history['d_loss'], label='Discriminator', color='blue')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.title('AC-GAN Training Losses')
plt.legend()
plt.grid(True, alpha=0.3)

plt.subplot(1, 3, 2)
plt.plot(history.history['d_loss_adv'], label='Adversarial', color='green')
plt.plot(history.history['d_loss_aux'], label='Auxiliary', color='orange')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.title('Discriminator Component Losses')
plt.legend()
plt.grid(True, alpha=0.3)

plt.subplot(1, 3, 3)
plt.plot(history.history['class_accuracy'], label='Class Accuracy', color='purple')
plt.xlabel('Epoch')
plt.ylabel('Accuracy')
plt.title('Classification Accuracy')
plt.legend()
plt.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('acgan_output/training_analysis.png', dpi=150)
plt.show()

# Save models
generator.save('acgan_output/acgan_generator.h5')
discriminator.save('acgan_output/acgan_discriminator.h5')

print("\n=== AC-GAN Training Complete ===")
```
