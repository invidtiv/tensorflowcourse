def download_model(model_name: str, download_dir: str = './models') -> str:
    """
    Download a pre-trained model from the TensorFlow Model Zoo.
    
    Args:
        model_name: Key from MODEL_ZOO
        download_dir: Directory to save the model
    
    Returns:
        Path to downloaded model
    """
    if model_name not in MODEL_ZOO:
        raise ValueError(f"Model {model_name} not found. Available: {list(MODEL_ZOO.keys())}")
    
    model_info = MODEL_ZOO[model_name]
    model_url = model_info['url']
    
    # Create download directory
    download_path = Path(download_dir)
    download_path.mkdir(parents=True, exist_ok=True)
    
    # Download
    filename = model_url.split('/')[-1]
    filepath = download_path / filename
    
    if not filepath.exists():
        print(f"Downloading {model_info['name']}...")
        print(f"URL: {model_url}")
        urllib.request.urlretrieve(model_url, filepath)
        print(f"Downloaded to {filepath}")
    else:
        print(f"Model already exists at {filepath}")
    
    # Extract
    extract_dir = download_path / filename.replace('.tar.gz', '')
    if not extract_dir.exists():
        print(f"Extracting to {extract_dir}...")
        with tarfile.open(filepath, 'r:gz') as tar:
            tar.extractall(download_path)
        print("Extraction complete")
    
    return str(extract_dir)

def load_tf_od_model(model_path: str):
    """
    Load a TensorFlow Object Detection model.
    
    Args:
        model_path: Path to saved model directory
    
    Returns:
        Loaded model
    """
    print(f"Loading model from {model_path}...")
    
    # Load saved model
    model = tf.saved_model.load(model_path)
    
    print("Model loaded successfully!")
    return model


def run_inference(model, image: np.ndarray) -> dict:
    """
    Run inference on a single image.
    
    Args:
        model: Loaded detection model
        image: Input image as numpy array [H, W, 3]
    
    Returns:
        Detection results dictionary
    """
    # Convert to tensor
    input_tensor = tf.convert_to_tensor(image)
    input_tensor = input_tensor[tf.newaxis, ...]
    
    # Run inference
    detections = model(input_tensor)
    
    # Convert to numpy
    results = {
        'detection_boxes': detections['detection_boxes'][0].numpy(),
        'detection_classes': detections['detection_classes'][0].numpy().astype(np.int64),
        'detection_scores': detections['detection_scores'][0].numpy(),
        'num_detections': int(detections['num_detections'][0])
    }
    
    return results

def visualize_detections(image: np.ndarray,
                         detections: dict,
                         score_threshold: float = 0.5,
                         max_boxes: int = 50):
    """
    Visualize detection results.
    
    Args:
        image: Input image
        detections: Detection results from model
        score_threshold: Minimum score to display
        max_boxes: Maximum boxes to draw
    """
    fig, ax = plt.subplots(1, 1, figsize=(14, 10))
    ax.imshow(image)
    
    height, width = image.shape[:2]
    
    boxes = detections['detection_boxes']
    classes = detections['detection_classes']
    scores = detections['detection_scores']
    
    colors = plt.cm.rainbow(np.linspace(0, 1, 91))
    
    box_count = 0
    for i in range(len(boxes)):
        if scores[i] < score_threshold or box_count >= max_boxes:
            continue
        
        box_count += 1
        
        # Convert normalized coords to pixel coords
        # boxes are in [y_min, x_min, y_max, x_max] format
        y_min, x_min, y_max, x_max = boxes[i]
        
        x_min = int(x_min * width)
        x_max = int(x_max * width)
        y_min = int(y_min * height)
        y_max = int(y_max * height)
        
        class_id = classes[i]
        score = scores[i]
        label = get_label(class_id)
        
        color = colors[class_id % len(colors)]
        
        # Draw box
        rect = patches.Rectangle(
            (x_min, y_min), x_max - x_min, y_max - y_min,
            linewidth=2, edgecolor=color, facecolor='none'
        )
        ax.add_patch(rect)
        
        # Draw label
        label_text = f'{label}: {score:.2f}'
        ax.text(x_min, y_min - 5, label_text,
               color='white', fontsize=10, fontweight='bold',
               bbox=dict(facecolor=color, alpha=0.7, edgecolor='none', pad=2))
    
    ax.set_title(f'Detections (threshold={score_threshold}): {box_count} objects')
    ax.axis('off')
    plt.tight_layout()
    plt.show()
    
    return box_count

def benchmark_model(model, image_size: tuple = (640, 640, 3), num_runs: int = 10):
    """
    Benchmark model inference speed.
    
    Args:
        model: Loaded model
        image_size: Input image size
        num_runs: Number of inference runs
    
    Returns:
        Average inference time in ms
    """
    import time
    
    # Warm-up
    dummy_input = np.random.rand(1, *image_size).astype(np.float32)
    for _ in range(3):
        _ = model(dummy_input)
    
    # Benchmark
    times = []
    for _ in range(num_runs):
        start = time.time()
        _ = model(dummy_input)
        times.append((time.time() - start) * 1000)  # Convert to ms
    
    avg_time = np.mean(times)
    std_time = np.std(times)
    
    print(f"\nBenchmark Results ({num_runs} runs):")
    print(f"  Average: {avg_time:.2f} ms")
    print(f"  Std Dev: {std_time:.2f} ms")
    print(f"  FPS: {1000/avg_time:.1f}")
    
    return avg_time

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Protobuf version conflict at import | TF OD API requires `protobuf < 4.0`; newer TensorFlow installs protobuf 4.x by default | Pin `protobuf==3.20.*` in your requirements; run `pip install protobuf==3.20.3` before installing TF OD API |
| `tf.gfile` deprecation errors | Legacy TF1 code in older TF OD API versions uses `tf.gfile` which was removed in TF2 | Use TF OD API `>= 0.1.0` (tf2 branch) or apply the `tf.io.gfile` compatibility shim |
| Model zoo checkpoint vs. SavedModel confusion | The zoo provides both frozen graphs and SavedModels; loading a `.ckpt` path with `tf.saved_model.load` fails silently | Always point `load_tf_od_model` to the `saved_model/` subdirectory, not the checkpoint root |
| Input image dtype mismatch | TF OD API models expect `uint8` tensors in [0, 255]; passing float32 in [0, 1] produces near-zero confidence scores | Cast the input: `tf.cast(image * 255, tf.uint8)` before passing to the model |
| Pipeline config `.config` file missing | Training and evaluation both require the pipeline config; forgetting to include it in the job directory causes silent config-not-found errors | Always copy `pipeline.config` alongside the checkpoint when moving models between directories |
