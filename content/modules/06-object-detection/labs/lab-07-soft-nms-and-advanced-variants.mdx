# Soft-NMS and Advanced Variants

## Objectives
- Understand limitations of standard NMS
- Implement Soft-NMS algorithm
- Implement DIoU-NMS
- Compare different NMS variants

## Theory

### Soft-NMS
Instead of completely removing overlapping boxes, Soft-NMS decays their scores:
```
s_i = s_i * f(IoU(M, b_i))
```
where f can be linear or Gaussian decay.

### DIoU-NMS
Incorporates center point distance in the suppression decision:
```
s_i = s_i * (1 - IoU) * (center_distance / enclosing_diagonal)
```

def soft_nms_numpy(boxes: np.ndarray,
                   scores: np.ndarray,
                   method: str = 'gaussian',
                   sigma: float = 0.5,
                   iou_threshold: float = 0.3,
                   score_threshold: float = 0.001,
                   max_detections: int = 100) -> Tuple[np.ndarray, np.ndarray]:
    """
    Soft-NMS implementation.
    
    Args:
        boxes: Bounding boxes [N, 4]
        scores: Confidence scores [N]
        method: 'linear' or 'gaussian' decay
        sigma: Sigma for Gaussian decay
        iou_threshold: IoU threshold for linear decay
        score_threshold: Minimum score to keep
        max_detections: Maximum detections
    
    Returns:
        kept_indices, updated_scores
    """
    N = len(boxes)
    indices = np.arange(N)
    
    # Sort by scores
    sorted_indices = np.argsort(scores)[::-1]
    boxes = boxes[sorted_indices]
    scores = scores[sorted_indices].copy()
    indices = indices[sorted_indices]
    
    keep = []
    
    while len(boxes) > 0 and len(keep) < max_detections:
        # Select box with highest score
        current_box = boxes[0]
        current_score = scores[0]
        current_idx = indices[0]
        
        if current_score < score_threshold:
            break
        
        keep.append(current_idx)
        
        if len(boxes) == 1:
            break
        
        # Compute IoU with remaining boxes
        remaining_boxes = boxes[1:]
        
        x1 = np.maximum(current_box[0], remaining_boxes[:, 0])
        y1 = np.maximum(current_box[1], remaining_boxes[:, 1])
        x2 = np.minimum(current_box[2], remaining_boxes[:, 2])
        y2 = np.minimum(current_box[3], remaining_boxes[:, 3])
        
        intersect_w = np.maximum(0, x2 - x1)
        intersect_h = np.maximum(0, y2 - y1)
        intersect_area = intersect_w * intersect_h
        
        current_area = (current_box[2] - current_box[0]) * (current_box[3] - current_box[1])
        remaining_areas = (remaining_boxes[:, 2] - remaining_boxes[:, 0]) * \
                         (remaining_boxes[:, 3] - remaining_boxes[:, 1])
        
        union_area = current_area + remaining_areas - intersect_area
        iou = intersect_area / (union_area + 1e-7)
        
        # Apply decay
        if method == 'linear':
            # Linear decay
            decay = np.where(iou > iou_threshold, 1 - iou, 1.0)
        else:  # gaussian
            # Gaussian decay
            decay = np.exp(-(iou * iou) / sigma)
        
        scores[1:] = scores[1:] * decay
        
        # Remove processed box
        boxes = boxes[1:]
        scores = scores[1:]
        indices = indices[1:]
        
        # Re-sort by updated scores
        sorted_indices = np.argsort(scores)[::-1]
        boxes = boxes[sorted_indices]
        scores = scores[sorted_indices]
        indices = indices[sorted_indices]
    
    return np.array(keep), scores


def soft_nms_tensorflow(boxes: tf.Tensor,
                        scores: tf.Tensor,
                        method: str = 'gaussian',
                        sigma: float = 0.5,
                        iou_threshold: float = 0.3,
                        max_detections: int = 100) -> tf.Tensor:
    """
    TensorFlow implementation of Soft-NMS.
    """
    N = tf.shape(boxes)[0]
    
    # Sort by scores
    sorted_indices = tf.argsort(scores, direction='DESCENDING')
    boxes = tf.gather(boxes, sorted_indices)
    scores = tf.gather(scores, sorted_indices)
    
    keep = tf.TensorArray(tf.int32, size=0, dynamic_size=True)
    keep_count = 0
    
    def condition(i, boxes, scores, keep, keep_count):
        return tf.less(i, max_detections) & tf.greater(tf.shape(boxes)[0], 0)
    
    def body(i, boxes, scores, keep, keep_count):
        # Get current box
        current_box = boxes[0]
        current_score = scores[0]
        
        # Add to keep
        keep = keep.write(keep_count, sorted_indices[i])
        keep_count = keep_count + 1
        
        # Compute IoU with remaining
        remaining_boxes = boxes[1:]
        
        x1 = tf.maximum(current_box[0], remaining_boxes[:, 0])
        y1 = tf.maximum(current_box[1], remaining_boxes[:, 1])
        x2 = tf.minimum(current_box[2], remaining_boxes[:, 2])
        y2 = tf.minimum(current_box[3], remaining_boxes[:, 3])
        
        intersect_w = tf.maximum(0.0, x2 - x1)
        intersect_h = tf.maximum(0.0, y2 - y1)
        intersect_area = intersect_w * intersect_h
        
        current_area = (current_box[2] - current_box[0]) * (current_box[3] - current_box[1])
        remaining_areas = (remaining_boxes[:, 2] - remaining_boxes[:, 0]) * \
                         (remaining_boxes[:, 3] - remaining_boxes[:, 1])
        
        union_area = current_area + remaining_areas - intersect_area
        iou = intersect_area / (union_area + 1e-7)
        
        # Apply decay
        if method == 'linear':
            decay = tf.where(iou > iou_threshold, 1.0 - iou, tf.ones_like(iou))
        else:
            decay = tf.exp(-tf.square(iou) / sigma)
        
        new_scores = scores[1:] * decay
        
        # Filter low scores
        valid_mask = new_scores > 0.001
        new_boxes = tf.boolean_mask(remaining_boxes, valid_mask)
        new_scores = tf.boolean_mask(new_scores, valid_mask)
        
        return i + 1, new_boxes, new_scores, keep, keep_count
    
    _, _, _, keep, _ = tf.while_loop(
        condition, body,
        [0, boxes, scores, keep, 0],
        shape_invariants=[
            tf.TensorShape([]),
            tf.TensorShape([None, 4]),
            tf.TensorShape([None]),
            tf.TensorShape(None),
            tf.TensorShape([])
        ]
    )
    
    return keep.stack()

# Section 2: DIoU-NMS Implementation

def diou_nms_numpy(boxes: np.ndarray,
                   scores: np.ndarray,
                   iou_threshold: float = 0.5,
                   beta: float = 0.6,
                   max_detections: int = 100) -> np.ndarray:
    """
    DIoU-NMS: Incorporates center point distance.
    
    Args:
        boxes: Bounding boxes [N, 4]
        scores: Confidence scores [N]
        iou_threshold: IoU threshold
        beta: Weight for DIoU term
        max_detections: Maximum detections
    
    Returns:
        Indices of kept boxes
    """
    indices = np.argsort(scores)[::-1]
    keep = []
    
    while len(indices) > 0 and len(keep) < max_detections:
        current = indices[0]
        keep.append(current)
        
        if len(indices) == 1:
            break
        
        current_box = boxes[current]
        remaining_boxes = boxes[indices[1:]]
        
        # Calculate IoU
        x1 = np.maximum(current_box[0], remaining_boxes[:, 0])
        y1 = np.maximum(current_box[1], remaining_boxes[:, 1])
        x2 = np.minimum(current_box[2], remaining_boxes[:, 2])
        y2 = np.minimum(current_box[3], remaining_boxes[:, 3])
        
        intersect_w = np.maximum(0, x2 - x1)
        intersect_h = np.maximum(0, y2 - y1)
        intersect_area = intersect_w * intersect_h
        
        current_area = (current_box[2] - current_box[0]) * (current_box[3] - current_box[1])
        remaining_areas = (remaining_boxes[:, 2] - remaining_boxes[:, 0]) * \
                         (remaining_boxes[:, 3] - remaining_boxes[:, 1])
        
        union_area = current_area + remaining_areas - intersect_area
        iou = intersect_area / (union_area + 1e-7)
        
        # Calculate center distance
        current_center = np.array([(current_box[0] + current_box[2]) / 2,
                                   (current_box[1] + current_box[3]) / 2])
        remaining_centers = np.stack([
            (remaining_boxes[:, 0] + remaining_boxes[:, 2]) / 2,
            (remaining_boxes[:, 1] + remaining_boxes[:, 3]) / 2
        ], axis=1)
        
        center_dist = np.sqrt(np.sum((remaining_centers - current_center) ** 2, axis=1))
        
        # Calculate diagonal of enclosing box
        c_x1 = np.minimum(current_box[0], remaining_boxes[:, 0])
        c_y1 = np.minimum(current_box[1], remaining_boxes[:, 1])
        c_x2 = np.maximum(current_box[2], remaining_boxes[:, 2])
        c_y2 = np.maximum(current_box[3], remaining_boxes[:, 3])
        
        c_diag = np.sqrt((c_x2 - c_x1) ** 2 + (c_y2 - c_y1) ** 2)
        
        # DIoU = IoU - center_dist^2 / c_diag^2
        diou = iou - center_dist ** 2 / (c_diag ** 2 + 1e-7)
        
        # Suppression criteria: use DIoU instead of IoU
        # Higher DIoU means boxes are closer (both in overlap and center)
        mask = diou < iou_threshold
        indices = indices[1:][mask]
    
    return np.array(keep)

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Using Gaussian decay sigma too large (> 0.75) | Decays scores so slowly that almost all boxes survive, flooding the output | Tune sigma on a validation split; typical range is 0.3–0.6 |
| Applying Soft-NMS per-class vs. globally | Running NMS across all classes simultaneously suppresses boxes of different classes that happen to overlap spatially | Always run NMS independently per class before combining results |
| DIoU-NMS ignores objectness score ordering | DIoU suppression must still start from the highest-scoring box; skipping score-sort causes weak detections to incorrectly survive | Sort boxes by score descending before entering the DIoU-NMS loop |
| NMS threshold too low in dense scenes | Setting `iou_threshold = 0.3` removes legitimate detections of touching objects (e.g., a row of cars) | Use 0.5 for standard COCO evaluation; lower only for isolated-object datasets |
| Score threshold removed during Soft-NMS | After score decay, some near-zero boxes may slip through if no final score filter is applied | Always apply a minimum score threshold (e.g., 0.001) after the decay loop to discard ghost detections |
