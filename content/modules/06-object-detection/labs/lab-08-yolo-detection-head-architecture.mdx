# YOLO Detection Head Architecture

## Objectives
- Understand YOLO architecture
- Implement YOLO detection head
- Create feature extraction backbone
- Implement multi-scale predictions

## Theory

YOLO (You Only Look Once) is a single-stage detector that:
1. Divides image into S×S grid
2. Each grid cell predicts B bounding boxes
3. Each box has: (x, y, w, h, confidence, class probabilities)

import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
import numpy as np
import matplotlib.pyplot as plt

# =============================================================================
# Section 1: YOLO Backbone (Feature Extractor)
# =============================================================================

def create_darknet_block(filters: int, num_blocks: int = 2) -> keras.Model:
    """
    Create a DarkNet residual block.
    
    Args:
        filters: Number of filters
        num_blocks: Number of residual blocks
    
    Returns:
        Keras Model
    """
    inputs = layers.Input(shape=(None, None, filters // 2))
    x = inputs
    
    for _ in range(num_blocks):
        residual = x
        
        # 1x1 conv
        x = layers.Conv2D(filters // 2, 1, padding='same', use_bias=False)(x)
        x = layers.BatchNormalization()(x)
        x = layers.LeakyReLU(0.1)(x)
        
        # 3x3 conv
        x = layers.Conv2D(filters, 3, padding='same', use_bias=False)(x)
        x = layers.BatchNormalization()(x)
        x = layers.LeakyReLU(0.1)(x)
        
        # Residual connection
        x = layers.Add()([x, residual])
    
    return keras.Model(inputs, x, name=f'darknet_block_{filters}')


def create_yolo_backbone(input_shape: tuple = (416, 416, 3),
                          num_filters: list = [32, 64, 128, 256, 512]) -> keras.Model:
    """
    Create simplified YOLO backbone (DarkNet-style).
    
    Args:
        input_shape: Input image shape
        num_filters: List of filter sizes for each stage
    
    Returns:
        Backbone model with multiple output scales
    """
    inputs = layers.Input(shape=input_shape)
    x = inputs
    
    # Initial conv
    x = layers.Conv2D(32, 3, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.LeakyReLU(0.1)(x)
    
    outputs = []
    
    for i, filters in enumerate(num_filters):
        # Downsample
        x = layers.Conv2D(filters, 3, strides=2, padding='same', use_bias=False)(x)
        x = layers.BatchNormalization()(x)
        x = layers.LeakyReLU(0.1)(x)
        
        # Residual blocks
        if i > 0:
            block = create_darknet_block(filters, num_blocks=2 if i < 3 else 4)
            x = block(x)
        
        # Save output for detection at different scales
        if i >= 2:  # Use last 3 scales
            outputs.append(x)
    
    return keras.Model(inputs, outputs, name='yolo_backbone')


# =============================================================================
# Section 2: YOLO Detection Head
# =============================================================================

def create_detection_head(input_filters: int,
                          num_anchors: int,
                          num_classes: int) -> keras.Model:
    """
    Create YOLO detection head.
    
    Args:
        input_filters: Number of input filters
        num_anchors: Number of anchor boxes per grid cell
        num_classes: Number of object classes
    
    Returns:
        Detection head model
    """
    inputs = layers.Input(shape=(None, None, input_filters))
    
    # Detection conv layers
    x = layers.Conv2D(input_filters, 3, padding='same', use_bias=False)(inputs)
    x = layers.BatchNormalization()(x)
    x = layers.LeakyReLU(0.1)(x)
    
    x = layers.Conv2D(input_filters, 3, padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.LeakyReLU(0.1)(x)
    
    # Output layer: (x, y, w, h, objectness, class_probs) * num_anchors
    # 5 + num_classes per anchor
    output_filters = num_anchors * (5 + num_classes)
    outputs = layers.Conv2D(output_filters, 1, padding='same')(x)
    
    return keras.Model(inputs, outputs, name=f'detection_head_{input_filters}')


class YOLOHead(layers.Layer):
    """
    YOLO detection head that processes feature maps and outputs predictions.
    """
    
    def __init__(self, num_classes: int, num_anchors: int = 3, **kwargs):
        super().__init__(**kwargs)
        self.num_classes = num_classes
        self.num_anchors = num_anchors
        self.num_outputs = 5 + num_classes  # x, y, w, h, conf, classes
    
    def build(self, input_shape):
        filters = input_shape[-1]
        
        self.conv1 = layers.Conv2D(filters, 3, padding='same', use_bias=False)
        self.bn1 = layers.BatchNormalization()
        self.conv2 = layers.Conv2D(filters, 3, padding='same', use_bias=False)
        self.bn2 = layers.BatchNormalization()
        self.output_conv = layers.Conv2D(
            self.num_anchors * self.num_outputs, 1, padding='same'
        )
    
    def call(self, x, training=None):
        x = self.conv1(x)
        x = self.bn1(x, training=training)
        x = tf.nn.leaky_relu(x, 0.1)
        
        x = self.conv2(x)
        x = self.bn2(x, training=training)
        x = tf.nn.leaky_relu(x, 0.1)
        
        x = self.output_conv(x)
        return x

# Section 3: Complete YOLO Model
# =============================================================================

class YOLODetector(keras.Model):
    """
    Complete YOLO detector model.
    """
    
    def __init__(self,
                 input_shape: tuple = (416, 416, 3),
                 num_classes: int = 80,
                 num_anchors: int = 3,
                 strides: list = [8, 16, 32],
                 **kwargs):
        super().__init__(**kwargs)
        
        self.input_shape = input_shape
        self.num_classes = num_classes
        self.num_anchors = num_anchors
        self.strides = strides
        self.num_outputs = 5 + num_classes
        
        # Backbone
        self.backbone = self._build_backbone()
        
        # Detection heads for each scale
        self.heads = []
        for i, stride in enumerate(strides):
            filters = 256 if i == 0 else 512 if i == 1 else 1024
            self.heads.append(YOLOHead(num_classes, num_anchors, name=f'head_p{i+3}'))
    
    def _build_backbone(self):
        """Build feature extraction backbone."""
        inputs = layers.Input(shape=self.input_shape)
        x = inputs
        
        # Simple backbone for demonstration
        filters_list = [32, 64, 128, 256, 512]
        outputs = []
        
        for i, f in enumerate(filters_list):
            x = layers.Conv2D(f, 3, strides=2 if i > 0 else 1, 
                             padding='same', use_bias=False)(x)
            x = layers.BatchNormalization()(x)
            x = layers.LeakyReLU(0.1)(x)
            
            if i >= 2:  # P3, P4, P5
                outputs.append(x)
        
        return keras.Model(inputs, outputs, name='backbone')
    
    def call(self, inputs, training=None):
        # Extract features at multiple scales
        features = self.backbone(inputs, training=training)
        
        # Apply detection heads
        outputs = []
        for feature, head in zip(features, self.heads):
            pred = head(feature, training=training)
            outputs.append(pred)
        
        return outputs
    
    def decode_predictions(self, predictions: list, anchors: list) -> tf.Tensor:
        """
        Decode raw predictions to bounding boxes.
        
        Args:
            predictions: List of prediction tensors from each scale
            anchors: List of anchor boxes for each scale
        
        Returns:
            Decoded predictions [batch, num_predictions, 5 + num_classes]
        """
        all_predictions = []
        
        for pred, stride, anchor in zip(predictions, self.strides, anchors):
            batch_size = tf.shape(pred)[0]
            grid_h = tf.shape(pred)[1]
            grid_w = tf.shape(pred)[2]
            
            # Reshape: [batch, H, W, num_anchors * outputs] -> [batch, H, W, num_anchors, outputs]
            pred = tf.reshape(pred, [batch_size, grid_h, grid_w, 
                                     self.num_anchors, self.num_outputs])
            
            # Create grid
            grid_x = tf.range(grid_w, dtype=tf.float32)
            grid_y = tf.range(grid_h, dtype=tf.float32)
            grid_x, grid_y = tf.meshgrid(grid_x, grid_y)
            grid = tf.stack([grid_x, grid_y], axis=-1)  # [H, W, 2]
            grid = tf.expand_dims(grid, 2)  # [H, W, 1, 2]
            
            # Split predictions
            xy_pred = pred[..., 0:2]  # Center offsets
            wh_pred = pred[..., 2:4]  # Size predictions
            conf_pred = pred[..., 4:5]  # Objectness
            class_pred = pred[..., 5:]  # Class probabilities
            
            # Apply sigmoid to xy and conf
            xy_pred = tf.sigmoid(xy_pred)
            conf_pred = tf.sigmoid(conf_pred)
            class_pred = tf.sigmoid(class_pred)
            
            # Decode xy: grid + sigmoid(pred)
            xy_decoded = (xy_pred + grid) * stride
            
            # Decode wh: anchor * exp(pred)
            anchor_tensor = tf.constant(anchor, dtype=tf.float32)
            anchor_tensor = tf.reshape(anchor_tensor, [1, 1, 1, self.num_anchors, 2])
            wh_decoded = anchor_tensor * tf.exp(wh_pred)
            
            # Combine
            decoded = tf.concat([xy_decoded, wh_decoded, conf_pred, class_pred], axis=-1)
            
            # Reshape to [batch, H*W*num_anchors, outputs]
            decoded = tf.reshape(decoded, [batch_size, -1, self.num_outputs])
            all_predictions.append(decoded)
        
        # Concatenate all scales
        return tf.concat(all_predictions, axis=1)

def visualize_feature_maps(model: keras.Model, input_image: tf.Tensor):
    """Visualize feature maps from backbone."""
    # Get intermediate outputs
    feature_model = keras.Model(
        inputs=model.backbone.input,
        outputs=model.backbone.outputs
    )
    
    features = feature_model(input_image)
    
    fig, axes = plt.subplots(1, len(features), figsize=(15, 5))
    if len(features) == 1:
        axes = [axes]
    
    for ax, feature, i in zip(axes, features, range(len(features))):
        # Take first image and average channels
        feature_map = tf.reduce_mean(feature[0], axis=-1).numpy()
        
        ax.imshow(feature_map, cmap='viridis')
        ax.set_title(f'P{i+3} Feature Map\n{feature.shape[1:3]}')
        ax.axis('off')
    
    plt.tight_layout()
    plt.show()

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Anchor mismatch between training and inference | Anchors are hardcoded in the head but not saved with the model weights | Save anchor values in the model config (e.g., a JSON sidecar) and load them at inference time |
| Grid-cell assignment ambiguity for large objects | Objects spanning multiple grid cells are assigned to the cell containing their centre — but the centre may shift after augmentation | Apply random crop augmentation carefully; re-compute GT assignments after each crop, not before |
| Objectness loss weight imbalance | The ratio of background cells to foreground cells is ~99:1; unweighted BCE floods gradients from negatives | Apply focal loss or weight the objectness BCE by foreground/background ratio (typical: λ_noobj = 0.5) |
| Sigmoid vs. softmax for multi-label classes | Using softmax forces mutually exclusive class predictions; an object can only belong to one class | Use independent sigmoid per class for multi-label detection (person + cyclist); only use softmax for single-label datasets |
| Feature upsampling artefacts in the neck | Nearest-neighbour upsampling (used in YOLOv3 neck) creates checkerboard artefacts that corrupt small-object features | Use bilinear upsampling or transposed convolutions with kernel_size=2, stride=2 to avoid the artefact |
