# YOLO Inference and Post-Processing

## Objectives
- Implement YOLO inference pipeline
- Decode predictions to bounding boxes
- Apply NMS to filter detections
- Visualize detection results

## Complete Implementation

```python
"""
Lab 6.9: YOLO Inference and Post-Processing
Object Detection Systems - Module 6
"""

import tensorflow as tf
from tensorflow import keras
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.patches as patches
from typing import Tuple, List

# =============================================================================
# Section 1: Prediction Decoder
# =============================================================================

class YOLODecoder:
    """
    Decode YOLO predictions to bounding boxes.
    """
    
    def __init__(self,
                 num_classes: int = 80,
                 num_anchors: int = 3,
                 strides: List[int] = [8, 16, 32],
                 anchors: List[np.ndarray] = None):
        self.num_classes = num_classes
        self.num_anchors = num_anchors
        self.strides = strides
        
        if anchors is None:
            # Default COCO anchors
            self.anchors = [
                np.array([[10, 13], [16, 30], [33, 23]]),
                np.array([[30, 61], [62, 45], [59, 119]]),
                np.array([[116, 90], [156, 198], [373, 326]])
            ]
        else:
            self.anchors = anchors
    
    def decode(self, predictions: List[tf.Tensor],
               conf_threshold: float = 0.25) -> tf.Tensor:
        """
        Decode raw predictions to bounding boxes.
        
        Args:
            predictions: List of tensors from each scale
            conf_threshold: Confidence threshold
        
        Returns:
            Decoded boxes [num_detections, 6] (x1, y1, x2, y2, conf, class)
        """
        all_boxes = []
        
        for pred, stride, anchors in zip(predictions, self.strides, self.anchors):
            boxes = self._decode_scale(pred, stride, anchors, conf_threshold)
            all_boxes.append(boxes)
        
        # Concatenate all scales
        all_boxes = tf.concat(all_boxes, axis=0)
        
        return all_boxes
    
    def _decode_scale(self, pred: tf.Tensor, stride: int,
                      anchors: np.ndarray,
                      conf_threshold: float) -> tf.Tensor:
        """Decode predictions for a single scale."""
        batch_size = tf.shape(pred)[0]
        grid_h = tf.shape(pred)[1]
        grid_w = tf.shape(pred)[2]
        
        # Reshape predictions
        pred = tf.reshape(pred, [batch_size, grid_h, grid_w, 
                                 self.num_anchors, 5 + self.num_classes])
        
        # Create grid
        grid_x = tf.range(grid_w, dtype=tf.float32)
        grid_y = tf.range(grid_h, dtype=tf.float32)
        grid_x, grid_y = tf.meshgrid(grid_x, grid_y)
        grid = tf.stack([grid_x, grid_y], axis=-1)
        grid = tf.expand_dims(grid, 2)
        
        # Split predictions
        xy_pred = tf.sigmoid(pred[..., 0:2])
        wh_pred = pred[..., 2:4]
        conf_pred = tf.sigmoid(pred[..., 4:5])
        cls_pred = tf.sigmoid(pred[..., 5:])
        
        # Decode coordinates
        xy_decoded = (xy_pred + grid) * stride
        
        anchors = tf.constant(anchors, dtype=tf.float32)
        anchors = tf.reshape(anchors, [1, 1, 1, self.num_anchors, 2])
        wh_decoded = tf.exp(wh_pred) * anchors
        
        # Convert to x1, y1, x2, y2
        x1 = xy_decoded[..., 0:1] - wh_decoded[..., 0:1] / 2
        y1 = xy_decoded[..., 1:2] - wh_decoded[..., 1:2] / 2
        x2 = xy_decoded[..., 0:1] + wh_decoded[..., 0:1] / 2
        y2 = xy_decoded[..., 1:2] + wh_decoded[..., 1:2] / 2
        
        # Combine
        boxes = tf.concat([x1, y1, x2, y2, conf_pred, cls_pred], axis=-1)
        
        # Reshape and filter by confidence
        boxes = tf.reshape(boxes, [-1, 5 + self.num_classes])
        
        # Get max class confidence
        cls_conf = tf.reduce_max(boxes[:, 5:], axis=1)
        cls_id = tf.argmax(boxes[:, 5:], axis=1, output_type=tf.int32)
        
        # Combined confidence
        conf = boxes[:, 4] * cls_conf
        
        # Filter
        mask = conf >= conf_threshold
        filtered_boxes = tf.boolean_mask(boxes[:, :4], mask)
        filtered_conf = tf.boolean_mask(conf, mask)
        filtered_cls = tf.boolean_mask(cls_id, mask)
        
        # Combine results
        result = tf.concat([
            filtered_boxes,
            tf.expand_dims(filtered_conf, 1),
            tf.cast(tf.expand_dims(filtered_cls, 1), tf.float32)
        ], axis=1)
        
        return result


# =============================================================================
# Section 2: NMS Post-Processing
# =============================================================================

def apply_nms(boxes: tf.Tensor,
              iou_threshold: float = 0.45,
              max_detections: int = 300) -> tf.Tensor:
    """
    Apply NMS to decoded predictions.
    
    Args:
        boxes: [N, 6] tensor (x1, y1, x2, y2, conf, class)
        iou_threshold: IoU threshold for NMS
        max_detections: Maximum detections to keep
    
    Returns:
        Filtered boxes after NMS
    """
    # Separate components
    box_coords = boxes[:, :4]
    scores = boxes[:, 4]
    classes = tf.cast(boxes[:, 5], tf.int32)
    
    # Apply class-aware NMS using offset trick
    max_class = tf.reduce_max(classes)
    offset = tf.cast(classes, tf.float32) * (max_class + 1) * 1000
    offset_boxes = box_coords + tf.expand_dims(offset, 1)
    
    # Apply NMS
    keep_indices = tf.image.non_max_suppression(
        offset_boxes, scores,
        max_output_size=max_detections,
        iou_threshold=iou_threshold
    )
    
    # Gather kept boxes
    result = tf.gather(boxes, keep_indices)
    
    return result


# =============================================================================
# Section 3: Inference Pipeline
# =============================================================================

class YOLOInference:
    """
    Complete YOLO inference pipeline.
    """
    
    def __init__(self,
                 model: keras.Model,
                 decoder: YOLODecoder,
                 conf_threshold: float = 0.25,
                 iou_threshold: float = 0.45,
                 max_detections: int = 300):
        self.model = model
        self.decoder = decoder
        self.conf_threshold = conf_threshold
        self.iou_threshold = iou_threshold
        self.max_detections = max_detections
        
        # Class names (COCO classes)
        self.class_names = [
            'person', 'bicycle', 'car', 'motorcycle', 'airplane', 'bus',
            'train', 'truck', 'boat', 'traffic light', 'fire hydrant',
            'stop sign', 'parking meter', 'bench', 'bird', 'cat', 'dog',
            'horse', 'sheep', 'cow', 'elephant', 'bear', 'zebra', 'giraffe',
            'backpack', 'umbrella', 'handbag', 'tie', 'suitcase', 'frisbee',
            'skis', 'snowboard', 'sports ball', 'kite', 'baseball bat',
            'baseball glove', 'skateboard', 'surfboard', 'tennis racket',
            'bottle', 'wine glass', 'cup', 'fork', 'knife', 'spoon', 'bowl',
            'banana', 'apple', 'sandwich', 'orange', 'broccoli', 'carrot',
            'hot dog', 'pizza', 'donut', 'cake', 'chair', 'couch',
            'potted plant', 'bed', 'dining table', 'toilet', 'tv', 'laptop',
            'mouse', 'remote', 'keyboard', 'cell phone', 'microwave', 'oven',
            'toaster', 'sink', 'refrigerator', 'book', 'clock', 'vase',
            'scissors', 'teddy bear', 'hair drier', 'toothbrush'
        ]
    
    def predict(self, image: tf.Tensor) -> np.ndarray:
        """
        Run inference on a single image.
        
        Args:
            image: Input image tensor [H, W, 3]
        
        Returns:
            Detections [N, 6] (x1, y1, x2, y2, conf, class)
        """
        # Add batch dimension
        if len(image.shape) == 3:
            image = tf.expand_dims(image, 0)
        
        # Run model
        predictions = self.model(image, training=False)
        
        # Decode predictions
        boxes = self.decoder.decode(predictions, self.conf_threshold)
        
        # Apply NMS
        if len(boxes) > 0:
            boxes = apply_nms(boxes, self.iou_threshold, self.max_detections)
        
        return boxes.numpy()
    
    def predict_batch(self, images: tf.Tensor) -> List[np.ndarray]:
        """
        Run inference on a batch of images.
        
        Args:
            images: Input images [batch, H, W, 3]
        
        Returns:
            List of detections for each image
        """
        predictions = self.model(images, training=False)
        
        results = []
        for i in range(len(images)):
            # Extract predictions for this image
            img_preds = [p[i:i+1] for p in predictions]
            
            # Decode and apply NMS
            boxes = self.decoder.decode(img_preds, self.conf_threshold)
            if len(boxes) > 0:
                boxes = apply_nms(boxes, self.iou_threshold, self.max_detections)
            
            results.append(boxes.numpy())
        
        return results


# =============================================================================
# Section 4: Visualization
# =============================================================================

def visualize_detections(image: np.ndarray,
                         detections: np.ndarray,
                         class_names: List[str],
                         score_threshold: float = 0.5):
    """
    Visualize detection results on image.
    
    Args:
        image: Input image [H, W, 3]
        detections: Detections [N, 6] (x1, y1, x2, y2, conf, class)
        class_names: List of class names
        score_threshold: Minimum score to display
    """
    fig, ax = plt.subplots(1, 1, figsize=(12, 12))
    
    # Display image
    ax.imshow(image)
    
    h, w = image.shape[:2]
    
    # Colors for different classes
    colors = plt.cm.rainbow(np.linspace(0, 1, len(class_names)))
    
    for det in detections:
        x1, y1, x2, y2, conf, cls = det
        
        if conf < score_threshold:
            continue
        
        # Convert normalized coords to pixel coords
        x1, x2 = x1 * w, x2 * w
        y1, y2 = y1 * h, y2 * h
        
        cls_id = int(cls)
        color = colors[cls_id % len(colors)]
        
        # Draw box
        rect = patches.Rectangle(
            (x1, y1), x2 - x1, y2 - y1,
            linewidth=2, edgecolor=color, facecolor='none'
        )
        ax.add_patch(rect)
        
        # Draw label
        label = f'{class_names[cls_id]}: {conf:.2f}'
        ax.text(x1, y1 - 5, label, color='white', fontsize=10,
               bbox=dict(facecolor=color, alpha=0.7, edgecolor='none'))
    
    ax.set_title(f'Detections: {len(detections)} objects')
    ax.axis('off')
    plt.tight_layout()
    plt.show()


def create_dummy_model(input_shape: Tuple[int, int, int] = (416, 416, 3),
                       num_classes: int = 80) -> keras.Model:
    """Create a dummy model for testing inference pipeline."""
    inputs = keras.Input(shape=input_shape)
    x = inputs
    
    # Simple backbone
    for f in [32, 64, 128, 256, 512]:
        x = keras.layers.Conv2D(f, 3, strides=2, padding='same', activation='relu')(x)
    
    # Multi-scale outputs
    outputs = []
    for i, (h, w) in enumerate([(52, 52), (26, 26), (13, 13)]):
        out = keras.layers.Conv2D(3 * (5 + num_classes), 1, padding='same')(x)
        outputs.append(out)
        if i < 2:
            x = keras.layers.UpSampling2D(2)(x)
    
    return keras.Model(inputs, outputs)

                  f"at ({x1:.2f}, {y1:.2f}, {x2:.2f}, {y2:.2f})")
    
    # Visualize
```
