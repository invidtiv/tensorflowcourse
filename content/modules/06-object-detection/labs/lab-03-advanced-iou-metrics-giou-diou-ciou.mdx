# Advanced IoU Metrics (GIoU, DIoU, CIoU)

## Objectives
- Understand limitations of standard IoU
- Implement Generalized IoU (GIoU)
- Implement Distance IoU (DIoU)
- Implement Complete IoU (CIoU)
- Compare different IoU metrics

## Theory

### Limitations of Standard IoU
1. IoU = 0 when boxes don't overlap (no gradient information)
2. Doesn't consider distance between non-overlapping boxes
3. Doesn't consider aspect ratio differences

### GIoU (Generalized IoU)
Addresses the zero-gradient problem by incorporating the smallest enclosing box:
```
GIoU = IoU - |C - (A ∪ B)| / |C|
```
where C is the smallest convex box enclosing both A and B.

### DIoU (Distance IoU)
Incorporates the normalized distance between box centers:
```
DIoU = IoU - ρ²(b, b^gt) / c²
```
where ρ is Euclidean distance between centers, c is diagonal of enclosing box.

### CIoU (Complete IoU)
Combines overlap, distance, and aspect ratio:
```
CIoU = IoU - ρ²/c² - αv
```
where v measures aspect ratio consistency, α is weighting parameter.

def calculate_giou(boxes1: tf.Tensor, boxes2: tf.Tensor) -> tf.Tensor:
    """
    Calculate Generalized IoU (GIoU) between two sets of boxes.
    
    GIoU = IoU - |C - Union| / |C|
    
    Args:
        boxes1: Tensor of shape [N, 4] in [x_min, y_min, x_max, y_max]
        boxes2: Tensor of shape [M, 4] in [x_min, y_min, x_max, y_max]
    
    Returns:
        GIoU matrix of shape [N, M]
    """
    # Expand dimensions for broadcasting
    boxes1 = tf.expand_dims(boxes1, 1)  # [N, 1, 4]
    boxes2 = tf.expand_dims(boxes2, 0)  # [1, M, 4]
    
    # Unpack coordinates
    x1_min, y1_min, x1_max, y1_max = tf.unstack(boxes1, axis=-1)
    x2_min, y2_min, x2_max, y2_max = tf.unstack(boxes2, axis=-1)
    
    # Calculate intersection
    intersect_xmin = tf.maximum(x1_min, x2_min)
    intersect_ymin = tf.maximum(y1_min, y2_min)
    intersect_xmax = tf.minimum(x1_max, x2_max)
    intersect_ymax = tf.minimum(y1_max, y2_max)
    
    intersect_width = tf.maximum(0.0, intersect_xmax - intersect_xmin)
    intersect_height = tf.maximum(0.0, intersect_ymax - intersect_ymin)
    intersection_area = intersect_width * intersect_height
    
    # Calculate areas
    box1_area = (x1_max - x1_min) * (y1_max - y1_min)
    box2_area = (x2_max - x2_min) * (y2_max - y2_min)
    
    # Calculate union
    union_area = box1_area + box2_area - intersection_area
    
    # Calculate IoU
    epsilon = 1e-7
    iou = intersection_area / (union_area + epsilon)
    
    # Calculate enclosing box (C)
    c_xmin = tf.minimum(x1_min, x2_min)
    c_ymin = tf.minimum(y1_min, y2_min)
    c_xmax = tf.maximum(x1_max, x2_max)
    c_ymax = tf.maximum(y1_max, y2_max)
    
    c_width = tf.maximum(0.0, c_xmax - c_xmin)
    c_height = tf.maximum(0.0, c_ymax - c_ymin)
    c_area = c_width * c_height
    
    # Calculate GIoU
    giou = iou - (c_area - union_area) / (c_area + epsilon)
    
    return giou

def calculate_diou(boxes1: tf.Tensor, boxes2: tf.Tensor) -> tf.Tensor:
    """
    Calculate Distance IoU (DIoU) between two sets of boxes.
    
    DIoU = IoU - ρ²(b, b^gt) / c²
    
    Args:
        boxes1: Tensor of shape [N, 4] in [x_min, y_min, x_max, y_max]
        boxes2: Tensor of shape [M, 4] in [x_min, y_min, x_max, y_max]
    
    Returns:
        DIoU matrix of shape [N, M]
    """
    # Expand dimensions
    boxes1 = tf.expand_dims(boxes1, 1)
    boxes2 = tf.expand_dims(boxes2, 0)
    
    # Unpack coordinates
    x1_min, y1_min, x1_max, y1_max = tf.unstack(boxes1, axis=-1)
    x2_min, y2_min, x2_max, y2_max = tf.unstack(boxes2, axis=-1)
    
    # Calculate intersection
    intersect_xmin = tf.maximum(x1_min, x2_min)
    intersect_ymin = tf.maximum(y1_min, y2_min)
    intersect_xmax = tf.minimum(x1_max, x2_max)
    intersect_ymax = tf.minimum(y1_max, y2_max)
    
    intersect_width = tf.maximum(0.0, intersect_xmax - intersect_xmin)
    intersect_height = tf.maximum(0.0, intersect_ymax - intersect_ymin)
    intersection_area = intersect_width * intersect_height
    
    # Calculate areas
    box1_area = (x1_max - x1_min) * (y1_max - y1_min)
    box2_area = (x2_max - x2_min) * (y2_max - y2_min)
    
    # Calculate union
    union_area = box1_area + box2_area - intersection_area
    
    # Calculate IoU
    epsilon = 1e-7
    iou = intersection_area / (union_area + epsilon)
    
    # Calculate center points
    center_x1 = (x1_min + x1_max) / 2
    center_y1 = (y1_min + y1_max) / 2
    center_x2 = (x2_min + x2_max) / 2
    center_y2 = (y2_min + y2_max) / 2
    
    # Calculate distance between centers
    center_distance_sq = tf.square(center_x1 - center_x2) + tf.square(center_y1 - center_y2)
    
    # Calculate diagonal of enclosing box
    c_xmin = tf.minimum(x1_min, x2_min)
    c_ymin = tf.minimum(y1_min, y2_min)
    c_xmax = tf.maximum(x1_max, x2_max)
    c_ymax = tf.maximum(y1_max, y2_max)
    
    c_diagonal_sq = tf.square(c_xmax - c_xmin) + tf.square(c_ymax - c_ymin)
    
    # Calculate DIoU
    diou = iou - center_distance_sq / (c_diagonal_sq + epsilon)
    
    return diou

def calculate_ciou(boxes1: tf.Tensor, boxes2: tf.Tensor) -> tf.Tensor:
    """
    Calculate Complete IoU (CIoU) between two sets of boxes.
    
    CIoU = IoU - ρ²/c² - αv
    where v = (4/π²) * (arctan(w_gt/h_gt) - arctan(w/h))²
    
    Args:
        boxes1: Tensor of shape [N, 4] in [x_min, y_min, x_max, y_max]
        boxes2: Tensor of shape [M, 4] in [x_min, y_min, x_max, y_max]
    
    Returns:
        CIoU matrix of shape [N, M]
    """
    # Expand dimensions
    boxes1 = tf.expand_dims(boxes1, 1)
    boxes2 = tf.expand_dims(boxes2, 0)
    
    # Unpack coordinates
    x1_min, y1_min, x1_max, y1_max = tf.unstack(boxes1, axis=-1)
    x2_min, y2_min, x2_max, y2_max = tf.unstack(boxes2, axis=-1)
    
    # Calculate intersection
    intersect_xmin = tf.maximum(x1_min, x2_min)
    intersect_ymin = tf.maximum(y1_min, y2_min)
    intersect_xmax = tf.minimum(x1_max, x2_max)
    intersect_ymax = tf.minimum(y1_max, y2_max)
    
    intersect_width = tf.maximum(0.0, intersect_xmax - intersect_xmin)
    intersect_height = tf.maximum(0.0, intersect_ymax - intersect_ymin)
    intersection_area = intersect_width * intersect_height
    
    # Calculate areas
    box1_area = (x1_max - x1_min) * (y1_max - y1_min)
    box2_area = (x2_max - x2_min) * (y2_max - y2_min)
    
    # Calculate union
    union_area = box1_area + box2_area - intersection_area
    
    # Calculate IoU
    epsilon = 1e-7
    iou = intersection_area / (union_area + epsilon)
    
    # Calculate center points
    center_x1 = (x1_min + x1_max) / 2
    center_y1 = (y1_min + y1_max) / 2
    center_x2 = (x2_min + x2_max) / 2
    center_y2 = (y2_min + y2_max) / 2
    
    # Calculate distance between centers
    center_distance_sq = tf.square(center_x1 - center_x2) + tf.square(center_y1 - center_y2)
    
    # Calculate diagonal of enclosing box
    c_xmin = tf.minimum(x1_min, x2_min)
    c_ymin = tf.minimum(y1_min, y2_min)
    c_xmax = tf.maximum(x1_max, x2_max)
    c_ymax = tf.maximum(y1_max, y2_max)
    
    c_diagonal_sq = tf.square(c_xmax - c_xmin) + tf.square(c_ymax - c_ymin)
    
    # Calculate aspect ratio consistency term (v)
    w1 = x1_max - x1_min
    h1 = y1_max - y1_min
    w2 = x2_max - x2_min
    h2 = y2_max - y2_min
    
    # Add epsilon to avoid log(0)
    v = (4 / (np.pi ** 2)) * tf.square(
        tf.atan(w2 / (h2 + epsilon)) - tf.atan(w1 / (h1 + epsilon))
    )
    
    # Calculate alpha
    alpha = v / ((1 - iou) + v + epsilon)
    
    # Calculate CIoU
    ciou = iou - center_distance_sq / (c_diagonal_sq + epsilon) - alpha * v
    
    return ciou

def giou_loss(pred_boxes: tf.Tensor, target_boxes: tf.Tensor) -> tf.Tensor:
    """
    GIoU loss for bounding box regression.
    
    Args:
        pred_boxes: Predicted boxes [N, 4]
        target_boxes: Target boxes [N, 4]
    
    Returns:
        Loss value
    """
    giou = calculate_giou(pred_boxes, target_boxes)
    # GIoU ranges from -1 to 1, loss is 1 - GIoU
    loss = 1 - tf.linalg.diag_part(giou)
    return tf.reduce_mean(loss)


def diou_loss(pred_boxes: tf.Tensor, target_boxes: tf.Tensor) -> tf.Tensor:
    """
    DIoU loss for bounding box regression.
    """
    diou = calculate_diou(pred_boxes, target_boxes)
    loss = 1 - tf.linalg.diag_part(diou)
    return tf.reduce_mean(loss)


def ciou_loss(pred_boxes: tf.Tensor, target_boxes: tf.Tensor) -> tf.Tensor:
    """
    CIoU loss for bounding box regression.
    """
    ciou = calculate_ciou(pred_boxes, target_boxes)
    loss = 1 - tf.linalg.diag_part(ciou)
    return tf.reduce_mean(loss)

## Key Insights

1. **GIoU**: Provides gradient even when boxes don't overlap (-1 to 1 range)
2. **DIoU**: Penalizes based on center distance, good for convergence
3. **CIoU**: Best overall, considers overlap, distance, and aspect ratio
