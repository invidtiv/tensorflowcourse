def download_glove(glove_dir='glove', embedding_dim=100):
    """
    Download GloVe embeddings.
    
    Args:
        glove_dir: Directory to save embeddings
        embedding_dim: GloVe dimension (50, 100, 200, 300)
        
    Returns:
        Path to downloaded file
    """
    import urllib.request
    import zipfile
    
    os.makedirs(glove_dir, exist_ok=True)
    
    glove_file = f'glove.6B.{embedding_dim}d.txt'
    glove_path = os.path.join(glove_dir, glove_file)
    
    if not os.path.exists(glove_path):
        print(f"Downloading GloVe {embedding_dim}d embeddings...")
        url = f'http://nlp.stanford.edu/data/glove.6B.zip'
        zip_path = os.path.join(glove_dir, 'glove.6B.zip')
        
        urllib.request.urlretrieve(url, zip_path)
        
        print("Extracting...")
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            zip_ref.extract(glove_file, glove_dir)
        
        os.remove(zip_path)
        print("Done!")
    else:
        print(f"GloVe embeddings already downloaded.")
    
    return glove_path


def load_glove_embeddings(glove_path):
    """
    Load GloVe embeddings from file.
    
    Args:
        glove_path: Path to GloVe file
        
    Returns:
        Dictionary of {word: embedding_vector}
    """
    embeddings_index = {}
    
    print(f"Loading GloVe embeddings from {glove_path}...")
    
    with open(glove_path, 'r', encoding='utf-8') as f:
        for line in f:
            values = line.split()
            word = values[0]
            coefs = np.asarray(values[1:], dtype='float32')
            embeddings_index[word] = coefs
    
    print(f"Loaded {len(embeddings_index)} word vectors.")
    
    return embeddings_index


# For demonstration, create synthetic GloVe-like embeddings
def create_synthetic_glove(vocab_size=10000, embedding_dim=100):
    """
    Create synthetic embeddings for demonstration.
    """
    print("Creating synthetic GloVe-like embeddings...")
    
    embeddings_index = {}
    
    # Common words with meaningful embeddings
    words = ['the', 'a', 'an', 'is', 'are', 'was', 'were', 'be', 'been',
             'good', 'bad', 'great', 'terrible', 'excellent', 'poor',
             'happy', 'sad', 'joy', 'sorrow', 'love', 'hate',
             'king', 'queen', 'man', 'woman', 'prince', 'princess',
             'big', 'small', 'large', 'tiny', 'huge', 'little',
             'fast', 'slow', 'quick', 'rapid', 'swift', 'speedy']
    
    for word in words:
        # Create embedding with some structure
        embedding = np.random.randn(embedding_dim).astype('float32')
        embeddings_index[word] = embedding
    
    # Add random words
    for i in range(vocab_size - len(words)):
        word = f'word_{i}'
        embedding = np.random.randn(embedding_dim).astype('float32')
        embeddings_index[word] = embedding
    
    print(f"Created {len(embeddings_index)} synthetic embeddings.")
    
    return embeddings_index


# ============================================================
# PART 2: Create Embedding Matrix

def create_embedding_matrix(word_index, embeddings_index, embedding_dim):
    """
    Create embedding matrix from pretrained embeddings.
    
    Args:
        word_index: {word: index} mapping from tokenizer
        embeddings_index: {word: vector} from GloVe
        embedding_dim: Embedding dimension
        
    Returns:
        Embedding matrix (vocab_size, embedding_dim)
    """
    vocab_size = len(word_index) + 1  # +1 for index 0 (padding)
    
    # Initialize with random values
    embedding_matrix = np.random.randn(vocab_size, embedding_dim).astype('float32') * 0.01
    
    # Fill in pretrained embeddings
    found = 0
    for word, idx in word_index.items():
        if word in embeddings_index:
            embedding_matrix[idx] = embeddings_index[word]
            found += 1
    
    print(f"Found embeddings for {found}/{len(word_index)} words "
          f"({100*found/len(word_index):.1f}%)")
    
    return embedding_matrix

def create_model_with_pretrained(embedding_matrix, max_length, trainable=False):
    """
    Create model with pretrained embeddings.
    
    Args:
        embedding_matrix: Pretrained embedding matrix
        max_length: Maximum sequence length
        trainable: Whether to fine-tune embeddings
        
    Returns:
        Compiled model
    """
    vocab_size, embedding_dim = embedding_matrix.shape
    
    model = tf.keras.Sequential([
        # Embedding layer with pretrained weights
        tf.keras.layers.Embedding(
            vocab_size, embedding_dim,
            weights=[embedding_matrix],
            input_length=max_length,
            trainable=trainable,
            mask_zero=True
        ),
        
        tf.keras.layers.LSTM(64, dropout=0.2),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(1, activation='sigmoid')
    ])
    
    model.compile(
        optimizer='adam',
        loss='binary_crossentropy',
        metrics=['accuracy']
    )
    
    return model


def create_model_with_random(vocab_size, embedding_dim, max_length):
    """
    Create model with random embeddings (baseline).
    """
    model = tf.keras.Sequential([
        tf.keras.layers.Embedding(
            vocab_size, embedding_dim,
            input_length=max_length,
            mask_zero=True
        ),
        tf.keras.layers.LSTM(64, dropout=0.2),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(1, activation='sigmoid')
    ])
    
    model.compile(
        optimizer='adam',
        loss='binary_crossentropy',
        metrics=['accuracy']
    )
    
    return model


# Compare models (using subset for speed)
print("\nComparing models...")

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| GloVe file line encoding error | GloVe `.txt` files can contain non-UTF-8 bytes for obscure vocabulary entries; `open(file, 'r')` without specifying encoding raises `UnicodeDecodeError` on some platforms | Always open GloVe files with `open(file, 'r', encoding='utf-8', errors='ignore')` to skip malformed lines silently |
| Embedding matrix index off by one | The tokenizer may assign index 0 to `<PAD>` and index 1 to `<UNK>`, so word tokens start at index 2; if `embedding_matrix` is indexed from 0 each word gets the wrong embedding | Build `embedding_matrix` with size `vocab_size + 2` (or use `tokenizer.word_index` directly starting at index 2) |
| Fine-tuning GloVe from epoch 1 disrupts pretrained structure | Allowing full gradient flow through the GloVe embedding table from the very first batch, before the classification head has learned anything, causes the embedding table to drift rapidly and lose pretrained semantic geometry | Freeze the embedding layer (`embedding.trainable = False`) for the first 2–3 warm-up epochs, then unfreeze — this gives the classification head time to calibrate before the embeddings move |
| OOV words initialized to all-zeros share a "vector" | Multiple OOV words all receive the zero vector; during inference, the model treats them as identical, which is semantically wrong | Initialize OOV embeddings to small random vectors (`np.random.uniform(-0.05, 0.05, embed_dim)`) or to the mean of all GloVe vectors — both approaches are better than all-zeros |
| Wrong GloVe dimension for a fine-tuned model checkpoint | Saving and reloading a model trained with GloVe-100 but loading GloVe-200 at inference time causes a weight shape mismatch | Store the embedding dimension alongside the vocabulary file; assert `embed_dim == embedding_matrix.shape[1]` before `model.set_weights` |
