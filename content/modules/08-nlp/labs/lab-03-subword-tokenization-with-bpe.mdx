# PART 1: BPE Algorithm from Scratch

class BytePairEncoder:
    """
    Byte Pair Encoding implementation from scratch.
    
    BPE works by:
    1. Starting with character vocabulary
    2. Iteratively merging most frequent pairs
    3. Building subword vocabulary
    """
    
    def __init__(self, vocab_size=1000):
        self.vocab_size = vocab_size
        self.vocab = {}
        self.merges = []
        
    def train(self, texts: List[str]):
        """
        Train BPE on corpus.
        
        Args:
            texts: List of training texts
        """
        # Initialize word frequencies with character-level tokens
        word_freqs = collections.defaultdict(int)
        
        for text in texts:
            words = text.lower().split()
            for word in words:
                # End-of-word symbol
                word_with_eow = ' '.join(list(word)) + ' </w>'
                word_freqs[tuple(word_with_eow.split())] += 1
        
        print(f"Initial unique words: {len(word_freqs)}")
        
        # Get initial character vocabulary
        all_chars = set()
        for word_tokens in word_freqs.keys():
            all_chars.update(word_tokens)
        
        self.vocab = {token: idx for idx, token in enumerate(sorted(all_chars))}
        
        # BPE training loop
        num_merges = self.vocab_size - len(self.vocab)
        
        for i in range(num_merges):
            # Count all pairs
            pairs = collections.defaultdict(int)
            for word_tokens, freq in word_freqs.items():
                for j in range(len(word_tokens) - 1):
                    pairs[(word_tokens[j], word_tokens[j+1])] += freq
            
            if not pairs:
                break
                
            # Find most frequent pair
            best_pair = max(pairs, key=pairs.get)
            
            # Add to merges
            self.merges.append(best_pair)
            
            # Add merged token to vocabulary
            merged_token = best_pair[0] + best_pair[1]
            if merged_token not in self.vocab:
                self.vocab[merged_token] = len(self.vocab)
            
            # Update word frequencies by applying merge
            word_freqs = self._apply_merge(word_freqs, best_pair)
            
            if (i + 1) % 100 == 0:
                print(f"  Merge {i+1}/{num_merges}: {best_pair} -> {merged_token}")
        
        print(f"\nFinal vocabulary size: {len(self.vocab)}")
        
    def _apply_merge(self, word_freqs, pair):
        """Apply a merge operation to all words."""
        new_word_freqs = {}
        bigram = re.escape(' '.join(pair))
        pattern = re.compile(r'(?<!\S)' + bigram + r'(?!\S)')
        
        for word_tokens, freq in word_freqs.items():
            word_str = ' '.join(word_tokens)
            new_word_str = pattern.sub(''.join(pair), word_str)
            new_word_tokens = tuple(new_word_str.split())
            new_word_freqs[new_word_tokens] = freq
            
        return new_word_freqs
    
    def encode(self, text: str) -> List[str]:
        """
        Encode text using learned BPE merges.
        
        Args:
            text: Input text
            
        Returns:
            List of subword tokens
        """
        words = text.lower().split()
        encoded = []
        
        for word in words:
            # Start with character tokens
            word_tokens = list(word) + ['</w>']
            
            # Apply merges in order
            for merge in self.merges:
                i = 0
                while i < len(word_tokens) - 1:
                    if word_tokens[i] == merge[0] and word_tokens[i+1] == merge[1]:
                        word_tokens = word_tokens[:i] + [merge[0] + merge[1]] + word_tokens[i+2:]
                    else:
                        i += 1
            
            encoded.extend(word_tokens)
            
        return encoded
    
    def decode(self, tokens: List[str]) -> str:
        """
        Decode tokens back to text.
        
        Args:
            tokens: List of subword tokens
            
        Returns:
            Reconstructed text
        """
        text = ''.join(tokens)
        text = text.replace('</w>', ' ')
        return text.strip()

def create_subword_tokenizer(texts, target_vocab_size=1000):
    """
    Create a complete subword tokenization pipeline.
    
    Args:
        texts: Training texts
        target_vocab_size: Target vocabulary size
        
    Returns:
        Tokenizer and vocabulary
    """
    # Use TensorFlow Text's WordPiece
    # First, we need to generate the vocabulary
    
    # Tokenize to get words
    tokenizer = tf_text.UnicodeScriptTokenizer()
    
    # Flatten all tokens
    all_tokens = []
    for text in texts:
        tokens = tokenizer.tokenize(text)
        all_tokens.extend(tokens.numpy().tolist())
    
    # Build vocabulary from most frequent tokens
    token_counts = collections.Counter(all_tokens)
    vocab = ['[PAD]', '[UNK]', '[CLS]', '[SEP]', '[MASK]']
    
    # Add most frequent tokens
    for token, _ in token_counts.most_common(target_vocab_size - len(vocab)):
        if isinstance(token, bytes):
            vocab.append(token.decode('utf-8'))
        else:
            vocab.append(str(token))
    
    return vocab

# Create vocabulary

token_to_id = {token: idx for idx, token in enumerate(vocab)}

def encode_text(text, token_to_id, max_length=50):
    """Encode text to token IDs."""
    tokenizer = tf_text.UnicodeScriptTokenizer()
    tokens = tokenizer.tokenize(text)
    
    # Convert to IDs
    token_ids = []
    for token in tokens.numpy():
        token_str = token.decode('utf-8') if isinstance(token, bytes) else str(token)
        token_ids.append(token_to_id.get(token_str, token_to_id['[UNK]']))
    
    # Pad or truncate
    if len(token_ids) < max_length:
        token_ids = token_ids + [token_to_id['[PAD]']] * (max_length - len(token_ids))
    else:
        token_ids = token_ids[:max_length]
    
    return token_ids

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| BPE merges applied in wrong order during inference | Merges must be applied in the exact order they were learned during training; applying them in a different order produces different subword segmentations | Store `self.merges` as an ordered list and iterate it sequentially in `encode`; never use a set or dict for merge storage |
| `</w>` end-of-word marker leaks into decoded output | `decode` replaces `</w>` with a space but misses cases where it is fused into a longer merged token (e.g., `est</w>` becomes `est` without a trailing space) | After the replacement loop, strip extra whitespace and ensure the final `strip()` call removes only leading/trailing spaces, not internal ones |
| Vocabulary size target exceeded | `num_merges = vocab_size - len(initial_chars)` can be negative if `vocab_size` is smaller than the initial character vocabulary | Add a guard: `num_merges = max(0, vocab_size - len(initial_chars))` and warn if the requested vocab_size is too small |
| OOV subword token for unseen characters | If the test text contains characters (e.g., accented letters, digits) absent from the training corpus, the character is not in `self.vocab` | Add a Unicode normalization step (`unicodedata.normalize('NFKD', text)`) before training and a fallback `<UNK>` character for truly unseen bytes |
| SentencePiece vs BPE segmentation mismatch | SentencePiece treats the entire text as a byte stream and includes whitespace as a special character (`▁`), so its segmentation of the same word differs from standard BPE | When comparing tokenizers, always decode back to text and compare the surface string, not the token sequences, since boundary conventions differ |
