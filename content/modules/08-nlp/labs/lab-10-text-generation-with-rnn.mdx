# Create character vocabulary
chars = sorted(list(set(text)))
char_to_idx = {c: i for i, c in enumerate(chars)}
idx_to_char = {i: c for i, c in enumerate(chars)}

vocab_size = len(chars)

print(f"Characters: {chars}")

# Encode text
encoded = np.array([char_to_idx[c] for c in text])
print(f"Encoded text shape: {encoded.shape}")


# ============================================================
# PART 2: Create Training Sequences
# ============================================================

seq_length = 100  # Length of input sequences
step = 3  # Step size for creating sequences

# Create sequences
sequences = []
next_chars = []

for i in range(0, len(encoded) - seq_length, step):
    sequences.append(encoded[i:i + seq_length])
    next_chars.append(encoded[i + seq_length])

X = np.array(sequences)
y = np.array(next_chars)

print(f"Number of sequences: {len(sequences)}")
print(f"X shape: {X.shape}")
print(f"y shape: {y.shape}")


# ============================================================
# PART 3: Build Text Generation Model
# ============================================================

def build_text_generation_model(vocab_size, seq_length, embedding_dim=64, lstm_units=128):
    """
    Build LSTM model for text generation.
    
    Args:
        vocab_size: Number of unique characters
        seq_length: Length of input sequences
        embedding_dim: Character embedding dimension
        lstm_units: LSTM hidden units
        
    Returns:
        Compiled model
    """
    model = tf.keras.Sequential([
        # Character embedding
        tf.keras.layers.Embedding(vocab_size, embedding_dim, input_length=seq_length),
        
        # LSTM layers
        tf.keras.layers.LSTM(lstm_units, return_sequences=True),
        tf.keras.layers.Dropout(0.2),
        tf.keras.layers.LSTM(lstm_units),
        tf.keras.layers.Dropout(0.2),
        
        # Output layer
        tf.keras.layers.Dense(vocab_size, activation='softmax')
    ])
    
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss='sparse_categorical_crossentropy'
    )
    
    return model

# Build model
model = build_text_generation_model(vocab_size, seq_length)

print("\nModel Architecture:")
model.summary()


# ============================================================
# PART 4: Training
# ============================================================

# Callbacks

    tf.keras.callbacks.ModelCheckpoint(

        monitor='loss',

        monitor='loss',
        patience=5,

history = model.fit(
    X, y,
    batch_size=128,
    epochs=50,
    callbacks=callbacks,
    verbose=1
)

# ============================================================
# PART 5: Text Generation
# ============================================================

def generate_text(model, seed_text, length=500, temperature=1.0):
    """
    Generate text using trained model.
    
    Args:
        model: Trained text generation model
        seed_text: Starting text (must be at least seq_length characters)
        length: Number of characters to generate
        temperature: Sampling temperature (higher = more random)
        
    Returns:
        Generated text
    """
    generated = seed_text
    
    for _ in range(length):
        # Encode the last seq_length characters
        encoded_seed = [char_to_idx.get(c, 0) for c in generated[-seq_length:]]
        
        # Pad if necessary
        while len(encoded_seed) < seq_length:
            encoded_seed = [0] + encoded_seed
        
        # Predict
        x_pred = np.array([encoded_seed])
        preds = model.predict(x_pred, verbose=0)[0]
        
        # Apply temperature
        preds = np.log(preds + 1e-10) / temperature
        exp_preds = np.exp(preds)
        preds = exp_preds / np.sum(exp_preds)
        
        # Sample next character
        next_idx = np.random.choice(len(preds), p=preds)
        next_char = idx_to_char[next_idx]
        
        generated += next_char
    
    return generated

def generate_text_top_k(model, seed_text, length=500, k=10):
    """
    Generate text using top-k sampling.
    
    Only sample from the top k most likely characters.
    """
    generated = seed_text
    
    for _ in range(length):
        encoded_seed = [char_to_idx.get(c, 0) for c in generated[-seq_length:]]
        while len(encoded_seed) < seq_length:
            encoded_seed = [0] + encoded_seed
        
        x_pred = np.array([encoded_seed])
        preds = model.predict(x_pred, verbose=0)[0]
        
        # Get top k
        top_k_idx = np.argsort(preds)[-k:]
        top_k_probs = preds[top_k_idx]
        top_k_probs = top_k_probs / np.sum(top_k_probs)
        
        # Sample from top k
        next_idx = np.random.choice(top_k_idx, p=top_k_probs)
        next_char = idx_to_char[next_idx]
        
        generated += next_char
    
    return generated


def generate_text_nucleus(model, seed_text, length=500, p=0.9):
    """
    Generate text using nucleus (top-p) sampling.
    
    Sample from the smallest set of characters whose cumulative 
    probability exceeds p.
    """
    generated = seed_text
    
    for _ in range(length):
        encoded_seed = [char_to_idx.get(c, 0) for c in generated[-seq_length:]]
        while len(encoded_seed) < seq_length:
            encoded_seed = [0] + encoded_seed
        
        x_pred = np.array([encoded_seed])
        preds = model.predict(x_pred, verbose=0)[0]
        
        # Sort by probability
        sorted_idx = np.argsort(preds)[::-1]
        sorted_probs = preds[sorted_idx]
        
        # Find nucleus
        cumsum = np.cumsum(sorted_probs)
        nucleus_size = np.searchsorted(cumsum, p) + 1
        
        # Sample from nucleus
        nucleus_idx = sorted_idx[:nucleus_size]
        nucleus_probs = preds[nucleus_idx]
        nucleus_probs = nucleus_probs / np.sum(nucleus_probs)
        
        next_idx = np.random.choice(nucleus_idx, p=nucleus_probs)
        next_char = idx_to_char[next_idx]
        
        generated += next_char
    
    return generated

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| BPTT truncation mismatch between training sequence length and generation seed length | The model is trained on fixed-length windows of `seq_length` characters, but generation feeds a seed of arbitrary length — a longer seed causes shape errors with stateless LSTM | Always truncate or pad the generation seed to exactly `seq_length` before the first forward pass, or use a stateful LSTM that accepts one character at a time |
| Vanishing gradient through long character sequences | Backpropagating through 100+ character steps in a vanilla LSTM still suffers gradient attenuation for very early positions, causing the model to rely on only the last ~20 characters of context | Clip gradients with `clipnorm=1.0`, use an orthogonal initializer for recurrent weights, or reduce `seq_length` to 50–80 characters and rely on the statefulness trick |
| Temperature division by zero or near-zero | Passing `temperature=0` to the softmax scaling produces `logits / 0` — `inf` or `nan` in the distribution | Guard with `temperature = max(temperature, 1e-6)` and implement greedy decoding as a special case (`np.argmax`) rather than passing `temperature→0` |
| Repetition loops in greedy and near-greedy generation | The model memorizes the most frequent n-gram continuation and gets stuck in "the the the" loops | Use nucleus sampling (`top_p=0.9`) or add a repetition penalty: divide already-generated token logits by a `penalty > 1.0` factor before sampling |
| Perplexity computed on training sequence instead of held-out text | Reporting perplexity on the same sequence used for training gives an overly optimistic score — a severely overfit model can reach perplexity < 1.5 on training data but generate nonsense | Always compute perplexity on a separate 20% held-out slice of the corpus; expect perplexity 5–15 on the toy corpus and 2.5–4.0 on Shakespeare after proper training |

## Stretch Goals

1. **Beam search decoding with length normalization:** Implement beam search with `beam_width=5`. At each step, keep the top-5 partial sequences by log-probability. After generating `max_length` characters, apply length normalization: divide total log-probability by `length^alpha` with `alpha=0.7` (the value used in Google's NMT system). Compare greedy, beam (no normalization), and beam (normalized) on three seeds. Length normalization should prevent beams from preferring short completions and produce more complete sentences.

2. **Transformer-XL segment-level recurrence for long-range generation:** Replace the LSTM with a 2-layer Transformer decoder augmented with Transformer-XL's segment-level memory (Dai et al., 2019). The memory caches the hidden states from the previous segment and allows the current segment to attend to them, giving the model an effective context of `num_layers × segment_length` without quadratic self-attention cost. Train on the full Shakespeare corpus (~5M characters) and compare perplexity at context lengths 50, 200, and 500 characters.

3. **Controllable generation with CTRL-style condition tokens:** Prepend a control token to each training sequence indicating the genre of the text (e.g., `[FORMAL]`, `[CASUAL]`, `[POETIC]`). During training, the model learns to condition its outputs on the control token. At generation time, choose the desired genre by prepending the corresponding control token to the seed. Evaluate coherence by asking 5 people to classify 20 generated samples (10 per genre) without knowing the condition — inter-rater agreement should be > 70% if the conditioning is effective.

4. **BLEU score evaluation for generation quality:** Implement a BLEU-2 and BLEU-3 evaluation (bigram and trigram precision with brevity penalty) comparing generated text against held-out reference text from the same corpus. Generate 50 continuations from random seeds (length=100 characters each) and compute the corpus-level BLEU score. Baseline (greedy decoding, temperature=1.0): expect BLEU-2 ≈ 0.15–0.25 on the toy corpus. Compare top-k (`k=10`) and nucleus (`p=0.9`) sampling — nucleus should score 5–10% higher on BLEU-3 due to less repetition.
