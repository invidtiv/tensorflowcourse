attention_concepts = """
Attention Mechanism:

The attention mechanism allows the model to focus on different 
parts of the input sequence when producing each output.

Key Components:
1. Query (Q): What we're looking for
2. Key (K): What we match against
3. Value (V): What we extract information from

Attention Score = f(Q, K)
Attention Weights = softmax(Attention Score)
Output = sum(Attention Weights * V)

Common Score Functions:
- Dot Product: Q · K^T
- Scaled Dot Product: (Q · K^T) / sqrt(d_k)
- Additive: v^T * tanh(W_q * Q + W_k * K)
"""

print(attention_concepts)

# PART 2: Dot-Product Attention
# ============================================================

class DotProductAttention(tf.keras.layers.Layer):
    """
    Dot-product attention mechanism.
    
    score(Q, K) = Q · K^T
    """
    
    def __init__(self, **kwargs):
        super(DotProductAttention, self).__init__(**kwargs)
    
    def call(self, query, key, value, mask=None):
        """
        Compute attention.
        
        Args:
            query: (batch_size, query_len, d_k)
            key: (batch_size, key_len, d_k)
            value: (batch_size, value_len, d_v)
            mask: Optional mask (batch_size, query_len, key_len)
            
        Returns:
            output: (batch_size, query_len, d_v)
            attention_weights: (batch_size, query_len, key_len)
        """
        # Compute attention scores
        # score = Q · K^T
        scores = tf.matmul(query, key, transpose_b=True)  # (batch, query_len, key_len)
        
        # Scale scores
        d_k = tf.cast(tf.shape(key)[-1], tf.float32)
        scores = scores / tf.math.sqrt(d_k)
        
        # Apply mask if provided
        if mask is not None:
            scores += (mask * -1e9)
        
        # Compute attention weights
        attention_weights = tf.nn.softmax(scores, axis=-1)
        
        # Apply attention to values
        output = tf.matmul(attention_weights, value)
        
        return output, attention_weights

# PART 3: Scaled Dot-Product Attention
# ============================================================

class ScaledDotProductAttention(tf.keras.layers.Layer):
    """
    Scaled dot-product attention (used in Transformer).
    
    score(Q, K) = (Q · K^T) / sqrt(d_k)
    """
    
    def __init__(self, dropout_rate=0.0, **kwargs):
        super(ScaledDotProductAttention, self).__init__(**kwargs)
        self.dropout_rate = dropout_rate
        self.dropout = tf.keras.layers.Dropout(dropout_rate)
    
    def call(self, query, key, value, mask=None, training=False):
        """
        Compute scaled dot-product attention.
        """
        d_k = tf.cast(tf.shape(query)[-1], tf.float32)
        
        # Compute attention scores
        scores = tf.matmul(query, key, transpose_b=True)
        scores = scores / tf.math.sqrt(d_k)
        
        # Apply mask
        if mask is not None:
            scores += (mask * -1e9)
        
        # Softmax
        attention_weights = tf.nn.softmax(scores, axis=-1)
        
        # Apply dropout
        attention_weights = self.dropout(attention_weights, training=training)
        
        # Apply attention
        output = tf.matmul(attention_weights, value)
        
        return output, attention_weights

class SelfAttention(tf.keras.layers.Layer):
    """
    Self-attention layer.
    
    Query, Key, and Value all come from the same input.
    """
    
    def __init__(self, d_model, **kwargs):
        super(SelfAttention, self).__init__(**kwargs)
        self.d_model = d_model
        
        # Projection layers
        self.W_q = tf.keras.layers.Dense(d_model)
        self.W_k = tf.keras.layers.Dense(d_model)
        self.W_v = tf.keras.layers.Dense(d_model)
        
        self.attention = ScaledDotProductAttention()
    
    def call(self, x, mask=None, training=False):
        """
        Apply self-attention.
        
        Args:
            x: Input tensor (batch, seq_len, d_model)
            mask: Optional mask
            
        Returns:
            output, attention_weights
        """
        # Project to Q, K, V
        query = self.W_q(x)
        key = self.W_k(x)
        value = self.W_v(x)
        
        # Apply attention
        output, weights = self.attention(
            query, key, value, mask=mask, training=training
        )
        
        return output, weights

class AttentionClassifier(tf.keras.Model):
    """
    Text classifier using attention mechanism.
    """
    
    def __init__(self, vocab_size, embedding_dim, hidden_units, num_classes):
        super(AttentionClassifier, self).__init__()
        
        self.embedding = tf.keras.layers.Embedding(vocab_size, embedding_dim)
        
        # Bidirectional LSTM
        self.lstm = tf.keras.layers.Bidirectional(
            tf.keras.layers.LSTM(hidden_units, return_sequences=True)
        )
        
        # Attention
        self.attention = SelfAttention(hidden_units * 2)
        
        # Classification head
        self.global_pool = tf.keras.layers.GlobalAveragePooling1D()
        self.dense1 = tf.keras.layers.Dense(64, activation='relu')
        self.dropout = tf.keras.layers.Dropout(0.5)
        self.output_layer = tf.keras.layers.Dense(num_classes, activation='softmax')
    
    def call(self, inputs, training=False):
        # Embed
        x = self.embedding(inputs)
        
        # LSTM encoding
        lstm_out = self.lstm(x)
        
        # Apply attention
        attended, attention_weights = self.attention(lstm_out, training=training)
        
        # Global pooling
        pooled = self.global_pool(attended)
        
        # Classification
        x = self.dense1(pooled)
        x = self.dropout(x, training=training)
        output = self.output_layer(x)
        
        return output, attention_weights

**Reference numbers to validate against:**

- **Softmax row-sum invariant**: `tf.reduce_sum(attention_weights, axis=-1)` must equal 1.0 for every query position. If any row sums to 0 → a whole row was masked (check mask polarity: this code uses `scores += mask * -1e9`, so `mask=1` means *blocked*, `mask=0` means *visible* — the opposite convention from PyTorch's `masked_fill`). If rows sum to `NaN` → `-inf + -inf` in a fully-masked row; replace `-1e9` with a large finite value and ensure at least one position is unmasked.
- **Scaling sanity check**: at `d_k=64`, raw dot-product scores have stddev ≈ √64 = 8; softmax over scores with stddev 8 concentrates > 99% mass on the argmax and kills gradients through all other keys. With `/sqrt(d_k)` the stddev is ≈ 1 and attention stays smooth. If loss plateaus immediately and only one token gets attended, you forgot the scale.
- **Self-attention output shape equals input shape**: `(batch, seq_len, d_model)` in → `(batch, seq_len, d_model)` out. If output is `(batch, seq_len, d_k)` you forgot the output projection `W_o` (not strictly required for single-head, but required before stacking layers).
- **Gradient norm**: `tf.linalg.global_norm(tape.gradient(loss, layer.trainable_variables))` should be in `[0.1, 10]` for the first few steps. > 100 → missing residual + LayerNorm (see Pitfall 3). < 1e-4 → softmax has saturated from unscaled scores.

## ⚠️ Common Pitfalls

1. **Forgetting the `1/√d_k` scale factor:** Without scaling, at `d_k=64` the pre-softmax scores have stddev ≈ 8, which pushes softmax into a near-one-hot regime. Attention weights then concentrate on a single key, all other V vectors get ~0 gradient, and training stalls. *Symptom*: loss drops for ~50 steps then freezes, and `attention_weights` histograms show bimodal mass at 0 and 1. *Fix*: divide scores by `tf.math.sqrt(tf.cast(d_k, tf.float32))` BEFORE the softmax — never after.

2. **Mask polarity confusion:** This lab uses `scores += (mask * -1e9)`, so `mask=1` means *blocked* and `mask=0` means *visible*. PyTorch's `masked_fill(mask, -inf)` and Keras' `Attention(use_causal_mask=True)` use the opposite convention in different places. *Symptom*: the model ignores the actual tokens and attends to padding, or a causal mask leaks future positions. *Fix*: always unit-test your mask by printing `attention_weights[0]` on a toy sequence where you know which positions should be zero. Also prefer a large finite penalty (`-1e9`) over `-inf` to avoid `NaN` in fully-masked rows.

3. **Stacking attention without residual + LayerNorm:** Naive stacking of self-attention layers without the transformer's `x = LayerNorm(x + Attention(x))` wrapper causes the signal magnitude to drift (either vanish or explode) after 2–3 layers. *Symptom*: gradient norm collapses to < 1e-5 or blows up to > 1e3, and training loss is flat. *Fix*: wrap every sublayer as `x = x + Dropout(Sublayer(LayerNorm(x)))` (pre-norm, more stable) or `x = LayerNorm(x + Dropout(Sublayer(x)))` (post-norm, as in the original paper). This lab demonstrates a *single* attention block, so it works without norm — but do not copy-paste it into a multi-layer model.

4. **Using `tf.keras.layers.Attention` thinking it's scaled dot-product:** The Keras built-in `Attention` layer does **not** scale by `1/√d_k` by default (the `use_scale` flag defaults to `False` in older TF 2.x). For transformer-equivalent behavior, either build it from scratch as in this lab or use `tf.keras.layers.MultiHeadAttention(num_heads=1, key_dim=d_k)`, which does scale correctly.

5. **Quadratic memory blowup at long sequences:** The attention-weights tensor is `(batch, seq_len, seq_len)`. At `batch=32, seq_len=2048` that's ~0.5 GB in fp32 per layer. *Symptom*: OOM on a T4 (16 GB) with `seq_len > 1024` even for tiny `d_model`. *Fix*: use `tf.keras.mixed_precision.set_global_policy('mixed_float16')` to halve the memory, then reach for `FlashAttention` / chunked attention / local windows for `seq_len > 4096`.

## 🚀 Stretch Goals

1. **Causal (auto-regressive) mask:** Build a lower-triangular mask `tf.linalg.band_part(tf.ones((L, L)), -1, 0)` (positions `j > i` set to 0), invert to match this lab's polarity, and apply inside `ScaledDotProductAttention`. Verify by printing `attention_weights` for a length-5 sequence: the upper triangle must be exactly 0. This single change turns the block into a GPT-style decoder layer.

2. **Scaled vs unscaled ablation:** Run `ScaledDotProductAttention` and a copy with the `/sqrt(d_k)` line removed on a toy copy task (input = random tokens, target = same tokens shifted by 1). Plot train loss vs step for both, varying `d_k ∈ {8, 32, 128, 512}`. Expected finding: the gap widens with `d_k`, and at `d_k=512` the unscaled variant fails to learn at all.

3. **KV-cache toy:** Implement incremental decoding. Instead of recomputing K and V for the full prefix at each step, cache them: `K_cache = tf.concat([K_cache, new_k], axis=1)`. Measure the speedup for generating a 128-token sequence vs the naive recompute-everything-each-step loop. Expected: ~`L/2` × speedup at position `L` (O(L²) → O(L) per step).

4. **Additive (Bahdanau) vs dot-product comparison:** Implement `score = v^T tanh(W_q Q + W_k K)` (broadcast over positions) and compare on a small machine-translation task (e.g., tatoeba en-es pairs). Dot-product is faster per step but needs the `1/√d_k` scale; additive is slower (extra `Dense` + `tanh`) but can be more stable at very high `d_k`.

5. **Attention-weight entropy as a training signal:** Track `H(alpha) = -sum(alpha * log(alpha))` across training. Healthy transformer training shows entropy starting high (~log(L), nearly uniform), dropping to moderate values (~2–4 nats) as the model commits to specific patterns. If entropy collapses to 0 in < 100 steps the model has mode-collapsed onto one token; if it never drops the attention is just an expensive identity — rescale initialization or add warmup.
