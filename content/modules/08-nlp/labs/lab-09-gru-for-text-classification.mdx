"""
Lab 8.8: GRU for Text Classification

Implement GRU (Gated Recurrent Unit) and compare with LSTM.

import tensorflow as tf
import numpy as np

# ============================================================
# PART 1: GRU Cell from Scratch
# ============================================================

class GRUCell(tf.keras.layers.Layer):
    """
    GRU cell implemented from scratch.
    
    GRU equations (simplified):
    z_t = sigmoid(W_z · [h_{t-1}, x_t] + b_z)  # Update gate
    r_t = sigmoid(W_r · [h_{t-1}, x_t] + b_r)  # Reset gate
    h~_t = tanh(W · [r_t * h_{t-1}, x_t] + b)  # Candidate activation
    h_t = (1 - z_t) * h_{t-1} + z_t * h~_t     # Hidden state update
    
    GRU has fewer gates than LSTM (2 vs 4), making it faster.
    """
    
    def __init__(self, units, **kwargs):
        super(GRUCell, self).__init__(**kwargs)
        self.units = units
        self.state_size = units
        self.output_size = units
    
    def build(self, input_shape):
        input_dim = input_shape[-1] + self.units
        
        # Update gate
        self.W_z = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_z'
        )
        self.b_z = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_z'
        )
        
        # Reset gate
        self.W_r = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_r'
        )
        self.b_r = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_r'
        )
        
        # Candidate activation
        self.W_h = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_h'
        )
        self.b_h = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_h'
        )
        
        super(GRUCell, self).build(input_shape)
    
    def call(self, inputs, states):
        """
        Single GRU step.
        
        Args:
            inputs: (batch, input_dim)
            states: [h_prev]
            
        Returns:
            h_t, [h_t]
        """
        h_prev = states[0]
        
        # Concatenate input and previous hidden state
        concat = tf.concat([inputs, h_prev], axis=1)
        
        # Update gate
        z_t = tf.sigmoid(tf.matmul(concat, self.W_z) + self.b_z)
        
        # Reset gate
        r_t = tf.sigmoid(tf.matmul(concat, self.W_r) + self.b_r)
        
        # Candidate activation with reset
        concat_reset = tf.concat([inputs, r_t * h_prev], axis=1)
        h_tilde = tf.tanh(tf.matmul(concat_reset, self.W_h) + self.b_h)
        
        # Update hidden state
        h_t = (1 - z_t) * h_prev + z_t * h_tilde
        
        return h_t, [h_t]
    
    def get_initial_state(self, inputs=None, batch_size=None, dtype=None):
        return [tf.zeros((batch_size, self.units), dtype=dtype)]

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Reset gate applied to full concatenation instead of previous hidden state only | The candidate activation uses `r_t * h_{t-1}` (reset gate applied only to the previous hidden state before concatenation with input), not `r_t * [h_{t-1}; x_t]` | Compute `h_candidate = tanh(W * concat([r_t * h_prev, x_t]) + b)` — reset the hidden state before concatenating with the current input |
| Update gate polarity flipped | Using `h_t = z_t * h_{t-1} + (1-z_t) * h_candidate` instead of the correct `(1-z_t) * h_{t-1} + z_t * h_candidate` swaps the interpolation: when `z_t→1` the model should favour the new candidate but this mistake makes it favour the old hidden state | Use `h_t = (1 - z_t) * h_prev + z_t * h_candidate` so `z_t→1` adopts the candidate and `z_t→0` retains the previous state |
| GRU `reset_after=True` vs `reset_after=False` mismatch with cuDNN | TensorFlow's built-in `GRU` layer defaults to `reset_after=True` (recurrent dropout is applied after the matrix multiplication) for cuDNN compatibility; the custom cell typically implements `reset_after=False` — the two variants are mathematically different | Use `tf.keras.layers.GRU(..., reset_after=True)` to match cuDNN behaviour when comparing your custom cell to the built-in; document which convention you use |
| Stateless comparison between GRU and LSTM inflated by random seed | If GRU and LSTM are initialized with different random seeds, a lucky seed can make GRU look faster to converge than it actually is | Fix `tf.random.set_seed(42)` and `np.random.seed(42)` before constructing each model; re-run with 3 different seeds and report mean ± std |
| Layer normalization applied per gate instead of per time step | Adding `LayerNormalization` inside the GRU cell should normalize each gate's pre-activation independently — but if you normalize after all gates are combined the normalization scale is wrong | Apply a separate `LayerNormalization` to each gate's linear pre-activation (`z_pre`, `r_pre`, `h_pre`) before the `sigmoid`/`tanh`, as in the Layer Normalization paper (Ba et al., 2016) |

## Stretch Goals
