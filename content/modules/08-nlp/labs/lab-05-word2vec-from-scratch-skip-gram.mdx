"""

Implement the Skip-gram Word2Vec model from scratch.

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from collections import Counter
import random
import re

# ============================================================
# PART 1: Data Preparation
# ============================================================

class Word2VecDataGenerator:
    """
    Generate training data for Word2Vec Skip-gram model.
    """
    
    def __init__(self, texts, window_size=2, min_count=1):
        """
        Initialize data generator.
        
        Args:
            texts: List of text strings
            window_size: Context window size (on each side)
            min_count: Minimum word frequency
        """
        self.window_size = window_size
        
        # Build vocabulary
        word_counts = Counter()
        for text in texts:
            words = self._tokenize(text)
            word_counts.update(words)
        
        # Filter by min_count
        self.vocab = ['<PAD>', '<UNK>']
        self.word_counts = {}
        
        for word, count in word_counts.items():
            if count >= min_count:
                self.vocab.append(word)
                self.word_counts[word] = count
        
        self.word_to_id = {word: idx for idx, word in enumerate(self.vocab)}
        self.id_to_word = {idx: word for word, idx in self.word_to_id.items()}
        self.vocab_size = len(self.vocab)
        
        print(f"Vocabulary size: {self.vocab_size}")
        
        # Generate training pairs
        self.training_data = self._generate_pairs(texts)
        print(f"Training pairs: {len(self.training_data)}")
        
    def _tokenize(self, text):
        """Simple tokenization."""
        text = text.lower()
        text = re.sub(r'[^a-z0-9\s]', '', text)
        return text.split()
    
    def _generate_pairs(self, texts):
        """Generate (target, context) pairs."""
        pairs = []
        
        for text in texts:
            words = self._tokenize(text)
            word_ids = [self.word_to_id.get(w, self.word_to_id['<UNK>']) 
                       for w in words]
            
            for i, target_id in enumerate(word_ids):
                # Get context window
                start = max(0, i - self.window_size)
                end = min(len(word_ids), i + self.window_size + 1)
                
                for j in range(start, end):
                    if i != j:
                        context_id = word_ids[j]
                        pairs.append((target_id, context_id))
        
        return pairs
    
    def generate_batch(self, batch_size, negative_samples=5):
        """
        Generate a batch of training data with negative sampling.
        
        Args:
            batch_size: Number of positive pairs
            negative_samples: Number of negative samples per positive
            
        Returns:
            target_words, context_words, labels
        """
        # Sample positive pairs
        batch_pairs = random.sample(self.training_data, batch_size)
        
        targets = []
        contexts = []
        labels = []
        
        for target, context in batch_pairs:
            # Positive sample
            targets.append(target)
            contexts.append(context)
            labels.append(1)
            
            # Negative samples
            for _ in range(negative_samples):
                negative_context = random.randint(0, self.vocab_size - 1)
                targets.append(target)
                contexts.append(negative_context)
                labels.append(0)
        
        return (np.array(targets), np.array(contexts)), np.array(labels)


# ============================================================
# PART 2: Skip-gram Model Implementation
# ============================================================

class SkipGramModel(tf.keras.Model):
    """
    Skip-gram Word2Vec model with negative sampling.
    """
    
    def __init__(self, vocab_size, embedding_dim):
        super(SkipGramModel, self).__init__()
        
        self.vocab_size = vocab_size
        self.embedding_dim = embedding_dim
        
        # Target word embeddings (input)
        self.target_embedding = tf.keras.layers.Embedding(
            vocab_size, embedding_dim, name='target_embedding'
        )
        
        # Context word embeddings (output)
        self.context_embedding = tf.keras.layers.Embedding(
            vocab_size, embedding_dim, name='context_embedding'
        )
        
    def call(self, inputs):
        """
        Forward pass.
        
        Args:
            inputs: Tuple of (target_words, context_words)
            
        Returns:
            Similarity scores
        """
        target_words, context_words = inputs
        
        # Get embeddings
        target_embed = self.target_embedding(target_words)  # (batch, embed_dim)
        context_embed = self.context_embedding(context_words)  # (batch, embed_dim)
        
        # Compute dot product similarity
        dots = tf.reduce_sum(target_embed * context_embed, axis=1)
        
        return dots
    
    def get_word_embedding(self, word_id):
        """Get embedding for a word."""
        return self.target_embedding(tf.constant([word_id]))[0]


def create_skipgram_model(vocab_size, embedding_dim=100):
    """
    Create and compile Skip-gram model.
    """
    model = SkipGramModel(vocab_size, embedding_dim)
    
    # Custom loss for negative sampling
    def negative_sampling_loss(y_true, y_pred):
        """
        Negative sampling loss.
        
        For positive samples (y=1): maximize log(sigmoid(pred))
        For negative samples (y=0): maximize log(sigmoid(-pred))
        """
        return tf.reduce_mean(
            tf.nn.sigmoid_cross_entropy_with_logits(
                labels=tf.cast(y_true, tf.float32),
                logits=y_pred
            )
        )
    
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss=negative_sampling_loss,
        metrics=['accuracy']
    )
    
    return model

# Find similar words
def find_similar_words(word, top_k=5):
    """Find most similar words using cosine similarity."""
    if word not in data_gen.word_to_id:
        return []
    
    word_id = data_gen.word_to_id[word]
    word_vec = embeddings[word_id]
    
    # Compute cosine similarities
    similarities = []
    for i, vec in enumerate(embeddings):
        if i != word_id and i > 1:  # Skip special tokens
            sim = np.dot(word_vec, vec) / (np.linalg.norm(word_vec) * np.linalg.norm(vec))
            similarities.append((data_gen.id_to_word[i], sim))
    
    # Sort by similarity
    similarities.sort(key=lambda x: x[1], reverse=True)
    
    return similarities[:top_k]

# Test similarity

def word_analogy(word_a, word_b, word_c, top_k=3):
    """
    Solve analogy: word_a is to word_b as word_c is to ?
    
    Formula: result = word_c + (word_b - word_a)
    """
    if not all(w in data_gen.word_to_id for w in [word_a, word_b, word_c]):
        return []
    
    # Get vectors
    vec_a = embeddings[data_gen.word_to_id[word_a]]
    vec_b = embeddings[data_gen.word_to_id[word_b]]
    vec_c = embeddings[data_gen.word_to_id[word_c]]
    
    # Compute target vector
    target = vec_c + (vec_b - vec_a)
    
    # Find closest words
    similarities = []
    for i, vec in enumerate(embeddings):
        if i > 1 and data_gen.id_to_word[i] not in [word_a, word_b, word_c]:
            sim = np.dot(target, vec) / (np.linalg.norm(target) * np.linalg.norm(vec))
            similarities.append((data_gen.id_to_word[i], sim))
    
    similarities.sort(key=lambda x: x[1], reverse=True)
    
    return similarities[:top_k]

## ⚠️ Common Pitfalls

1. **Uniform instead of frequency-based negative sampling:** The `generate_batch` method here samples negative words uniformly at random (`random.randint(0, vocab_size - 1)`). The original Word2Vec paper samples from a unigram distribution raised to the `3/4` power, which down-weights high-frequency function words ('the', 'a', 'is') as negatives. On large corpora, uniform sampling means high-frequency words appear so often as negatives that their embeddings get pushed apart from everything else. *Fix*: build a `noise_distribution = word_freq ** 0.75 / (word_freq ** 0.75).sum()` and use `np.random.choice(vocab_size, size=negative_samples, p=noise_distribution, replace=True)`. On this toy corpus the difference is small, but on real datasets it can be the difference between meaningful and meaningless embeddings.

2. **Corpus too small for meaningful semantic geometry:** This lab uses 10 sentences repeated 50 times (~500 total). Word2Vec requires millions of tokens to learn the linear analogy structure it's famous for (king − man + woman ≈ queen). With 500 tokens, you get co-occurrence embeddings, not true distributional semantics. *Symptom*: all word similarities cluster in a narrow band (0.6–0.8) with no clear separation, and analogy arithmetic fails. *Fix for real use*: load `datasets.load_dataset('wikitext', 'wikitext-2-raw-v1')` (~2M tokens) and filter to `min_count=5`; expect vocab ~30k and training time ~10 min on CPU.

3. **Discarding the context embedding matrix:** This model trains *two* embedding matrices: `target_embedding` (input) and `context_embedding` (output). The lab uses only `target_embedding` for downstream tasks. Some practitioners average both (`(W_in + W_out) / 2`) to get slightly better similarity scores. More importantly, if you accidentally call `model.context_embedding.weights[0]` instead of `model.target_embedding.weights[0]` the resulting similarities will be meaningless — the two matrices serve different roles during training even if their shapes are identical.

4. **Low-frequency word embedding instability:** Words with very few training pairs receive almost no gradient signal. In a vocabulary of 47 words where 'quick' appears only 3 times, its embedding can be nearly random after training. *Symptom*: `find_similar_words('quick')` returns words that intuitively have nothing in common with it. *Fix*: increase `min_count` to exclude ultra-rare words, or use `subsampling` to down-weight high-frequency words (freeing gradient budget for rare ones), or simply accept that rare-word embeddings are unreliable and document this to downstream users.

5. **`<PAD>` and `<UNK>` token embeddings leaking into similarity searches:** Indices 0 and 1 are reserved for `<PAD>` and `<UNK>`. The `find_similar_words` function correctly skips `i > 1`, but if you copy this function and forget that check you'll find that 'machine' is "most similar" to `<PAD>` on some runs. In production code, always exclude special tokens from similarity lookups *and* from analogy arithmetic. For PCA plots, note that the code already slices `embeddings[2:]` — but the indices then no longer align with `word_to_id`, causing annotation offsets; compute `idx = data_gen.word_to_id[word] - 2` as the lab does, and never use raw `word_to_id` values as indices into the 2D array.

## 🚀 Stretch Goals

1. **Frequency-based unigram negative sampling:** Implement the 3/4-power noise distribution described in Pitfall 1. Build `noise_dist = np.array([count ** 0.75 for word, count in data_gen.word_counts.items()])`, normalize it, and pass it to `np.random.choice`. Train both variants for 200 epochs on the toy corpus and compare the top-5 similar words for 'learning'. On a larger corpus (wikitext-2), the difference in analogy accuracy should be measurable (expect +3–8 percentage points on the Google analogy dataset).

2. **Dynamic context window:** The original Word2Vec samples the context window size uniformly from `[1, max_window]` for each training pair, which gives nearer context words higher effective frequency and subtly helps grammatical word pairs. Modify `_generate_pairs` to use `actual_window = random.randint(1, self.window_size)` instead of the fixed `self.window_size`. Compare the similarity structure before and after: dynamic windows tend to tighten syntactic clusters ('quick'/'brown'/'lazy') more than semantic ones.

3. **FastText-style subword extension:** Extend the `Word2VecDataGenerator` to decompose each word into character n-grams (e.g., 'learning' → `<le`, `lea`, `ear`, `arn`, `rni`, `nin`, `ing`, `ng>`). Build a joint vocabulary of words + n-grams, and have each word's embedding be the mean of its own embedding plus all its n-gram embeddings (as in the original FastText paper). Evaluate OOV handling: split your corpus into train/test at the sentence level, drop 3 words from the training vocabulary, and check that the OOV words receive reasonable embeddings based on their character structure (e.g., 'tensorflow' should embed near 'tensor' if the model saw 'tensor' but not 'tensorflow').

4. **Analogy evaluation on Google analogy dataset:** Download the Google analogy dataset (19,544 analogies across semantic and syntactic categories) and evaluate the trained embeddings using the `3CosAdd` metric: for each `a : b :: c : ?` quadruple, compute `emb[b] - emb[a] + emb[c]` and rank all vocabulary words by cosine similarity to this vector. Report separate accuracy for semantic analogies (capital cities, currency pairs) and syntactic analogies (comparative adjectives, past tense verbs). On the toy corpus in this lab, expect near-0% accuracy — that is the expected result and motivates scaling to wikitext-2 where a 100-d Skip-gram model typically achieves 30–50% semantic and 40–60% syntactic analogy accuracy after 5 epochs.
