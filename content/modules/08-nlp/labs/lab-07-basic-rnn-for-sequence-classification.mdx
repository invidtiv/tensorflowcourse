import tensorflow as tf
import numpy as np

# ============================================================
# PART 1: RNN Cell from Scratch
# ============================================================

class SimpleRNNCell(tf.keras.layers.Layer):
    """
    Simple RNN cell implemented from scratch.
    
    h_t = tanh(W_x * x_t + W_h * h_{t-1} + b)
    """
    
    def __init__(self, units, **kwargs):
        super(SimpleRNNCell, self).__init__(**kwargs)
        self.units = units
        self.state_size = units
        self.output_size = units
    
    def build(self, input_shape):
        # Input weights
        self.W_x = self.add_weight(
            shape=(input_shape[-1], self.units),
            initializer='glorot_uniform',
            name='W_x'
        )
        
        # Hidden state weights
        self.W_h = self.add_weight(
            shape=(self.units, self.units),
            initializer='orthogonal',
            name='W_h'
        )
        
        # Bias
        self.b = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b'
        )
        
        super(SimpleRNNCell, self).build(input_shape)
    
    def call(self, inputs, states):
        """
        Single RNN step.
        
        Args:
            inputs: (batch, input_dim)
            states: [(batch, units)] - previous hidden state
            
        Returns:
            output, new_states
        """
        prev_h = states[0]
        
        # Compute new hidden state
        h = tf.matmul(inputs, self.W_x) + tf.matmul(prev_h, self.W_h) + self.b
        h = tf.tanh(h)
        
        return h, [h]
    
    def get_initial_state(self, inputs=None, batch_size=None, dtype=None):
        """Initialize hidden state to zeros."""
        return [tf.zeros((batch_size, self.units), dtype=dtype)]


# ============================================================
# PART 2: RNN Model for Classification
# ============================================================

def create_rnn_classifier(vocab_size, embedding_dim=64, rnn_units=64, num_classes=2):
    """
    Create RNN-based text classifier.
    
    Args:
        vocab_size: Size of vocabulary
        embedding_dim: Embedding dimension
        rnn_units: RNN hidden units
        num_classes: Number of output classes
        
    Returns:
        Compiled model
    """
    model = tf.keras.Sequential([
        # Embedding layer
        tf.keras.layers.Embedding(vocab_size, embedding_dim, mask_zero=True),
        
        # RNN layer (using our custom cell)
        tf.keras.layers.RNN(
            SimpleRNNCell(rnn_units),
            return_sequences=False,
            return_state=False
        ),
        
        # Output layers
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ])
    
    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model


# ============================================================
# PART 3: Using Built-in SimpleRNN
# ============================================================

def create_builtin_rnn_classifier(vocab_size, embedding_dim=64, rnn_units=64, num_classes=2):
    """
    Create classifier using built-in SimpleRNN.
    """
    model = tf.keras.Sequential([
        tf.keras.layers.Embedding(vocab_size, embedding_dim, mask_zero=True),
        tf.keras.layers.SimpleRNN(rnn_units, dropout=0.2),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ])
    
    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model

# Manual RNN forward pass demonstration
def manual_rnn_step(x_t, h_prev, W_x, W_h, b):
    """
    Single RNN step computation.
    
    Args:
        x_t: Input at time t (batch, input_dim)
        h_prev: Hidden state at t-1 (batch, units)
        W_x, W_h, b: RNN parameters
        
    Returns:
        h_t: New hidden state
    """
    # Linear transformation
    linear = tf.matmul(x_t, W_x) + tf.matmul(h_prev, W_h) + b
    
    # Non-linearity
    h_t = tf.tanh(linear)
    
    return h_t

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Vanishing gradients on sequences longer than ~20 tokens | The `tanh` nonlinearity and recurrent weight matrix `W_h` multiply the gradient at every step; if the spectral radius of `W_h` is < 1 the gradient decays exponentially | Initialize `W_h` with the orthogonal initializer (spectral radius = 1); switch to LSTM or GRU for sequences > 50 tokens |
| Loss spikes to NaN on real-text sequences | Without gradient clipping, occasional large gradients produce catastrophic weight updates in long sequences | Add `clipnorm=1.0` to the Adam optimizer: `tf.keras.optimizers.Adam(learning_rate=0.001, clipnorm=1.0)` |
| Padding tokens affect the final hidden state | `mask_zero=True` propagates a mask through `SimpleRNN` / `LSTM`, but a custom `Lambda` layer or `Flatten` between the RNN and `Dense` silently drops it | Call `model.compute_mask(input, None)` at each layer to confirm the mask survives; avoid non-mask-aware layers between the RNN and the head |
| Stateful RNN carries hidden state across unrelated batches | `stateful=True` is designed for splitting a single long document across batches; it corrupts classification tasks where each batch is independent | Use `stateful=False` (the default) for standard classification; if you need `stateful=True`, call `model.reset_states()` at the start of each epoch |
| `return_sequences=True` causes shape mismatch with `Dense` | With `return_sequences=True` the RNN outputs `(batch, seq_len, units)` and a following `Dense(2)` produces `(batch, seq_len, 2)` instead of `(batch, 2)` | Insert `GlobalAveragePooling1D()` or `Lambda(lambda x: x[:, -1, :])` between the RNN and the classification head |

## ⚠️ Common Pitfalls

1. **`tanh` saturation and vanishing gradients on longer sequences:** The vanilla RNN cell uses `tanh` as its nonlinearity. At `seq_length=20` with `rnn_units=64`, gradient magnitudes at time step 1 are multiplied by the recurrent weight matrix `W_h` 19 times. If `W_h` has spectral radius > 1, gradients explode; < 1, they vanish. *Symptom*: loss decreases quickly for 2–3 epochs then plateaus; early time steps in a gradient analysis plot show near-zero gradient. *Diagnostic*: initialize `W_h` with `orthogonal` initializer (as this lab does — spectral radius = 1.0) and confirm with `np.linalg.svd(model.layers[1].cell.W_h.numpy())[1].max()`. For sequences > 50 tokens, switch to LSTM or GRU — the gate mechanisms explicitly preserve gradient paths.

2. **Missing gradient clipping on long or real-text sequences:** This lab trains on synthetic data with `seq_length=20`, which is forgiving. On IMDB (`seq_length=256`) or document classification (`seq_length=512`), vanilla RNN without gradient clipping will occasionally produce loss spikes (`NaN` or `inf`) within the first 5 epochs. *Fix*: always add `clipnorm=1.0` to the Adam optimizer: `tf.keras.optimizers.Adam(learning_rate=0.001, clipnorm=1.0)`. Gradient clipping has almost no downside and prevents catastrophic weight updates.

3. **`mask_zero=True` on Embedding but masking unaware in downstream layers:** Setting `mask_zero=True` on the `Embedding` layer tells Keras to propagate a boolean padding mask downstream. `SimpleRNN` and `LSTM` are mask-aware and will correctly skip zero-padded positions. However, if you insert a custom `Lambda` layer or use `tf.keras.layers.Flatten` between the RNN and the `Dense`, the mask can be silently dropped without error. *Symptom*: padding tokens contribute to the final hidden state, slightly degrading accuracy in a way that is hard to attribute. *Diagnostic*: call `model.compute_mask(input, mask=None)` at each layer; the mask should be `None` only after a layer that intentionally consumes it (like a final RNN step with `return_sequences=False`).

4. **Stateful RNN (`stateful=True`) without resetting state between unrelated batches:** `SimpleRNN` and `LSTM` have an optional `stateful=True` flag that carries the hidden state from one batch to the next — useful for processing a single long document split across batches, but destructive when batches are *different* sequences. *Symptom*: validation accuracy drops erratically epoch-to-epoch; the model appears to "remember" the previous batch. *Fix*: call `model.reset_states()` at the start of each epoch (in a `Callback`), or just leave `stateful=False` (the default) for standard classification tasks.

5. **`return_sequences=True` shape mismatch with a following `Dense`:** If you change `return_sequences=False` to `True` to get per-step outputs, the RNN now outputs `(batch, seq_len, units)`. A subsequent `Dense(2)` layer *will* still compile (Keras broadcasts Dense over the time dimension), but it then produces `(batch, seq_len, 2)` instead of `(batch, 2)`. Calling `model.compile(loss='sparse_categorical_crossentropy')` with integer labels of shape `(batch,)` will then fail at training time with a shape mismatch error. *Fix*: after `return_sequences=True`, add `tf.keras.layers.GlobalAveragePooling1D()` (mean across time) or `tf.keras.layers.Lambda(lambda x: x[:, -1, :])` (last step only) before the classification head.
