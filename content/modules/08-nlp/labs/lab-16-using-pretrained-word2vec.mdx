def load_word2vec_pretrained(model_name='word2vec-google-news-300'):
    """
    Load pretrained Word2Vec using gensim.
    
    Args:
        model_name: Name of pretrained model
        
    Returns:
        Word2Vec model
    """
    print(f"Loading {model_name}...")
    print("(This may take a while for large models)")
    
    try:
        model = api.load(model_name)
        print(f"Loaded model with {len(model.key_to_index)} words")
        return model
    except Exception as e:
        print(f"Error loading model: {e}")

def create_word2vec_matrix(word_index, word2vec_model, embedding_dim):
    """
    Create embedding matrix from Word2Vec model.
    
    Args:
        word_index: {word: index} mapping
        word2vec_model: Loaded Word2Vec model
        embedding_dim: Embedding dimension
        
    Returns:
        Embedding matrix
    """
    vocab_size = len(word_index) + 1
    embedding_matrix = np.zeros((vocab_size, embedding_dim))
    
    found = 0
    for word, idx in word_index.items():
        if word in word2vec_model:
            embedding_matrix[idx] = word2vec_model[word]
            found += 1
        else:
            # Random initialization for OOV words
            embedding_matrix[idx] = np.random.normal(0, 0.01, embedding_dim)
    
    print(f"Found embeddings for {found}/{len(word_index)} words "
          f"({100*found/len(word_index):.1f}%)")
    
    return embedding_matrix

class SyntheticWord2Vec:
    """
    Synthetic Word2Vec-like embeddings for demonstration.
    """
    
    def __init__(self, vocab_size=10000, embedding_dim=100):
        self.embedding_dim = embedding_dim
        self.embeddings = {}
        
        # Create embeddings with some structure
        np.random.seed(42)
        
        # Semantic groups
        groups = {
            'positive': ['good', 'great', 'excellent', 'amazing', 'wonderful', 'fantastic'],
            'negative': ['bad', 'terrible', 'awful', 'horrible', 'poor', 'worst'],
            'royalty': ['king', 'queen', 'prince', 'princess', 'monarch', 'royal'],
            'gender': ['man', 'woman', 'boy', 'girl', 'male', 'female'],
            'country': ['france', 'germany', 'italy', 'spain', 'england', 'america'],
            'capital': ['paris', 'berlin', 'rome', 'madrid', 'london', 'washington']
        }
        
        # Create group centroids
        group_centroids = {}
        for group_name in groups:
            group_centroids[group_name] = np.random.randn(embedding_dim)
        
        # Assign words to groups with noise
        for group_name, words in groups.items():
            centroid = group_centroids[group_name]
            for word in words:
                noise = np.random.randn(embedding_dim) * 0.3
                self.embeddings[word] = centroid + noise
        
        # Add random words
        for i in range(vocab_size - sum(len(w) for w in groups.values())):
            word = f'word_{i}'
            self.embeddings[word] = np.random.randn(embedding_dim) * 0.5
    
    def __contains__(self, word):
        return word in self.embeddings
    
    def __getitem__(self, word):
        return self.embeddings.get(word, np.random.randn(self.embedding_dim) * 0.1)
    
    def similarity(self, word1, word2):
        """Compute cosine similarity between two words."""
        if word1 not in self or word2 not in self:
            return 0.0
        
        vec1 = self[word1]
        vec2 = self[word2]
        
        return np.dot(vec1, vec2) / (np.linalg.norm(vec1) * np.linalg.norm(vec2))
    
    def most_similar(self, word, topn=5):
        """Find most similar words."""
        if word not in self:
            return []
        
        similarities = []
        for other_word in self.embeddings:
            if other_word != word:
                sim = self.similarity(word, other_word)
                similarities.append((other_word, sim))
        
        similarities.sort(key=lambda x: x[1], reverse=True)
        return similarities[:topn]

def word_analogy(model, word_a, word_b, word_c, topn=3):
    """
    Solve word analogy: word_a is to word_b as word_c is to ?
    
    Formula: result = word_c + (word_b - word_a)
    """
    if not all(w in model for w in [word_a, word_b, word_c]):
        return []
    
    vec_a = model[word_a]
    vec_b = model[word_b]
    vec_c = model[word_c]
    
    # Compute target vector
    target = vec_c + (vec_b - vec_a)
    
    # Find closest words
    similarities = []
    for word in model.embeddings:
        if word not in [word_a, word_b, word_c]:
            vec = model[word]
            sim = np.dot(target, vec) / (np.linalg.norm(target) * np.linalg.norm(vec))
            similarities.append((word, sim))
    
    similarities.sort(key=lambda x: x[1], reverse=True)
    return similarities[:topn]

def create_model_with_discriminative_lr(embedding_matrix, max_length):
    """
    Create model with different learning rates for embeddings vs other layers.
    """
    vocab_size, embedding_dim = embedding_matrix.shape
    
    # Build model
    inputs = tf.keras.Input(shape=(max_length,))
    
    # Embedding layer (will have lower learning rate)
    x = tf.keras.layers.Embedding(
        vocab_size, embedding_dim,

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| `gensim` model download fails or is too slow | The Google News Word2Vec model is ~1.6 GB; `gensim.downloader.load` fetches it at runtime, which times out on slow connections | Cache the model locally with `api.load('word2vec-google-news-300', return_path=True)` and reload from disk with `gensim.models.KeyedVectors.load_word2vec_format` |
| Case mismatch between tokenizer and Word2Vec vocabulary | Google News Word2Vec is case-sensitive ('Apple' != 'apple') but the tokenizer may lowercase all tokens, causing 'Apple' (company name) to miss the Word2Vec entry | Try both `word` and `word.capitalize()` and `word.upper()` lookups; fall back to lowercase entry if none found, then to zero/random initialization |
| Word2Vec and GloVe embedding dimensions differ | Word2Vec-Google-News uses 300-d; GloVe comes in 50/100/200/300-d; mixing them in a comparison without standardizing dimensions makes the comparison unfair | Always use the same dimension (100-d for both, via GloVe-100 and Word2Vec truncated to 100-d via PCA or trained at 100-d) when comparing embedding sources |
| `embedding.trainable = True` from epoch 1 corrupts Word2Vec geometry | Allowing gradient flow through 300-d Word2Vec embeddings before the classifier head has converged causes rapid drift away from the pretrained space | Freeze the embedding layer for 2 warm-up epochs, then set `embedding.trainable = True` with a reduced learning rate (10× smaller than the rest of the model) to preserve semantic structure while allowing task adaptation |
| Analogy arithmetic fails for domain-specific words | The Word2Vec analogy `king - man + woman = queen` works on the Google News corpus but fails for technical domains (e.g., `LSTM - memory + gating = ?` returns unrelated words) because the general-domain model lacks domain-specific co-occurrence | For domain-specific analogy tasks, fine-tune a Word2Vec model on domain text (e.g., arXiv ML abstracts) rather than relying on Google News; use `gensim.models.Word2Vec` with `min_count=5` and `vector_size=100` |
