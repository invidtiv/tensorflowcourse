# PART 1: Custom Word Tokenizer (From Scratch)
# ============================================================

class CustomWordTokenizer:
    """
    Custom word tokenizer demonstrating tokenization concepts.
    """
    
    def __init__(self, num_words=None, oov_token='<OOV>'):
        self.num_words = num_words
        self.oov_token = oov_token
        self.word_index = {}
        self.index_word = {}
        self.word_counts = Counter()
        
    def fit_on_texts(self, texts):
        """
        Build vocabulary from texts.
        
        Args:
            texts: List of strings
        """
        # Tokenize all texts and count word frequencies
        for text in texts:
            tokens = self._tokenize(text)
            self.word_counts.update(tokens)
        
        # Build word index (reserve 0 for padding, 1 for OOV)
        self.word_index = {self.oov_token: 1}
        self.index_word = {1: self.oov_token}
        
        # Sort by frequency and take top num_words
        sorted_words = self.word_counts.most_common(self.num_words)
        
        for idx, (word, count) in enumerate(sorted_words, start=2):
            self.word_index[word] = idx
            self.index_word[idx] = word
            
    def _tokenize(self, text):
        """
        Tokenize text into words.
        
        Strategy:
        1. Convert to lowercase
        2. Remove special characters
        3. Split on whitespace
        """
        # Convert to lowercase
        text = text.lower()
        
        # Remove special characters (keep alphanumeric and spaces)
        text = re.sub(r'[^a-z0-9\s]', '', text)
        
        # Split on whitespace
        tokens = text.split()
        
        return tokens
    
    def texts_to_sequences(self, texts):
        """
        Convert texts to sequences of integers.
        """
        sequences = []
        for text in texts:
            tokens = self._tokenize(text)
            sequence = []
            for token in tokens:
                if token in self.word_index:
                    sequence.append(self.word_index[token])
                else:
                    sequence.append(self.word_index[self.oov_token])
            sequences.append(sequence)
        return sequences
    
    def sequences_to_texts(self, sequences):
        """
        Convert sequences back to texts.
        """
        texts = []
        for sequence in sequences:
            words = []
            for idx in sequence:
                if idx in self.index_word:
                    words.append(self.index_word[idx])
                else:
                    words.append(self.oov_token)
            texts.append(' '.join(words))
        return texts

class AdvancedTokenizer(CustomWordTokenizer):
    """
    Advanced tokenizer with contraction handling.
    """
    
    # Common English contractions
    CONTRACTIONS = {
        "n't": "not",
        "'re": "are",
        "'s": "is",
        "'d": "would",
        "'ll": "will",
        "'ve": "have",
        "'m": "am"
    }
    
    def _tokenize(self, text):
        """Tokenize with contraction handling."""
        text = text.lower()
        
        # Handle contractions before removing punctuation
        for contraction, expansion in self.CONTRACTIONS.items():
            text = text.replace(contraction, f" {expansion}")
        
        # Remove remaining special characters
        text = re.sub(r'[^a-z0-9\s]', '', text)
        
        return text.split()

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Vocabulary grows beyond `num_words` | `most_common(self.num_words)` is called but `num_words=None` is the default, so slicing returns the full vocabulary | Always supply an explicit `num_words` value; guard with `if self.num_words else sorted_words` before slicing |
| Index 0 silently used as a real word | The loop starts at `idx=2` but if an upstream caller uses `word_index.get(word, 0)` for unknown words AND a real word also maps to index 0, the pad token collides with that word | Reserve index 0 exclusively for padding by starting the sorted-word enumeration at `idx=2` and never assigning 0 to any real word |
| `sequences_to_texts` returns `<OOV>` for pad zeros | `index_word` does not contain a mapping for index 0, so the `else` branch appends `self.oov_token` for every padding zero when decoding | Add `self.index_word[0] = '<PAD>'` during `fit_on_texts`, or skip zeros explicitly in `sequences_to_texts` |
| `TextVectorization` vocabulary includes empty-string pad token | The layer reserves index 0 for `''` (empty) and index 1 for `[UNK]`; calling `get_vocabulary()[0]` returns `''` not a real word | When mapping output indices to words, offset by 2: `vocab[idx]` where `idx >= 2` maps to real tokens |
| Post-padding truncation discards end of sequence | `pad_sequences(..., truncating='post')` removes tokens from the *end*, losing the sentence terminator or key final tokens | Use `truncating='pre'` to drop leading context instead, or increase `maxlen` to avoid truncation on typical sentences |
