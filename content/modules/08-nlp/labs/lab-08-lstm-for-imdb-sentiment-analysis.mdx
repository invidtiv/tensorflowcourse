# PART 2: LSTM Cell from Scratch
# ============================================================

class LSTMCell(tf.keras.layers.Layer):
    """
    LSTM cell implemented from scratch.
    
    LSTM equations:
    f_t = sigmoid(W_f · [h_{t-1}, x_t] + b_f)  # Forget gate
    i_t = sigmoid(W_i · [h_{t-1}, x_t] + b_i)  # Input gate
    c~_t = tanh(W_c · [h_{t-1}, x_t] + b_c)    # Candidate cell state
    c_t = f_t * c_{t-1} + i_t * c~_t           # Cell state update
    o_t = sigmoid(W_o · [h_{t-1}, x_t] + b_o)  # Output gate
    h_t = o_t * tanh(c_t)                      # Hidden state
    """
    
    def __init__(self, units, **kwargs):
        super(LSTMCell, self).__init__(**kwargs)
        self.units = units
        self.state_size = [units, units]  # h and c
        self.output_size = units
    
    def build(self, input_shape):
        # Combined weights for all gates
        # Input dimension = input_dim + hidden_dim (concatenated)
        input_dim = input_shape[-1] + self.units
        
        # Forget gate
        self.W_f = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_f'
        )
        self.b_f = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_f'
        )
        
        # Input gate
        self.W_i = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_i'
        )
        self.b_i = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_i'
        )
        
        # Candidate cell state
        self.W_c = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_c'
        )
        self.b_c = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_c'
        )
        
        # Output gate
        self.W_o = self.add_weight(
            shape=(input_dim, self.units),
            initializer='glorot_uniform',
            name='W_o'
        )
        self.b_o = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            name='b_o'
        )
        
        super(LSTMCell, self).build(input_shape)
    
    def call(self, inputs, states):
        """
        Single LSTM step.
        
        Args:
            inputs: (batch, input_dim)
            states: [h_prev, c_prev]
            
        Returns:
            h_t, [h_t, c_t]
        """
        h_prev, c_prev = states
        
        # Concatenate input and previous hidden state
        concat = tf.concat([inputs, h_prev], axis=1)
        
        # Forget gate
        f_t = tf.sigmoid(tf.matmul(concat, self.W_f) + self.b_f)
        
        # Input gate
        i_t = tf.sigmoid(tf.matmul(concat, self.W_i) + self.b_i)
        
        # Candidate cell state
        c_tilde = tf.tanh(tf.matmul(concat, self.W_c) + self.b_c)
        
        # Update cell state
        c_t = f_t * c_prev + i_t * c_tilde
        
        # Output gate
        o_t = tf.sigmoid(tf.matmul(concat, self.W_o) + self.b_o)
        
        # Update hidden state
        h_t = o_t * tf.tanh(c_t)
        
        return h_t, [h_t, c_t]
    
    def get_initial_state(self, inputs=None, batch_size=None, dtype=None):
        """Initialize hidden and cell states to zeros."""
        return [
            tf.zeros((batch_size, self.units), dtype=dtype),
            tf.zeros((batch_size, self.units), dtype=dtype)
        ]

# PART 3: LSTM Sentiment Classifier
# ============================================================

def create_lstm_classifier(vocab_size, embedding_dim, lstm_units, max_length):
    """
    Create LSTM-based sentiment classifier.
    """
    model = tf.keras.Sequential([
        # Embedding layer
        tf.keras.layers.Embedding(
            vocab_size, embedding_dim, 
            input_length=max_length,
            mask_zero=True
        ),
        
        # LSTM layer (using built-in for efficiency)
        tf.keras.layers.LSTM(
            lstm_units,
            dropout=0.2,
            recurrent_dropout=0.2,
            return_sequences=False
        ),
        
        # Dense layers
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dropout(0.5),
        tf.keras.layers.Dense(1, activation='sigmoid')
    ])
    
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),
        loss='binary_crossentropy',
        metrics=['accuracy']
    )
    
    return model

def create_model_variant(variant='lstm'):
    """Create model with different RNN variants."""
    model = tf.keras.Sequential()
    model.add(tf.keras.layers.Embedding(vocab_size, embedding_dim, input_length=max_length))
    
    if variant == 'lstm':
        model.add(tf.keras.layers.LSTM(lstm_units, dropout=0.2))
    elif variant == 'gru':
        model.add(tf.keras.layers.GRU(lstm_units, dropout=0.2))
    elif variant == 'bilstm':
        model.add(tf.keras.layers.Bidirectional(
            tf.keras.layers.LSTM(lstm_units // 2, dropout=0.2)
        ))
    elif variant == 'stacked':
        model.add(tf.keras.layers.LSTM(lstm_units, dropout=0.2, return_sequences=True))
        model.add(tf.keras.layers.LSTM(lstm_units // 2, dropout=0.2))
    
    model.add(tf.keras.layers.Dense(64, activation='relu'))
    model.add(tf.keras.layers.Dropout(0.5))
    model.add(tf.keras.layers.Dense(1, activation='sigmoid'))
    
    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
    
    return model

- **Validation diagnostic tree** if test acc is off-band:
  1. Test acc 0.50 ± 0.02 → labels not aligned with sequences (random predictions). Re-check `y_train.shape == (25000,)` and dtype `int32/float32`.
  2. Test acc 0.65–0.80 → probably training without the embedding mask, or `maxlen` too short (< 100). Re-pad with `padding='post'` and `mask_zero=True`.
  3. Test acc 0.86–0.88 → expected. Ship it.
  4. Test acc 0.92+ → suspicious: confirm you're evaluating on `x_test, y_test`, not on a re-split of `x_train`.
  5. Loss = NaN within first 100 steps → exploding gradients. Add `clipnorm=1.0` to Adam, or drop LR to 5e-4.

## ⚠️ Common Pitfalls

1. **Padding without masking:** Calling `pad_sequences(..., padding='pre')` and then feeding into `Embedding(..., mask_zero=False)` means the LSTM consumes hundreds of zero tokens before the real input. The final hidden state is biased toward the padding region and accuracy drops 3–6 pp. Always set `mask_zero=True` and use `padding='post'` so the meaningful state is the most recent.

2. **`recurrent_dropout > 0` silently disables cuDNN:** The LSTM falls back to the generic kernel and is 3–5× slower on GPU. If your T4 epoch jumps from ~20 s to ~80 s after a "small" hyperparameter change, this is almost always why. Keep recurrent_dropout=0 and apply `Dropout` AFTER the LSTM instead.

3. **Binary task wired as 2-class softmax:** IMDB is binary, so the head should be `Dense(1, activation='sigmoid')` with `binary_crossentropy` and labels of shape `(N,)`. Wiring it as `Dense(2, activation='softmax')` with `sparse_categorical_crossentropy` works numerically but doubles the output params, breaks downstream code expecting a single probability, and is the #1 source of shape errors when adding attention or custom metrics later.

4. **Treating test accuracy as transferable:** IMDB train/test are drawn from the same distribution. A model that hits 87% on IMDB typically loses 5–15 pp when zero-shot evaluated on Yelp or Amazon reviews due to domain shift in vocabulary, length, and tone. Don't claim "sentiment LSTM" performance from a single benchmark.

5. **Forgetting to set seeds:** Test accuracy varies by ±1.5 pp across runs purely from LSTM kernel initialization. For reproducible numbers in the report, set `tf.random.set_seed(42)`, `np.random.seed(42)`, and `os.environ['PYTHONHASHSEED'] = '0'` BEFORE the first `import tensorflow`.

## 🚀 Stretch Goals

1. **Bidirectional LSTM:** Wrap the layer as `Bidirectional(LSTM(64))`. Test accuracy typically improves to 0.88–0.89 because the model can use future context for words like "but" / "however" that flip sentiment late in the review. Compare wall-clock and parameter counts.

2. **Pretrained GloVe embeddings (frozen, then fine-tuned):** Load 100-d GloVe vectors for the IMDB vocabulary, initialize the embedding matrix, and freeze it for the first 3 epochs (`embedding.trainable = False`), then unfreeze. Measure how the validation curve changes — pretrained embeddings usually reach the 0.86 band in 1–2 epochs instead of 5.

3. **Length-bucketed batching:** Group reviews of similar length into the same batch (`tf.data.Dataset.bucket_by_sequence_length`). For variable-length IMDB you can drop padding waste by ~40% and shave per-epoch time by 25–35% with no accuracy change.

4. **Attention probe:** Replace the final hidden state with a single-head attention pool over the full sequence: `score = Dense(1)(h_t); alpha = softmax(score); context = sum(alpha_t * h_t)`. Visualize `alpha` for a few correctly-classified and a few misclassified reviews — the attended tokens are usually adjective/negation words and are a useful debugging tool when the model gets a review wrong.

5. **Label-smoothing for calibration:** Replace `binary_crossentropy` with `BinaryCrossentropy(label_smoothing=0.1)`. Test accuracy is unchanged, but the predicted probabilities become better calibrated (Brier score drops ~15%), which matters if you ever feed these scores into a downstream decision system.
