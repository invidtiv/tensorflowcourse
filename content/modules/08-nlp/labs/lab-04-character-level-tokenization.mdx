class CharacterTokenizer:
    """
    Character-level tokenizer.
    Handles any text by breaking into individual characters.
    """
    
    def __init__(self, char_set=None):
        """
        Initialize character tokenizer.
        
        Args:
            char_set: Set of characters to include in vocabulary.
                     If None, uses printable ASCII.
        """
        if char_set is None:
            # Standard character set
            char_set = string.printable
        
        # Create vocabulary
        self.char_to_id = {'<pad>': 0, '<unk>': 1, '<start>': 2, '<end>': 3}
        self.id_to_char = {0: '<pad>', 1: '<unk>', 2: '<start>', 3: '<end>'}
        
        for idx, char in enumerate(char_set, start=4):
            self.char_to_id[char] = idx
            self.id_to_char[idx] = char
            
        self.vocab_size = len(self.char_to_id)
        
    def encode(self, text, add_special_tokens=True):
        """
        Encode text to character IDs.
        
        Args:
            text: Input string
            add_special_tokens: Whether to add <start> and <end>
            
        Returns:
            List of character IDs
        """
        ids = []
        
        if add_special_tokens:
            ids.append(self.char_to_id['<start>'])
        
        for char in text:
            ids.append(self.char_to_id.get(char, self.char_to_id['<unk>']))
            
        if add_special_tokens:
            ids.append(self.char_to_id['<end>'])
            
        return ids
    
    def decode(self, ids, remove_special_tokens=True):
        """
        Decode character IDs to text.
        
        Args:
            ids: List of character IDs
            remove_special_tokens: Whether to remove special tokens
            
        Returns:
            Reconstructed string
        """
        chars = []
        
        for id in ids:
            char = self.id_to_char.get(id, '<unk>')
            
            if remove_special_tokens and char in ['<pad>', '<start>', '<end>']:
                continue
                
            chars.append(char)
            
        return ''.join(chars)
    
    def encode_batch(self, texts, max_length=None, padding=True):
        """
        Encode a batch of texts.
        
        Args:
            texts: List of strings
            max_length: Maximum sequence length
            padding: Whether to pad sequences
            
        Returns:
            Numpy array of shape (batch_size, max_length)
        """
        encoded = [self.encode(text) for text in texts]
        
        if max_length is None:
            max_length = max(len(seq) for seq in encoded)
        
        if padding:
            padded = []
            for seq in encoded:
                if len(seq) < max_length:
                    seq = seq + [self.char_to_id['<pad>']] * (max_length - len(seq))
                else:
                    seq = seq[:max_length]
                padded.append(seq)
            encoded = padded
            
        return np.array(encoded)

def create_char_lm_dataset(text, seq_length=100, tokenizer=None):
    """
    Create dataset for character-level language modeling.
    
    Args:
        text: Training text
        seq_length: Length of input sequences
        tokenizer: Character tokenizer
        
    Returns:
        X, y tensors for training
    """
    if tokenizer is None:
        tokenizer = CharacterTokenizer()
    
    # Encode full text
    encoded = tokenizer.encode(text, add_special_tokens=False)
    
    # Create sequences
    sequences = []
    next_chars = []
    
    for i in range(0, len(encoded) - seq_length):
        sequences.append(encoded[i:i + seq_length])
        next_chars.append(encoded[i + seq_length])
    
    X = np.array(sequences)
    y = np.array(next_chars)
    
    return X, y, tokenizer

def build_char_lstm_model(vocab_size, embedding_dim=64, lstm_units=128):
    """
    Build character-level LSTM for language modeling.
    
    Args:
        vocab_size: Size of character vocabulary
        embedding_dim: Embedding dimension
        lstm_units: LSTM hidden units
        
    Returns:
        Compiled model
    """
    model = tf.keras.Sequential([
        # Embedding layer
        tf.keras.layers.Embedding(vocab_size, embedding_dim),
        
        # LSTM layers
        tf.keras.layers.LSTM(lstm_units, return_sequences=True),
        tf.keras.layers.LSTM(lstm_units),
        
        # Output layer
        tf.keras.layers.Dense(vocab_size, activation='softmax')
    ])
    
    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model

def generate_text(model, tokenizer, seed_text, length=200, temperature=1.0):
    """
    Generate text using trained character model.
    
    Args:
        model: Trained model
        tokenizer: Character tokenizer
        seed_text: Starting text
        length: Number of characters to generate
        temperature: Sampling temperature
        
    Returns:
        Generated text
    """
    generated = seed_text
    
    for _ in range(length):
        # Encode current text
        encoded = tokenizer.encode(generated, add_special_tokens=False)
        
        # Take last seq_length characters
        encoded = encoded[-50:]
        
        # Pad if needed
        while len(encoded) < 50:
            encoded = [tokenizer.char_to_id['<pad>']] + encoded
        
        # Predict
        x_pred = np.array([encoded])
        preds = model.predict(x_pred, verbose=0)[0]
        
        # Apply temperature
        preds = np.log(preds + 1e-10) / temperature
        exp_preds = np.exp(preds)
        preds = exp_preds / np.sum(exp_preds)
        
        # Sample
        next_index = np.random.choice(len(preds), p=preds)
        next_char = tokenizer.id_to_char[next_index]
        
        generated += next_char
    
    return generated

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---|---|---|
| Vocabulary built on training set only but test text has unseen characters | Fitting the `CharacterTokenizer` on training data misses Unicode characters that appear only in test data (e.g., accented letters, emoji) | Add a `byte_fallback=True` option: for unknown characters, encode each UTF-8 byte as a separate token using a 256-entry byte vocabulary |
| Sequence length explosion for long documents | Character-level encoding inflates a 500-word document to ~2500 characters, making attention quadratic in cost | Apply `maxlen` truncation from the left (`truncating='pre'`) to preserve sentence endings, or use a hierarchical encoder (characters → words → sentences) |
| `Dense` output dimension mismatch after character embedding | The character embedding `(batch, seq, embed_dim)` is fed directly to a `Dense(num_classes)` which applies over the last axis only — producing `(batch, seq, num_classes)` instead of `(batch, num_classes)` | Add a `GlobalAveragePooling1D()` or `LSTM(..., return_sequences=False)` before the final `Dense` to collapse the time dimension |
| Unicode normalization differences between train and test | `"é"` can be encoded as a single precomposed character (U+00E9) or as `e + combining accent` (U+0065 + U+0301); these are different tokens if normalization is not applied | Apply `unicodedata.normalize('NFC', text)` to all text before tokenization to collapse equivalent Unicode representations |
| Confusion matrix distorted by padding tokens | When computing per-class accuracy for NER, padding positions (index 0) are included in the macro average, artificially inflating the `O` class precision | Mask padding positions using `sample_weight = (y_true != 0).astype(float)` when calling `model.evaluate` or computing metrics manually |

1. **Byte-level BPE (GPT-2 style):** Replace the character vocabulary with a 256-entry byte vocabulary and run BPE merges on byte sequences. 2. **Hierarchical character-to-word encoder:** Build a two-level encoder: first a 1D CNN or small LSTM reads character embeddings within each word (kernel sizes 2, 3, 4; max-pool to a fixed-size word vector), then a second LSTM reads the resulting word vectors for classification. This is the ELMo predecessor architecture (Kim et al., 2015, "Character-Aware Neural Language Models"). Evaluate on the CoNLL 2003 NER dataset and compare F1 against a word-level baseline — character models typically gain +1–3 F1 on rare proper nouns.

3. **Character-level language model with nucleus sampling:** Train a character-level LSTM LM on a text corpus (e.g., the complete works of Shakespeare, ~5M characters) and implement nucleus sampling (`top_p` sampling): at each step, keep only the top tokens whose cumulative probability exceeds `p`, renormalize, and sample. Compare generated text quality for `p ∈ {0.5, 0.9, 0.95, 1.0}` (greedy, restricted, permissive, uniform). Measure character-level perplexity on a 50k-character test split; a well-trained 2-layer LSTM should reach perplexity < 3.5 on Shakespeare.

4. **Contextual character embeddings via ELMo comparison:** After training the character CNN encoder, extract its output word vectors for the 500 most common words and compare them with GloVe 100-d vectors using cosine similarity. Compute the Spearman correlation between the two similarity matrices on the WordSim-353 benchmark. Character-based vectors should correlate moderately (ρ ≈ 0.5–0.65) with GloVe; lower correlation indicates that character structure alone does not capture full distributional semantics, motivating the combination approach used in ELMo.
