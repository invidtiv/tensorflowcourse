import numpy as np

def benchmark_numpy_vs_tf(operation_name, numpy_fn, tf_fn, sizes):
    """Benchmark NumPy vs TensorFlow operations."""
    numpy_times = []
    tf_times = []
    
    for size in sizes:
        # NumPy
        x_np = np.random.randn(size, size).astype(np.float32)
        start = time.time()
        for _ in range(10):
            _ = numpy_fn(x_np)
        numpy_times.append((time.time() - start) / 10)
        
        # TensorFlow
        x_tf = tf.constant(x_np)
        # Warmup
        _ = tf_fn(x_tf)
        start = time.time()
        for _ in range(10):
            _ = tf_fn(x_tf)
        tf_times.append((time.time() - start) / 10)
    
    return numpy_times, tf_times

# Matrix multiplication
sizes = [100, 500, 1000, 2000]
numpy_times, tf_times = benchmark_numpy_vs_tf(
    "Matrix Multiplication",
    lambda x: x @ x,
    lambda x: tf.matmul(x, x),
    sizes
)

print(f"\nMatrix Multiplication (X @ X):")
print(f"{'Size':<10} {'NumPy (ms)':<15} {'TensorFlow (ms)':<15} {'Speedup':<10}")
print("-" * 55)
for i, size in enumerate(sizes):
    speedup = numpy_times[i] / tf_times[i] if tf_times[i] > 0 else float('inf')
    print(f"{size:<10} {numpy_times[i]*1000:<15.3f} {tf_times[i]*1000:<15.3f} {speedup:<10.2f}x")

def broadcasting_benchmark():
    """Compare broadcasting vs explicit operations."""
    sizes = [100, 500, 1000, 2000, 5000]
    broadcast_times = []
    explicit_times = []
    
    for size in sizes:
        A = np.random.randn(size, size).astype(np.float32)
        b = np.random.randn(size).astype(np.float32)
        
        # Broadcasting
        start = time.time()
        for _ in range(100):
            _ = A + b
        broadcast_times.append((time.time() - start) / 100)
        
        # Explicit (tile)
        start = time.time()
        for _ in range(100):
            _ = A + np.tile(b, (size, 1))
        explicit_times.append((time.time() - start) / 100)
    
    return sizes, broadcast_times, explicit_times

sizes, broadcast_times, explicit_times = broadcasting_benchmark()

print(f"\nBroadcasting vs Explicit (A + b):")
print(f"{'Size':<10} {'Broadcast (ms)':<20} {'Explicit (ms)':<20} {'Speedup':<10}")
print("-" * 65)
for i, size in enumerate(sizes):
    speedup = explicit_times[i] / broadcast_times[i]
    print(f"{size:<10} {broadcast_times[i]*1000:<20.3f} {explicit_times[i]*1000:<20.3f} {speedup:<10.2f}x")

def memory_layout_benchmark():
    """Compare C-order vs F-order performance."""
    size = 5000
    
    # C-order (row-major)
    arr_c = np.random.randn(size, size).astype(np.float32)
    
    # F-order (column-major)
    arr_f = np.asfortranarray(arr_c)
    
    # Row-wise sum (faster for C-order)
    start = time.time()
    for _ in range(100):
        _ = np.sum(arr_c, axis=1)
    row_c_time = (time.time() - start) / 100
    
    start = time.time()
    for _ in range(100):
        _ = np.sum(arr_f, axis=1)
    row_f_time = (time.time() - start) / 100
    
    # Column-wise sum (faster for F-order)
    start = time.time()
    for _ in range(100):
        _ = np.sum(arr_c, axis=0)
    col_c_time = (time.time() - start) / 100
    
    start = time.time()
    for _ in range(100):
        _ = np.sum(arr_f, axis=0)
    col_f_time = (time.time() - start) / 100
    
    return row_c_time, row_f_time, col_c_time, col_f_time

row_c, row_f, col_c, col_f = memory_layout_benchmark()

print(f"\nMemory Layout Performance (5000x5000 matrix):")
print(f"  Row-wise sum - C-order: {row_c*1000:.3f} ms, F-order: {row_f*1000:.3f} ms")
print(f"  Column-wise sum - C-order: {col_c*1000:.3f} ms, F-order: {col_f*1000:.3f} ms")
print(f"  Row-wise speedup (C vs F): {row_f/row_c:.2f}x")
print(f"  Column-wise speedup (F vs C): {col_c/col_f:.2f}x")

def vectorization_patterns():
    """Demonstrate common vectorization patterns."""
    n = 10000
    
    # Pattern 1: Distance matrix
    points = np.random.randn(n, 3).astype(np.float32)
    
    # Loop version
    start = time.time()
    dist_loop = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            dist_loop[i, j] = np.sqrt(np.sum((points[i] - points[j])**2))
    loop_time = time.time() - start
    
    # Vectorized version
    start = time.time()
    sum_sq = np.sum(points**2, axis=1)
    dist_vec = np.sqrt(sum_sq[:, None] + sum_sq[None, :] - 2 * points @ points.T)
    vec_time = time.time() - start
    
    print(f"\nDistance Matrix ({n}x{n}):")
    print(f"  Loop time: {loop_time:.3f}s")
    print(f"  Vectorized time: {vec_time:.3f}s")
    print(f"  Speedup: {loop_time/vec_time:.1f}x")
    
    # Pattern 2: Batch operations
    batch_size = 1000
    features = 100
    X = np.random.randn(batch_size, features).astype(np.float32)
    w = np.random.randn(features).astype(np.float32)
    
    # Loop version
    start = time.time()
    for _ in range(100):
        result_loop = np.zeros(batch_size)
        for i in range(batch_size):
            result_loop[i] = np.dot(X[i], w)
    loop_time = (time.time() - start) / 100
    
    # Vectorized version
    start = time.time()
    for _ in range(100):
        result_vec = X @ w
    vec_time = (time.time() - start) / 100
    
    print(f"\nBatch Dot Product ({batch_size}x{features}):")
    print(f"  Loop time: {loop_time*1000:.3f} ms")
    print(f"  Vectorized time: {vec_time*1000:.3f} ms")
    print(f"  Speedup: {loop_time/vec_time:.1f}x")

vectorization_patterns()
