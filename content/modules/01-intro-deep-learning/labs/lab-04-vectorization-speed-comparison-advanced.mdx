def benchmark_numpy_vs_tf(operation_name, numpy_fn, tf_fn, sizes):
    """Benchmark NumPy vs TensorFlow operations."""
    numpy_times = []
    tf_times = []
    
    for size in sizes:
        # NumPy
        x_np = np.random.randn(size, size).astype(np.float32)
        start = time.time()
        for _ in range(10):
            _ = numpy_fn(x_np)
        numpy_times.append((time.time() - start) / 10)
        
        # TensorFlow
        x_tf = tf.constant(x_np)
        # Warmup
        _ = tf_fn(x_tf)
        start = time.time()
        for _ in range(10):
            _ = tf_fn(x_tf)
        tf_times.append((time.time() - start) / 10)
    
    return numpy_times, tf_times

def broadcasting_benchmark():
    """Compare broadcasting vs explicit operations."""
    sizes = [100, 500, 1000, 2000, 5000]
    broadcast_times = []
    explicit_times = []
    
    for size in sizes:
        A = np.random.randn(size, size).astype(np.float32)
        b = np.random.randn(size).astype(np.float32)
        
        # Broadcasting
        start = time.time()
        for _ in range(100):
            _ = A + b
        broadcast_times.append((time.time() - start) / 100)
        
        # Explicit (tile)
        start = time.time()
        for _ in range(100):
            _ = A + np.tile(b, (size, 1))
        explicit_times.append((time.time() - start) / 100)
    
    return sizes, broadcast_times, explicit_times

def memory_layout_benchmark():

    start = time.time()
    for _ in range(100):
        _ = np.sum(arr_f, axis=0)
    col_f_time = (time.time() - start) / 100

    sum_sq = np.sum(points**2, axis=1)

    print(f"  Vectorized time: {vec_time:.3f}s")
    print(f"  Speedup: {loop_time/vec_time:.1f}x")

    features = 100

    w = np.random.randn(features).astype(np.float32)

## Common Pitfalls

| Pitfall | Why It Happens | Fix |
|---------|----------------|-----|
| **Benchmarking TensorFlow before the first call completes tracing** | TF uses `@tf.function` tracing on the first call, adding ~0.5–2 s overhead. Naively timing the very first call inflates TF timings by orders of magnitude. | Always run one warm-up call before starting the timer: `_ = tf_fn(x); start = time.time(); ...`. |
| **Measuring loop overhead with small arrays** | When arrays are tiny (e.g., 10×10), Python overhead dominates and vectorized code shows no advantage — even appearing slower in some cases. | Use sizes ≥ 500×500 to measure meaningful speedups; always plot speedup *vs. size* to see the crossover point. |
| **`np.tile` copying data vs broadcasting in-place** | `np.tile` allocates a full expanded array before the addition, doubling peak memory. Broadcasting never materializes the repeated copy. | Prefer broadcasting whenever the semantics allow; only fall back to `np.tile` when you explicitly need the repeated array for indexing or assignment. |
| **Forgetting `astype(np.float32)` before benchmarking** | Default NumPy arrays are float64. TensorFlow and GPUs are optimized for float32. Mixed-precision comparisons give misleading speedup numbers. | Cast both NumPy and TF tensors to float32 before timing. Add `assert arr.dtype == np.float32` at the start of each benchmark. |
| **C-order vs F-order confusion in multi-axis reductions** | Row-wise reductions (`axis=1`) are cache-friendly for C-order arrays; column-wise reductions (`axis=0`) are cache-friendly for F-order. Applying the wrong reduction axis to the wrong layout causes unexpected slowdowns. | Use `np.isfortran(arr)` to check memory layout. Default NumPy arrays are C-order — reshape and transpose before heavy column-wise reductions if performance is critical. |
