BROADCASTING RULES:
====================

1. Align arrays on the right (trailing dimensions)
2. Dimensions are compatible when:
   - They are equal, OR
   - One of them is 1
3. The result has the maximum size along each dimension

Example:
  Array A shape: (3, 4, 1, 5)
  Array B shape:    (1, 6, 5)
  Aligned:       (3, 4, 1, 5)
                 (   1, 6, 5)
  Result:        (3, 4, 6, 5)

print("\n--- Example 5: 3D Broadcasting ---")
image = np.random.rand(10, 10, 3)  # RGB image

print("\n--- Application 1: Feature Normalization ---")
data = np.random.randn(100, 5) * [10, 5, 2, 1, 0.5] + [5, 0, -5, 10, 2]
mean = np.mean(data, axis=0)
std = np.std(data, axis=0)

# Example 1: Scalar broadcasting
scalar_tf = tf.constant(5)
vector_tf = tf.constant([1, 2, 3, 4])

print(f"Scalar + Vector: {result_tf}")

matrix_tf = tf.ones([3, 4])
row_tf = tf.constant([1, 2, 3, 4])
result_tf = matrix_tf + row_tf
print(f"Matrix + Row:\n{result_tf}")

# Example 3: Softmax using broadcasting
def softmax(x, axis=-1):
    """Softmax using broadcasting."""
    x_max = tf.reduce_max(x, axis=axis, keepdims=True)
    x_shifted = x - x_max
    exp_x = tf.exp(x_shifted)
    return exp_x / tf.reduce_sum(exp_x, axis=axis, keepdims=True)

logits = tf.constant([[1.0, 2.0, 3.0], [1.0, 2.0, 3.0]])
probs = softmax(logits)
print(f"\nSoftmax input:\n{logits}")
print(f"Softmax output:\n{probs}")
print(f"Sum along axis: {tf.reduce_sum(probs, axis=-1)}")

axes[0, 1].set_title('2D Gaussian (Broadcasting)')

axes[0, 2].set_xlabel('b')

# Distance matrix visualization

dist_matrix = np.sqrt(np.sum((points[:, None, :] - points[None, :, :])**2, axis=2))
im = axes[1, 0].imshow(dist_matrix, cmap='hot', aspect='auto')
axes[1, 0].set_title('Distance Matrix')

axes[1, 0].set_ylabel('Point Index')
for i in range(len(points)):
    for j in range(len(points)):
        axes[1, 0].text(j, i, f'{dist_matrix[i, j]:.1f}', 
                       ha='center', va='center', color='white', fontsize=8)
plt.colorbar(im, ax=axes[1, 0])

axes[1, 1].legend()

for i in range(3):
    axes[1, 2].hist(normalized[:, i], alpha=0.3, bins=20, label=f'Feature {i+1} (norm)')
axes[1, 2].set_title('Normalized Data Distribution')
axes[1, 2].legend()
